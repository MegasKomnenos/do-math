37def requirement_change_instance(root, shared, *, completed):
38 """``restore_template`` build: synthetic active state, solved premise, published requirement ``writeup``.
39
40 Mathematical proof construction is covered by ``solved_instance``. Only that
41 premise is synthetic (``_solution_basis`` returns ``SYNTHETIC_BASIS``, a
42 process-local patch each test sets again on its engine); controls, followup
43 and storage are real. ``completed`` adds requirement ``remaining``, which
44 keeps the instance open for public requirement changes, and completes the
45 ``writeup`` task through the real assessment rules.
46 """
47 from state_fixtures import active_instance, complete_academic_task
48 instance_id = active_instance(root, shared, config={})
49 engine = ResearchEngine(root / "repository", instance_id)
50 with patch.object(engine, "_solution_basis", return_value=SYNTHETIC_BASIS):
51 engine.instance["goal_status"] = "mathematically_solved"
52 engine._save()
53 requirements = [requirement(["draft account"])]
54 if completed:
55 requirements.append({**requirement(["Independent remaining account"]), "id": "remaining"})
56 send_control(engine, "decision", config={"change": charter_change(engine,
57 academic_requirements=requirements)})
58 engine._ensure_academic_followup()
59 engine._save()
60 if completed:
61 complete_academic_task(engine, academic_tasks(engine)[0], SYNTHETIC_BASIS)
62 return instance_id
