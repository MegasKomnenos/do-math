44def provider_adapter(provider: str, cli_path: str | None = None) -> LocalCLIAdapter:
45 """The adapter of a provider; its CLI defaults to the executable named like the provider."""
46 if provider not in ADAPTERS:
47 raise ValueError("unsupported profile provider")
48 return ADAPTERS[provider](cli_path or shutil.which(provider) or provider)

50
51def request_provider(request: dict) -> str:
52 """The provider whose worker request carries this executable key."""
53 providers = [name for name, adapter in ADAPTERS.items() if adapter.executable_key in request]
54 if len(providers) != 1:
55 raise ValueError("worker request names no single provider executable")
56 return providers[0]
