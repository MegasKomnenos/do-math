79def decision_basis_changes(basis: dict, instance: dict, runtime: dict) -> list[dict]:
80 from . import knowledge, planning

82 return [{"id": None, "field": "decision_basis", "reason": "missing_basis"}]

97 if prefix_hash != row["results_sha256"]:
98 add(identifier, "results_sha256", row["results_sha256"], prefix_hash)
99 else:
100 current = knowledge.decision_reference(record, instance)
101 for field in ("revision", "content_sha256", "use_suspended", "trust_environment_id", "trust_environment_state", "source_basis"):
102 if current.get(field) != row.get(field):
103 add(identifier, field, row.get(field), current.get(field))
104 if current["status"] != row["status"] and (row["status"], current["status"]) != ("reviewing", "verified"):
105 add(identifier, "status", row["status"], current["status"])
106 if row["usable"] and not current["usable"]:
107 add(identifier, "usable", True, False)
108 if not sources_current(row["source_basis"], instance):
109 add(identifier, "source_content", row["source_basis"], "changed")
110 if not sources_current(basis.get("sources", []), instance):
111 add(None, "sources", basis["sources"], "changed")
