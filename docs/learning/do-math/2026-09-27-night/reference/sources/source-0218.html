23def enqueue(h,role,target=None,**fields):
24 """Queue through the product and return the row it added or merged into."""
25 h._enqueue(role,target,**fields)
26 return next(a for a in reversed(h.runtime['queue']) if a['role']==role and a.get('target')==target)
