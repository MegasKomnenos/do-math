31def available_input_refs(instance: dict, runtime: dict, *, lean_available: bool) -> list[str]:
32 """Input references a role may copy into a task, derived on demand and never stored.
33
34 Fixed documents, configured tools, knowledge, tasks and source aliases come
35 from state. Whether a local Lean was discovered is a host fact the caller passes.
36 """
37 from .input_basis import aliases
38 config = instance.get("config", {})
39 result = set(FIXED_INPUT_REFS)
40 if config.get("lean_command") or lean_available:
41 result.add("tool:lean")
42 if config.get("scholarly_enabled") and config.get("scholarly_server"):
43 result.add("tool:scholarly")
44 result.update(instance.get("knowledge", {}))
45 result.update(runtime.get("tasks", {}))
46 for source in instance.get("sources", []):
47 result.update(aliases(source))
48 return sorted(result)

69def _distinct_ids(rows: list[dict], name: str) -> None:
70 values = [row.get("id") for row in rows if isinstance(row, dict)]
71 if len(values) != len(rows) or any(not isinstance(value, str) or not value for value in values) or len(values) != len(set(values)):

106def judge_coverage_errors(role: str, execution_id: str, task: dict, assessment: dict, families: dict) -> list[str]:
107 """A fulfilment by an assessor outside an obligation's judges must cite its judges' evidence.
108
109 `families` maps the cited IDs the caller can see to their record family. The
110 producer derives it from its catalog; the receiver from current records and
111 then also checks that the cited judgment is current (task_scope.designated_judgment).
112 """
113 obligations = {row["id"]: row for row in contract_obligations(task)}
114 errors = []
115 for row in assessment.get("item_dispositions", []):
116 obligation = obligations.get(row.get("obligation_id"))
117 if row.get("kind") != "fulfilled" or obligation is None:
118 continue
119 judges = obligation["judges"]
120 if role in judges or execution_id in judges:
121 continue
122 wanted = {JUDGE_EVIDENCE_FAMILIES[judge] for judge in judges if judge in JUDGE_EVIDENCE_FAMILIES}
123 if not any(families.get(ref) in wanted for ref in row.get("evidence_refs", [])):
124 errors.append("fulfilled obligation " + row["obligation_id"] + " needs its designated judge "
125 + repr(sorted(judges)) + " or cited evidence of the family " + repr(sorted(wanted)))
126 return errors

129def _row_by_obligation(assessment: dict) -> dict[str, dict]:
130 rows = assessment.get("item_dispositions", [])
131 if not isinstance(rows, list):
132 raise ValueError("item_dispositions must be a list")
133 result = {}
134 for row in rows:
135 if not isinstance(row, dict):
136 raise ValueError("item disposition must be an object")
137 identifier = row.get("obligation_id")
138 if not isinstance(identifier, str) or not identifier or identifier in result:
139 raise ValueError("item dispositions must identify each obligation at most once")
140 if row.get("kind") not in DISPOSITION_KINDS:
141 raise ValueError("unknown item disposition")
142 if "evidence_refs" not in row:
143 raise ValueError("item disposition requires evidence_refs")
144 evidence = row["evidence_refs"]
145 if not isinstance(evidence, list) or not all(isinstance(value, str) and value for value in evidence) or len(evidence) != len(set(evidence)):
146 raise ValueError("item disposition evidence_refs must be distinct references")
147 result[identifier] = row
148 return result

151def validate_item_dispositions(task: dict, assessment: dict, *, require_complete: bool = False) -> None:
152 validate_task_contract(task)
153 if not isinstance(assessment, dict):
154 raise ValueError("assessment must be an object")
155 terminal = assessment.get('decision') in {'completed', 'closed'}
156 require_complete = require_complete or terminal
157 if task.get('status') in {'completed', 'closed'} and assessment.get('decision') != task['status']:
158 raise ValueError('terminal task status differs from its assessment decision')
159 obligations = {row["id"]: row for row in contract_obligations(task)}
160 rows = _row_by_obligation(assessment)
161 if set(rows) - set(obligations):
162 raise ValueError("item disposition names an unknown obligation")
163 unresolved = assessment.get("unresolved", [])
164 if not isinstance(unresolved, list):
165 raise ValueError("unresolved must be a list")
166 unresolved_ids = set()
167 for row in unresolved:
168 if not isinstance(row, dict) or set(row) != {"obligation_id", "missing_inputs", "next_owner"}:
169 raise ValueError("unresolved inspection needs obligation_id, missing_inputs, and next_owner")
170 identifier = row["obligation_id"]
171 if identifier not in obligations or identifier in unresolved_ids or identifier in rows:
172 raise ValueError("unresolved inspection must name one undisposed obligation")
173 unresolved_ids.add(identifier)
174 if not isinstance(row["missing_inputs"], list) or not row["missing_inputs"] or any(not isinstance(v,str) or not v for v in row['missing_inputs']):
175 raise ValueError("unresolved inspection needs missing inputs")
176 require_text(row["next_owner"], "unresolved.next_owner")
177 for identifier, row in rows.items():
178 obligation = obligations[identifier]
179 required = set(obligation.get("evidence_required", []))
180 if row["kind"] == "fulfilled" and not required <= set(row["evidence_refs"]):
181 raise ValueError("item disposition lacks required evidence: " + identifier)
182 if row["kind"] == "waived":
183 if not isinstance(row.get("authority_ref"), str) or not row["authority_ref"]:
184 raise ValueError("waived obligation needs authority_ref")
185 if row["kind"] == "transferred":
186 if not all(isinstance(row.get(key), str) and row[key] for key in ("authority_ref", "target_task_id", "target_obligation_id")):
187 raise ValueError("transferred obligation needs authority and target obligation")
188 if set(obligations) != set(rows) | unresolved_ids:
189 raise ValueError("every obligation requires a disposition or unresolved inspection")
190 if require_complete and (set(obligations) - set(rows) or unresolved_ids):
191 raise ValueError("unresolved or undisposed completion obligations remain")
192 if require_complete and any(row['kind'] == 'open' for row in rows.values()):
193 raise ValueError("terminal assessment retains open obligations")
194 if assessment.get('decision') == 'completed' and any(row['kind'] == 'transferred' for row in rows.values()):
195 raise ValueError('completed assessment retains transferred obligations; use closed for authorized scope termination')
196 if assessment.get('decision') == 'closed' and not any(row['kind'] in {'waived', 'transferred'} for row in rows.values()):
197 raise ValueError('closure needs an authoritative scope disposition')

200def remaining_obligations(task: dict, assessment: dict | None = None) -> list[str]:
201 """Compute residual task scope from current obligation dispositions."""
202 obligations = {row["id"] for row in contract_obligations(task)}
203 if assessment:
204 validate_item_dispositions(task, assessment)
205 rows = _row_by_obligation(assessment or {})
206 closed = {identifier for identifier, row in rows.items() if row["kind"] in {"fulfilled", "waived", "transferred"}}
207 return sorted(obligations - closed)

224def validate_plan_dispositions(plan: dict, tasks: dict, ready_ids: list[str]) -> None:
225 """The meaning of a plan the research_plan wire schema has already shaped (PLAN_DISPOSITION)."""
226 if set(ready_ids)-set(tasks):
227 raise ValueError('ready population names an unknown task')
228 supplied = [row['task_id'] for row in plan['dispositions']]
229 counts = Counter(supplied)
230 if len(supplied) != len(counts) or set(counts) != set(ready_ids):
231 raise ValueError("plan must dispose every derived-ready task exactly once: "
232 f"missing={sorted(set(ready_ids) - set(counts))}; "
233 f"unexpected={sorted(set(counts) - set(ready_ids))}; "
234 f"duplicate={sorted(identifier for identifier, count in counts.items() if count > 1)}")
235 for row in plan['dispositions']:
236 if row['kind'] == 'selected':
237 require_text(row.get('expected_research_change'), 'expected_research_change')
238 else:
239 revisit = row.get('revisit', {})
240 if not revisit.get('event_refs') and not revisit.get('at'):
241 raise ValueError('nonselection requires a reobservation condition')
242 if revisit.get('at'):
243 from .file_io import parse_utc
244 parse_utc(revisit['at'])
245 if row['kind'] == 'not_assessable' and not row.get('missing_inputs'):
246 raise ValueError('not_assessable requires missing inputs')
247 parallel = row.get('parallel_task_ids', [])
248 if len(parallel) != len(set(parallel)) or set(parallel) - set(tasks) or row['task_id'] in parallel:
249 raise ValueError('parallel references must identify distinct other tasks')
250 assessments = plan['task_assessments']
251 assessed = [row['task_id'] for row in assessments]
252 if len(assessed) != len(set(assessed)) or set(assessed) != set(returned_task_ids(tasks)):
253 raise ValueError('plan must assess every returned task exactly once')
254 if set(selected_task_ids(plan)) & set(assessed):
255 raise ValueError('returned tasks require disposition before a later selection')
256 for assessment in assessments:
257 task = tasks.get(assessment["task_id"])
258 if task is None:
259 raise ValueError("assessment names an unknown task")
260 validate_item_dispositions(task, assessment, require_complete=assessment.get("decision") in {"completed", "closed"})

263def repetition_lineages(runtime: dict, task_id: str | None = None, *,
264 include_events: bool = True) -> dict[str, dict]:
265 """Count independent histories without turning a count into an authority decision."""
266 rows = runtime.get("history", [])
267 result = {name: {"count": 0, "owner": "ai", "events": []} for name in REPEAT_LINEAGES}
268 baselines={}
269 for row in rows:
270 if not isinstance(row, dict) or row.get('kind') != 'task_repetition' or row.get('lineage') not in result:
271 continue
272 if task_id is not None and row['lineage'] == 'contract_revision' and row['lineage_id'] != task_id:
273 continue
274 name = row['lineage']
275 counts = result[name].setdefault('by_lineage', {})
276 key = row['lineage_id']
277 if row['outcome']=='continued':baselines[name,key]=row.get('baseline_revision',0)
278 if row['outcome'] in {'accepted', 'contracts_preserved', 'continued'}:
279 counts[key] = 0
280 elif 'version' in row:
281 counts[key] = row['version']-baselines.get((name,key),0)
282 elif row['outcome'] not in {'premises_changed', 'capacity_wait', 'input_overflow', 'waiting', 'transport_failure','unclassified_failure'}:
283 counts[key] = counts.get(key, 0) + 1
284 result[name]["count"] = sum(counts.values())
285 if include_events:
286 result[name]["events"].append(deepcopy(row))
287 if row.get("owner") == "user":
288 result[name]["owner"] = "user"
289 return result
