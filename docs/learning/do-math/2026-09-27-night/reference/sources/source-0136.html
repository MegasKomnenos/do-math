66 def test_receiver_rejects_bad_payload_once_then_processes_following_valid_control(self):
67 f = self
68 bad_ids = []
69 for index, (action, config) in enumerate([('pause', [1]), *BAD_PAYLOADS]):
70 identifier = 'UC-invalid-' + str(index)
71 control(f.git, QUERY_INSTANCE_ID, 'pause', request_id=identifier)
72 path = f.workspace / 'control' / QUERY_INSTANCE_ID / (identifier + '.json')
73 request = json.loads(path.read_bytes())
74 request.update(action=action, config=config)
75 path.write_bytes(canonical_json(request))
76 bad_ids.append(identifier)
77 control(f.git, QUERY_INSTANCE_ID, 'pause', text='valid later request', request_id='UC-good')
78 engine = ResearchEngine(f.git, QUERY_INSTANCE_ID)
79 engine.process_controls()
80 for identifier in bad_ids:
81 self.assertEqual(engine.runtime['processed_controls'].count(identifier), 1)
82 self.assertEqual(engine.control_result(identifier)['effect'], 'rejected')
83 self.assertEqual(engine.control_result('UC-good')['effect'], 'applied')
84 engine = ResearchEngine(f.git, QUERY_INSTANCE_ID)
85 engine.process_controls()
86 self.assertTrue(all(engine.runtime['processed_controls'].count(identifier) == 1 for identifier in bad_ids))
