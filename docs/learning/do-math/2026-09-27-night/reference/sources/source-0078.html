14def question_view(instance, runtime, read_many):
15 """Question links with the text each resolves to in its original source.
16
17 ``read_many(paths)`` reads the source artifacts of one snapshot. A source
18 that no longer resolves leaves the text None and marks it unavailable.
19 """
20 rows = deepcopy(runtime.get("question_links", []))
21 refs = list(dict.fromkeys(row["source_ref"] for row in rows if row.get("source_ref")))
22 sources = {ref: json.loads(raw) for ref, raw in read_many(refs).items()} if refs else {}
23 for row in rows:
24 decision = instance.get('decisions', {}).get(row.get('resolution_ref'))
25 if decision:
26 row['resolution'] = {key: deepcopy(decision[key]) for key in ('id', 'kind', 'change', 'source_ref', 'created_at') if key in decision}
27 try:
28 source = (sources[row["source_ref"]] if row.get("source_ref") else
29 runtime["intent_imports"][row["source_id"]] if row.get("source_kind") == "intent_import" else
30 runtime["review_cycles"][row["source_id"]] if row.get("source_kind") == "review_cycle" else
31 runtime["history"] if row.get("source_kind") == "history" else
32 runtime["executions"][row["origin_execution_id"]])
33 field = row["origin_field"]
34 row["text"] = resolve_pointer(source, field)
35 if row.get("question_kind") == "grant":
36 row["text"] = "External action requires a user grant: " + row["text"]
37 except (KeyError, IndexError, ValueError, TypeError):
38 row["text"] = None
39 row["source_status"] = "unavailable"
40 return rows

52 def _record_output_questions(self, action, output, execution_id, *, public=False, source_ref=None):
53 # v5 intent questions are published only by governance.apply_import after check.
54 if action.get("role") == "intent_interpretation":
55 return
56 if not output.get("questions") or (action.get("review_cycle_id") and not public):
57 return
58 execution = self.runtime["executions"][execution_id]
59 source = None
60 review_source = None
61 if public and action.get("review_cycle_id"):
62 cycle = self.runtime["review_cycles"][action["review_cycle_id"]]
63 review_source = next(key for key, row in cycle["results"].items() if row["execution_id"] == execution_id)
64 else:
65 source = source_ref or (execution.get("submission_intent", {}).get("output_ref") or
66 f"runtime/artifacts/{execution['artifact_bundle']}/output.json")
67 routes = {row["question_index"]: row for row in execution.get("extensions", {}).get("question_routes", [])}
68 default_owner = "operations" if action["role"] in questions.OPERATIONS_ROLES else "research"
69 for index, text in enumerate(output.get("questions", [])):
70 route = routes.get(index, {})
71 owner = route.get("owner", default_owner)
72 targets = route.get("target_ids", [self.instance["id"]] if owner == "user" else [action["target"]] if action.get("target") else [])
73 field = ("results/" + review_source + "/output/" if review_source else "") + "questions/" + str(index)
74 row = questions.link(self.runtime, execution_id=execution_id, field=field,
75 owner=owner, targets=targets, source_ref=source, reason=route.get("reason"), blocks=route.get("blocks", False))
76 if review_source:
77 row.update(source_kind="review_cycle", source_id=action["review_cycle_id"])

79 def _apply_question_resolutions(self, execution_id):
80 extension = self.runtime["executions"][execution_id].get("extensions", {})
81 rows = deepcopy(self.runtime.get("question_links", []))
82 by_id = {row["id"]: row for row in rows}
83 role = self.runtime["executions"][execution_id].get("role")
84 seen = set()
85 for resolution in extension.get("question_resolutions", []):
86 identifier = resolution["question_id"]
87 # The receiver's schema already fixed disposition, reason and evidence.
88 error = questions.resolution_error(None if identifier in seen else by_id.get(identifier), role)
89 if error:
90 raise ValueError(error)
91 seen.add(identifier)
92 row = by_id[identifier]
93 for ref in resolution["evidence_refs"]:
94 academic.evidence_record(self.instance, self.runtime, ref, self._academic_evidence)
95 questions.close(row, kind="ai_resolution", status=resolution["disposition"], resolution_ref=execution_id,
96 affected_ids=questions.open_targets(row),
97 evidence={"evidence_refs": resolution["evidence_refs"], "reason": resolution["reason"]})
98 self.runtime["question_links"] = rows
