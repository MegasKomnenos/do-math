40 def test_blocked_machine_does_not_hide_a_smaller_ready_machine(self):
41 instance,selected,constraint=host_capacity_fixture()
42 big=profiles.with_host_resources(selected['formal_check'],constraint,
43 {'jobs':1,'memory_bytes':384*1024**2,'cpu_rate_basis_points':6000})
44 original=next(p for p in instance['execution_profiles'].values() if p['role']=='trust_check')
45 small=profiles.with_host_resources(original,constraint,

50 held=selected['research_work']
51 self.assertTrue(slots.acquire('EX-held',

53 scheduler=bare_engine(instance,runtime)
54 scheduler._execution_profile=create_autospec(scheduler._execution_profile,
55 side_effect=lambda action:profiles.resolve_profile(instance['execution_profiles'],action['profile_ref']))
56 scheduler._action_lifecycle_gate=create_autospec(scheduler._action_lifecycle_gate,
57 side_effect=lambda action:scheduler._execution_profile(action))
58 scheduler._revision_dispatch_waits=create_autospec(scheduler._revision_dispatch_waits,return_value=[])

75class ModelCallTests(unittest.TestCase):
76 def test_parallel_call_carries_a_distinct_execution_id(self):
77 events = []
78 class Runner:
79 def run(self, **arguments):
80 arguments["on_started"]({"pid": 7})
81 return {"status": "completed"}
82 def operation():
83 result = yield {"on_started": lambda observed: events.append(observed)}
84 events.append(result)
85 generator = operation(); arguments = next(generator)
86 calls = ModelCalls()
87 try:
88 calls.start_call(generator, Runner().run, arguments, execution_id="EX-1")
89 deadline = time.monotonic() + 10
90 while not calls.collect():
91 self.assertLess(time.monotonic(), deadline,
92 "EX-1 did not complete; pending callbacks=" + str(calls.events.qsize()))
93 time.sleep(.01)
94 finally:
95 calls.close()
96 self.assertEqual(events, [{"pid": 7}, {"status": "completed"}])
