27 def test_completion_records_input_and_held_submission_before_finishing_activities(self):
28 engine = self.engine()
29 incoming = {'id': 'A-input', 'role': 'intent_interpretation', 'target': 'UI-late',
30 'owner_activity_id': 'AC-input'}
31 review = {'id': 'A-review', 'role': 'maintenance', 'stage': 'impact_review',
32 'target': 'EX-producer', 'owner_activity_id': 'AC-review'}
33 held = {'action': {'id': 'A-producer', 'role': 'research_work',
34 'owner_activity_id': 'AC-producer'}, 'kind': 'impact_review',
35 'execution_id': 'EX-producer', 'review_requests': {'IB-one': {'payload_sha256': 'sealed'}}}
36 engine.runtime['queue'] = [incoming, review]
37 engine.runtime['waiting_actions']['A-producer'] = deepcopy(held)
38 engine._settle_goal_completion()
39 self.assertEqual(engine.runtime['queue'], [])
40 self.assertEqual(engine.runtime['waiting_actions'], {})
41 events = engine.runtime['history']
42 self.assertEqual({row['action']['id'] for row in events}, {'A-input', 'A-review', 'A-producer'})
43 self.assertEqual(next(row['waiting'] for row in events if row['action']['id'] == 'A-producer'), held)
44 self.assertTrue(any('UI-late' in question for question in engine.runtime['questions']))
45 for activity in engine.runtime['activities'].values():
46 self.assertEqual(activity['status'], 'finished')
47 self.assertTrue(activity['unresolved'])
48 self.assertEqual(activity['followups'][-1]['next_owner'], 'user')
49 before = deepcopy(engine.runtime)
50 engine._settle_goal_completion()
51 self.assertEqual(engine.runtime, before)
