82def lookup_definition(instance, family, token):
83 matches = [row for row in instance.get("kind_definitions", {}).values()
84 if row.get("family") == family and row.get("token") == token]
85 current = [row for row in matches if row.get("status") != "withdrawn"]
86 if len(current) > 1:
87 raise ValueError("multiple current definitions of one kind")
88 return current[0] if current else next((row for row in reversed(matches)), None)

115def register_kind_definition(instance: dict, definition: dict, *, source_ref: str,
116 actor_ref: str) -> tuple[dict, str]:
117 require_text(source_ref, "source_ref"); require_text(actor_ref, "actor_ref")
118 if not isinstance(definition, dict) or set(definition) != {"family", "token", "meaning", "constraints"}:
119 raise ValueError("kind definition requires family, token, meaning and constraints")
120 family, token = definition["family"], definition["token"]
121 if family not in {"knowledge", "relation"} or not isinstance(token, str) or not _TOKEN.fullmatch(token):
122 raise ValueError("kind definition has an invalid family or token")
123 require_text(definition["meaning"], "meaning")
124 constraints = definition["constraints"]
125 if not isinstance(constraints, dict) or set(constraints) - {
126 "mathematical_claim", "requires_evidence", "requires_formal_results", "requires_assessments",
127 "allowed_epistemic_statuses", "required_fields"}:
128 raise ValueError("kind definition has unsupported constraints")
129 for key, value in constraints.items():
130 if key in {"allowed_epistemic_statuses", "required_fields"}:
131 if not isinstance(value, list) or not value or any(not isinstance(v, str) or not v for v in value) or len(value) != len(set(value)):
132 raise ValueError("kind list constraints require distinct nonempty strings")
133 if key == "required_fields" and set(value) - _KNOWLEDGE_FIELDS:
134 raise ValueError("required fields may not grant authority or add unknown fields")
135 elif not isinstance(value, bool):
136 raise ValueError("kind boolean constraint expected")
137 if constraints.get("mathematical_claim"):
138 definition = deepcopy(definition)
139 definition["constraints"].update(requires_formal_results=True, requires_assessments=True)
140 output = ensure_ontology(instance)
141 previous = lookup_definition(output, family, token)
142 if previous and previous["status"] != "withdrawn":
143 raise ValueError("kind token is already defined")
144 identifier = new_id("KD")
145 output["kind_definitions"][identifier] = {**deepcopy(definition), "id": identifier, "status": "candidate", "builtin": False,
146 "source_ref": source_ref, "actor_ref": actor_ref,
147 "created_at": datetime.now(timezone.utc).isoformat()}
148 if previous:
149 output["kind_definitions"][identifier]["supersedes"] = previous["id"]
150 return output, token

153def check_kind_definition(instance, family, token, *, check_ref, actor_ref):
154 require_text(check_ref, "check_ref"); require_text(actor_ref, "actor_ref")
155 output = deepcopy(instance)
156 record = lookup_definition(output, family, token)
157 if record is None:
158 raise ValueError("unknown kind definition")
159 if record["status"] not in {"candidate", "checked"}:
160 raise ValueError("only candidate definitions can be checked")
161 record.update(status="checked", check_ref=check_ref, checked_by=actor_ref)
162 return output

177def withdraw_kind_definition(instance, runtime, family, token, *, reason, actor_ref):
178 require_text(reason, "reason"); require_text(actor_ref, "actor_ref")
179 output, new_runtime = deepcopy(instance), deepcopy(runtime)
180 definition = lookup_definition(output, family, token)
181 if definition is None:
182 raise ValueError("unknown kind definition")
183 definition.update(status="withdrawn", withdrawn_reason=reason, withdrawn_by=actor_ref)
184 roots = set()
185 for row in output.get("knowledge", {}).values() if family == "knowledge" else output.get("relations", []):
186 if row.get("kind_definition_id") == definition["id"]:
187 roots.add(row["id"])
188 if family == "relation":
189 row["status"] = "suspended"
190 else:
191 row["kind_use_suspended"] = True
192 if roots:
193 _propagate(output, new_runtime, roots, reason)
194 return output, new_runtime

205def candidate_shape_errors(candidate: dict) -> list[str]:
206 """Shape rules of one knowledge candidate: the producer's submission check and import share them.
207
208 How dependencies are named (new titles in a submission, IDs on import) and
209 whether they exist stay with each caller."""
210 errors = []
211 for field in ("kind", "title", "statement", "explanation", "summary", "lean_code"):
212 if not isinstance(candidate.get(field), str) or (field != "lean_code" and not candidate[field].strip()):
213 errors.append(f"candidate {field} is invalid")
214 declarations, dependencies = candidate.get("declaration_names", []), candidate.get("dependencies", [])
215 if (not isinstance(declarations, list) or any(not isinstance(value, str) or not value for value in declarations)
216 or len(declarations) != len(set(declarations))):
217 errors.append("declaration_names must contain distinct strings")
218 elif not all(is_lean_identifier(value) for value in declarations):
219 errors.append("declaration_names must be Lean identifiers")
220 if (not isinstance(dependencies, list) or any(not isinstance(value, str) or not value for value in dependencies)
221 or len(dependencies) != len(set(dependencies))):
222 errors.append("dependencies must contain distinct references")
223 if isinstance(candidate.get("lean_code"), str) and candidate["lean_code"] and not declarations:
224 errors.append("a formal candidate requires exported declaration names")
225 return errors

228def validate_insight(instance: dict, runtime: dict, insight: dict) -> None:
229 """An incidental candidate needs a registered kind and real evidence refs."""
230 for field in ("kind", "title", "statement", "summary"):
231 require_text(insight.get(field), "insight " + field)
232 _definition_for(instance, "knowledge", insight["kind"])
233 refs = insight.get("evidence_ids", [])
234 if not isinstance(refs, list) or any(not isinstance(ref, str) or not ref for ref in refs) or len(refs) != len(set(refs)):
235 raise ValueError("insight evidence must contain distinct references")
236 known = set(record_index(instance, runtime, source_aliases=True))
237 for key in ("inputs", "events", "external_rejections"):
238 known.update(row["id"] for row in runtime.get(key, []) if row.get("id"))
239 unknown = set(refs) - known
240 if unknown:
241 raise ValueError("unknown insight evidence: " + ", ".join(sorted(unknown)))

244def import_candidate(instance: dict, runtime: dict, candidate: dict, *, source_ref: str,
245 actor_ref: str, reserved_dependency_ids=(), plan_nonformal=True) -> tuple[dict, dict, str]:
246 require_text(source_ref, "source_ref"); require_text(actor_ref, "actor_ref")
247 if not isinstance(candidate, dict) or set(candidate) - _KNOWLEDGE_FIELDS:
248 raise ValueError("candidate has unsupported fields")
249 if "uncertainty_assessment" in candidate and not isinstance(candidate["uncertainty_assessment"], (str, dict)):
250 raise ValueError("uncertainty_assessment must be text or an object")
251 output, new_runtime = ensure_ontology(instance), deepcopy(runtime)
252 candidate = deepcopy(candidate)
253 reuse_formal = candidate.get("formal_result_ref")
254 if reuse_formal:
255 formal = output.get("formal_results", {}).get(reuse_formal)
256 if not formal or not knowledge_records.formal_usable(output, formal):
257 raise ValueError("reused formal result must be current and accepted")
258 if candidate.get("lean_code") and candidate["lean_code"] != formal["lean_code"]:
259 raise ValueError("reused formal source differs")
260 if candidate.get("declaration_names") and candidate["declaration_names"] != formal["declaration_names"]:
261 raise ValueError("reused formal declarations differ")
262 candidate.update(lean_code=formal["lean_code"], declaration_names=deepcopy(formal["declaration_names"]))
263 # A reused formal result has filled its verified source fields before the shared shape check.
264 errors = candidate_shape_errors(candidate)
265 if errors:
266 raise ValueError(errors[0])
267 definition = _definition_for(output, "knowledge", candidate["kind"])
268 dependencies = candidate.get("dependencies", [])
269 record_id = candidate.get("id") or new_id("KN")
270 if not isinstance(record_id, str) or not is_new_id("KN", record_id):
271 raise ValueError("knowledge ID must be a canonical KN-prefixed ULID")
272 if record_id in dependencies:
273 raise ValueError("candidate cannot depend on itself")
274 reserved = set(reserved_dependency_ids)
275 missing = [value for value in dependencies
276 if value not in output.get("knowledge", {}) and value not in reserved]
277 if missing:
278 raise ValueError("candidate dependencies do not exist: " + ", ".join(missing))
279 if record_id in knowledge_records.all_records(output):
280 raise ValueError("record ID already exists; revisions require a new ID")
281 constraints = definition.get("constraints", {})
282 allowed = constraints.get("allowed_epistemic_statuses")
283 if allowed is not None and candidate.get("epistemic_status") not in allowed:
284 raise ValueError("epistemic_status is not allowed by the kind definition")
285 if any(not candidate.get(field) for field in constraints.get("required_fields", [])):
286 raise ValueError("candidate is missing kind-required fields")
287 claim = candidate.get("relation_claim")
288 if claim is not None:
289 if not isinstance(claim, dict) or set(claim) != {"kind", "source", "target", "confirmed_scope"}:
290 raise ValueError("relation_claim needs kind, source, target and confirmed_scope")
291 _definition_for(output, "relation", claim["kind"])
292 if any(claim[key] not in output.get("knowledge", {}) for key in ("source", "target")):
293 raise ValueError("relation claim requires exact existing endpoints")
294 require_text(claim["confirmed_scope"], "confirmed_scope")
295 from .input_basis import capture_sources
296 source_refs = candidate.get("source_refs", [])
297 if not isinstance(source_refs, list) or any(not isinstance(ref, str) for ref in source_refs):
298 raise ValueError("source_refs must be a list of source references")
299 if set(source_refs) - set(record_index(output, {}, families=(), source_aliases=True)):
300 raise ValueError("required source does not exist")
301 output, record_id = knowledge_records.create_candidate(output, {**candidate, "id": record_id},
302 source_ref=source_ref, actor_ref=actor_ref)
303 record = output["knowledge"][record_id]
304 if actor_ref not in new_runtime.get('executions', {}):
305 # User/import provenance is not an execution identity. Review author
306 # exclusion links must resolve to real executions in this instance.
307 authored = [record, output['expositions'][record['exposition_id']]]
308 if record['formal_result_id'] and not reuse_formal:
309 authored.append(output['formal_results'][record['formal_result_id']])
310 for authored_record in authored:
311 authored_record.pop('producer_execution', None)
312 authored_record['origin_refs'] = list(dict.fromkeys([*authored_record.get('origin_refs', []), actor_ref]))
313 record["kind_definition_id"] = definition["id"]
314 record["source_basis"] = capture_sources(output, source_refs)
315 if claim:
316 record["relation_endpoint_basis"] = capture_basis(output, new_runtime, list(dict.fromkeys([claim["source"], claim["target"]])))
317 new_runtime.setdefault("history", []).append({"kind": "knowledge_candidate_imported",
318 "knowledge_id": record_id, "source_ref": source_ref})
319 next_role = "identity_review" if reuse_formal else "formal_check" if record["formal_result_id"] else "research_plan"
320 if next_role != 'research_plan' or plan_nonformal:
321 action_id = new_id("AC")
322 new_runtime.setdefault("queue", []).append({"id": action_id, "lineage_id": action_id, "role": next_role,
323 "target": record_id, "reason": "imported candidate requires verification", "source_ref": source_ref})
324 return output, new_runtime, record_id

327def validate_relation(instance: dict, relation: dict) -> dict:
328 reasons: list[str] = []
329 if not isinstance(relation, dict) or set(relation) - _RELATION_FIELDS:
330 return {"valid": False, "reasons": ["unsupported_relation_fields"], "definition": None}
331 kind = relation.get("kind")
332 try:
333 definition = _definition_for(instance, "relation", kind)
334 except ValueError as error:
335 return {"valid": False, "reasons": [str(error)], "definition": None}
336 knowledge = instance.get("knowledge", {})
337 for endpoint in ("source", "target"):
338 value = relation.get(endpoint)
339 if value not in knowledge:
340 reasons.append(f"missing_{endpoint}:{value}")
341 constraints = definition.get("constraints", {})
342 evidence = relation.get("evidence_refs", [])
343 if relation.get("evidence"):
344 evidence = [relation["evidence"], *evidence] if isinstance(evidence, list) else [relation["evidence"]]
345 if constraints.get("requires_evidence") and not evidence:
346 reasons.append("evidence_required")
347 establishing = {}
348 if constraints.get("requires_formal_results") or constraints.get("requires_assessments"):
349 # Effective matching judgments of the named formal results that claim exactly this relation.
350 formal_refs = relation.get("formal_result_refs", [])
351 for assessment_id, assessment in instance.get("identity_assessments", {}).items():
352 claim = assessment.get("relation_claim", {})
353 if (assessment.get("formal_result_id") in formal_refs and assessment.get("decision") == "match"
354 and isinstance(claim, dict)
355 and all(claim.get(key) == relation.get(key) for key in ("kind", "source", "target", "confirmed_scope"))
356 and assess_use(assessment, instance)["valid"]):
357 establishing[assessment_id] = assessment["formal_result_id"]
358 if constraints.get("requires_formal_results"):
359 refs = relation.get("formal_result_refs", [])
360 if not isinstance(refs, list) or not refs:
361 reasons.append("formal_result_refs_required")
362 else:
363 for ref in refs:
364 record = instance.get("formal_results", {}).get(ref)
365 if not record or not knowledge_records.formal_usable(instance, record):
366 reasons.append(f"formal_result_not_usable:{ref}")
367 if ref not in establishing.values():
368 reasons.append(f"formal_result_does_not_establish_this_relation:{ref}")
369 if constraints.get("requires_assessments"):
370 assessment_refs = relation.get("assessment_refs", [])
371 if not isinstance(assessment_refs, list) or not assessment_refs:
372 reasons.append("assessment_refs_required")
373 else:
374 for ref in assessment_refs:
375 if ref not in establishing:
376 reasons.append(f"assessment_not_effective:{ref}")
377 return {"valid": not reasons, "reasons": reasons, "definition": deepcopy(definition)}

380def import_relation(instance: dict, relation: dict, *, source_ref: str,
381 actor_ref: str) -> tuple[dict, str]:
382 require_text(source_ref, "source_ref"); require_text(actor_ref, "actor_ref")
383 result = validate_relation(instance, relation)
384 if not result["valid"]:
385 raise ValueError("invalid relation: " + ", ".join(result["reasons"]))
386 output = ensure_ontology(instance)
387 relation_id = relation.get("id") or new_id("RL")
388 if not isinstance(relation_id, str) or not is_new_id("RL", relation_id):
389 raise ValueError("relation ID must be a canonical RL-prefixed ULID")
390 old = next((value for value in output.setdefault("relations", []) if value.get("id") == relation_id), None)
391 record = {key: deepcopy(value) for key, value in relation.items() if key != "id"}
392 mathematical = result["definition"]["constraints"].get("mathematical_claim", False)
393 record.update(id=relation_id, revision=int(old.get("revision", 0)) + 1 if old else 1,
394 status="active", source_ref=source_ref, imported_by=actor_ref,
395 kind_definition_id=result["definition"]["id"],
396 authority="mathematical_connection" if mathematical else "research_connection",
397 imported_at=datetime.now(timezone.utc).isoformat())
398 if old:
399 record["history"] = [*deepcopy(old.get("history", [])),
400 {key: deepcopy(value) for key, value in old.items() if key != "history"}]
401 output["relations"].remove(old)
402 output["relations"].append(record)
403 # Ensure the stored structure itself is deterministic JSON data.
404 canonical_json(record)
405 return output, relation_id
