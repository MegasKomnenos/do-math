29def preserve_rejected_candidate(store, snapshot, *, raw, reason):
30 """Keep rejected bytes on the verified head, never publish the rejected state.
31
32 Loading ``snapshot`` validates it under the state contract; a head that
33 does not load is damaged and is not rewritten.
34 """
35 if not isinstance(raw, bytes):
36 raise TypeError('candidate raw bytes required')
37 try:
38 _, instance, runtime = store.load(snapshot)
39 except (ValueError, TypeError, KeyError, OSError, RuntimeError) as exc:
40 raise ValueError('damaged current state cannot be rewritten: ' + str(exc)) from exc
41 identity = new_id('RJ')
42 path = 'runtime/artifacts/' + identity + '/candidate.raw'
43 audit = 'runtime/artifacts/' + identity + '/rejection.json'
44 new_head = store.save_with_state(snapshot, instance, runtime, artifacts={path: raw, audit: canonical_json({
45 'id': identity, 'candidate_ref': path, 'sha256': hashlib.sha256(raw).hexdigest(),
46 'base_snapshot': snapshot, 'reason': reason, 'disposition': 'rejected_not_applied'})})[0]
47 return {'snapshot': new_head, 'raw_ref': path, 'audit_ref': audit}

68 def _clock_guard(self):
69 """A retained regression blocks authority until a verified re-observation."""
70 before = self.lifecycle_monotonic()
71 now = self._lifecycle_now()
72 after = self.lifecycle_monotonic()
73 persisted = self.runtime.get('last_observed_utc')
74 if not persisted:
75 return {'usable': False, 'reason': 'missing_utc_high_water'}
76 previous = timestamp(persisted)
77 # A reported tick is not an exact instant. Use a full reported resolution
78 # on either side (covers truncation or rounding), plus representation error.
79 # UTC intervals round outwards to datetime's microsecond precision; the
80 # monotonic interval also includes the float's ULP at this uptime.
81 wall_radius = timedelta(microseconds=math.ceil(
82 time.get_clock_info('time').resolution * 1_000_000) + 1)
83 mono_radius = time.get_clock_info('monotonic').resolution + max(math.ulp(before), math.ulp(after))
84 anchor = self._clock_anchor
85 reason = 'persisted_utc_regression' if now < previous else None
86 expected = None
87 if anchor is not None:
88 if now < anchor[0] and reason is None:
89 reason = 'live_utc_regression'
90 elapsed_lower = max(0., before - mono_radius - (anchor[1] + anchor[3]))
91 expected = anchor[0] - anchor[2] + timedelta(microseconds=math.floor(elapsed_lower * 1_000_000))
92 if now + wall_radius < expected and reason is None:
93 reason = 'live_monotonic_regression'
94 if reason:
95 self._clock_failure = {'usable': False, 'reason': reason,
96 'last_observed_utc': persisted, 'observed_utc': now.isoformat(),
97 'expected_at_least': (expected if reason == 'live_monotonic_regression'
98 else previous if reason == 'persisted_utc_regression'
99 else anchor[0]).isoformat()}
100 return self._clock_failure
101 if self._clock_failure:
102 return self._clock_failure
103 self._clock_anchor = (now, after, wall_radius, mono_radius)
104 return {'usable': True, 'observed_utc': now.isoformat()}
