4``run_tests.py`` refuses to run while a ``tests/test_*.py`` module is missing
5from it or it names a module that does not exist. The time targets (each shard

62def unregistered_modules(tests_dir):
63 """(test modules on disk missing from SHARDS, SHARDS entries with no module on disk)."""
64 present = {path.stem for path in Path(tests_dir).glob('test_*.py')}
65 return sorted(present - _SHARD_OF.keys()), sorted(_SHARD_OF.keys() - present)

68def test_group(label):
69 if label not in LABELS:
70 raise ValueError('unknown test label: ' + str(label))
71 def decorate(target):
72 target.do_math_test_group = label
73 return target
74 return decorate

77def isolation_for(test):
78 """'lean' or 'resource' when the test must run isolated, else None; a method label overrides its class."""
79 method = getattr(test, getattr(test, '_testMethodName', ''), None)
80 return getattr(method, 'do_math_test_group', None) or getattr(type(test), 'do_math_test_group', None)
