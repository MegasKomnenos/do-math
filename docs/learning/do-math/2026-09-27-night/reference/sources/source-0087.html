108def start_review(proposal, reviewers, judges=3, max_rounds=3, previous=None, *, policy=None,
109 scope_assessment=None):
110 """Open a finite review cycle; with scope_assessment it begins at its initial slots.
111
112 The engine always opens at the scope stage. A given scope_assessment is the
113 replay of an accepted scope (validate_sealed_review) or a test's start state.
114 """
115 if not isinstance(proposal, dict) or proposal.get("revision") is None:
116 raise ValueError("proposal requires an exact revision")
117 reviewer_slots, requested_judges = _slots(reviewers, "initial"), _slots(judges, "judge")
118 if len(requested_judges) != 3 or isinstance(max_rounds, bool) or not isinstance(max_rounds, int) or max_rounds < 1:
119 raise ValueError("exactly three judges and a positive round limit are required")
120 _unique(reviewer_slots + requested_judges + ["scope", "organize"], "requested role slot")
121 policy = build_review_policy(policy, reviewer_count=len(reviewer_slots), max_rounds=max_rounds,
122 judge_slots=requested_judges)
123 assessment = validate_scope_assessment(scope_assessment, policy) if scope_assessment is not None else None
124 actual_reviewer_slots = [slot["id"] for slot in assessment["initial_slots"]] if assessment else []
125 judge_slots, max_rounds = list(policy["judge_slots"]), policy["round_limit"]
126 _unique(actual_reviewer_slots + judge_slots + ["scope", "organize"], "role slot")
127 digest = exact_content_hash(proposal)
128 previous_reviews, turns, manifests = [], {}, {}
129 lineage = proposal.get("review_lineage_id", digest)
130 if previous is not None:
131 _check_proposal(previous)
132 if proposal["revision"] == previous["revision"]:
133 raise ValueError("new review requires a new proposal revision")
134 if assessment is not None and (not set(previous["reviewer_slots"]).issubset(actual_reviewer_slots) or judge_slots != previous["judge_slots"]):
135 raise ValueError("existing logical slots must be retained")
136 if proposal.get("review_lineage_id", previous["review_lineage_id"]) != previous["review_lineage_id"]:
137 raise ValueError("proposal changes the inherited review lineage")
138 if previous.get("review_policy") != policy:
139 raise ValueError("a revised review must retain its frozen review policy")
140 lineage = previous["review_lineage_id"]
141 turns = copy.deepcopy(previous["turns"])
142 manifests = copy.deepcopy(previous["transcript_manifest"])
143 previous_reviews = copy.deepcopy(previous["previous_reviews"])
144 previous_reviews.append({k: copy.deepcopy(previous[k]) for k in
145 ("revision", "proposal_digest", "proposal", "results", "sealed", "issues", "status", "decision")})
146 previous_reviews[-1]["research"] = copy.deepcopy(previous.get("research", {}))
147 return {"proposal": copy.deepcopy(proposal), "proposal_digest": digest,
148 "revision": proposal["revision"], "review_lineage_id": lineage,
149 "requested_reviewer_slots": reviewer_slots,
150 "retained_reviewer_slots": list(previous["reviewer_slots"]) if previous is not None else [],
151 "reviewer_slots": actual_reviewer_slots,
152 "judge_slots": judge_slots, "max_rounds": max_rounds, "round": 1,
153 "review_policy": policy, "scope_assessment": assessment, "owner_activity_id": None,
154 "activity_execution_limit": assessment["activity_execution_limit"] if assessment else policy["activity_execution_limit"],
155 "required_source_refs": sorted({ref for slot in assessment["initial_slots"] for ref in slot["required_source_refs"]}) if assessment else [],
156 "execution_accounting": {"executions_used": []},
157 "phase": "initial" if assessment is not None else "scope", "status": "working", "decision": None,
158 "results": {}, "sealed": [], "turns": turns, "transcript_manifest": manifests,
159 "issues": {}, "findings": {}, "pending_findings": [], "answer_requests": [],
160 "history": [], "previous_reviews": previous_reviews, "followup": [],
161 "deferred_actions": {}, "research": {}, "recovery_events": [],
162 "continuity_events": [], "blocked_actions": {}, "replacement_actions": {},
163 "replacement_receipts": []}

602def validate_resource_disposition(cycle, runtime, *, read_artifact=None, instance=None):
603 """Check machine closure evidence without manufacturing an identity judgment."""
604 record = resource_disposition(cycle)
605 if record is None:
606 return None
607 if sum(row.get('event') == 'review_resource_disposition' for row in cycle['history']) != 1:
608 raise ValueError('review has duplicate resource dispositions')
609 if record.get('state') not in {'draining', 'deferred', 'ended'}:
610 raise ValueError('unknown review resource disposition state')
611 if record['state'] == 'ended' and (cycle['status'], cycle['decision'], cycle['phase']) != ('unresolved', 'unresolved', 'done'):
612 raise ValueError('ended resource disposition must preserve an unresolved review')

614 or record['limit'] != cycle['activity_execution_limit']
615 or record['disposition'] != cycle['review_policy']['overflow_disposition']):

619 raise ValueError('review resource disposition changed execution accounting')

622 if execution.get('actor_kind') != 'ai' or execution.get('owner_activity_id') != cycle['owner_activity_id']:

631 if minimum_scope_executions(count, cycle['max_rounds'], len(used) + 1) <= record['limit']:

635 raise ValueError('review followup fits the available resources')
636 elif reason == 'review_input_overflow':

639 or overflow['without_restored_dialogue_bytes'] <= overflow['limit_bytes']
640 or detail['execution_id'] not in used):

643 raise ValueError('review overflow differs from preserved measurement')

648 (cycle.get('scope_assessment') or {}).get('selected_profile_ref')]

653 from .execution_profiles import match_profile_ref
654 approved = detail['adopted_host_ref']
655 if approved is not None and instance['capacity_constraints'][approved]['kind'] != 'host':
656 raise ValueError('review host change names no adopted host constraint')

658 if any(match_profile_ref(instance['execution_profiles'], ref).get('host_capacity_refs', []) == expected
659 for ref in detail['profile_refs']):
660 raise ValueError('review pinned profile already matches the host configuration')
661 else:
662 raise ValueError('unknown review resource disposition')
663 if record['state'] == 'ended' and any(runtime['executions'][eid]['status'] != 'ended' for eid in used):
664 raise ValueError('review closed before its attempts ended')
665 return record
