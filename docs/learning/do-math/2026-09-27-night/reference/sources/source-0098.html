41def read_sealed_submission(execution: dict, snapshot: str, read_artifact, *, git_dir=None):
42 """Read the exact bytes accepted by the central submission receipt.
43
44 With the repository's ``git_dir``, a sealed row this process already matched
45 (이음새-정본.md §3) is not hashed again.

49 intent = execution.get("submission_intent") or {}
50 receipt = execution.get("submission_receipt") or {}

118
119 The engine writes both into the immutable input bundle (engine._input_artifacts);

362def contract_check_errors(item: dict, entry: dict, dependencies: list[str], tasks: dict, known: set, revising: set) -> list[tuple[str, str]]:
363 """The mechanically decidable part of contract review, run by the producer before submission.
364
365 Each finding is (rule_id, message); the rule identifies the kind of defect,
366 so distinct defects of one candidate remain distinct issues.
367 """
368 text = "\n".join([item["title"], item["question"], *item["scope"]["included"], *item["scope"]["excluded"],
369 *item["expected"]["outputs"], item["expected"]["research_change"],
370 *(row["claim"] for row in item["completion_obligations"]), *(row["claim"] for row in item["progress_criteria"])])
371 errors = []
372 cited = set(RECORD_ID.findall(text)) - {item["revises_task_id"]}
373 errors += [("contract_check.cited_record", "cited record does not exist in the supplied input: " + ref)
374 for ref in sorted(cited - known - set(tasks))]
375 symbols = [row["symbol"] for row in entry["symbols"]]
376 errors += [("contract_check.symbol_meaning", "symbol has more than one declared meaning: " + symbol)
377 for symbol in sorted({x for x in symbols if symbols.count(x) > 1})]
378 errors += [("contract_check.symbol_absent", "declared symbol is absent from the contract text: " + symbol)
379 for symbol in symbols if symbol not in text]
380 outcomes = "\n".join(row["outcome"] for row in entry["outcome_branches"])
381 if len({row["outcome"] for row in entry["outcome_branches"]}) < 2:
382 errors.append(("contract_check.outcome_branches", "outcome branches must state at least two distinct outcomes"))
383 for dependency in dependencies:
384 names = {dependency, tasks.get(dependency, {}).get("title", dependency)}
385 if not any(name in outcomes for name in names):
386 errors.append(("contract_check.dependency_outcomes",
387 "no outcome branch names the normal outcomes of the dependency: " + dependency))
388 declared = {row["task_id"]: row for row in entry["sibling_boundaries"]}
389 if len(declared) != len(entry["sibling_boundaries"]):
390 errors.append(("contract_check.sibling_boundary", "sibling boundaries must name distinct tasks"))
391 siblings = {ref for ref in cited if ref.startswith("TK-")}
392 errors += [("contract_check.sibling_boundary", "cited sibling contract has no declared boundary: " + ref)
393 for ref in sorted(siblings - set(declared))]
394 for ref, row in declared.items():
395 sibling = tasks.get(ref)
396 if sibling is None:
397 errors.append(("contract_check.sibling_boundary", "sibling boundary names a task outside the supplied input: " + ref))
398 elif row["revision"] != sibling.get("revision"):
399 errors.append(("contract_check.sibling_revision",
400 f"sibling boundary was not read from the current revision {sibling.get('revision')} of {ref}"))
401 elif ((sibling.get("revision_requested") or ref in revising)
402 and not row["independent_of_next_revision"] and ref not in dependencies):
403 errors.append(("contract_check.sibling_under_revision",
404 "sibling contract is under revision: state a boundary that its next revision cannot move, "
405 "or wait for it through task_conditions dependencies: " + ref))
406 return errors

420 if error.validator == "required" and isinstance(error.instance, dict):
421 missing = [key for key in error.validator_value if key not in error.instance]
422 rows = [_issue(coordinate + "/" + key, "required property is missing", "schema.required") for key in missing]
423 else:
424 rows = [_issue(coordinate, error.message, "schema." + str(error.validator))]
425 for row in rows:
426 issues[row["issue_id"]] = row
427 return list(issues.values())

578def action_resolution_errors(blocked: dict, evidence_action: dict | None) -> list[str]:
579 """What the fixed failed action and the evidence's own action decide about a resolution.

581 The producer passes its own action for `current` evidence; the receiver passes the
582 evidence execution's action and separately checks acceptance and current premises.
583 """

585 return ["only an output_validation or execution_failure wait can be resolved; this action waits for "
586 + str(blocked.get("kind"))]
587 if evidence_action is None:
588 return []
589 same_target = (blocked.get("target") == evidence_action.get("target")
590 or blocked.get("target") in evidence_action.get("revision_targets", {}))
591 if not same_target or any(blocked.get(key) != evidence_action.get(key) for key in _JUDGMENT_SCOPE):
592 return ["resolution evidence must be an execution of the same judgment scope as the failed action"]
593 return []
