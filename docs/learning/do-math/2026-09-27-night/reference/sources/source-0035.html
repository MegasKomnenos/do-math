44def normalized_source(name, raw):
45 """The one admitted form of source bytes: exact body, hash and size."""
46 return {'name': name, 'body_base64': base64.b64encode(raw).decode('ascii'),
47 'sha256': hashlib.sha256(raw).hexdigest(), 'bytes': len(raw)}

50def normalized_source_bytes(spec):
51 """Return the exact bytes of a normalized source, or raise ValueError.
52
53 ENGINE's public ``control`` checks the source it produced with this rule,
54 and the owner applies the same rule to the inbox copy before import.
55 """
56 if not isinstance(spec, dict) or set(spec) != SOURCE_FIELDS:
57 raise ValueError('normalized source requires exact name, body, hash and size')
58 name = spec['name']
59 if not isinstance(name, str) or name in {'', '.', '..'} or '/' in name or '\\' in name:
60 raise ValueError('source name must be a file name')
61 try:
62 raw = base64.b64decode(spec['body_base64'], validate=True)
63 except (binascii.Error, TypeError) as error:
64 raise ValueError('source body is not strict base64') from error
65 if hashlib.sha256(raw).hexdigest() != spec['sha256'] or len(raw) != spec['bytes']:
66 raise ValueError('source bytes differ from their declared hash or size')
67 return raw

176def _check(validator, value, path):
177 error = next(validator.iter_errors(value), None)
178 if error is not None:
179 location = '/'.join([path, *(str(part) for part in error.absolute_path)])
180 raise ValueError(location + ': invalid ' + error.validator + ' (expected ' +
181 str(error.validator_value) + ')')

192def validate_control_payload(action, text, config):
193 if not isinstance(action, str) or action not in {*_VALIDATORS, 'configure'}:
194 raise ValueError('unknown control action')
195 if not isinstance(text, str):
196 raise ValueError('control text must be a string')
197 _check(_configure_validator() if action == 'configure' else _VALIDATORS[action], config, 'config')
198 if action == 'trust' and config.get('operation') == 'register':
199 from .trust_validation import register_request_errors
200 errors = register_request_errors(config)
201 if errors:
202 raise ValueError('config: ' + '; '.join(errors))
