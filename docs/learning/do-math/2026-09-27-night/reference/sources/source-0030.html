58def _parse_jsonl(raw: bytes | None) -> dict[str, Any]:
59 if raw is None:
60 return {"parsed_bytes": 0, "unparsed_tail_bytes": 0, "events": [], "thread_id": None,
61 "turn_completed": False, "turn_failed": False, "usage": [], "unknown_events": ["stdout_missing"],
62 "parse_complete": False}
63 events, usage, unknown = [], [], []
64 offset = 0
65 thread_id = None
66 completed = failed = False
67 message = None # (event index, text) of the latest completed agent message
68 for line in raw.splitlines(keepends=True):
69 body = line.rstrip(b"\r\n")
70 if not body:
71 offset += len(line)
72 continue
73 try:
74 event = strict_json_loads(body)
75 except (UnicodeError, ValueError):
76 unknown.append({"kind": "invalid_jsonl", "offset": offset})
77 return {"parsed_bytes": offset, "unparsed_tail_bytes": len(raw) - offset, "events": events,
78 "thread_id": thread_id, "turn_completed": completed, "turn_failed": failed,
79 "usage": usage, "unknown_events": unknown, "parse_complete": False}
80 if not isinstance(event, dict) or not isinstance(event.get("type"), str):
81 unknown.append({"kind": "unknown_jsonl_event", "offset": offset})
82 else:
83 event_type = event["type"]
84 events.append(event_type)
85 if not (event_type in {"thread.started", "turn.started", "turn.completed", "turn.failed", "error"}
86 or event_type.startswith("item.")):
87 unknown.append({"kind": "unknown_event_type", "offset": offset, "length": len(line), "type": event_type})
88 if event_type == "thread.started" and isinstance(event.get("thread_id"), str):
89 thread_id = event["thread_id"]
90 completed |= event_type == "turn.completed"
91 failed |= event_type == "turn.failed"
92 if "usage" in event:
93 usage.append(event["usage"])
94 item = event.get("item")
95 if (event_type == "item.completed" and isinstance(item, dict) and item.get("type") == "agent_message"
96 and isinstance(item.get("text"), str)):
97 message = (len(events) - 1, item["text"])
98 offset += len(line)
99 terminals = [index for index, item in enumerate(events) if item in {"turn.completed", "turn.failed"}]
100 if not thread_id or len(events) < 3 or events[0] != "thread.started" or events[1] != "turn.started":
101 unknown.append({"kind": "invalid_lifecycle_order"})
102 if len(terminals) != 1 or terminals and terminals[0] != len(events) - 1:
103 unknown.append({"kind": "invalid_terminal_count_or_position"})
104 if any(not item.startswith("item.") for item in events[2:-1]):
105 unknown.append({"kind": "invalid_event_between_turn_start_and_terminal"})
106 parsed = {"parsed_bytes": len(raw), "unparsed_tail_bytes": 0, "events": events, "thread_id": thread_id,
107 "turn_completed": completed, "turn_failed": failed, "usage": usage, "unknown_events": unknown,
108 "parse_complete": not unknown}
109 # The final response is the agent message immediately before the terminal event.
110 if message is not None and message[0] == len(events) - 2:
111 parsed["final_text"] = message[1]
112 return parsed

115class LocalCLIAdapter:

144 def capability_identity(self) -> dict:
145 """Current local identity, without claiming any behavioral capability."""
146 from . import file_io
147 path = Path(shutil.which(self.cli_path) or self.cli_path).resolve()
148 # Digests are recomputed only when a file's fingerprint changes (이음새-정본.md §2).

150 "module_sha256": {name: file_io.cached_sha256(importlib.import_module("do_math." + name).__file__)
151 for name in sorted(self.identity_modules)},
152 "host": {"system": platform.system(), "release": platform.release(), "version": platform.version(),
153 "machine": platform.machine(), "node": platform.node(), "python": platform.python_version()}}

156 """One batch may share facts only for this exact adapter/tool selection."""
157 return (type(self).__module__, type(self).__qualname__,
158 str(Path(shutil.which(self.cli_path) or self.cli_path).resolve()),

161 @measured('capability_surface')
162 def describe(self) -> dict[str, Any]:
163 def run(arguments):
164 return subprocess.run([self.cli_path, *arguments], check=False, shell=False, capture_output=True,
165 timeout=self.describe_timeout, creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0))
166 try:
167 version = run(("--version",))
168 helps = [(run(arguments), flags) for arguments, flags in self.help_commands]
169 except (OSError, subprocess.TimeoutExpired) as exc:
170 return {"cli_path": self.cli_path, "verified": False, "error_type": type(exc).__name__}
171 verified = version.returncode == 0 and all(result.returncode == 0 and all(flag in result.stdout for flag in flags)
172 for result, flags in helps)
173 help_bytes = helps[-1][0].stdout
174 return {"scope": "cli_command_surface_only", "identity": self.capability_identity(), "provider": self.provider,
175 "cli_path": self.cli_path, "version_bytes": version.stdout,
176 "version_sha256": hashlib.sha256(version.stdout).hexdigest(),
177 "exec_help_bytes": help_bytes, "exec_help_sha256": hashlib.sha256(help_bytes).hexdigest(),
178 "verified": verified,
179 "supported": {**{name: verified for name in self.surface_capabilities}, "local_job_stop": True},
180 "unknown": list(self.unknown_capabilities)}

182 def start(self, request_path: Path, *, cwd: Path, env: Mapping[str, str], job_name=None) -> CLIExecution:
183 request = self.worker_module.load_request(request_path)
184 prompt_path, stdout_path, stderr_path, status_path = (Path(request[name]) for name in
185 ("prompt_path", "stdout_path", "stderr_path", "status_path"))
186 final_path = Path(request["final_path"]) if "final_path" in request else None
187 if Path(request[self.executable_key]).resolve() != Path(self.cli_path).resolve():
188 raise ValueError("request CLI executable differs from adapter executable")
189 outputs = [path for path in (stdout_path, stderr_path, final_path, status_path) if path is not None]
190 paths = [request_path, prompt_path, *outputs]
191 if "output_schema_path" in request:
192 paths.append(Path(request["output_schema_path"]))
193 if len({path.resolve() for path in paths}) != len(paths):
194 raise ValueError("request, prompt, and output paths must be distinct")
195 if any(path.exists() for path in outputs):
196 raise FileExistsError("output paths must be newly allocated")
197 workspace = Path(request["cwd"]).resolve()
198 if any(path.resolve().is_relative_to(workspace) for path in outputs):
199 raise ValueError("worker outputs and status must be outside the model workspace")
200 prompt_raw = prompt_path.read_bytes()
201 stage = Path(tempfile.mkdtemp(prefix="do-math-cli-", dir=workspace.parent))
202 snapshot_prompt = stage / "prompt.bin"; snapshot_prompt.write_bytes(prompt_raw)
203 snapshot_schema = None
204 schema_sha256 = None
205 if "output_schema_path" in request:
206 snapshot_schema = stage / "schema.json"; schema_bytes = Path(request["output_schema_path"]).read_bytes()
207 snapshot_schema.write_bytes(schema_bytes); schema_sha256 = hashlib.sha256(schema_bytes).hexdigest()
208 worker_request = dict(request); worker_request["prompt_path"] = str(snapshot_prompt)
209 if snapshot_schema is not None: worker_request["output_schema_path"] = str(snapshot_schema)
210 snapshot_raw = compact_json_bytes(worker_request)
211 snapshot_request = stage / "request.json"; snapshot_request.write_bytes(snapshot_raw)
212 worker = Path(self.worker_module.__file__)
213 worker_argv = (sys.executable, str(worker), str(snapshot_request))
214 job = WindowsJob(worker_argv, cwd=cwd, env=self.worker_module.sanitized_environment(env),
215 name=job_name or "Local\\DoMath-" + uuid.uuid4().hex, resource_limits=request.get("resource_limits"))
216 return CLIExecution(hashlib.sha256(snapshot_raw).hexdigest(), hashlib.sha256(prompt_raw).hexdigest(), worker_argv,
217 stdout_path, stderr_path, final_path, status_path, job, schema_sha256,
218 snapshot_raw, prompt_raw)

231 def wait_for_termination(self, execution: CLIExecution, timeout: float | None) -> bool:
232 """Observe the Job barrier; a signaled root alone is insufficient."""
233 deadline = None if timeout is None else time.monotonic() + timeout
234 while True:
235 state = execution.job.inspect()
236 if state.root_signaled and state.active_processes == 0 and state.process_ids == ():
237 return True
238 if deadline is not None and time.monotonic() >= deadline:
239 return False
240 time.sleep(0.02)

242 def collect(self, execution: CLIExecution) -> dict[str, Any]:
243 # The private request and prompt snapshots are the bytes ``start`` wrote and hashed.
244 request_raw, prompt_raw = execution.request_raw, execution.prompt_raw
245 request = json.loads(request_raw)
246 job_state = execution.job.inspect()
247 barrier = job_state.root_signaled and job_state.active_processes == 0 and job_state.process_ids == ()
248 if not barrier:
249 return {"result": "incomplete", "reasons": ["termination_barrier_unconfirmed"],
250 "termination_barrier": {"active_processes": job_state.active_processes,
251 "process_ids": job_state.process_ids, "root_signaled": job_state.root_signaled, "confirmed": False},
252 "artifacts_read": False, "provider_operation_ids": "unknown", "remote_termination": "unknown",
253 "provider_observation": _provider_observation(request, {}, None)}
254 stdout, stderr, final_file, status_raw = (_read(path) for path in (execution.stdout_path,
255 execution.stderr_path, execution.final_path, execution.status_path))
256 status = None
257 if status_raw is not None:
258 try:
259 status = json.loads(status_raw)
260 except (UnicodeDecodeError, json.JSONDecodeError):
261 status = {"state": "invalid_status_json"}
262 parsed = self.parse_response(stdout, request)
263 final = self.final_bytes(parsed) if self.final_from_stream else final_file
264 reasons = []
265 if not isinstance(status, dict): reasons.append("worker_status_missing")
266 elif status.get("state") == "timeout": reasons.append("max_runtime_elapsed")
267 elif status.get("state") != "exited": reasons.append("worker_not_exited")
268 elif status.get("cli_exit_code") != 0: reasons.append("cli_nonzero_exit")
269 elif status.get("stdin_sha256") != execution.prompt_sha256: reasons.append("worker_stdin_hash_mismatch")
270 elif status.get("request_sha256") != execution.request_sha256: reasons.append("worker_request_hash_mismatch")
271 elif status.get("schema_sha256") != execution.schema_sha256: reasons.append("worker_schema_hash_mismatch")
272 if isinstance(status, dict):
273 for key in cli_worker.STATUS_BINDING_KEYS:
274 if key in request and status.get(key) != request[key]:
275 reasons.append("worker_" + key + "_mismatch")
276 if parsed["turn_failed"]: reasons.append("turn_failed")
277 if not parsed["turn_completed"]: reasons.append("turn_completed_missing")
278 if not parsed["parse_complete"]: reasons.append("jsonl_parse_incomplete")
279 if final is None: reasons.append("final_message_missing")
280 else: reasons.extend(self.validate_final(parsed, final))
281 return {"source_request": _artifact(request_raw, execution.request_sha256),
282 "source_prompt": _artifact(prompt_raw, execution.prompt_sha256),
283 "host_resources": execution.job.resource_usage() if request.get("resource_limits") is not None else None,
284 "stdout": _artifact(stdout), "stderr": _artifact(stderr), "final": _artifact(final),
285 "worker_status": status, "jsonl": parsed, "worker_argv": execution.worker_argv,
286 "process_identity": process_identity(execution.job),
287 "stop_receipts": execution.stop_receipts,
288 "result": "output_collected" if not reasons else "incomplete",
289 "reasons": reasons, "termination_barrier": {"active_processes": job_state.active_processes,
290 "process_ids": job_state.process_ids, "root_signaled": job_state.root_signaled, "confirmed": barrier}, "provider_operation_ids": "unknown", "remote_termination": "unknown",
291 "provider_stop_reason": parsed.get("stop_reason", "unknown"),
292 "execution_id": request.get("execution_id"), "profile_ref": request.get("profile_ref"),
293 "provider_observation": _provider_observation(request, parsed, stdout)}
