60 from record_fixtures import approved_task, approve_task, task_disposition

62 engine = context_engine(instance, runtime)
63 config = engine.instance['config']
64 config['session_limits'] = deepcopy(model_policy.LIMITS)
65 config['session_limits']['work'][config['tier_models']['high']] = 0
66 config['role_tiers']['research_work'] = 'low'
67 engine.instance.update(execution_profiles.build_profiles(config))
68 task = engine.runtime['tasks'][task['id']]
69 task['profile_refs'] = deepcopy(engine.instance['role_profile_refs']['research_work'])
70 task['model_assessment'] = {'difficulty': 'low', 'importance': 'low', 'reason': 'Initial bounded work'}
71 approve_task(task, engine.instance, engine.runtime)
72 before = deepcopy(task)
73 high = {'difficulty': 'high', 'importance': 'high', 'reason': 'New plan needs stronger reasoning'}
74 engine.runtime['executions']['EX-plan'] = {'id': 'EX-plan', 'role': 'research_plan', 'status': 'ended',
75 'model_decisions': {task['id']: high}}
76 engine._apply_task_plan({'dispositions': [task_disposition(task['id'])], 'task_assessments': []},
77 'EX-plan', deepcopy(engine.runtime['tasks']), [task['id']])
78 action = next(a for a in engine.runtime['queue'] if a['role'] == 'research_work')
79 selected = engine._select_model(action)
80 self.assertEqual(selected['tier'], 'high')
81 self.assertEqual(selected['model'], config['tier_models']['high'])
82 self.assertEqual(action['model_assessment'], high)
83 self.assertEqual(planning.contract_hash(engine.runtime['tasks'][task['id']]), planning.contract_hash(before))
84 self.assertEqual(engine.runtime['tasks'][task['id']]['reviews'], before['reviews'])
85 high['reason'] = 'Mutation of the source must not change the request'
86 self.assertNotEqual(action['model_assessment'], high)
87 from do_math.session_slots import SessionSlots

92 transition_execution(execution, 'starting', at='2026-09-23T00:00:00+00:00')
93 slots = SessionSlots(engine.runtime, engine.instance)
94 self.assertFalse(slots.acquire('EX-no-slot', profile=profile, execution=execution))
95 self.assertEqual(engine._select_model(action)['tier'], 'high')

201class DuplicateRevisionConnections(unittest.TestCase):
202 def test_same_task_replacements_are_rejected_before_any_receiver_mutation(self):
203 from record_fixtures import approved_task
204 from research_fixtures import candidate
205 from do_math import revisions
206 task, instance, runtime = approved_task()
207 instance.update(config={"lean_command": ["unused-fixture"], "scholarly_enabled": False},
208 knowledge={})
209 runtime.update(cycles=0, waiting_actions={})
210 engine = routing_engine()
211 engine.instance, engine.runtime = instance, runtime
212 engine._evaluation_context = create_autospec(engine._evaluation_context, return_value={})
213 action = {"id": "AC-revise", "role": "research_preparation",
214 "revision_targets": revisions.capture_targets(instance, runtime, [task["id"]])}
