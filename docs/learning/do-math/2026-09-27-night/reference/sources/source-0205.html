13 def test_large_manifest_is_indexed_once_without_authorizing_the_projection(self):
14 from do_math import submission
15 with tempfile.TemporaryDirectory() as directory:
16 root = Path(directory)
17 paths = [f"environment/module-{i}.olean" for i in range(2000)]
18 context = {"manifest": {"files": [{"path": p} for p in paths]},
19 "reference_catalog": [{"path": "invented"}],
20 "proposal": {"valid": {}},
21 "other": [{"ref": "proposal:missing"}]}
22 with patch.object(submission, '_reference_locators', wraps=submission._reference_locators) as locators:
23 values = {row['value'] for row in reference_catalog(context, root)}
24 self.assertEqual(locators.call_count, 1)
25 self.assertTrue(set(paths).issubset(values))
26 self.assertNotIn('invented', values)
27 self.assertNotIn('proposal:missing', values)
28 context['manifest']['files'].pop()

30 validate_reference(paths[-1], context, root)
