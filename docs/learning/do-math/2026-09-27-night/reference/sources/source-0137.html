231 ControlTransport.worker_submit(

233 payload={"manifest": [f"{prefix}-{index}.json"]})
234 except Exception as error:
235 errors.append(error)
236
237 # A broken I/O implementation must not keep the test process alive.
238 threads = [threading.Thread(target=producer, args=(prefix,), daemon=True) for prefix in ("left", "right")]
239 for thread in threads:

241 start.wait(timeout=10)

243 while any(thread.is_alive() for thread in threads):
244 if time.monotonic() >= deadline:
245 errors.append(TimeoutError("request producers did not finish; inbox=" +
246 repr(sorted(p.name for p in (self.endpoint / "inbox").glob("*.json")))))

249 self.transport.pump(lambda _: {"state": "rejected", "reason": "stress observation"})
250 except Exception as error:
251 errors.append(error)
252 break
253 time.sleep(.01)
254 stopping.set()
255 for thread in threads:
