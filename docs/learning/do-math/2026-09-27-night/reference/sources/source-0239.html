69 def wait_for(self, predicate, timeout: float = 15.0) -> None:
70 deadline = time.monotonic() + timeout
71 while time.monotonic() < deadline:
72 if predicate():
73 return
74 time.sleep(0.02)

77 def test_root_is_suspended_until_explicit_resume(self) -> None:
78 marker = self.path / "started"
79 code = f"from pathlib import Path; Path({str(marker)!r}).write_text('ok')"
80 with WindowsJob([sys.executable, "-c", code]) as job:
81 self.assertFalse(marker.exists())
82 self.assertGreater(job.creation_filetime, 0)
83 initial = job.inspect()
84 self.assertEqual(initial.active_processes, 1)
85 self.assertIn(job.pid, initial.process_ids)
86 self.assertFalse(initial.root_signaled)
87
88 job.resume()
89 self.wait_for(marker.exists)

94 self.assertEqual(completed.process_ids, ())
95 self.assertTrue(completed.root_signaled)

97 def test_wait_timeout_keeps_job_owned_until_explicit_close(self) -> None:
98 kernel = ctypes.WinDLL("kernel32", use_last_error=True)
99 kernel.OpenProcess.argtypes = [ctypes.c_uint32, ctypes.c_int, ctypes.c_uint32]
100 kernel.OpenProcess.restype = ctypes.c_void_p
101 kernel.WaitForSingleObject.argtypes = [ctypes.c_void_p, ctypes.c_uint32]
102 kernel.WaitForSingleObject.restype = ctypes.c_uint32
103 kernel.TerminateProcess.argtypes = [ctypes.c_void_p, ctypes.c_uint32]
104 kernel.CloseHandle.argtypes = [ctypes.c_void_p]
105 job = WindowsJob([sys.executable, "-c", "import time; time.sleep(60)"])
106 # Retain the process, not the Job: close must release the last Job handle.
107 root_handle = kernel.OpenProcess(0x00100001, False, job.pid)
108 try:
109 self.assertTrue(root_handle, f"OpenProcess failed: {ctypes.get_last_error()}")
110 job.resume()
111 self.assertFalse(job.wait(0.01))
112 inspection = job.inspect()
113 self.assertGreaterEqual(inspection.active_processes, 1)
114 self.assertIn(job.pid, inspection.process_ids)
115 job.close()
116 self.assertEqual(kernel.WaitForSingleObject(root_handle, 5000), 0,
117 "a timed-out wait leaked a live root after close")
118 finally:
119 job.close()
120 if root_handle:
121 try:
122 if kernel.WaitForSingleObject(root_handle, 0) == 258:
123 kernel.TerminateProcess(root_handle, 1)
124 kernel.WaitForSingleObject(root_handle, 5000)
125 finally:
126 kernel.CloseHandle(root_handle)

168 def test_owner_process_loss_closes_job_and_ends_root(self) -> None:
169 root_pid_file = self.path / "root-pid"
170 started_marker = self.path / "root-started"
171 module_root = str(SCRIPTS)
172 root_code = (
173 "import time; from pathlib import Path; "
174 f"Path({str(started_marker)!r}).write_text('running'); "
175 "time.sleep(60)"
176 )
177 owner_code = (
178 "import os, sys; from pathlib import Path; "
179 f"sys.path.insert(0, {module_root!r}); "
180 "from do_math.windows_job import WindowsJob; "
181 f"job=WindowsJob([sys.executable, '-c', {root_code!r}]); "
182 "job.resume(); "
183 f"Path({str(root_pid_file)!r}).write_text(str(job.pid)); "
184 "sys.stdin.buffer.read(1); os._exit(0)"
185 )
186 kernel = ctypes.WinDLL("kernel32", use_last_error=True)
187 kernel.OpenProcess.argtypes = [ctypes.c_uint32, ctypes.c_int, ctypes.c_uint32]
188 kernel.OpenProcess.restype = ctypes.c_void_p
189 kernel.WaitForSingleObject.argtypes = [ctypes.c_void_p, ctypes.c_uint32]
190 kernel.WaitForSingleObject.restype = ctypes.c_uint32
191 kernel.TerminateProcess.argtypes = [ctypes.c_void_p, ctypes.c_uint32]
192 kernel.TerminateProcess.restype = ctypes.c_int
193 kernel.CloseHandle.argtypes = [ctypes.c_void_p]
194 kernel.CloseHandle.restype = ctypes.c_int
195 owner = subprocess.Popen(
196 [sys.executable, "-c", owner_code], stdin=subprocess.PIPE,
197 creationflags=subprocess.CREATE_NO_WINDOW,
198 )
199 root_handle = None
200 try:
201 self.wait_for(lambda: root_pid_file.exists() and bool(root_pid_file.read_text()))
202 # Hold only the process handle; retaining a Job handle would prevent
203 # the owner-loss condition from closing the last Job handle.
204 root_handle = kernel.OpenProcess(0x00100001, False, int(root_pid_file.read_text()))
205 self.assertTrue(root_handle, f"OpenProcess failed: {ctypes.get_last_error()}")
206 self.wait_for(started_marker.exists)
207 self.assertIsNone(owner.poll())
208 self.assertEqual(kernel.WaitForSingleObject(root_handle, 0), 258)
209 owner.stdin.write(b"x")
210 owner.stdin.flush()
211 self.assertEqual(owner.wait(timeout=10), 0)
212 self.assertEqual(
213 kernel.WaitForSingleObject(root_handle, 5000), 0,
214 "the running root did not terminate after its Job owner exited",
215 )
216 finally:
217 if owner.poll() is None:
218 owner.kill()
219 owner.wait(timeout=10)
220 owner.stdin.close()
221 if root_handle:
222 try:
223 if kernel.WaitForSingleObject(root_handle, 0) == 258:
224 kernel.TerminateProcess(root_handle, 1)
225 kernel.WaitForSingleObject(root_handle, 5000)
226 finally:
227 kernel.CloseHandle(root_handle)

229 def test_job_handle_is_not_inheritable_and_breakaway_is_disabled(self) -> None:
230 with WindowsJob([sys.executable, "-c", "import time; time.sleep(30)"]) as job:
231 flags = ctypes.c_uint32()
232 self.assertTrue(_kernel32.GetHandleInformation(job.job_handle, ctypes.byref(flags)))
233 self.assertEqual(flags.value & _HANDLE_FLAG_INHERIT, 0)
234
235 limits = _EXTENDED_LIMITS()
236 returned = ctypes.c_uint32()
237 self.assertTrue(_kernel32.QueryInformationJobObject(
238 job.job_handle,
239 _JobObjectExtendedLimitInformation,
240 ctypes.byref(limits),
241 ctypes.sizeof(limits),
242 ctypes.byref(returned),
243 ))
244 limit_flags = limits.BasicLimitInformation.LimitFlags
245 self.assertTrue(limit_flags & _JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE)
246 self.assertFalse(limit_flags & _JOB_OBJECT_LIMIT_BREAKAWAY_OK)
247 self.assertFalse(limit_flags & _JOB_OBJECT_LIMIT_SILENT_BREAKAWAY_OK)
