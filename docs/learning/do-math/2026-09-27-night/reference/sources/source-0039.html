105def _installation_listing(root, documents):
106 """Fingerprints of the files an installation manifest hashes, from directory listings.
107
108 One ``scandir`` pass per directory below ``scripts/do_math`` and per
109 document directory yields every ``.py`` file and listed document with its
110 (size, modification, birth) fingerprint, without a stat call per file. An
111 added or removed file changes the key set and a rewrite its fingerprint.
112 A Windows directory entry shows a rewrite once its writer closes the file.
113 """
114 rows = {}
115
116 def walk(directory, prefix):
117 with os.scandir(directory) as entries:
118 for entry in entries:
119 if entry.is_dir(follow_symlinks=False):
120 walk(entry.path, prefix + entry.name + "/")
121 elif entry.name.lower().endswith(".py"):
122 rows[prefix + entry.name] = stat_fingerprint(entry.stat())
123
124 walk(root / "scripts" / "do_math", "scripts/do_math/")
125 by_directory = {}
126 for relative in documents:
127 parent, _, name = relative.rpartition("/")
128 by_directory.setdefault(parent, {})[os.path.normcase(name)] = relative
129 for parent, names in by_directory.items():
130 with os.scandir(root / parent) as entries:
131 for entry in entries:
132 relative = names.get(os.path.normcase(entry.name))
133 if relative is not None:
134 rows[relative] = stat_fingerprint(entry.stat())
135 return rows

207def _next_control_order(inbox: Path) -> str:
208 """Persist an order that remains comparable after a restart or clock change."""
209 counter = inbox / ".acceptance-counter"
210 if counter.exists():
211 previous = int(counter.read_text(encoding="ascii"))
212 else:
213 # The inbox retains processed requests. Recover if only the counter was
214 # lost, without allowing a new request to overtake an older pending one.
215 previous = max((int(json.loads(path.read_bytes())["accepted_order"])
216 for path in inbox.glob("*.json")), default=0)
217 if previous < 0:
218 raise ValueError("control acceptance counter is invalid")
219 next_order = previous + 1
220 write_bytes_atomic(counter, str(next_order).encode("ascii"), fsync=True)
221 return f"{next_order:020d}"

350def query_records(repository: Path, instance_id: str, *, snapshot: str | None = None,
351 kind: str | None = None, text: str = "", limit: int = 50,
352 cursor: str | None = None, related_to: str | None = None, direction: str = "both") -> dict:
353 if isinstance(limit, bool) or not isinstance(limit, int) or not 1 <= limit <= 1000:
354 raise ValueError("limit must be between 1 and 1000")
355 if direction not in {"incoming", "outgoing", "both"}:
356 raise ValueError("direction must be incoming, outgoing or both")
357 parameters = {"instance_id": instance_id, "kind": kind, "text": text, "related_to": related_to, "direction": direction}
358 last = None
359 if cursor:
360 try:
361 decoded = json.loads(base64.urlsafe_b64decode(cursor).decode("utf-8"))
362 if decoded["parameters"] != parameters or (snapshot is not None and snapshot != decoded["snapshot"]):
363 raise ValueError("cursor query or snapshot changed")
364 snapshot, last = decoded["snapshot"], decoded["last"]
365 except (KeyError, UnicodeError, json.JSONDecodeError, binascii.Error) as error:
366 raise ValueError("invalid cursor") from error
367 store = ResearchRepository(repository, instance_id)
368 head, instance, runtime, rows = store.load_records(snapshot)
369 descriptors = {row["id"]: row for row in rows if row["id"]}
370 records = sorted((_query_record(row) for row in descriptors.values()), key=lambda row: row["id"])
371 if kind:
372 records = [r for r in records if r["family"] == kind or r.get("kind") == kind]
373 if text:
374 records = [r for r in records if text.casefold() in canonical_json(r).decode("utf-8").casefold()]
375 if related_to:
376 relations = [r for r in instance["relations"] if
377 (direction in {"outgoing", "both"} and r["source"] == related_to) or
378 (direction in {"incoming", "both"} and r["target"] == related_to)]
379 ids = {r["id"] for r in relations} | {r["target"] for r in relations if r["source"] == related_to} | {
380 r["source"] for r in relations if r["target"] == related_to}
381 records = [r for r in records if r["id"] in ids]
382 total = len(records)
383 if last is not None:
384 if last not in {r["id"] for r in records}:
385 raise ValueError("cursor does not name a record in the fixed query")
386 records = [r for r in records if r["id"] > last]
387 selected = records[:limit]
388 truncated = len(records) > limit
389 next_cursor = None
390 if truncated:
391 next_cursor = base64.urlsafe_b64encode(canonical_json({"parameters": parameters, "snapshot": head,
392 "last": selected[-1]["id"]})).decode("ascii")
393 return {"snapshot": head, "total": total, "loaded": len(selected), "truncated": truncated,
394 "items": [{"record": r, **_record_evidence(descriptors[r["id"]])} for r in selected],
395 "query_basis": {"parameters": parameters, "limit": limit, "complete": not truncated,
396 "snapshot": head, "loaded_ids": [row["id"] for row in selected]},
397 "next_cursor": next_cursor}

462 @staticmethod
463 def _installation_manifest() -> dict:
464 """Hash the installed code and role documents; unchanged files reuse their digests.
465
466 While the directory listings of the last hashed edition are unchanged,
467 that edition's manifest is returned without reading the role map,
468 globbing or hashing. The listing is taken before the files are found
469 and hashed, so a later change always differs from it; a listing with a
470 fingerprint too recent to be settled is not reused. Callers only
471 compare or copy the result.
472 """
473 global _HASHED_INSTALLATION
474 root = _SKILL_ROOT
475 if _HASHED_INSTALLATION is not None:
476 documents, listing, manifest = _HASHED_INSTALLATION
477 try:
478 if _installation_listing(root, documents) == listing:
479 return manifest
480 except OSError:
481 pass # A removed directory is found or reported by hashing below.
482 documents = dict.fromkeys(relative for paths in reading_map(skill_root=root).values() for relative in paths)
483 documents.update(dict.fromkeys(OPERATING_DOCUMENTS))
484 listing = _installation_listing(root, documents)
485 paths = [*sorted((root / "scripts" / "do_math").rglob("*.py")), *(root / relative for relative in documents)]
486 files = {path.relative_to(root).as_posix(): cached_sha256(path) for path in paths}
487 manifest = {"files": files, "sha256": exact_content_hash(files)}
488 if all(fingerprint_settled(fingerprint) for fingerprint in listing.values()):
489 _HASHED_INSTALLATION = (tuple(documents), listing, manifest)
490 return manifest

541 def _register_execution(self, action, execution, *, profile=None):

543

545 in this owner step; without it the profile is selected here.

555 continuation_of=action.get("continuation_of"), context_origin=action.get("context_origin", []))

561 execution.update(model=profile["ai"]["model"], effort=profile["ai"]["effort"])
562 else:
563 for field in ("model", "effort", "model_selection"):
564 execution.pop(field, None)
565 held = activity.get("waiting", {}).get("reason") in {"review_input_restoration", "review_recovery"}
566 admission = ({"disposition": "review_input_restoration"} if held else self._admit_review_execution(action, execution))
567 if admission["disposition"] not in {"admitted", "already_admitted", "not_applicable"}:
568 if admission['disposition'] in {'review_budget_exhausted', 'review_scope_unaffordable',
569 'review_followup_reserved', 'review_resource_pending'}:
570 self._request_review_resource_disposition(action, admission['disposition'], admission)

575 condition=admission, next_owner="review_policy", at=now())

577 return None
578 if activity["status"] == "waiting":
579 activities.resume_activity(self.runtime, activity["id"], condition=activity["waiting"]["condition"], at=now())

586 condition=deepcopy(action["dispatch_wait"]), next_owner=action["dispatch_wait"]["next_owner"], at=now())

588 action.pop("dispatch_wait", None)
589 execution["action"] = deepcopy(action)

591 elif requested_state in {"starting", "running"}:
592 execution_lifecycle.transition_execution(execution, "starting", at=now())
593 if requested_state == "running" and not (machine_admission and profile.get("host_resources")):
594 execution_lifecycle.transition_execution(execution, "running", at=now())
595 activities.assign_execution(self.runtime, activity["id"], execution)
596 revisions.record_import_rework(self.instance, self.runtime, action, execution['id'])

608 self._settle_review_resource_dispositions()
609 execution = self.runtime["executions"][execution_id]
610 aid = execution["owner_activity_id"]
611 activity = self.runtime["activities"][aid]
612 cycle = self.runtime.get("review_cycles", {}).get(execution.get("action", {}).get("review_cycle_id"), {})

614 return

628 condition={"operational_disposition_ids": repairs}, next_owner="engine", at=now())

631 waiting = [row for row in self.runtime.get("waiting_actions", {}).values()
632 if row.get("action", {}).get("owner_activity_id") == aid]

637 return
638 if waiting:
639 wait = waiting[0]
640 reason = (wait['kind'] if wait.get('kind') in {'review_input_restoration', 'review_recovery', 'review_authority'}
641 else wait.get('reason') or wait['kind'])
642 activities.wait_for(self.runtime, aid, reason=reason,
643 condition={"action_ids": [row["action"]["id"] for row in waiting]},
644 next_owner=waiting[0].get("next_owner", "engine"), at=now())
645 return
646 preserved = [row for row in activity['unresolved'] if isinstance(row, dict)
647 and row.get('reason') == 'goal_completed_before_dispatch']
648 followups = [row for row in activity['followups'] if isinstance(row, dict)
649 and row.get('reason') == 'goal_completed_before_dispatch']
650 failed = ([] if execution["outcome"] == "output_accepted" or execution.get("formal_status") == "verified"

653 unresolved=[*preserved, *failed], followups=followups, at=now())

769 def _inbox(self) -> list[dict]:
770 path = Path(self.instance["workspace"]) / "control" / self.instance["id"]
771 processed = set(self.runtime["processed_controls"])
772 results = []
773 for file in sorted(path.glob("*.json")):
774 if file.stem in processed:
775 continue
776 request = json.loads(file.read_text(encoding="utf-8"))
777 if not isinstance(request, dict) or request.get("id") != file.stem or request.get("instance_id") != self.instance["id"]:
778 raise ValueError("control identity mismatch")
779 if any(not isinstance(request.get(key), str) for key in ('created_at', 'accepted_order')):
780 raise ValueError('control ordering envelope is invalid')
781 results.append(request)
782 # ULID suffixes are random within a millisecond; they do not preserve
783 # the order in which a pause, release, or newer user control arrived.
784 # Distinct calls can share even the complete wall-clock timestamp.
785 # Preserve their host acceptance order, including equal wall-clock timestamps.
786 return sorted(results, key=lambda request: (request["accepted_order"], request["created_at"], request["id"]))

838 reason = "Required source content changed: " + spec["name"]
839 if task["status"] != "completed":
840 task["status"] = "planned"

844 for execution in self.runtime["executions"].values():

855 if self.runtime["status"] in {"waiting", "capacity_waiting"}: self.runtime["status"] = "ready"
856 self.runtime["processed_controls"].append(request["id"])
857 self._record_control_result(request, source_id=entry["id"])

862 handled = apply_control(self.runtime, request) or self._review_recovery_control(request) or self._knowledge_io_control(request) or self._academic_control(request) or self._governance_control(request) or self._apply_extended_control(request)
863 except (ValueError, KeyError, TypeError) as error:
864 self._reject_control(request, str(error))
865 continue
866 if handled:
867 if action == 'quiesce' and not (self._calls is not None and self._calls.futures):
868 settle(self.runtime)

990 def _preserve_outputs(self, execution_id, workspace, bundle, *, max_bytes, evidence_dir=None):
991 """Seal one execution's outputs, control files and checkpoint into one artifact bundle.
992
993 The checkpoint is recorded in runtime. ``evidence_dir`` adds the provider
994 evidence as the checkpoint's observation sources. Returns the artifacts,
995 output file rows, checkpoint reference and whether every output fit.
996 """
997 prefix = f"runtime/artifacts/{bundle}"
998 files = (self._sealed_submission_files(self.runtime["executions"][execution_id])
999 or collect_outputs(workspace, max_bytes=max_bytes))
1000 control_files = collect_outputs(workspace / ".control", max_bytes=max_bytes)
1001 artifacts = {f"{prefix}/output-manifest.json": canonical_json(files["manifest"]),
1002 f"{prefix}/omitted.json": canonical_json(files["omitted"]),
1003 f"{prefix}/control-manifest.json": canonical_json(control_files["manifest"]),
1004 f"{prefix}/control-omitted.json": canonical_json(control_files["omitted"])}
1005 output_files = []
1006 for path, raw in files["files"].items():
1007 reference = f"{prefix}/files/{path}"
1008 artifacts[reference] = raw
1009 output_files.append({"path": path, "artifact_ref": reference,
1010 "sha256": hashlib.sha256(raw).hexdigest(), "bytes": len(raw)})
1011 for path, raw in control_files["files"].items():
1012 artifacts[f"{prefix}/control/{path}"] = raw
1013 checkpoint, checkpoint_artifacts = preserve_checkpoint(workspace, artifact_prefix=f"{prefix}/checkpoint",
1014 max_bytes=max_bytes, previous=self.runtime.get("checkpoints", {}).get(execution_id))
1015 artifacts.update(checkpoint_artifacts)
1016 if evidence_dir is not None:
1017 evidence = collect_outputs(evidence_dir, max_bytes=max_bytes)
1018 checkpoint["observation_sources"] = []
1019 for path, raw in evidence["files"].items():
1020 reference = f"{prefix}/provider-evidence/{path}"
1021 artifacts[reference] = raw
1022 checkpoint["observation_sources"].append({"artifact_ref": reference,
1023 "sha256": hashlib.sha256(raw).hexdigest(), "bytes": len(raw), "path": path})
1024 checkpoint["observation_evidence_complete"] = evidence["complete"]
1025 checkpoint["observation_evidence_omitted"] = deepcopy(evidence["omitted"])
1026 self.runtime.setdefault("checkpoints", {})[execution_id] = checkpoint
1027 checkpoint_ref = f"{prefix}/checkpoint-recovery.json"
1028 artifacts[checkpoint_ref] = canonical_json(checkpoint)
1029 return artifacts, output_files, checkpoint_ref, files["complete"]
