38def open_activity(runtime: dict, *, activity_id: str, activity_kind: str, owner_ref: Any,
39 input_refs: list[Any], opened_at: str, stage: str | None = None,
40 execution_limit: int | None = None) -> dict:
41 """Create an activity, or return its exact idempotent prior declaration."""
42 require_text(activity_id, "activity_id")
43 require_text(activity_kind, "activity_kind")
44 require_text(opened_at, "opened_at")
45 if owner_ref is None:
46 raise ValueError("owner_ref is required")
47 if stage is not None:
48 require_text(stage, "stage")
49 if execution_limit is not None and (isinstance(execution_limit, bool) or not isinstance(execution_limit, int) or execution_limit < 1):
50 raise ValueError("execution_limit must be a positive integer or null")
51 record = {
52 "id": activity_id,
53 "activity_kind": activity_kind,
54 "owner_ref": deepcopy(owner_ref),
55 "opened_at": opened_at,
56 "input_refs": _references(input_refs, "input_refs"),
57 "status": "ready",
58 "stage": stage,
59 "execution_limit": execution_limit,
60 "assigned_execution_ids": [],
61 "result_refs": [],
62 "unresolved": [],
63 "followups": [],
64 }
65 activities = _activities(runtime)
66 existing = activities.get(activity_id)
67 if existing is not None:
68 declaration = ("id", "activity_kind", "owner_ref", "opened_at", "input_refs", "stage", "execution_limit")
69 if {key: existing.get(key) for key in declaration} != {key: record[key] for key in declaration}:
70 raise ValueError(f"activity {activity_id} was already opened with different facts")
71 return existing
72 activities[activity_id] = record

76def assign_execution(runtime: dict, activity_id: str, execution: dict, *, stage: str | None = None) -> dict:
77 """Attach one execution attempt to a non-finished activity atomically."""
78 activity = _activities(runtime).get(activity_id)
79 if activity is None:
80 raise ValueError(f"unknown activity {activity_id}")
81 if activity["status"] == "finished":
82 raise ValueError("a finished activity cannot receive another execution")
83 if activity["status"] == "waiting":
84 raise ValueError("resolve the recorded activity wait before assigning an execution")
85 execution_id = require_text(execution.get("id"), "execution.id")
86 if execution.get("owner_activity_id") != activity_id:
87 raise ValueError("execution owner_activity_id differs from activity")
88 if stage is not None and activity.get("stage") not in {None, stage}:
89 raise ValueError("execution stage differs from activity stage")
90 assigned = activity["assigned_execution_ids"]
91 if execution_id not in assigned:
92 limit = activity.get("execution_limit")
93 if limit is not None and len(assigned) >= limit:
94 raise ValueError("activity execution limit is exhausted")
95 assigned.append(execution_id)
96 if stage is not None:
97 activity["stage"] = stage
98 _executions(runtime)[execution_id] = execution
99 if activity["status"] == "ready":
100 activity["status"] = "working"
101 activity.pop("waiting", None)
102 return activity

105def bind_action(runtime: dict, action: dict, activity_id: str) -> dict:
106 """Bind the working copy and its same-ID queued request before preparation."""
107 activity = _activities(runtime).get(activity_id)
108 if activity is None or activity["status"] == "finished":
109 raise ValueError("action requires an existing unfinished activity")
110 action_id = require_text(action.get("id"), "action.id")
111 queued = [row for row in runtime.get("queue", []) if row.get("id") == action_id]
112 if len(queued) > 1:
113 raise ValueError("action has duplicate queued requests")
114 for row in [action, *queued]:
115 if row.get("owner_activity_id", activity_id) != activity_id:
116 raise ValueError("action is already bound to another activity")
117 action["owner_activity_id"] = activity_id
118 for row in queued:
119 row["owner_activity_id"] = activity_id
120 return activity

123def successor_state(runtime: dict, execution_id: str) -> dict | None:
124 """Read explicit continuation edges; unrelated success never resolves a failure."""
125 executions = runtime.get('executions', {})
126 target = executions[execution_id]
127 owner = target.get('owner_activity_id')
128 descendants, pending = set(), {execution_id}
129 while pending:
130 parent = pending.pop()
131 for eid, execution in executions.items():
132 previous = execution.get('continuation_of') or execution.get('action', {}).get('continuation_of')
133 if (eid != execution_id and eid not in descendants and previous == parent
134 and execution.get('owner_activity_id') == owner):
135 descendants.add(eid)
136 pending.add(eid)
137 queued = [row['id'] for row in runtime.get('queue', [])
138 if row.get('continuation_of') in {execution_id, *descendants}
139 and row.get('owner_activity_id', owner) == owner]
140 if not descendants and not queued:
141 return None
142 live = sorted(eid for eid in descendants if executions[eid]['status'] != 'ended')
143 return {'state': 'in_progress' if live or queued else 'continued',
144 'execution_ids': sorted(descendants), 'live_execution_ids': live,
145 'action_ids': sorted(set(queued))}

148def continue_activity(runtime: dict, execution_id: str, *, at: str, reason: str,
149 condition: Any = None, next_owner: str = "engine") -> dict:
150 """Preserve the exact ended attempt's obligation and request its successor.
151
152 A pause condition is an activity wait, even though the pending action stays
153 queued. The central lifecycle gate must resolve it before dispatch. An old
154 checkpoint remains keyed by continuation_of; it is not evidence that a new
155 conversation has inherited the old context.
156 """
157 require_text(at, "at")
158 require_text(reason, "reason")
159 require_text(next_owner, "next_owner")
160 execution = _executions(runtime)[execution_id]
161 if execution["status"] != "ended":
162 raise ValueError("only an ended execution can request continuation")
163 activity = _activities(runtime)[execution["owner_activity_id"]]
164 action = deepcopy(execution.get("action", {}))
165 action_id = require_text(action.get("id"), "execution.action.id")
166 live = [eid for eid in activity["assigned_execution_ids"] if _executions(runtime)[eid]["status"] != "ended"]
167 if live:
168 raise ValueError("continuation requires the activity's current attempts to end")
169 bind_action(runtime, action, activity["id"])
170 queued = next((row for row in runtime.get("queue", []) if row.get("id") == action_id), None)
171 if queued is not None:
172 # Keep current task/authority inputs; only the execution relation changes.
173 action = deepcopy(queued)
174 action.update(owner_activity_id=activity["id"], continuation_of=execution_id)
175 action.setdefault("context_origin", [])
176 if queued is None:
177 runtime.setdefault("queue", []).append(action)
178 else:
179 queued.update(action)
180 action = queued
181 obligation = {"action_id": action_id, "execution_id": execution_id, "reason": reason}
182 activity["unresolved"] = [row for row in activity["unresolved"]
183 if not isinstance(row, dict) or row.get("action_id") != action_id] + [obligation]
184 followup = {**obligation, "next_owner": next_owner}
185 activity["followups"] = [row for row in activity["followups"]
186 if not isinstance(row, dict) or row.get("action_id") != action_id] + [followup]
187 if condition is not None:
188 wait_for(runtime, activity["id"], reason=reason, condition=condition, next_owner=next_owner, at=at)
189 elif activity["status"] != "waiting":
190 activity["status"] = "ready"
191 return action

194def wait_for(runtime: dict, activity_id: str, *, reason: str, condition: Any,
195 next_owner: Any, at: str) -> dict:
196 """Record a durable wait after every currently assigned attempt ended."""
197 activity = _activities(runtime).get(activity_id)
198 if activity is None:
199 raise ValueError(f"unknown activity {activity_id}")
200 if activity["status"] == "finished":
201 raise ValueError("a finished activity cannot wait")
202 require_text(reason, "reason")
203 require_text(at, "at")
204 if condition is None or next_owner is None:
205 raise ValueError("condition and next_owner are required")
206 executions = _executions(runtime)
207 live = [eid for eid in activity["assigned_execution_ids"]
208 if executions.get(eid, {}).get("status") != "ended"]
209 if live:
210 raise ValueError("an activity may wait only after its assigned executions ended")
211 activity["status"] = "waiting"
212 activity["waiting"] = {"reason": reason, "condition": deepcopy(condition),
213 "next_owner": deepcopy(next_owner), "at": at}
214 return activity

233def finish_activity(runtime: dict, activity_id: str, *, result_refs: list[Any], unresolved: list[Any],
234 followups: list[Any], at: str) -> dict:
235 """Seal a work unit without equating it to task or mathematical completion."""
236 activity = _activities(runtime).get(activity_id)
237 if activity is None:
238 raise ValueError(f"unknown activity {activity_id}")
239 require_text(at, "at")
240 executions = _executions(runtime)
241 live = [eid for eid in activity["assigned_execution_ids"]
242 if executions.get(eid, {}).get("status") != "ended"]
243 if live:
244 raise ValueError("an activity with a live execution cannot finish")
245 sealed = {"result_refs": _references(result_refs, "result_refs"),
246 "unresolved": _references(unresolved, "unresolved"),
247 "followups": _references(followups, "followups"), "finished_at": at}
248 if activity["status"] == "finished":
249 if {key: activity.get(key) for key in sealed} != sealed:
250 raise ValueError("a finished activity is immutable")
251 return activity
252 activity.update(sealed, status="finished")
253 activity.pop("waiting", None)
254 return activity

257def validate_activities(runtime: dict) -> list[str]:
258 """Return structural errors; callers choose the persistent validation policy."""
259 errors: list[str] = []
260 activities = runtime.get("activities", {})
261 executions = runtime.get("executions", {})
262 if not isinstance(activities, dict) or not isinstance(executions, dict):
263 return ["activities and executions must be dictionaries"]
264 assigned_by_execution: dict[str, str] = {}
265 for activity_id, activity in activities.items():
266 if not isinstance(activity, dict) or activity.get("id") != activity_id:
267 errors.append(f"{activity_id}: invalid activity record")
268 continue
269 for execution_id in activity["assigned_execution_ids"]:
270 execution = executions.get(execution_id)
271 if execution is None or execution.get("owner_activity_id") != activity_id:
272 errors.append(f"{activity_id}: execution binding is invalid for {execution_id}")
273 elif execution_id in assigned_by_execution:
274 errors.append(f"{execution_id}: assigned to multiple activities")
275 else:
276 assigned_by_execution[execution_id] = activity_id
277 if activity.get("status") == "waiting" and not isinstance(activity.get("waiting"), dict):
278 errors.append(f"{activity_id}: waiting state lacks its condition")
279 if activity.get("status") == "finished" and "finished_at" not in activity:
280 errors.append(f"{activity_id}: finished state lacks finished_at")
281 for execution_id, execution in executions.items():
282 owner = execution.get("owner_activity_id") if isinstance(execution, dict) else None
283 if owner not in activities or assigned_by_execution.get(execution_id) != owner:
284 errors.append(f"{execution_id}: execution has no matching activity")
285 continue
286 predecessor = execution.get("continuation_of")
287 if predecessor is not None:
288 previous = executions.get(predecessor)
289 if previous is None or previous.get("owner_activity_id") != owner or previous.get("status") != "ended":
290 errors.append(f"{execution_id}: continuation is not an ended execution in the same activity")
291 return errors
