197 import jsonschema
198 i, rt = audited_governance()
199 basis = gov.additional_audit_input(i, rt)
200 rt['executions']['AUDIT'] = {'role': 'goal_audit', 'additional_audit_input': basis}
201 row = {'audit_ref': 'AU', 'assessor': 'goal_audit', 'verdict': 'partial', 'reason': 'Only partial',
202 'input_hashes': {'CH1': exact_content_hash(i['governance']['CH1'])}}
203 jsonschema.Draft202012Validator(gov.additional_audit_schema(i)).validate([row])
204 mapped = gov.evaluate_goal_audits(i, rt, 'AUDIT', [row])
205 self.assertEqual(mapped[0]['decision'], 'not_satisfied')
206 with self.assertRaises(ValueError): gov.validate_goal_additional_audits(i, [{'execution_id': 'AUDIT', 'additional_audits': mapped}])
207 with self.assertRaises(ValueError): gov.evaluate_goal_audits(i, rt, 'AUDIT', [])
208 row['verdict'] = 'pass'
209 mapped = gov.evaluate_goal_audits(i, rt, 'AUDIT', [row])
210 gov.validate_goal_additional_audits(i, [{'execution_id': 'AUDIT', 'additional_audits': mapped}])
211 i['governance']['CH1']['content']['goal'] += ' changed'
