1"""Shared toolchain records: installation identity, fingerprints and content baselines.
2
3One record file describes one installed Lean/Mathlib identity. setup_lean.py
4writes it when a toolchain is installed or replaced, re-recorded or explicitly
5content-checked; research instances only read it. The threat model is
6accidental change (updates, reinstallation, synchronization, mistakes, disk
7damage). A fingerprint row is the scandir metadata of one judgment file, so
8bytes changed while size and both times are preserved are found only by an
9explicit content check against the recorded full hashes.
10"""

72def roots(identity):
73 """Named fingerprint roots: the Lean installation and the Lake project, if any."""
74 executable = Path(identity['executable'])
75 result = {'toolchain': executable.parent.parent if executable.parent.name.lower() == 'bin' else executable.parent}
76 if identity.get('project') is not None:
77 result['project'] = Path(identity['project']['source_identity']['root'])

85def identity_errors(identity):
86 """Lexical scope rules of an identity; no host path is resolved or read."""
87 from .formal import toolchain_errors
88 errors = toolchain_errors(identity)
89 if errors:
90 return errors
91 named = {name: Path(_key(path)) for name, path in roots(identity).items()}
92 if any(a != b and (named[a] == named[b] or named[a].is_relative_to(named[b])) for a in named for b in named):
93 errors.append('toolchain and project roots must not overlap')
94 for path in (identity['project'] or {}).get('search_paths', []):
95 if not any(Path(_key(path)).is_relative_to(root) for root in named.values()):
96 errors.append('Lean search path is outside the recorded toolchain and project roots: ' + path)
97 return errors

100def observe_fingerprint(identity):
101 """Rows (size, modification ns, creation ns) of every judgment file under the roots."""
102 rows = {}
103 for name, base in sorted(roots(identity).items()):
104 _scan(base, name, rows)
105 return {'files': rows, 'sha256': exact_content_hash(rows)}

108def _scan(directory, prefix, rows):
109 with os.scandir(directory) as entries:
110 for entry in entries:
111 relative = prefix + '/' + entry.name
112 if entry.is_dir(follow_symlinks=False):
113 if entry.name != '.git':
114 _scan(entry.path, relative, rows)
115 elif entry.name.lower().endswith(JUDGMENT_SUFFIXES):
116 rows[relative] = fingerprint_json(stat_fingerprint(entry.stat(follow_symlinks=False)))

123def content_hashes(identity, paths):
124 """Full SHA-256 of each named file; used at recording and explicit content checks only."""
125 base = roots(identity)
126 def one(relative):
127 name, _, rest = relative.partition('/')
128 return relative, sha256_file(base[name] / rest)
129 with ThreadPoolExecutor(max_workers=4) as pool:
130 files = dict(pool.map(one, sorted(paths)))
131 return {'files': files, 'sha256': exact_content_hash(files)}

165def record_errors(record):
166 """Schema and self-consistency of a record; the listed file hashes are not recomputed."""
167 error = next(_VALIDATOR.iter_errors(record), None)
168 if error is not None:
169 return ['toolchain record differs from ' + SCHEMA + ': ' + '/'.join(map(str, error.absolute_path))
170 + ': ' + error.message]
171 errors = identity_errors(record['identity'])
172 if record['identity_sha256'] != exact_content_hash(record['identity']):
173 errors.append('toolchain record identity digest differs from its identity')
174 elif record['id'] != record_id(record['identity'], record['baselines'][0]['content']['sha256']):
175 errors.append('toolchain record ID differs from its identity and first content baseline')
176 if any(check['baseline'] >= len(record['baselines']) for check in record['checks']):
177 errors.append('toolchain record check names an absent baseline')
178 return errors

181def usable_errors(record):
182 """A current baseline with a successful representative check and no later mismatch."""
183 current = len(record['baselines']) - 1
184 validation = record['baselines'][current]['validation']
185 errors = []
186 if validation['status'] != 'verified' or validation['exit_code'] != 0:
187 errors.append('current toolchain baseline has no successful representative check')
188 if any(check['baseline'] == current and check['result'] == 'mismatch' for check in record['checks']):
189 errors.append('a content check found files that differ from the current toolchain baseline')
190 return errors

211@contextmanager
212def writer_lock(directory):
213 """Exclude other record writers with the OS lock of ``directory/.lock``; ``OSError`` while another holds it.
214
215 The OS releases the lock when its holder ends, so a killed writer leaves nothing to remove.
216 """
217 directory = Path(directory)
218 directory.mkdir(parents=True, exist_ok=True)
219 with os_lock(directory / '.lock', blocking=False):
220 yield

223def records_for(directory, identity):
224 """Record files in a records directory that describe exactly this identity."""
225 wanted = exact_content_hash(identity)
226 found = []
227 for path in sorted(Path(directory).glob(record_path('', 'TC-*').name)):
228 record = read_record(path)
229 if record['identity_sha256'] == wanted:
230 found.append((path, record))
231 return found
