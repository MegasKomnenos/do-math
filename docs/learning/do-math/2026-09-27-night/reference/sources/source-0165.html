29def apply_limit(root: Path, repository: Path, identifier: str, limit: int, request_id: str) -> ResearchEngine:
30 """Configure ``max_input_bytes`` publicly and record the new profiles' capability evidence."""
31 config = root / (request_id + ".json")
32 config.write_bytes(canonical_json({"max_input_bytes": limit}))
33 accepted = cli_result(root, "configure", "--repository", str(repository), "--instance", identifier,
34 "--config-file", str(config), "--request-id", request_id)
35 if not accepted["accepted"]:
36 raise AssertionError(accepted)
37 engine = ResearchEngine(repository, identifier, runner=ScriptedRunner(), formal_verifier=formal_ok)
38 engine.process_controls()
39 evidence_store = engine.instance.setdefault("capability_evidence", {})
40 for reference in engine.instance["role_profile_refs"]["maintenance"]:
41 profile = execution_profiles.resolve_profile(engine.instance["execution_profiles"], reference)
42 if profile["actor_kind"] != "ai":
43 continue
44 evidence = execution_profiles.capability_evidence(profile, FixtureAdapter().describe(),
45 observed_at=engine.runtime["last_observed_utc"])
46 evidence_store[evidence["id"]] = evidence
47 engine._save()
48 return engine
