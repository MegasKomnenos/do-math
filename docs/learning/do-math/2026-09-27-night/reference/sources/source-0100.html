77def apply_task_dispositions(instance: dict, runtime: dict, dispositions: list[dict], *,
78 execution_id: str, input_tasks: dict) -> tuple[dict, dict]:
79 """Apply withdrawal requests after snapshot and provenance validation.
80
81 The caller owns role authorization. This function deliberately cannot alter
82 goals, policies, grants, academic requirements, or user task-scope records.
83 """
84 if not isinstance(execution_id, str) or not execution_id:
85 raise ValueError("task dispositions need an execution ID")
86 if not isinstance(dispositions, list) or not isinstance(input_tasks, dict):
87 raise ValueError("task dispositions and input tasks must be collections")
88 # The sealed extension schema and the submission check fixed shape and distinct targets.
89 requested = [(item["task_id"], item["reason"]) for item in dispositions]
90
91 new_instance, new_runtime = deepcopy(instance), deepcopy(runtime)
92 tasks = new_runtime.get("tasks", {})
93 for task_id, _ in requested:
94 task = tasks.get(task_id)
95 if not isinstance(task, dict):
96 raise ValueError("unknown task: " + task_id)
97 _validate(task, input_tasks.get(task_id), new_runtime, task_id, new_instance,
98 actor_execution_id=execution_id)
99
100 for task_id, reason in requested:
101 task = tasks[task_id]
102 removed_queue = [deepcopy(action) for action in new_runtime.get("queue", [])
103 if isinstance(action, dict) and action.get("target") == task_id]
104 new_runtime["queue"] = [action for action in new_runtime.get("queue", [])
105 if not isinstance(action, dict) or action.get("target") != task_id]
106 waiting_actions = new_runtime.get("waiting_actions", {})
107 removed_waiting = []
108 for action_id, entry in list(waiting_actions.items()):
109 action = entry.get("action", {}) if isinstance(entry, dict) else {}
110 if isinstance(action, dict) and action.get("target") == task_id:
111 removed_waiting.append({"id": action_id, "entry": deepcopy(entry)})
112 waiting_actions.pop(action_id)
113 previous_hash = contract_hash(task)
114 from .task_contracts import remaining_obligations
115 assessment = task.get('assessment')
116 remaining = remaining_obligations(task, assessment)
117 if assessment:
118 task.setdefault('assessment_history', []).append(deepcopy(assessment))
119 del task['assessment']
120 task["scope_disposition"] = {
121 "kind": "withdraw", "actor_execution": execution_id, "reason": reason,
122 "previous_revision": task.get("revision"), "previous_contract_sha256": previous_hash,
123 "at": datetime.now(timezone.utc).isoformat(), "cancelled_queue_actions": removed_queue,
124 "cancelled_waiting_actions": removed_waiting,
125 'results_sha256': exact_content_hash(task.get('results', [])),
126 'assessment_sha256': exact_content_hash(assessment) if assessment else None,
127 'remaining_obligation_ids': remaining,
128 }
129 task.pop('waiting', None)
130 task["status"] = "closed"
131 return new_instance, new_runtime

197def assessment_basis(task, assessment, instance, runtime, *, records=None):
198 """Content hashes of everything an assessment's evidence reaches.
199
200 records is planning.record_index(instance, runtime) when the caller already built it.
201 """
202 from .planning import record_index,task_reference_basis,condition_refs,READ_REFERENCE_FIELDS,READ_REFERENCE_ID_FIELDS
203 from .relation_refs import relation_references, relation_evidence_references
204 if records is None:
205 records=record_index(instance,runtime)
206 refs={ref for row in assessment.get('item_dispositions',[]) for ref in row.get('evidence_refs',[])}
207 refs.update(row['authority_ref'] for row in assessment.get('item_dispositions',[]) if row.get('authority_ref'))
208 refs.update(ref for row in assessment.get('unresolved',[]) for ref in row['missing_inputs'])
209 direct = set(refs)
210 strict = {ref for ref in direct if records.get(ref, (None,))[0] in
211 {'formal-result', 'identity-assessment', 'assessment-use', 'goal-judgment'}}
212 pending = list(refs)
213 visited = set()
214 while pending:
215 ref = pending.pop()
216 mode = (ref, ref in strict)
217 if mode in visited:
218 continue
219 visited.add(mode)
220 found = records.get(ref)
221 if not found:
222 continue
223 family, record = found
224 # The reference fields every read closure shares (planning.READ_REFERENCE_FIELDS).
225 linked = {item if isinstance(item, str) else item['id'] for key in READ_REFERENCE_FIELDS
226 if isinstance(record.get(key), list) for item in record[key]
227 if isinstance(item, str) or isinstance(item, dict) and isinstance(item.get('id'), str)}
228 fields = (*READ_REFERENCE_ID_FIELDS, 'source_ref')
229 if family == 'formal-result' and ref not in strict:
230 fields = tuple(key for key in fields if key != 'trust_environment_id')
231 linked.update(record[key] for key in fields if isinstance(record.get(key), str))
232 linked.update(row.get('id') or row['name'] for row in record.get('source_basis', [])
233 if isinstance(row.get('id') or row.get('name'), str))
234 linked.update(condition_refs(record.get('occurrence_condition')))
235 linked.update(relation_references(record.get('relation_claim') or {}))
236 if family == 'relation':
237 linked.update(relation_references(record))
238 # Explicit expert evidence has the same force here as a direct
239 # assessment citation; structural KN endpoints still bind content.
240 judgments = {item for item in relation_evidence_references(record)
241 if records.get(item, (None,))[0] in
242 {'formal-result', 'identity-assessment', 'assessment-use', 'goal-judgment'}}
243 pending.extend(judgments - strict)
244 strict.update(judgments)
245 for review in record.get('reviews', []):
246 linked.update(row['id'] for row in review.get('decision_basis', {}).get('records', []))
247 if family == 'identity-assessment':
248 linked.update(row['id'] for row in instance.get('assessment_uses', {}).values()
249 if row.get('assessment_id') == ref)
250 if family == 'knowledge' and ref in strict and record.get('formal_result_id'):
251 linked.update(row['id'] for row in instance.get('identity_assessments', {}).values()
252 if row.get('formal_result_id') == record['formal_result_id']
253 and row.get('exposition_id') == record.get('exposition_id'))
254 pending.extend(linked - refs)
255 if ref in strict:
256 pending.extend(linked - strict)
257 strict.update(linked)
258 refs.update(linked)
259 refs.discard(task['id'])
260 from .input_basis import aliases
261 result={}
262 for ref in sorted(refs):
263 found=records.get(ref)
264 if found is None:
265 source = next((source for source in instance.get('sources', [])
266 if ref in aliases(source)), None)
267 if source is not None:
268 found = ('source', source)
269 value=found[1] if found else next((r for r in task.get('results',[]) if ref in {r.get('id'),r.get('execution_id')}),None)
270 if found and found[0]=='task':value=task_reference_basis(value)
271 elif found and found[0] == 'knowledge':
272 value = {key: item for key, item in value.items()
273 if key not in {'review_cycle_id', 'previous_review_cycle_id'}}
274 elif found and found[0] == 'formal-result' and ref not in strict:
275 # A KN citation binds submitted mathematics, not a judgment that
276 # did not exist yet. Explicit FR citations still bind the full record.
277 value = {key: value[key] for key in ('id', 'family', 'lean_code', 'declaration_names',
278 'dependencies', 'producer_execution', 'origin_refs') if key in value}
279 result[ref]=exact_content_hash(value) if value is not None else None
280 return result

283def assessment_current(task,instance,runtime,*,evaluation_context=None,records=None):
284 assessment=task.get('assessment',{})
285 if (not assessment or assessment.get('current') is False or assessment.get('contract_sha256')!=contract_hash(task)
286 or assessment.get('results_sha256')!=exact_content_hash(task.get('results',[]))
287 or assessment.get('basis')!=assessment_basis(task,assessment,instance,runtime,records=records)):
288 return False
289 from .task_contracts import validate_item_dispositions

291 validate_item_dispositions(task, assessment, require_complete=assessment.get('decision') in {'completed', 'closed'})
292 except ValueError:
293 return False
294 for row in assessment.get('item_dispositions',[]):
295 if row['kind'] == 'fulfilled':
296 obligation = next(o for o in task['completion_obligations'] if o['id'] == row['obligation_id'])
297 actor = runtime.get('executions', {}).get(assessment.get('execution_id'), {})

303 try:validate_disposition_authority(task,row,instance,runtime,evaluation_context=evaluation_context)
304 except ValueError:return False

306 target=runtime.get('tasks',{}).get(row['target_task_id'],{})

308 expected={**original,'id':row['target_obligation_id']}
309 if expected not in target.get('completion_obligations',[]):return False
310 return True

337def designated_judgment(task, obligation, disposition, instance, runtime):
338 """Resolve existing specialist judgments; the planner still checks semantic coverage.
339
340 A judgment must concern a result of this task, be currently usable, and be
341 explicitly cited. A role name, peer cycle ID, or prose claim is not a receipt.
342 """
343 from .knowledge_records import formal_usable, usable_links
344 from .knowledge import validate_basis
345 produced = {row.get('execution_id') for row in task.get('results', [])} - {None}
346 records = [row for row in instance.get('knowledge', {}).values()
347 if row.get('task_id') == task['id'] or row.get('producer_execution') in produced]
348 judges = set(obligation['judges'])
349 cited = set(disposition['evidence_refs'])
350 for record in records:
351 fid = record.get('formal_result_id')
352 formal = instance.get('formal_results', {}).get(fid, {})
353 if ('formal_check' in judges and fid in cited and formal_usable(instance, formal)
354 and record.get('lifecycle') == 'active'):
355 return fid
356 if 'identity_review' in judges:
357 for link in usable_links(instance, record['id']):
358 if link['assessment_id'] in cited:
359 return link['assessment_id']
360 if 'goal_audit' in judges:
361 from .planning import governance_revisions
362 from .input_basis import sources_current
363 editions = governance_revisions(instance)
364 owned = {row['id'] for row in records}
365 for judgment in instance.get('goal_judgments', []):
366 if (judgment.get('id') in cited and judgment.get('decision') == 'satisfied'
367 and judgment.get('scope_complete') is True and judgment.get('current') is True
368 and all(judgment.get(key) == value for key, value in editions.items())
369 and sources_current(judgment.get('source_basis', []), instance)
370 and owned.intersection(judgment.get('dependencies', []))
371 and validate_basis(judgment.get('record_basis', []), instance, runtime)['valid']
372 and all(usable_links(instance, kid) for kid in judgment.get('dependencies', [])
373 if kid in instance.get('knowledge', {}))):
374 return judgment['id']
375 return None

453 target.pop('revision_requested', None)
454 for assessment in assessments:
