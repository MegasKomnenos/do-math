34def ready_work(root, shared, *, count, public, config):
35 """``restore_template`` build: ``count`` ready tasks whose work a real plan selected and queued.
36
37 By default: validated active state, a real preparation and sealed contract
38 approvals, since public activation and contract review have their own tests.
39 public=True keeps public activation and real reviews for tests whose outcome
40 depends on that state: check() of synthetic initialization executions
41 reports an unexamined role input, and activation leaves queue events.
42 The plan consumes the readiness events; work queued without it would leave
43 them pending, and the engine would schedule another plan for them. With
44 count=0 the queue is left as activation left it. Returns the instance ID.
45 """
46 from research_fixtures import FixtureAdapter, activated_instance
47 from state_fixtures import ACTIVE_LEAN_COMMAND, active_instance
48 base = {"operator_enabled": False, "scholarly_enabled": False}
49 if public:
50 instance_id = activated_instance(root, shared, goal="bounded research",
51 config={**base, "lean_command": ["test-lean"], **config})
52 else:
53 instance_id = active_instance(root, shared, config={**base, "lean_command": list(ACTIVE_LEAN_COMMAND), **config})
54 if not count:
55 return instance_id
56 engine = ResearchEngine(root / "repository", instance_id, runner=ScriptedRunner(
57 preparation_candidates=[candidate(str(i)) for i in range(count)],
58 model_rating={"difficulty": "high", "importance": "medium", "reason": "Independent novel fixture approaches"}))
59 with patch("do_math.initialization.adapter_for", return_value=FixtureAdapter()):
60 engine.run(max_steps=1)
61 if public:
62 engine.run(max_steps=count)
63 else:
64 # Seal the approval the review role returns; the engine judges readiness.
65 for task in engine.runtime["tasks"].values():
66 task["reviews"].append(seal_contract_review(task, contract_review({"task": task}),
67 engine.instance, engine.runtime, execution_id=new_id("EX")))
68 engine._refresh_readiness()
69 if len(engine.runtime["tasks"]) != count:
70 raise AssertionError(engine.runtime["tasks"])
71 if not all(row["state"] == "ready" for row in engine._task_readiness.values()):
72 raise AssertionError(engine._task_readiness)
73 # The sealed approvals answer the queued reviews; the plan then selects every ready task.
74 engine.runtime["queue"] = [a for a in engine.runtime["queue"] if a["role"] != "task_contract_review"]
75 engine._save()
76 engine.run(max_steps=1)
77 queued = sorted((a["role"], a["target"]) for a in engine.runtime["queue"])
78 if queued != sorted(("research_work", tid) for tid in engine.runtime["tasks"]):
79 raise AssertionError(queued)
80 if engine.runtime["pending_events"] != []:
81 raise AssertionError(engine.runtime["pending_events"])
82 return instance_id

85class ConcurrentRunner:
86 def __init__(self, expected=3, release=None):
87 self.expected = expected
88 self.release = release or threading.Event()
89 self.lock = threading.Lock()
90 self.active = 0
91 self.peak = 0
92 self.models = []
93 self.workspaces = []
94 self.started = threading.Condition(self.lock)
95 self.started_count = 0
96
97 @local_job_runner
98 def run(self, *, prompt, workspace, model, on_started, should_stop, **kwargs):
99 ctx = context_from(prompt, workspace)
100 with self.lock:
101 self.active += 1
102 self.peak = max(self.peak, self.active)
103 self.models.append(model)
104 self.workspaces.append(workspace)
105 if self.active == self.expected:
106 self.release.set()
107 try:
108 on_started()
109 with self.started:
110 self.started_count += 1
111 self.started.notify_all()
112 deadline = time.monotonic() + 120
113 while not self.release.wait(.01):
114 if should_stop():
115 return {"status": "stopped", "output": None, "usage": {}, "error": "stopped"}
116 if time.monotonic() > deadline:
117 raise AssertionError("independent model calls did not overlap")
118 (workspace / "same-name.txt").write_text(ctx["action"]["target"], encoding="utf-8")
119 out = common("research_work")
120 out.update(outcome="submitted", report="Completed the assigned bounded task.", checkpoint="done")
121 return producer_result({"status": "completed", "output": out, "usage": {}, "error": None}, {**kwargs, 'prompt': prompt, 'workspace': workspace, 'model': model, 'on_started': on_started, 'should_stop': should_stop})
122 finally:
123 with self.lock:
124 self.active -= 1
