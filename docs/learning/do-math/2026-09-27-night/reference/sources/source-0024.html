35def plan_reservations(
36 held: Iterable[Mapping[str, Any]],
37 constraints: Mapping[str, Mapping[str, Any]],
38 profile: Mapping[str, Any],
39 *,
40 owner_id: str,
41) -> tuple[dict[str, Any], ...]:
42 """The held records one execution needs on ``profile``, or ``BudgetConflict``.
43
44 ``profile`` reserves one ``accounting_unit`` of each AI ``capacity_refs``
45 constraint and each pinned host demand on its host constraint; a profile
46 with neither reserves nothing. ``held`` are the reservations still held,
47 each counted against the constraint ID it names; ``constraints`` maps IDs
48 to the adopted constraints. Nothing is mutated.
49 """
50 held = list(held)
51 demands = [(f"RS-{owner_id}-{index}", ref, constraints[ref]["accounting_unit"], 1)
52 for index, ref in enumerate(profile.get("ai", {}).get("capacity_refs", []))]
53 host = profile.get("host_resources")
54 if host is not None:
55 demands += [(f"RS-{owner_id}-host-{unit}", host["constraint_id"], unit, amount)
56 for unit, amount in host["demands"].items()]
57 for _, constraint_id, unit, amount in demands:
58 constraint = constraints[constraint_id]
59 used = outstanding_sum(held, constraint_id, unit)
60 if constraint["kind"] == "concurrency":
61 if used + amount > constraint["work_slots"]:
62 raise BudgetConflict(f"concurrency capacity unavailable: {constraint_id}",
63 constraint_id=constraint_id, constraint_kind="concurrency")
64 # The controller's reserve is never lent to an admitted child Job.
65 elif used + amount > constraint[_HOST_LIMITS[unit]] - constraint["control_reserve"][unit]:
66 raise BudgetConflict(f"host capacity unavailable: {constraint_id}/{unit}",
67 constraint_id=constraint_id, constraint_kind="host")
68 authority = {key: profile[key] for key in ("id", "revision", "sha256")}
69 return tuple({
70 "id": identifier,
71 "constraint_id": constraint_id,
72 "unit": unit,
73 "authority_refs": [dict(authority)],
74 "owner_kind": "execution",
75 "owner_id": owner_id,
76 "purpose": "normal",
77 "reserved": amount,
78 "confirmed_consumed": 0,
79 "returned": 0,
80 "state": "held",
81 "evidence_refs": [],
82 "reconcile_owner": "engine",
83 "reobserve": {"on": "local_termination_barrier"},
84 } for identifier, constraint_id, unit, amount in demands)

87def settle_reservation(
88 record: Mapping[str, Any],
89 *,
90 confirmed_consumed: int,
91 returned: int,
92 evidence_refs: Sequence[Any],
93) -> dict[str, Any]:
94 """Create the one complete actual settlement of a held reservation record.
95
96 A settlement must account for the whole reserved amount, once. Unknown
97 remote use remains ``held`` until a caller has actual evidence and invokes
98 this function.
99 """
100 if record["state"] != "held":
101 raise BudgetInputError("only a held reservation can be settled")
102 _nonnegative("confirmed_consumed", confirmed_consumed)
103 _nonnegative("returned", returned)
104 if confirmed_consumed + returned != record["reserved"]:
105 raise BudgetInputError("settlement must account for every reserved unit exactly once")
106 if not isinstance(evidence_refs, Sequence) or isinstance(evidence_refs, (str, bytes)) or not evidence_refs:
107 raise BudgetInputError("settlement requires actual evidence_refs")
108 settled = deepcopy(dict(record))
109 settled.update({
110 "confirmed_consumed": confirmed_consumed,
111 "returned": returned,
112 "state": "settled",
113 "evidence_refs": deepcopy(list(evidence_refs)),
114 })
115 settled.pop("reobserve", None)
116 return settled

119def validate_reservation(record: Mapping[str, Any]) -> None:
120 """The state gate's check of one stored reservation record."""
121 for name in ("id", "constraint_id", "unit", "state"):
122 if not isinstance(record.get(name), str) or not record[name]:
123 raise BudgetInputError(f"reservation lacks valid {name}")
124 if record["state"] not in {"held", "settled"}:
125 raise BudgetInputError("reservation has an invalid state")
126 for name in ("reserved", "confirmed_consumed", "returned"):
127 _nonnegative(name, record.get(name))
128 if record["confirmed_consumed"] + record["returned"] > record["reserved"]:
129 raise BudgetInputError("confirmed_consumed + returned exceeds reserved")
130 if record["state"] == "settled":
131 if record["confirmed_consumed"] + record["returned"] != record["reserved"]:
132 raise BudgetInputError("settled reservation must account for reserved exactly")
133 if not record.get("evidence_refs"):
134 raise BudgetInputError("settled reservation requires evidence_refs")
135 if record["state"] == "held" and "reobserve" not in record:
136 raise BudgetInputError("held reservation requires reobserve")

139def outstanding_sum(records: Iterable[Mapping[str, Any]], constraint_id: str, unit: str) -> int:
140 """Units of ``unit`` on ``constraint_id`` that ``records`` have not returned."""
141 return sum(record["reserved"] - record["returned"] for record in records
142 if record["constraint_id"] == constraint_id and record["unit"] == unit)
