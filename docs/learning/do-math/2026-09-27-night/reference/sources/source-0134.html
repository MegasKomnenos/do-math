67 def test_sibling_boundary_reads_the_exact_catalog_contract(self):
68 import hashlib
69 self.sibling(status='completed')
70 task = self.context['tasks'].pop()
71 raw = json.dumps(task).encode('utf-8')
72 root = Path(self.temp.name)
73 path = root / 'input' / 'details' / 'sibling.json'
74 path.parent.mkdir(parents=True)
75 path.write_bytes(raw)
76 self.context['snapshot'] = 'fixed-input'
77 self.context['task_catalog'] = [{'id': task['id'], 'title': task['title'],
78 'revision': task['revision'], 'status': task['status'], 'source_ref': {
79 'path': path.relative_to(root).as_posix(), 'snapshot': 'fixed-input',
80 'sha256': hashlib.sha256(raw).hexdigest(), 'bytes': len(raw)}}]
81 entry = {'task_id': task['id'], 'revision': task['revision'],
82 'boundary': 'k <= tau - 3', 'independent_of_next_revision': True}
83 self.assertEqual(self.issues(contract_check(sibling_boundaries=[entry])), '[]')
84 self.assertIn('revision', self.issues(contract_check(sibling_boundaries=[{**entry, 'revision': 1}])))
85 path.write_bytes(raw + b' ')
86 self.assertIn('sibling contract source', self.issues(contract_check(sibling_boundaries=[entry])))
