22class QueryFamilyNameTests(unittest.TestCase):
23 def test_query_family_names_are_the_condition_grammar_names(self) -> None:

25 if stored is not None:
26 with self.subTest(stored=stored):
27 # A kind definition's own "family" field moves to definition_family.
