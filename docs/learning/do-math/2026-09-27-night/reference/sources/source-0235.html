103 def test_submission_rejects_nonwithdraw_and_duplicate_requests(self):
104 """The sealed extension schema and the submission check own disposition shape and distinct targets."""
105 import tempfile
106 from do_math.contracts import schema_for
107 from do_math.repository import canonical_json
108 from do_math.submission import payload_issues
109 from research_fixtures import seal_response
110 from runner_fixtures import common
111 from do_math.knowledge import content_hash
112 # A disposition revises an existing task, so the fixed input supplies that task
113 # with its content hash, as the engine's task views do.
114 task = {**task_contract(), "id": "TK-AI", "status": "planned", "producer_execution": "EX-prep",
115 "results": [], "reviews": []}
116 context = {"action": {"role": "research_preparation"}, "budget_unit": "concurrent_sessions",
117 "tasks": [{**task, "content_sha256": content_hash(task)}], "knowledge_catalog": [],
118 "kind_definitions": {}, "sources": []}
119
120 def issues(dispositions):
121 with tempfile.TemporaryDirectory() as directory:
122 workspace, extensions = Path(directory), {"task_dispositions": dispositions}
123 (workspace / "activity-extensions.json").write_bytes(canonical_json(extensions))
124 output = common("research_preparation")
125 seal_response(output, context, workspace)
126 return payload_issues(output, schema_for("research_preparation"), context, extensions, workspace)
127
128 revise = issues([{"task_id": "TK-AI", "disposition": "revise", "reason": "x"}])
129 self.assertTrue(any(row["path"] == "activity-extensions.json/task_dispositions/0/disposition" for row in revise))
130 duplicate = issues([{"task_id": "TK-AI", "disposition": "withdraw", "reason": "x"},
131 {"task_id": "TK-AI", "disposition": "withdraw", "reason": "x"}])
132 self.assertTrue(any(row["path"] == "activity-extensions.json/task_dispositions"
133 and "distinct targets" in row["message"] for row in duplicate))

151class SourceCorrectionTest(unittest.TestCase):
152 def test_same_source_bytes_can_revise_a_misread_claim_and_suspend_dependents(self):
153 """A corrected reading changes knowledge, not the immutable source bytes."""
154 from types import SimpleNamespace
155 from do_math import knowledge,knowledge_io,knowledge_records as kr
156 engine=SimpleNamespace(instance={},runtime={'queue':[],'history':[]})
157 engine.instance["sources"] = [{"id": "SO-paper", "name": "paper.pdf", "path": "source.pdf",
158 "sha256": "a" * 64, "status": "active"}]
159 candidates = {
160 "KN-reading": {"id": "KN-reading", "revision": 1, "status": "verified", "statement": "The paper prints m/n.",
161 "explanation": "Earlier reading.", "summary": "Misread fraction.", "lean_code": "",
162 "declaration_names": [], "dependencies": [], "reviews": [],
163 "source_basis": [{"name": "paper.pdf", "path": "source.pdf", "sha256": "a" * 64}]},
164 "KN-dependent": {"id": "KN-dependent", "revision": 1, "status": "verified", "statement": "Derived claim.",
165 "explanation": "Depends on the reading.", "summary": "Dependent.", "lean_code": "",
166 "declaration_names": [], "dependencies": ["KN-reading"], "reviews": []},
167 }
168 ids={}
169 for name,row in candidates.items():
170 candidate={key:row[key] for key in ('statement','explanation','summary','lean_code','declaration_names','dependencies')}
171 candidate.update(kind='observation',title=name,dependencies=[ids.get(k,k) for k in row['dependencies']])
172 engine.instance,engine.runtime,ids[name]=knowledge_io.import_candidate(engine.instance,engine.runtime,candidate,source_ref='SO-paper',actor_ref='user')
173 changes={'statement':'The paper prints n/m.','explanation':'The original PDF is unchanged; prior transcription reversed it.'}
174 roots=knowledge.revision_roots(engine.instance,ids['KN-reading'])
175 impact=knowledge.plan_impact(engine.instance,engine.runtime,roots,[])
176 engine.instance,engine.runtime=knowledge.revise_knowledge(engine.instance,engine.runtime,ids['KN-reading'],changes,
177 reason='PDF layout audit reads n/m.',actor_ref='EX-correction',declared_impact=impact['computed_impact'])
178
179 self.assertEqual(engine.instance["sources"][0]["sha256"], "a" * 64)
180 old=engine.instance['knowledge'][ids['KN-reading']]
181 self.assertEqual(old['lifecycle'],'superseded')
182 self.assertEqual(kr.proposal_view(engine.instance,old['id'])['statement'],'The paper prints m/n.')
183 replacement=next(k for k,v in engine.instance['knowledge'].items() if v.get('supersedes')==old['id'])
184 self.assertEqual(kr.proposal_view(engine.instance,replacement)['statement'],'The paper prints n/m.')
185 self.assertFalse(knowledge.assess_use(engine.instance['knowledge'][ids['KN-dependent']],engine.instance)['valid'])
186 self.assertIn(ids['KN-dependent'],next(row for row in engine.runtime['history'] if row['kind']=='knowledge_reassessment_required')['affected_ids'])
