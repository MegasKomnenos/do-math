25def archived_run(root):
26 run = root / 'runs/EX-1'
27 run.mkdir(parents=True)
28 (run / 'output.txt').write_bytes(b'preserved result')
29 receipt = json.dumps({'execution_id': 'EX-1', 'local_terminated': True}).encode()
30 snapshot = 'a' * 40
31 archive = {'runtime/artifacts/EX-1/output.txt': b'preserved result',
32 'runtime/artifacts/EX-1/termination.json': receipt}
33 execution = {'id': 'EX-1', 'status': 'ended', 'termination': {
34 'local_terminated': True, 'evidence_ref': 'runtime/artifacts/EX-1/termination.json',
35 'snapshot': snapshot, 'sha256': hashlib.sha256(receipt).hexdigest(), 'bytes': len(receipt)},
36 'collection': {'retention_manifest': [{'path': 'output.txt', 'artifact_ref': 'runtime/artifacts/EX-1/output.txt',
37 'snapshot': snapshot, 'sha256': hashlib.sha256(b'preserved result').hexdigest(), 'bytes': 16}]}}
38 state = {'executions': {'EX-1': execution}, 'protected_references': set(),
39 'approved_execution_ids': {'EX-1'}, 'runtime': {}}
40 def read(path, commit):
41 if commit != snapshot:
42 raise ValueError('not an archived snapshot')
43 return archive[path]
44 return run, state, archive, read

115 def test_cleanup_hashes_local_files_and_archive_rows_once(self):
116 import uuid
117 from unittest.mock import Mock, patch
118 from do_math import file_io, retention
119 with tempfile.TemporaryDirectory() as directory:
120 root = Path(directory)
121 run, state, archive, read = archived_run(root)
122 reader, git_dir, marks = Mock(side_effect=read), 'git-' + uuid.uuid4().hex, {}
123 # A fresh file's birth time cannot be aged; treat any past fingerprint as settled.
124 with patch.object(file_io, 'RACY_NS', 0), \
125 patch.object(retention, 'read_stable', wraps=retention.read_stable) as local:
126 candidates = cleanup_candidates(root, **state, read_artifact=reader, git_dir=git_dir, marks=marks)
127 self.assertTrue(candidates[0]['eligible'], candidates)
128 self.assertEqual((local.call_count, reader.call_count), (1, 2))
129 result = cleanup(root, candidates, current_state=lambda: state, read_artifact=reader,
130 git_dir=git_dir, marks=marks)
131 self.assertEqual(result['deleted'], ['runs/EX-1'])
132 # The pre-deletion recheck used fingerprints and matched archive rows.
133 self.assertEqual(local.call_count, 1)
134 self.assertEqual(reader.call_count, 3) # only the termination receipt is parsed again

151 def test_cleanup_rechecks_reference_files_and_archive_before_deletion(self):
152 for mutation in ('reference', 'local', 'archive', 'termination', 'reservation', 'recovery'):
153 with self.subTest(mutation=mutation), tempfile.TemporaryDirectory() as d:
154 root = Path(d); run, state, archive, reader = archived_run(root)
155 candidates = cleanup_candidates(root, **state, read_artifact=reader)
156 self.assertTrue(candidates[0]['eligible'])
157 if mutation == 'reference': state['protected_references'].add('runs/EX-1/output.txt')
158 elif mutation == 'local': (run / 'new.txt').write_bytes(b'not archived')
159 elif mutation == 'archive': archive['runtime/artifacts/EX-1/output.txt'] = b'corrupt'
160 elif mutation == 'termination': archive['runtime/artifacts/EX-1/termination.json'] = b'{}'
161 elif mutation == 'reservation': state['runtime']['reservations'] = {'RS': {'owner_id': 'EX-1', 'state': 'held'}}
162 else: state['runtime']['recoveries'] = {'RY-1': {'id': 'RY-1',
163 'started_at': '2026-09-25T00:00:00Z', 'execution_ids': ['EX-1'], 'dispositions': []}}
164 original = deepcopy(state)
165 result = cleanup(root, candidates, current_state=lambda: state, read_artifact=reader)
166 self.assertEqual(result['status'], 'degraded')
167 self.assertTrue(run.exists())
168 self.assertEqual(state, original)
