21def routed_requests(calls):
22 """(provider, worker request) for each ``(model, claude_only_assigned_model)`` run of one runner."""
23 from provider_fixtures import Adapter
24 runner = model_runner.ModelRunner()
25 adapters = {provider: Adapter({"result": "output_collected", "final": {"raw": b"{}"}, "jsonl": {}}, provider)
26 for provider in runner.adapters}
27 runner.adapters.update(adapters)
28 requests = []
29 for model, lock in calls:
30 with tempfile.TemporaryDirectory() as directory:
31 root = Path(directory); workspace = root / "workspace"; workspace.mkdir()
32 for adapter in adapters.values():
33 adapter.__dict__.pop("request", None)
34 runner.run(model=model, prompt="fixture", output_schema={}, workspace=workspace,
35 evidence_dir=root / "evidence", effort="low", timeout_seconds=1,
36 claude_only_assigned_model=lock)
37 requests.extend((provider, adapter.request) for provider, adapter in adapters.items()
38 if "request" in adapter.__dict__)
39 return requests

214 def test_mixed_models_use_existing_selection_and_per_model_work_slots(self):
215 config = {**DEFAULT_CONFIG, "tier_models": {"high": "claude:opus", "medium": "gpt-5.6-sol", "low": "claude:haiku"},
216 "session_limits": {"work": {"claude:opus": 1, "gpt-5.6-sol": 1, "claude:haiku": 1}}}
217 validate_config(config)
218 self.assertEqual(selection(config, {"role": "research_plan"}, {}, {})["model"], "claude:opus")
219 self.assertEqual(selection(config, {"role": "task_contract_review"}, {}, {})["model"], "gpt-5.6-sol")
220 from do_math.execution_profiles import build_profiles
221 state = {**build_profiles(config), "config": config}
222 runtime = {"executions": {}, "reservations": {}}
223 with tempfile.TemporaryDirectory() as d:
224 slots = SessionSlots( runtime=runtime, instance=state)
225 def acquire(eid, role, model):
226 profile = next(p for p in state["execution_profiles"].values()
227 if p["role"] == role and p.get("ai", {}).get("model") == model)
228 return slots.acquire(eid, profile=profile,
229 execution={"id": eid, "role": role, "status": "queued"})
230 self.assertTrue(acquire("one", "research_work", "claude:opus"))
231 self.assertFalse(acquire("two", "research_work", "claude:opus"))
232 # An operational role reserves no work slot of its model.
233 self.assertTrue(acquire("three", "research_plan", "claude:opus"))
234 self.assertTrue(acquire("four", "research_work", "gpt-5.6-sol"))
235 self.assertTrue(slots.release("one"))
236 self.assertTrue(acquire("two", "research_work", "claude:opus"))
