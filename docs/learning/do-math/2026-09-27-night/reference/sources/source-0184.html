17 def test_default_bootstrap_reports_exact_missing_settings_before_activation(self):
18 with tempfile.TemporaryDirectory() as root:

20 instance.update(build_profiles(instance['config']))

28 self.assertNotIn('intent_interpretation', {e['role'] for e in errors})

30 # Initial-only adoption must not hide the role profiles dispatch uses later.

32 result = activation_requirements(instance, runtime)
33 self.assertIn('observation_configuration', result['missing'])
34 self.assertIn('research_work', {e['role'] for e in result['observation_errors']})

95 from do_math.engine import ResearchEngine, control

97 with tempfile.TemporaryDirectory() as directory:

99 repository = root / 'repo'
100 instance_id = new_instance(repository, goal='Preserve this original', workspace=root / 'work',

102 engine = ResearchEngine(repository, instance_id)
103 old = deepcopy(engine.instance['role_profile_refs']['research_work'][0])
104 request = control(repository, engine.instance['id'], 'decision', config={'change': {
105 'kind': 'initialization', 'adopt_current_config': True, 'profile_refs': [old]}})
106 engine.process_controls()
107 rejected = next(h for h in engine.runtime['history'] if h.get('kind') == 'rejected_user_control'
108 and h['request']['id'] == request['request_id'])
109 self.assertIn('tool_review_window_seconds', rejected['reason'])
110 self.assertIn('self_check_response_window_seconds', rejected['reason'])
111 self.assertEqual(engine.instance['lifecycle']['state'], 'initializing')
112 control(repository, engine.instance['id'], 'configure', config={
113 'tool_review_window_seconds': 60, 'self_check_response_window_seconds': 5})
114 engine.process_controls()
115 fresh = deepcopy(engine.instance['role_profile_refs']['research_work'][0])
116 self.assertNotEqual(fresh, old)
117 request = control(repository, engine.instance['id'], 'decision', config={'change': {
118 'kind': 'initialization', 'adopt_current_config': True, 'profile_refs': [fresh]}})
119 engine.process_controls()

121 self.assertIn(old['id'], engine.instance['execution_profiles'])
122 self.assertEqual(engine.instance['lifecycle']['immutable_input']['text'], 'Preserve this original')
