1"""Prepare the machine-wide Lean toolchain and record it for the research instances.
2
3Installation is idempotent: finished steps are skipped. A toolchain record is written
4only after `import Mathlib` compiles through the exact command the verifier will use,
5and the ready marker follows the record, so a half-downloaded cache is never selected.
6Record writers (installation, --verify-only, --rerecord, --check-contents) exclude each
7other with the lock file of the records directory. Exit codes: 0 success, 1 a refused,
8failed or mismatching result, 2 an execution error. Every mode prints one JSON object on stdout;
9installation progress and Lake output go to stderr.
10"""

40 """A record operation that must not proceed; exit code 1 with the stated result."""

55def verify_environment(command, evidence_root, identity):
56 """Compile the representative examples through the exact command; nothing is adopted."""
57 directory = evidence_root.resolve() / (datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%SZ') + '-' + uuid.uuid4().hex)
58 if any(directory.is_relative_to(root.resolve()) for root in records.roots(identity).values()):
59 raise ValueError('validation evidence must be outside the recorded toolchain and project roots')
60 directory.mkdir(parents=True)
61 source = directory / 'smoke.lean'
62 source_text, scope = records.validation_program(identity)
63 source.write_bytes(source_text.encode('utf-8'))
64 validation = {'status': 'failed', 'scope': scope, 'source': str(source), 'source_text': source_text,
65 'source_sha256': hashlib.sha256(source_text.encode('utf-8')).hexdigest(),
66 'command': [*command, str(source)], 'exit_code': None, 'stdout': '', 'stderr': '',
67 'started_at': now()}
68 try:
69 from do_math.formal import _job_run
70 from do_math.operation_trace import tracing, phase
71 with tracing(directory / 'operations'), phase('environment_smoke_compile') as metrics:
72 code, stdout, stderr, stopped = _job_run([*command, str(source)], directory,
73 on_started=lambda process: metrics.update(process=process))
74 validation.update(exit_code=code, stdout=stdout, stderr=stderr)
75 if not stopped and code == 0:
76 validation['status'] = 'verified'
77 except Exception as error:
78 validation['error'] = f'{type(error).__name__}: {error}'
79 validation['finished_at'] = now()
80 (directory / 'validation.json').write_text(json.dumps(validation, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
81 return validation

84def _baseline(command, evidence_root, identity, fingerprint, content):
85 validation = verify_environment(command, evidence_root, identity)
86 if validation['status'] != 'verified':
87 raise Refused({'status': 'failed', 'reason': 'the representative compile did not succeed', 'validation': validation})
88 if records.observe_fingerprint(identity)['sha256'] != fingerprint['sha256']:
89 raise Refused({'status': 'failed', 'reason': 'judgment files changed while the baseline was being recorded'})
90 return records.new_baseline(fingerprint, content, validation)

97def _settled_fingerprint(identity, attempts=3):
98 """Observe the judgment files once every fingerprint is older than the racy window.
99
100 A baseline is compared by metadata later, so it must not record a time that a
101 same-size rewrite could still share.
102 """
103 for _ in range(attempts):
104 observed = records.observe_fingerprint(identity)
105 if all(fingerprint_settled((size, int(modified), int(born)))
106 for size, modified, born in observed['files'].values()):
107 return observed
108 time.sleep(RACY_NS / 1e9)
109 raise RuntimeError('judgment files are still changing; record the installation once it is idle')

112def record_installation(command, records_dir, evidence_root):
113 """Create the record of an installed identity or confirm an unchanged one; never append a baseline."""
114 identity = observe_identity(command)
115 with records.writer_lock(records_dir):
116 fingerprint = _settled_fingerprint(identity)
117 for path, record in records.records_for(records_dir, identity):
118 if record['baselines'][-1]['fingerprint']['sha256'] == fingerprint['sha256'] and not records.usable_errors(record):
119 return _summary(path, record, status='recorded', created=False)
120 content = records.content_hashes(identity, fingerprint['files'])
121 path = records.record_path(records_dir, records.record_id(identity, content['sha256']))
122 if path.exists():
123 raise Refused({'status': 'rerecord_required', 'record_path': str(path),
124 'reason': 'the installed bytes match this record but its metadata or check history changed; '
125 'a user or operator decides whether to run --rerecord'})
126 record = records.new_record(identity, _baseline(command, evidence_root, identity, fingerprint, content))
127 records.write_record(path, record)
128 return _summary(path, record, status='recorded', created=True)

131def resolve_record(records_dir, record_id, identity):
132 """The named record, or the only record of the identity (a callable observes it on demand)."""
133 if record_id is not None:
134 path = records.record_path(records_dir, record_id)
135 return path, records.read_record(path)
136 candidates = records.records_for(records_dir, identity())
137 if len(candidates) != 1:
138 raise ValueError('name the record with --record; records of the current identity: '
139 + (', '.join(record['id'] for _, record in candidates) or 'none'))
140 return candidates[0]

143def rerecord(command, records_dir, evidence_root, record_id):
144 """Append a baseline after metadata-only changes. Users and operators only (D5)."""
145 identity = observe_identity(command)
146 with records.writer_lock(records_dir):
147 path, record = resolve_record(records_dir, record_id, lambda: identity)
148 if record['identity'] != identity:
149 raise Refused({'status': 'refused', 'record_path': str(path),
150 'reason': 'the installed identity differs from the record; create a new record and register and transition to it'})
151 fingerprint = _settled_fingerprint(identity)
152 content = records.content_hashes(identity, fingerprint['files'])
153 mismatched = records.changed_files(record['baselines'][-1]['content']['files'], content['files'])
154 if mismatched:
155 raise Refused({'status': 'refused', 'record_path': str(path), 'mismatched': mismatched,
156 'reason': 'judgment file contents differ from the current baseline; create a new record and '
157 'register and transition to it'})
158 record['baselines'].append(_baseline(command, evidence_root, identity, fingerprint, content))
159 records.write_record(path, record)
160 return _summary(path, record, status='recorded')

163def check_contents(records_dir, record_id, command):
164 """Hash every judgment file and append the comparison with the current baseline."""
165 with records.writer_lock(records_dir):
166 path, record = resolve_record(records_dir, record_id, lambda: observe_identity(command))
167 current = len(record['baselines']) - 1
168 fingerprint = records.observe_fingerprint(record['identity'])
169 content = records.content_hashes(record['identity'], fingerprint['files'])
170 mismatched = records.changed_files(record['baselines'][current]['content']['files'], content['files'])
171 check = {'checked_at': now(), 'baseline': current,
172 'result': 'mismatch' if mismatched else 'passed', 'mismatched': mismatched}
173 record['checks'].append(check)
174 records.write_record(path, record)
175 return _summary(path, record, status=check['result'], check=check)

178def write_trust_request(result, destination):
179 """A registration request binding the record's identity and one baseline."""
180 record = records.read_record(result['record_path'])
181 request = {'operation': 'register', 'toolchain': record['identity'], 'allowed_axioms': sorted(AXIOMS),
182 'toolchain_record': {'path': str(Path(result['record_path']).resolve()), 'id': record['id']},
183 'binding': result['binding']}
184 destination = destination.resolve()
185 destination.parent.mkdir(parents=True, exist_ok=True)
186 with destination.open('x', encoding='utf-8') as stream:
187 json.dump(request, stream, ensure_ascii=False, indent=2)
188 return {'path': str(destination), 'toolchain_record': request['toolchain_record'], 'binding': request['binding']}

197def install(root):
198 project = mathlib_project(root)
199 bin_dir = portable_bin(root)
200 if not (bin_dir / "lean.exe").is_file():
201 archive = bin_dir.parents[1] / f"lean-{LEAN_VERSION}-windows.zip"
202 archive.parent.mkdir(parents=True, exist_ok=True)
203 print("downloading", LEAN_ZIP, file=sys.stderr, flush=True)
204 urllib.request.urlretrieve(LEAN_ZIP, archive)
205 with zipfile.ZipFile(archive) as bundle:
206 bundle.extractall(archive.parent)
207 project.mkdir(parents=True, exist_ok=True)
208 for name, text in PROJECT_FILES.items():
209 if not (project / name).is_file() or (project / name).read_text(encoding="utf-8") != text:
210 (project / name).write_text(text, encoding="utf-8")
211 lake = str(bin_dir / "lake.exe")
212 # Mathlib's cache tool calls lean and leantar by name; there is no elan here to resolve them.
213 env = {**os.environ, "PATH": str(bin_dir) + os.pathsep + os.environ.get("PATH", "")}
214 if not (project / "lake-manifest.json").is_file():
215 run([lake, "update"], project, env)
216 run([lake, "exe", "cache", "get"], project, env)
217 return [lake, "--dir", str(project), "env", str(bin_dir / "lean.exe")]

220def main():
221 parser = argparse.ArgumentParser(description=__doc__)
222 mode = parser.add_mutually_exclusive_group()
223 mode.add_argument('--inspect-only', action='store_true', help='observe an installed command without installing or recording it')
224 mode.add_argument('--verify-only', action='store_true',
225 help='record an installed environment without downloading: identity, fingerprints, content hashes and a representative compile')
226 mode.add_argument('--rerecord', action='store_true',
227 help='users and operators only: append a baseline when only metadata changed; refused if any content differs')
228 mode.add_argument('--check-contents', action='store_true',
229 help='hash every judgment file, compare with the current baseline and append the result to the record')
230 parser.add_argument('--command-file', type=Path, help='UTF-8 JSON object with lean_command argv; not for installation')
231 parser.add_argument('--record', help='toolchain record ID (TC-...); default: the only record of the current identity')
232 parser.add_argument('--records-dir', type=Path, help='toolchain record directory; default: <toolchain root>/records')
233 parser.add_argument('--evidence-dir', type=Path, help='parent directory for representative compile evidence')
234 parser.add_argument('--trust-file', type=Path, help='write a new registration request for the resulting record baseline')
235 args = parser.parse_args()
236 installing = not (args.inspect_only or args.verify_only or args.rerecord or args.check_contents)
237 if args.command_file and installing:
238 parser.error('--command-file requires --inspect-only, --verify-only, --rerecord or --check-contents')
239 if args.inspect_only and (args.evidence_dir or args.trust_file or args.record or args.records_dir):
240 parser.error('--inspect-only neither records nor validates an environment')
241 if args.check_contents and (args.evidence_dir or args.trust_file):
242 parser.error('--check-contents runs no compile and writes no registration request')
243 if args.record and not (args.rerecord or args.check_contents):
244 parser.error('--record names the record for --rerecord or --check-contents')
245 root = toolchain_root()
246 try:
247 if installing:
248 if root is None:
249 raise ValueError("this installation has no repository root above it; set lean_command explicitly instead")
250 (mathlib_project(root) / READY_MARKER).unlink(missing_ok=True)
251 command = install(root)
252 else:
253 from do_math.submission import load_json
254 command = load_json(args.command_file.read_bytes())['lean_command'] if args.command_file else discover_lean()
255 if not command:
256 parser.error('Lean was not found; supply --command-file')
257 if args.inspect_only:
258 with tempfile.TemporaryDirectory() as directory:
259 print(json.dumps({'status': 'observed', 'lean_command': command,
260 'toolchain': inspect_toolchain(command, Path(directory))}))
261 return
262 if root is None and (args.records_dir is None or args.evidence_dir is None and not args.check_contents):
263 parser.error('this installation has no toolchain root; supply --records-dir and --evidence-dir')
264 records_dir = args.records_dir or root / 'records'
265 evidence_root = args.evidence_dir or (root / 'validation' if root else None)
266 if args.check_contents:
267 result = check_contents(records_dir, args.record, command)
268 elif args.rerecord:
269 result = rerecord(command, records_dir, evidence_root, args.record)
270 else:
271 result = record_installation(command, records_dir, evidence_root)
272 result['lean_command'] = command
273 if args.trust_file:
274 result['trust_request'] = write_trust_request(result, args.trust_file)
275 if installing:
276 (mathlib_project(root) / READY_MARKER).write_text(LEAN_VERSION, encoding="utf-8")
277 except Refused as refused:
278 print(json.dumps(refused.result, ensure_ascii=False))
279 raise SystemExit(1)
280 except (OSError, RuntimeError, ValueError, KeyError, subprocess.CalledProcessError) as error:
281 print(json.dumps({'status': 'error', 'error': f'{type(error).__name__}: {error}'}, ensure_ascii=False))
282 raise SystemExit(2)
283 print(json.dumps(result, ensure_ascii=False))
284 if result['status'] == 'mismatch':
285 raise SystemExit(1)

289 main()
