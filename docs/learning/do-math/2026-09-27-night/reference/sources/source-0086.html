22# The record fields whose values name records, artifacts or paths that a
23# cleanup must keep. This one vocabulary decides every reference a record holds.
24PROTECTED_REFERENCE_KEYS = frozenset({
25 "references", "dependencies", "source_refs", "origin_refs", "input_refs", "inputs", "reuse",
26 "artifact_ref", "path", "checkpoint_ref", "source_ref", "formal_result_id", "exposition_id",
27 "assessment_id", "evidence_refs", "authority_refs", "basis_refs", "record_refs", "result_ref",
28 "formal_result_refs", "assessment_refs"})
29# The fields of a referring object that name its target.
30_TARGET_FIELDS = ("id", "ref", "artifact_ref", "path")
31# Issued semantic references that name no materialized run file.
32_SEMANTIC_ALIASES = CONTEXT_INPUT_REFS | TOOL_INPUT_REFS

41def _run_path(workspace, execution_id):
42 if not isinstance(execution_id, str) or not execution_id or any(c in execution_id for c in '/\\\\:') or execution_id in {'.', '..'}:
43 raise ValueError('invalid execution path segment')
44 root = _physical(workspace)
45 path = _physical(root / 'runs' / execution_id)
46 if path.parent != root / 'runs':
47 raise ValueError('execution path outside workspace/runs')
48 return path

51def _strings(value):
52 if isinstance(value, str):
53 yield value
54 elif isinstance(value, dict):
55 for item in value.values():
56 yield from _strings(item)
57 elif isinstance(value, (list, tuple, set)):
58 for item in value:
59 yield from _strings(item)

62def _protected_paths(workspace, references):
63 """The physical paths of protected references, resolved once per cleanup check.
64
65 An alias can refer into a run even when its lexical spelling does not.
66 Uncertain/reparse reference paths are refused instead of being followed
67 and treating an external target as a new trusted root.
68 """
69 paths = []
70 for reference in _strings(references):
71 candidate = Path(reference.replace('\\\\', '/'))
72 candidate = candidate if candidate.is_absolute() else Path(workspace) / candidate
73 paths.append(_physical(candidate))
74 return paths

77def record_references(value):
78 """The reference strings held under PROTECTED_REFERENCE_KEYS anywhere in ``value``.
79
80 Under such a key a string is a reference, a list contributes its strings
81 and the target fields of its objects, and an object contributes its target
82 fields. Nested values are searched the same way. Task ``reuse`` lists record
83 IDs; grant ``reuse`` is a policy word and names nothing.
84 """
85 result = []
86 if isinstance(value, dict):
87 for key, child in value.items():
88 if key in PROTECTED_REFERENCE_KEYS and not (key == "reuse" and isinstance(child, str)):
89 if isinstance(child, str):
90 result.append(child)
91 elif isinstance(child, list):
92 for item in child:
93 if isinstance(item, str):
94 result.append(item)
95 elif isinstance(item, dict):
96 result.extend(item[name] for name in _TARGET_FIELDS if isinstance(item.get(name), str))
97 elif isinstance(child, dict):
98 result.extend(child[name] for name in _TARGET_FIELDS if isinstance(child.get(name), str))
99 if isinstance(child, (dict, list)):
100 result.extend(record_references(child))
101 elif isinstance(value, list):
102 for child in value:
103 result.extend(record_references(child))
104 return result

107def retention_inputs(instance, runtime, rows):
108 """The reference records and protected roots of one loaded snapshot.
109
110 ``rows`` are the snapshot's persisted record rows (``load_records``). Each
111 record maps to the references it holds; relations add their authored edges
112 (``relation_refs``). Research records, tasks, question links and review
113 cycles are protected roots, with every reference a waiting action holds.
114 """
115 records = {}
116 for row in rows:
117 if row.get("id"):
118 references = set(record_references(row["record"]))
119 if row["family"] == "relations":
120 references |= relation_references(row["record"])
121 records[row["id"]] = {"references": sorted(references)}
122 for source in instance.get("sources", []):
123 if source.get("id") in records:
124 for alias in (source.get("name"), "source:" + source.get("name", "")):
125 if alias:
126 records.setdefault(alias, {"references": [source["id"]]})
127 for alias in _SEMANTIC_ALIASES:
128 records.setdefault(alias, {"references": []})
129 protected = [row["id"] for row in rows if row.get("id") and
130 (row["root"] == "research" or row["family"] in {"tasks", "question_links", "review_cycles"})]
131 protected.extend(record_references(runtime.get("waiting_actions", {})))
132 return records, protected

135def reference_closure(references, records):
136 """Resolve record IDs to their explicit source/reference edges, cycle-safely.
137
138 Unknown identifiers are returned, never treated as an empty dependency set.
139 Paths are leaves; each record lists its edges under ``references``
140 (``retention_inputs``).
141 """
142 pending = list(_strings(references)); seen, paths, unresolved = set(), set(), set()
143 while pending:
144 reference = pending.pop()
145 if reference in seen:
146 continue
147 seen.add(reference)
148 if reference in records:
149 pending.extend(_strings(records[reference].get('references', [])))
150 elif '/' in reference or '\\' in reference or ':' in reference:
151 paths.add(reference)
152 else:
153 unresolved.add(reference)
154 return paths, unresolved

157def _inventory(path, marks=None, contents=None):
158 """Hash every local file once per cleanup.
159
160 ``marks`` maps a file to (settled fingerprint, proof) from an earlier read in
161 this cleanup; an unchanged fingerprint reuses the proof without reading.
162 ``contents`` receives the bytes that were read.
163 """
164 result = {}
165 if not path.is_dir():
166 raise ValueError('execution directory missing')
167 boundary = PhysicalBoundary(path)
168 def unreadable(error):
169 raise error
170 for base, dirs, files in os.walk(path, followlinks=False, onerror=unreadable):
171 for name in dirs + files:
172 _physical(Path(base) / name)
173 for name in files:
174 file = Path(base) / name
175 relative = file.relative_to(path).as_posix()
176 mark = file_fingerprint(file)
177 known = marks.get(file) if marks is not None else None
178 if known is not None and known[0] == mark:
179 result[relative] = dict(known[1])
180 continue
181 data = read_stable(file, boundary=boundary)
182 result[relative] = {'sha256': hashlib.sha256(data).hexdigest(), 'bytes': len(data)}
183 if contents is not None:
184 contents[relative] = data
185 if marks is not None and fingerprint_settled(mark) and file_fingerprint(file) == mark:
186 marks[file] = (mark, dict(result[relative]))
187 return result

190def archive_run(workspace, identifier, execution, prefix, marks=None):
191 """The archive of one locally terminated run below ``prefix``.
192
193 Returns the artifact bytes to publish, the retention manifest rows and the
194 termination evidence row. The archive holds exactly the bytes the inventory
195 hashed, so the caller may record every row as matched once it is published.
196 """
197 contents = {}
198 inventory = _inventory(_run_path(workspace, identifier), marks, contents)
199 artifacts, manifest = {}, []
200 for relative, proof in inventory.items():
201 ref = prefix + '/files/' + relative
202 artifacts[ref] = contents[relative]
203 manifest.append({'path': relative, 'artifact_ref': ref, **proof})
204 receipt = canonical_json({'execution_id': identifier, 'local_terminated': True,
205 'basis': deepcopy(execution['capacity']), 'remote_termination': 'not_established'})
206 ref = prefix + '/termination.json'
207 artifacts[ref] = receipt
208 return artifacts, manifest, {'local_terminated': True, 'evidence_ref': ref,
209 'sha256': hashlib.sha256(receipt).hexdigest(), 'bytes': len(receipt)}

219def cleanup_candidates(workspace, executions, *, protected_references,
220 approved_execution_ids=None, read_artifact=None, runtime=None, reference_records=None,
221 git_dir=None, marks=None):
222 """Verify every file against collection.retention_manifest rows.
223
224 Each row: path (relative to run), artifact_ref, snapshot, sha256, bytes.
225 read_artifact(path, snapshot) reads the archive, never the local run copy.
226 An archive row already matched in this process (repository.VERIFIED_ARTIFACTS
227 of ``git_dir``) is not read again; ``marks`` is passed to ``_inventory``.
228 """
229 candidates = []
230 references, unresolved = reference_closure(protected_references, reference_records or {})
231 # One unresolvable protected path makes every run ineligible.
232 try:
233 protected_paths, protected_error = _protected_paths(workspace, references), None
234 except (ValueError, OSError) as error:
235 protected_paths, protected_error = [], str(error)
236 for identifier, execution in sorted(executions.items()):
237 reasons, inventory = ['unresolved_reference: ' + ref for ref in sorted(unresolved)], {}
238 try:
239 path = _run_path(workspace, identifier)
240 if protected_error is not None:
241 raise ValueError(protected_error)
242 if any(candidate == path or candidate.is_relative_to(path) or path.is_relative_to(candidate)
243 for candidate in protected_paths):
244 reasons.append('protected_reference')
245 inventory = _inventory(path, marks)
246 except (ValueError, OSError) as error:
247 reasons.append(str(error))
248 if not isinstance(execution, dict) or execution.get('id') != identifier:
249 reasons.append('invalid_execution_record')
250 execution = {}
251 if execution.get('status') != 'ended':
252 reasons.append('execution_not_ended')
253 termination = execution.get('termination', {})
254 if not isinstance(termination, dict):
255 termination = {}
256 if termination.get('local_terminated') is not True or not termination.get('evidence_ref'):
257 reasons.append('termination_unconfirmed')
258 else:
259 try:
260 _archive_reference(termination['evidence_ref'], termination['snapshot'])
261 receipt = sealed_bytes(git_dir, termination['evidence_ref'], termination['sha256'], termination['bytes'],
262 lambda: read_artifact(termination['evidence_ref'], termination['snapshot']))
263 fact = json.loads(receipt)
264 if not isinstance(fact, dict) or fact.get('execution_id') != identifier or fact.get('local_terminated') is not True:
265 raise ValueError('termination receipt does not confirm this execution')
266 except (TypeError, ValueError, KeyError, OSError, RuntimeError) as error:
267 reasons.append('termination_unverified: ' + str(error))
268 # A recovery disposition lives only until the execution ends (execution_lifecycle.transition_execution),
269 # so an ended execution holds no recovery marker; pending recovery is read from the runtime below.
270 if runtime is None:
271 reasons.append('current_recovery_and_reservations_unexamined')
272 else:
273 for reservation in runtime.get('reservations', {}).values():
274 if reservation.get('owner_id') == identifier and reservation.get('state') != 'settled':
275 reasons.append('unsettled_reservation')
276 for waiting in runtime.get('waiting_actions', {}).values():
277 if identifier in set(_strings(waiting)):
278 reasons.append('recovery_pending')
279 for recovery in runtime.get('recoveries', {}).values():
280 if identifier in recovery.get('execution_ids', []) and not recovery.get('finished_at'):
281 reasons.append('recovery_pending')
282 for activity in runtime.get('activities', {}).values():
283 if identifier in activity.get('assigned_execution_ids', []) and (activity.get('status') != 'finished' or activity.get('unresolved')):
284 reasons.append('activity_obligation_pending')
285 if approved_execution_ids is None or identifier not in approved_execution_ids:
286 reasons.append('outside_approved_scope')
287 collection = execution.get('collection')
288 manifest = collection.get('retention_manifest') if isinstance(collection, dict) else None
289 if not isinstance(manifest, list) or read_artifact is None:
290 reasons.append('artifacts_not_verified')
291 else:
292 archived = {}
293 for row in manifest:
294 try:
295 local = row['path']; archive = row['artifact_ref']; snapshot = row['snapshot']
296 _archive_reference(archive, snapshot)
297 if local in archived:
298 raise ValueError('invalid archive reference')
299 proof = {'sha256': row['sha256'], 'bytes': row['bytes']}
300 sealed_bytes(git_dir, archive, proof['sha256'], proof['bytes'],
301 lambda: read_artifact(archive, snapshot), needed=False)
302 archived[local] = proof
303 except (ValueError, TypeError, KeyError, OSError, RuntimeError) as error:
304 reasons.append('archive_unverified: ' + str(error))
305 if inventory != archived:
306 reasons.append('archive_does_not_cover_current_files')
307 candidates.append({'execution_id': identifier, 'path': f'runs/{identifier}',
308 'eligible': not reasons, 'reasons': sorted(set(reasons)), 'inventory': inventory})
309 return candidates

312def cleanup(workspace, candidates, *, current_state=None, read_artifact=None, git_dir=None, marks=None):
313 """Re-read eligibility immediately before removal; never mutate research state.
314
315 current_state() supplies executions, protected_references,
316 approved_execution_ids, runtime under the caller's writer lock. With the
317 candidates' ``marks``, local files are re-examined by fingerprint.
318 """
319 deleted, failures = [], []
320 for candidate in candidates:
321 if not candidate.get('eligible'):
322 continue
323 try:
324 if current_state is None:
325 raise ValueError('current cleanup prerequisites are required')
326 state = current_state()
327 identifier = candidate['execution_id']
328 execution = state['executions'].get(identifier)
329 if execution is None:
330 raise ValueError('execution disappeared from current state')
331 refreshed = cleanup_candidates(workspace, {identifier: execution},
332 protected_references=state['protected_references'], approved_execution_ids=state['approved_execution_ids'],
333 read_artifact=read_artifact, runtime=state['runtime'], reference_records=state.get('reference_records'),
334 git_dir=git_dir, marks=marks)[0]
335 if not refreshed['eligible'] or refreshed['inventory'] != candidate['inventory']:
336 raise ValueError('cleanup prerequisites changed: ' + ', '.join(refreshed['reasons']))
337 shutil.rmtree(_run_path(workspace, identifier))
338 deleted.append(f'runs/{identifier}')
339 except (OSError, ValueError, KeyError, TypeError, RuntimeError) as error:
340 failures.append({'path': candidate.get('path'), 'error': str(error),
341 'status': 'degraded', 'next_owner': 'maintenance'})
342 return {'deleted': deleted, 'failures': failures, 'status': 'degraded' if failures else 'healthy'}

345def record_cleanup_interruption(runtime, execution_ids, execution_id, reason, *, history_start):
346 """Name unfinished paths without undoing deletions recorded in this attempt."""
347 completed = {path for event in runtime['history'][history_start:]
348 if event.get('kind') == 'maintenance_cleanup' and event.get('execution_id') == execution_id
349 for path in event.get('result', {}).get('deleted', [])}
350 runtime['history'].append({'kind': 'maintenance_cleanup_failed', 'execution_id': execution_id,
351 'paths': ['runs/' + identifier for identifier in execution_ids if 'runs/' + identifier not in completed],
352 'completed_paths': sorted(completed), 'reason': reason, 'next_owner': 'maintenance'})

355def pending_cleanup_failures(runtime):
356 """Derive unresolved path failures from the actual maintenance history.
357
358 An unrelated healthy attempt does not clear an older failure. Only deletion
359 of that exact path establishes cleanup completion; research effect is not
360 part of this operational projection.
361 """
362 pending = {}
363 for index, event in enumerate(runtime.get('history', [])):
364 if event.get('kind') == 'maintenance_cleanup_failed':
365 reference = 'runtime/history/' + str(index)
366 for path in event['paths']:
367 pending[path] = {'path': path, 'error': event.get('reason'), 'status': 'degraded',
368 'next_owner': 'maintenance', 'execution_id': event.get('execution_id'), 'evidence_ref': reference}
369 continue
370 if event.get('kind') != 'maintenance_cleanup':
371 continue
372 result = event.get('result', {})
373 for path in result.get('deleted', []):
374 pending.pop(path, None)
375 for failure in result.get('failures', []):
376 path = failure.get('path')
377 pending[path] = {**failure, 'status': 'degraded', 'next_owner': 'maintenance',
378 'execution_id': event.get('execution_id'),
379 'evidence_ref': 'runtime/history/' + str(index)}
380 return list(pending.values())
