42def seal_large(value):
43 """Seal large eligible leaves of ``value``; plain dicts and lists change in place.
44
45 The walk visits every node of a state on each save, so leaves that cannot
46 be sealed (scalars, short plain strings) are skipped without a call, and a
47 string already encoded below the limit is not encoded again. It rejects a
48 float, which the storage value domain excludes: a record equal to its
49 parent's under ``==`` (``1.0 == 1``) is neither schema-checked nor
50 serialized again (StateContract.validate, repository._encode_families).
51 """
52 from .repository import canonical_json

54 def walk(value):
55 if isinstance(value, FrozenText): return value
56 if isinstance(value, float):
57 raise ValueError("JSON floating-point numbers are not permitted")
58 if _eligible(value):
59 raw = canonical_json(value)
60 return seal(value, raw) if len(raw) >= LIMIT else value
61 if isinstance(value, dict):
62 for key, child in value.items():
63 kind = type(child)
64 if kind in _LEAVES or kind is str and len(child) < LIMIT // 6 or isinstance(child, FrozenText):
65 continue
66 sealed = walk(child)
67 if sealed is not child:
68 value[key] = sealed
69 elif isinstance(value, list):
70 for index, child in enumerate(value):
71 kind = type(child)
72 if kind in _LEAVES or kind is str and len(child) < LIMIT // 6:
73 continue
74 sealed = walk(child)
75 if sealed is not child:
76 value[index] = sealed
77 return value

79 return walk(value)

85def encode(envelope):
86 """Replace each sealed payload of the record by a reference to its blob.
87
88 Only sealed values carry payload bytes: plain scalars and strings are
89 copied without a call, plain dicts and lists skip the attribute probe, and
90 a payload's coordinate is copied from the walk's path only when found.
91 """
92 blobs, references, tokens = {}, [], []
93 def visit(value):
94 kind = type(value)
95 if kind is not dict and kind is not list and hasattr(value, '_payload_bytes'):
96 raw, digest = value._payload_bytes, value._payload_hash
97 path = 'runtime/artifacts/payload-' + digest + '/value.json'
98 blobs[path] = raw
99 references.append({'tokens':list(tokens), 'path':path, 'sha256':digest, 'bytes':len(raw)})
100 return None
101 if isinstance(value, dict):
102 result = {}
103 for key in sorted(value):
104 child = value[key]
105 if type(child) in _PLAIN:
106 result[key] = child
107 else:
108 tokens.append(key)
109 result[key] = visit(child)
110 tokens.pop()
111 return result
112 if isinstance(value, list):
113 result = []
114 for index, child in enumerate(value):
115 if type(child) in _PLAIN:
116 result.append(child)
117 else:
118 tokens.append(index)
119 result.append(visit(child))
120 tokens.pop()
121 return result
122 return value
123 result = {**envelope, 'record':visit(envelope['record'])}
124 if references: result['payloads'] = references
125 return result, blobs

128def decode(envelope, read_payload, git_dir=None):
129 """Restore externalized payloads; ``git_dir`` names the repository they are read from.
130
131 A payload blob this process already matched in that repository
132 (repository.VERIFIED_ARTIFACTS) is parsed but not hashed or re-checked
133 for canonical encoding again: its path is its digest and bundles are
134 immutable.
135 """
136 from .file_io import is_sha256, reject_constant, unique_pairs
137 from .repository import canonical_json, artifact_verified, mark_artifact_verified
138 if set(envelope) not in ({'contract_id','family','record'}, {'contract_id','family','record','payloads'}):
139 raise ValueError('invalid record envelope fields')
140 value = deepcopy(envelope['record'])
141 references = envelope.get('payloads', [])
142 if not isinstance(references, list): raise ValueError('payload references must be a list')
143 for ref in references:
144 if (not isinstance(ref, dict) or set(ref) != {'tokens','path','sha256','bytes'} or not isinstance(ref['tokens'], list)
145 or any(type(t) not in {str,int} or type(t) is int and t < 0 for t in ref['tokens'])
146 or not is_sha256(ref['sha256'])
147 or type(ref['bytes']) is not int or ref['bytes'] < LIMIT
148 or ref['path'] != 'runtime/artifacts/payload-' + ref['sha256'] + '/value.json'):
149 raise ValueError('invalid immutable payload reference')
150 cache = getattr(read_payload, 'decoded', {})
151 key = (ref['path'], ref['sha256'], ref['bytes'])
152 if key not in cache:
153 raw = read_payload(ref['path'])
154 known = git_dir is not None and artifact_verified(git_dir, ref['path'], ref['sha256'], ref['bytes'])
155 if len(raw) != ref['bytes'] or not known and hashlib.sha256(raw).hexdigest() != ref['sha256']:
156 raise ValueError('immutable payload bytes differ')
157 decoded = json.loads(raw, object_pairs_hook=unique_pairs, parse_constant=reject_constant)
158 if not known:
159 if canonical_json(decoded) != raw:
160 raise ValueError('noncanonical immutable payload')
161 if git_dir is not None:
162 mark_artifact_verified(git_dir, ref['path'], ref['sha256'], len(raw))
163 if not _eligible(decoded):
164 raise ValueError('payload is not an externalized immutable field')
165 cache[key] = seal(decoded, raw, ref['sha256']) # matched above or earlier
166 parent = value
167 try:
168 if not ref['tokens']:
169 if value is not None or len(references) != 1: raise ValueError('payload root overlaps')
170 value = cache[key]
171 continue
172 for token in ref['tokens'][:-1]: parent = parent[token]
173 last = ref['tokens'][-1]
174 if parent[last] is not None: raise ValueError('payload coordinate overlaps')
175 parent[last] = cache[key]
176 except (KeyError, IndexError, TypeError) as error:
177 raise ValueError('invalid payload coordinate') from error
178 # Enforce the one current encoding, including mandatory externalization of
179 # large immutable fields. No implicit migration of old inline records.
180 rebuilt, _ = encode({**{k:envelope[k] for k in ('contract_id','family')}, 'record':seal_large(value)})
181 if rebuilt != envelope:
182 raise ValueError('record differs from canonical payload encoding')
183 return value

186def json_safe(value: Any) -> Any:
187 """A JSON-safe projection of a reported value: bytes, paths, unsafe integers and floats made explicit."""
188 if isinstance(value, bytes):
189 return bytes_evidence(value)
190 if isinstance(value, dict):
191 return {str(k): json_safe(v) for k, v in value.items()}
192 if isinstance(value, (tuple, list)):
193 return [json_safe(v) for v in value]
194 if isinstance(value, Path):
195 return str(value)
196 if isinstance(value, int) and not isinstance(value, bool) and abs(value) > (1 << 53) - 1:
197 return str(value)
198 if isinstance(value, float):
199 # Opaque provider reports retain their original artifact bytes. State
200 # projections represent decimals explicitly without integer rounding.
201 return {"reported_decimal": repr(value)}
202 return value
