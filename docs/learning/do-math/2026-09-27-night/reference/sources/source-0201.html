17class QuestionFlowTests(ActivatedPurposeHarness):
18 def test_answer_closes_only_its_question_and_reaches_the_next_worker(self):
19 # From a prepared task: its worker's question route is the subject.
20 engine, _ = purpose_copy(self, PurposeRunner(), until='prepared',
21 config=self.fixture_config(operator_enabled=False, scholarly_enabled=False))
22 action = next(a for a in engine.runtime['queue'] if a['role'] == 'research_work')
23 task = engine.runtime['tasks'][action['target']]
24 engine.runtime['queue'] = [action]
25 class Runner:
26 @local_job_runner
27 def run(inner, **args):
28 args['on_started']({'fixture': True})
29 context = parse_context(args['prompt'], args['workspace'])
30 output = role_output('research_work')
31 output.update(outcome='waiting', report='A user choice is needed.', remaining_work=['Use the chosen option.'],
32 checkpoint='Waiting for the user.', questions=['Choose option one or two?', 'Research the remaining lemma.'])
33 (args['workspace'] / 'activity-extensions.json').write_bytes(canonical_json({'question_routes': [
34 {'question_index': 0, 'owner': 'user', 'target_ids': [task['id']], 'reason': 'The user owns this choice.'}]}))
35 return producer_result({'status': 'completed', 'output': output, 'usage': {}}, args)
36 engine.runner = Runner()
37 engine._save()
38 engine.run(max_steps=1)
39 rows = engine._question_view()
40 self.assertTrue(any(row['owner'] == 'user' for row in rows),
41 [{key: ex.get(key) for key in ('role', 'outcome', 'error')}
42 for ex in engine.runtime['executions'].values()])
43 user = next(row for row in rows if row['owner'] == 'user')
44 research = next(row for row in rows if row['owner'] == 'research')
45 self.assertEqual(engine.status()['questions'], ['Choose option one or two?'])
46 control(self.repo, engine.instance['id'], 'input', text='Unrelated information.')
47 engine.process_controls()
48 self.assertEqual(next(row for row in engine.runtime['question_links'] if row['id'] == user['id'])['status'], 'open')
49 control(self.repo, engine.instance['id'], 'decision', config={'change': {
50 'kind': 'question_answer', 'text': 'Use option one.', 'target_ids': [task['id']],
51 'resolves_questions': [user['id']]}})
52 engine.process_controls()

54 answered = next(row for row in context['question_links'] if row['id'] == user['id'])
55 self.assertEqual(answered['status'], 'resolved')
56 self.assertEqual(answered['resolution']['change']['text'], 'Use option one.')
57 self.assertEqual(next(row for row in context['question_links'] if row['id'] == research['id'])['status'], 'open')
58 self.assertEqual(engine.status()['questions'], [])

94 validate_candidate=lambda instance, runtime, bundle: engine._validate_bundle_candidate(
95 instance, runtime, bundle, artifacts=artifacts))
96 self.assertEqual(preview['status'], 'accepted', preview['issues'])
