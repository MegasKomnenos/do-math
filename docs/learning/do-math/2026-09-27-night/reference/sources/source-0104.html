18def _source_paths(root: Path, file_io):

48def verify_snapshot(manifest_path: Path):
49 """Verify the frozen copy; hash only files whose settled fingerprint changed.
50
51 The first verification hashes every file and records settled fingerprints
52 beside the manifest. Later starts compare fingerprints (accidental-change
53 threat model, design D1) and hash a file again only when it differs.
54 """
55 file_io = _file_io()
56 # Proves the snapshot root and its ancestors; files are checked below the root.
57 file_io.no_reparse(manifest_path)
58 manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
59 if not isinstance(manifest, dict) or set(manifest) != {"format", "installation_root", "files", "snapshot_sha256"}:
60 raise ValueError("invalid bootstrap manifest")
61 if manifest["format"] != FORMAT or not isinstance(manifest["files"], dict):
62 raise ValueError("unsupported bootstrap manifest")
63 basis = {key: manifest[key] for key in ("format", "installation_root", "files")}
64 if file_io.json_digest(basis) != manifest["snapshot_sha256"]:
65 raise ValueError("bootstrap manifest identity mismatch")
66 root = manifest_path.parent.resolve(strict=True)
67 if root.name != manifest["snapshot_sha256"]:
68 raise ValueError("bootstrap directory identity mismatch")
69 recorded, settled = _marks(root), {}
70 for name, digest in manifest["files"].items():
71 relative = PurePosixPath(name)
72 if relative.is_absolute() or relative.as_posix() != name or ".." in relative.parts or "\\" in name:
73 raise ValueError("invalid bootstrap file path")
74 target = root.joinpath(*relative.parts)
75 file_io.no_reparse(target, below=root)
76 if not target.resolve(strict=True).is_relative_to(root) or not target.is_file():
77 raise ValueError("bootstrap file escaped its snapshot")
78 mark = file_io.file_fingerprint(target)
79 stored = file_io.fingerprint_json(mark)
80 if recorded.get(name) != stored or not file_io.fingerprint_settled(mark):
81 if file_io.sha256_file(target) != digest:
82 raise ValueError("bootstrap file hash mismatch: " + name)
83 if file_io.fingerprint_settled(mark):
84 settled[name] = stored
85 if set(_source_paths(root, file_io)) != set(manifest["files"]):
86 raise ValueError("bootstrap snapshot has unlisted or missing files")
87 if settled != recorded:
88 file_io.write_bytes_atomic(root / MARKS, json.dumps(settled, sort_keys=True).encode("utf-8"))
89 return manifest

92def freeze(installation_root: Path, workspace: Path):
93 """Publish a content-addressed runtime copy; no research state or tasks change."""
94 file_io = _file_io()
95 file_io.no_reparse(installation_root)
96 file_io.no_reparse(workspace)
97 installation_root = installation_root.resolve(strict=True)
98 workspace = workspace.resolve()
99 if not (installation_root / "scripts/do_math/scheduled_operator.py").is_file():
100 raise ValueError("installation has no supervisor")
101 files = _source_files(installation_root, file_io)
102 basis = {"format": FORMAT, "installation_root": str(installation_root), "files": files}
103 identity = file_io.json_digest(basis)
104 parent = workspace / "operator/bootstrap"
105 file_io.no_reparse(parent)
106 parent.mkdir(parents=True, exist_ok=True)
107 destination = parent / identity
108 if destination.exists():
109 verify_snapshot(destination / MANIFEST)
110 return {"entrypoint": str(destination / "scripts/operator_bootstrap.py"),
111 "manifest": str(destination / MANIFEST), "snapshot_sha256": identity}
112 staging = parent / (".building-" + uuid.uuid4().hex)
113 staging.mkdir()
114 try:
115 for name in files:
116 source = installation_root.joinpath(*PurePosixPath(name).parts)
117 target = staging.joinpath(*PurePosixPath(name).parts)
118 target.parent.mkdir(parents=True, exist_ok=True)
119 shutil.copyfile(source, target)
120 if _source_files(staging, file_io) != files or _source_files(installation_root, file_io) != files:
121 raise ValueError("installation changed while freezing supervisor")
122 (staging / MANIFEST).write_text(json.dumps({**basis, "snapshot_sha256": identity},
123 ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
124 try:
125 staging.rename(destination)
126 except FileExistsError:
127 verify_snapshot(destination / MANIFEST)
128 finally:
129 if staging.exists():
130 # Only the fresh staging directory created above can be removed.
131 file_io.no_reparse(staging)
132 if staging.resolve().parent != parent.resolve() or not staging.name.startswith(".building-"):
133 raise ValueError("invalid bootstrap staging cleanup target")
134 shutil.rmtree(staging)
135 verify_snapshot(destination / MANIFEST)
136 return {"entrypoint": str(destination / "scripts/operator_bootstrap.py"),
137 "manifest": str(destination / MANIFEST), "snapshot_sha256": identity}
