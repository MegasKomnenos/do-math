2
3The repository writer lock and single CAS own admission. Local Job evidence
4settles local capacity; it proves neither remote termination nor provider use.
5Only work roles reserve a model slot. Operational AI roles are admitted on
6their pinned model without a slot; host limits still bound every capped role.

15# Admission wait kinds recorded in a queued request's ``dispatch_wait``.
16PROVIDER_WAIT = 'provider_limit'
17# Slot waits: the instance is capacity_waiting when every queued request has one.
18SLOT_WAITS = frozenset({'capacity', 'host_capacity'})
19# Resource waits: a changed resource configuration may release them.
20RESOURCE_WAITS = SLOT_WAITS | {'resource_unclassified'}
21# Every admission wait, as status reports show scheduling waits.
22ADMISSION_WAITS = RESOURCE_WAITS | {PROVIDER_WAIT}

54 database is authoritative for this path.

85 def _admission(self, profile, execution_id, current):
86 """The one admission calculation shared by the check and the real acquire.
87
88 Returns the registered profile and the reservation records to publish,
89 or None after recording ``admission_wait``. Nothing is written.
90 """
91 from .execution_profiles import (match_profile_ref, profile_ref, job_resource_limits,
92 validate_host_constraint, UNCAPPED_ROLES)
93 from .budget import plan_reservations, BudgetConflict
94 self.admission_wait = None
95 profile = match_profile_ref(self.instance["execution_profiles"], profile_ref(profile))
96 approved_host = self.instance.get("host_capacity_ref")
97 uncapped_lean = (profile['actor_kind']=='machine'
98 and profile['role'] in UNCAPPED_ROLES
99 and profile.get('host_resources') is None)
100 if profile.get("host_capacity_refs", []) != ([approved_host] if approved_host and not uncapped_lean else []):
101 raise ValueError("execution profile must bind the instance's current adopted host constraint")
102 job_resource_limits(profile) # Reject a profile bound to another host before any mutation.
103 ai = profile.get("ai", {})
104 model = ai.get("model")
105 self.admission_wait = self.provider_wait(profile, current_time=current)
106 if self.admission_wait is not None:
107 return None
108 # A new capacity edition must still count executions reserved under older
109 # editions of the same model. Limits are not independent resource pools.
110 constraints = self.instance["capacity_constraints"]
111 # Rebase only each held row's constraint ID; nested evidence stays read-only.
112 held = [dict(row) for row in held_reservations(self.runtime)]
113 capacity_refs = list(ai.get("capacity_refs", []))
114 adopted = self.instance.get("role_profile_refs")
115 if adopted and capacity_refs:
116 # An execution keeps its immutable profile, but a new reservation
117 # must obey the currently adopted limit for the same resource.
118 active_constraints = set()
119 for role_refs in adopted.values():
120 for reference in role_refs:
121 active_profile = self.instance["execution_profiles"][reference["id"]]
122 active_ai = active_profile.get("ai", {})
123 active_constraints.update(active_ai.get("capacity_refs", []))
124
125 def current_ref(reference):
126 previous = constraints[reference]
127 matches = [cid for cid in active_constraints if constraints[cid]["kind"] == "concurrency"
128 and constraints[cid]["model"] == previous["model"]]
129 return matches[0] if len(matches) == 1 else None
130
131 capacity_refs = [current_ref(ref) for ref in capacity_refs]
132 if None in capacity_refs:
133 self.admission_wait = {"kind": "capacity", "model": model,
134 "profile_ref": profile_ref(profile), "reason": "pinned profile has no current adopted accounting constraint",
135 "next_owner": "user"}
136 return None
137 refs = capacity_refs
138 accounting_profile = {**profile, "ai": {**ai, "capacity_refs": capacity_refs}} if ai else profile
139 host_refs = profile.get("host_capacity_refs", [])
140 for ref in host_refs:
141 validate_host_constraint(constraints[ref])
142 if any(profile["host_resources"][k] != constraints[ref][k] for k in ("scope", "host_id")):
143 raise ValueError("profile and approved host scope differ")
144 for row in held:
145 previous = constraints.get(row["constraint_id"], {})
146 for ref in refs + host_refs:
147 constraint = constraints[ref]
148 same_model = (previous.get("kind") == constraint["kind"] == "concurrency"
149 and previous.get("model") == constraint["model"])
150 same_host = (previous.get("kind") == constraint["kind"] == "host"
151 and previous.get("scope") == constraint.get("scope") == "instance_on_host"
152 and previous.get("host_id") == constraint.get("host_id"))
153 if same_model or same_host:
154 row["constraint_id"] = ref
155 try:
156 records = plan_reservations(held, constraints, accounting_profile, owner_id=execution_id)
157 except BudgetConflict as error:
158 kind, owner = {'concurrency': ('capacity', 'scheduler'),
159 'host': ('host_capacity', 'operations')}.get(
160 error.constraint_kind, ('resource_unclassified', 'operations'))
161 constraint = constraints.get(error.constraint_id, {})
162 if error.constraint_kind == 'concurrency' and constraint.get('work_slots') == 0:
163 owner = 'user'
164 self.admission_wait = {'kind': kind, 'model': model,
165 'profile_ref': profile_ref(profile), 'constraint_id': error.constraint_id,
166 'constraint_kind': error.constraint_kind, 'reason': str(error), 'next_owner': owner}
167 return None
168 return profile, records

180 def acquire(self, execution_id, *, profile, execution, current_time=None):
181 """Admit one registered execution on its pinned profile, or record the wait."""
182 self._writable()
183 from .execution_profiles import bind_execution
184 from .execution_lifecycle import transition_execution
185 from .launch import ticket
186 from datetime import datetime, timezone
187 if execution.get("id") != execution_id:
188 raise ValueError("execution identity differs from admission")
189 existing = self.runtime.get("executions", {}).get(execution_id)
190 if existing is not None and (existing is not execution or "capacity" in existing
191 or existing.get("status") not in {"queued", "starting"}):
192 raise ValueError("execution IDs are single-use; only the registered unreserved execution can be admitted")
193 current = time.time() if current_time is None else current_time
194 admitted = self._admission(profile, execution_id, current)
195 if admitted is None:
196 return False
197 profile, records = admitted
198 candidate = deepcopy(execution)
199 bind_execution(candidate, profile)
200 candidate['launch_ticket'] = ticket(execution_id, profile)
201 admitted_at = (datetime.now(timezone.utc) if current_time is None
202 else datetime.fromtimestamp(current, timezone.utc))
203 transition_execution(candidate, "starting", at=admitted_at.isoformat())
204 provider = profile.get("ai", {}).get("provider")
205 candidate["capacity"] = {"reservation_ids": [r["id"] for r in records],
206 "provider": provider, "process": None, "local_termination": "unknown"}
207 self.runtime.setdefault("reservations", {}).update({r["id"]: r for r in records})
208 self.runtime.setdefault("executions", {})[execution_id] = candidate
209 gate = self._gate(provider)
210 if gate and gate["state"] == "waiting":
211 gate.update(state="checking", probe_owner=execution_id, probe_started_at=int(current))
212 execution.clear()
213 execution.update(candidate)
214 self.runtime["executions"][execution_id] = execution
215 return True

217 def started(self, execution_id, identity, *, at):
218 """Record the started named Job, its running transition and start time."""
219 self._writable()
220 from .execution_lifecycle import transition_execution
221 execution = self.runtime["executions"][execution_id]
222 capacity = execution["capacity"]
223 launch = execution['launch_ticket']
224 if launch['kind'] != 'native_job' or identity.get('job_name') != launch['job_name']:
225 raise ValueError('started Job differs from the published launch ticket')
226 if capacity["process"] is not None:
227 if capacity["process"] == identity:
228 return execution
229 raise ValueError("execution process identity is already fixed")
230 capacity["process"] = deepcopy(identity)
231 transition_execution(execution, "running", at=at)
232 execution["started_at"] = at
233 return execution

235 def release(self, execution_id, *, reason="local runner returned after termination"):
236 self._writable()
237 from .budget import settle_reservation
238 from .launch import termination_confirmed
239 execution = self.runtime.get("executions", {}).get(execution_id)
240 if not execution or "capacity" not in execution:
241 return False
242 capacity = execution["capacity"]
243 if capacity["local_termination"] == "confirmed":
244 return True
245 identity = capacity["process"]
246 # The launch ticket, not missing process metadata, establishes the
247 # barrier; for a native Job it already requires the named Job empty.
248 # A suspended root may exist before on_started is saved.
249 if not execution.get('launch_ticket') or not termination_confirmed(execution['launch_ticket']):
250 capacity['local_termination'] = 'unknown'
251 return False
252 evidence = [{"execution_id": execution_id, "job_name": identity.get("job_name") if identity else None,
253 "kind": "local_job_empty" if identity else "never_resumed", "reason": reason}]
254 updates = {}
255 for rid in capacity["reservation_ids"]:
256 record = self.runtime["reservations"][rid]
257 updates[rid] = settle_reservation(record, confirmed_consumed=0, returned=record["reserved"],
258 evidence_refs=evidence)
259 self.runtime["reservations"].update(updates)
260 capacity["local_termination"] = "confirmed"
261 for gate in self.runtime.get("provider_waits", {}).values():
262 if gate["state"] == "checking" and gate["probe_owner"] == execution_id:
263 # Releasing a local slot is no evidence that a provider recovered.
264 gate.update(state="waiting", probe_owner=None, probe_started_at=None)
265 return True

267 def provider_result_from_run(self, execution_id, result, config, *, current_time=None):
268 """Apply one runner result to the gate of the execution's admitted provider.
269
270 Only a provider_limit failure closes the gate. The probe owner reopens
271 it when the provider answered at all: a completed call, or a reported
272 failed or completed turn.
273 """
274 self._writable()
275 from .provider_wait import retry_time
276 current = time.time() if current_time is None else current_time
277 initial_seconds = config["provider_retry_initial_seconds"]
278 maximum_seconds = config["provider_retry_max_seconds"]
279 execution = self.runtime.get("executions", {}).get(execution_id, {})
280 provider = execution.get("capacity", {}).get("provider")
281 if provider is None:
282 raise ValueError("provider result has no matching admitted execution")
283 parsed = result.get("execution", {}).get("collect", {}).get("jsonl", {})
284 failure = result.get("failure") if result.get("error_kind") == "provider_limit" else None
285 response_observed = result.get("status") == "completed" or bool(
286 parsed.get("turn_failed") or parsed.get("turn_completed"))
287 rows = self.runtime.setdefault("provider_waits", {})
288 row = rows.get("PW-" + provider)
289 if failure is not None:
290 if failure.get("kind") != "provider_limit" or failure.get("provider") != provider:
291 raise ValueError("provider failure differs from admission")
292 attempts = 1 if not row or row["state"] == "available" else row["attempts"] + int(
293 row["state"] == "checking" and row["probe_owner"] == execution_id)
294 retry_at = math.ceil(retry_time(failure, attempts, current, initial_seconds, maximum_seconds))
295 if row and row["state"] == "waiting":
296 retry_at = max(retry_at, row["retry_at"])
297 rows["PW-" + provider] = {"id": "PW-" + provider, "provider": provider, "generation": row["generation"] + 1 if row else 1,
298 "state": "waiting", "retry_at": retry_at, "evidence": {
299 "execution_id": execution_id, "kind": failure["kind"], "provider": provider,
300 "evidence_ref": failure.get("evidence_ref"),
301 "reported_fields_json": json.dumps(failure, ensure_ascii=False, allow_nan=False)},
302 "probe_owner": None, "probe_started_at": None, "attempts": attempts}
303 elif row and row["state"] == "checking" and row["probe_owner"] == execution_id:
304 if response_observed:
305 row.update(state="available", retry_at=None, attempts=0, probe_owner=None, probe_started_at=None)
306 else:
307 retry_at = math.ceil(retry_time({}, row["attempts"] + 1, current, initial_seconds, maximum_seconds))
308 row.update(state="waiting", retry_at=retry_at, attempts=row["attempts"] + 1,
309 probe_owner=None, probe_started_at=None)
310 return self.provider_snapshot(current_time=current)["providers"].get(provider,
311 {"provider": provider, "state": "available", "generation": 0})

314 from .provider_wait import gate_open, utc
315 current = time.time() if current_time is None else current_time
316 rows = deepcopy(self.runtime.get("provider_waits", {}))

323 def snapshot(self):
324 """Work slots of each currently adopted model; operational roles hold none."""
325 constraints = self.instance["capacity_constraints"]
326 active_refs = {ref["id"] for refs in self.instance.get("role_profile_refs", {}).values() for ref in refs}
327 active_constraints = {ref for pid in active_refs for ref in self.instance["execution_profiles"][pid].get("ai", {}).get("capacity_refs", [])}
328 result = {constraints[cid]["model"]: {"limit": constraints[cid]["work_slots"], "used": 0, "execution_ids": []}
329 for cid in active_constraints}
330 for row in held_reservations(self.runtime):
331 if row["unit"] != "execution":
332 continue
333 model = constraints[row["constraint_id"]]["model"]
334 cell = result.setdefault(model, {"limit": 0, "used": 0, "execution_ids": []})
335 cell["used"] += row["reserved"]
336 cell["execution_ids"].append(row["owner_id"])
337 for cell in result.values():
338 cell["available"] = max(0, cell["limit"] - cell["used"])
339 return result
