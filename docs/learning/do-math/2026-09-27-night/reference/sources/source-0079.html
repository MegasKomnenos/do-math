35def link(runtime, *, execution_id, field, owner, targets, source_ref=None, reason=None, blocks=False, checked_by=None):
36 if owner not in QUESTION_OWNERS or type(blocks) is not bool:
37 raise ValueError("question owner and blocking decision required")
38 if not isinstance(targets, list) or any(not isinstance(t, str) or not t for t in targets) or (blocks and not targets):
39 raise ValueError("blocking questions require explicit affected targets")
40 rows = runtime.setdefault("question_links", [])
41 existing = next((row for row in rows if row["origin_execution_id"] == execution_id and row["origin_field"] == field), None)
42 if existing:
43 if (existing["owner"], existing["target_ids"], existing.get("blocks", False)) != (owner, list(dict.fromkeys(targets)), blocks):
44 raise ValueError("question source cannot be rerouted silently")
45 return existing
46 row = {"id": new_id("QU"), "origin_execution_id": execution_id, "origin_field": field,
47 "owner": owner, "target_ids": list(dict.fromkeys(targets)), "unresolved_target_ids": list(dict.fromkeys(targets)),
48 "blocks": blocks, "status": "open", "resolution_ref": None, "checked_by": checked_by}
49 if source_ref:
50 row["source_ref"] = source_ref
51 if reason:

57def close(row, *, kind, resolution_ref, affected_ids, status="resolved", evidence=None):
58 """The only resolving transition: resolve affected targets, closing the question with its last one.
59
60 Every closing path appends the same history entry shape."""
61 remaining = set(open_targets(row)) - set(affected_ids)
62 row["unresolved_target_ids"] = [target for target in row["target_ids"] if target in remaining]
63 entry = {"kind": kind, "resolution_ref": resolution_ref,
64 "affected_ids": sorted(set(affected_ids) & set(row["target_ids"]))}
65 if evidence is not None:
66 entry["evidence"] = deepcopy(evidence)
67 row.setdefault("resolution_history", []).append(entry)
68 if not remaining:
69 row.update(status=status, resolution_ref=resolution_ref)

72def resolution_error(row, role):
73 """Why an AI extension cannot resolve this question row now; None when it can.
74
75 Submission checks call this on the fixed input, and application calls it again
76 because another execution may have closed the question in between."""
77 if not row or row["status"] != "open" or row["owner"] not in QUESTION_OWNERS - {"user"}:
78 return "only distinct open research or operations questions can be resolved by an AI extension"
79 if row["owner"] == "operations" and role not in OPERATIONS_ROLES:

84def resolve_user(runtime, identifiers, *, resolution_ref, affected_ids):
85 questions = {row["id"]: row for row in runtime.get("question_links", [])}
86 if not isinstance(identifiers, list) or len(identifiers) != len(set(identifiers)):
87 raise ValueError("resolves_questions must be distinct question IDs")
88 for identifier in identifiers:
89 row = questions.get(identifier)
90 if not row or row["owner"] != "user" or row["status"] != "open":
91 raise ValueError("user decision must name an open user question: " + str(identifier))
92 if not set(open_targets(row)) & set(affected_ids):
93 raise ValueError("the applied decision does not affect the question target: " + identifier)
94 for identifier in identifiers:
95 close(questions[identifier], kind="user_decision", resolution_ref=resolution_ref, affected_ids=affected_ids)

98def validate_extensions(output, extensions, context):
