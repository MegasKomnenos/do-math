9def case_basis_reference_ids(case):
10 """Return IDs from the closed string-or-``{"id": ID}`` case-reference form."""
11 refs = case.get("basis_refs", [])
12 if not isinstance(refs, list):
13 raise ValueError("case basis refs must be a list")
14 identifiers = []
15 for reference in refs:
16 if isinstance(reference, str):
17 identifier = reference
18 elif isinstance(reference, dict) and set(reference) == {"id"}:
19 identifier = reference["id"]
20 else:
21 raise ValueError("case basis reference must be a record ID or an exact {'id': ID} object")
22 if not isinstance(identifier, str) or not identifier:
23 raise ValueError("case basis refs must name distinct record IDs")
24 identifiers.append(identifier)
25 if len(identifiers) != len(set(identifiers)):
26 raise ValueError("case basis refs must name distinct record IDs")
27 return identifiers

30def validate_delegation(delegation, *, now_iso=None):
31 if not isinstance(delegation, dict): raise ValueError("delegation record required")
32 for key in ("id", "decision_kind", "original_decision_id", "source_ref", "decision_id"):
33 if not isinstance(delegation.get(key), str) or not delegation[key].strip(): raise ValueError("missing delegation " + key)
34 if delegation.get("status") != "active": raise ValueError("delegation inactive")
35 # The design prohibits these decisions; its other examples are not a closed kind list.
36 if delegation["decision_kind"] in {"proposal_adoption", "contract_limit_continuation"}:
37 raise ValueError("this decision requires a new explicit user decision")
38 for key in ("included_targets", "excluded_targets", "external_effects"):
39 value = delegation.get(key)
40 if not isinstance(value, list) or any(not isinstance(x, str) or not x for x in value) or len(value) != len(set(value)):
41 raise ValueError("invalid delegation " + key)
42 if not delegation["included_targets"]: raise ValueError("delegation needs explicit included scope")
43 if not isinstance(delegation.get("resource_limits", {}), dict): raise ValueError("resource limits must be mapped")
44 if any(type(v) is not int or v < 0 for v in delegation.get("resource_limits", {}).values()):
45 raise ValueError("invalid delegation resource limit")
46 if delegation.get("expires_at") and parse_utc(delegation["expires_at"]) <= (parse_utc(now_iso) if now_iso is not None else datetime.now(timezone.utc)): raise ValueError("delegation expired")
47 return deepcopy(delegation)

50def _basis(instance, runtime, delegation_id, case, now_iso=None):
51 delegation = validate_delegation(instance.get("delegations", {}).get(delegation_id), now_iso=now_iso)
52 original = instance.get("decisions", {}).get(delegation["original_decision_id"], {})
53 approval = instance.get("decisions", {}).get(delegation["decision_id"], {})
54 if original.get("actor") != "user" or original.get("kind") != delegation["decision_kind"]:
55 raise ValueError("original decision is absent or incompatible")
56 if approval.get("actor") != "user":
57 raise ValueError("delegation has no current user approval")
58 approved = approval.get("change", {}).get("delegation")
59 declared = {k:v for k,v in delegation.items() if k not in {"source_ref", "decision_id", "created_at"}}
60 if approval.get("kind") != "delegation" or approved != declared or approval.get("source_ref") != delegation["source_ref"]:
61 raise ValueError("approval does not bind the exact delegated scope")
62 source = instance.get("governance_sources", {}).get(delegation["source_ref"])
63 from .governance import validate_source
64 validate_source(source or {})
65 original_source = instance.get("governance_sources", {}).get(original.get("source_ref"))
66 validate_source(original_source or {})
67 target = case.get("target_id")
68 if case.get("decision_kind") != delegation["decision_kind"] or target not in delegation["included_targets"] or target in delegation["excluded_targets"]:
69 raise ValueError("case outside delegation scope")
70 effects, resources = case.get("external_effects"), case.get("resources")
71 if not isinstance(effects, list) or any(x not in delegation["external_effects"] for x in effects): raise ValueError("external effect outside delegation")
72 limits = delegation.get("resource_limits", {})
73 if not isinstance(resources, dict) or any(k not in limits or type(v) is not int or v < 0 or v > limits[k] for k, v in resources.items()):
74 raise ValueError("resource request outside delegation")
75 if not isinstance(case.get("effect"), dict) or not case.get("id"): raise ValueError("case must preserve exact proposed effect")
76 # Use the same globally unique record population as task/currentness checks.
77 from .planning import record_index
78 records = record_index(instance, runtime)
79 target_record = records.get(target, (None, None))[1]
80 if target == instance.get("id"): target_record = {"id": target, "governance": instance.get("governance")}
81 if target_record is None: raise ValueError("case target does not resolve")
82 refs = case_basis_reference_ids(case)
83 if any(ref not in records for ref in refs): raise ValueError("case basis reference does not resolve")
84 return {"delegation": delegation, "original_decision": deepcopy(original), "approval": deepcopy(approval),
85 "source": deepcopy(source), "original_source":deepcopy(original_source), "case": deepcopy(case),
86 "target": deepcopy(target_record), "references": {ref: deepcopy(records[ref][1]) for ref in refs}}

135def _checked(instance, runtime, application_id, now_iso=None):

137 if record is None: raise ValueError("application not found")
138 basis = _basis(instance, runtime, record["delegation_id"], record["basis"]["case"], now_iso)

140 expected = receipt.get("post_basis_sha256",record["input_sha256"])
141 if receipt and receipt.get("sha256") != exact_content_hash({k:v for k,v in receipt.items() if k != "sha256"}): raise ValueError("delegation receipt changed")
142 if exact_content_hash(basis) != expected or exact_content_hash(record["basis"]) != record["input_sha256"]: raise ValueError("delegation application stale")
143 judgments = record["judgments"]
144 if set(judgments) != {"interpret", "check"} or len({j["execution_id"] for j in judgments.values()}) != 2: raise ValueError("two independent judgments required")
145 from .governance import validate_independent_execution
146 validate_independent_execution(runtime,judgments["check"]["execution_id"],[judgments["interpret"]["execution_id"]])
147 for judgment in judgments.values():
148 _check_judgment(runtime, record, judgment["execution_id"], judgment["output"],
