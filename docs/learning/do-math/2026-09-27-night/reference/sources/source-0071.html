1"""Run blocking adapters or machine calls; the owner thread judges stops and applies callbacks.
2
3A calculation reads only its stop flag. The owner thread runs each call's stop
4judgment after every head movement (and when a time-bound reason falls due)
5and sets the flag once the judgment is true. ``on_started`` remains a
6synchronous owner round trip because the identity must be published before
7resume. Observations are queued per call without waiting; the owner applies a
8call's observations in order, and all of them before it collects that call.
9"""
10from __future__ import annotations

12from collections import deque
13from concurrent.futures import Future
14from contextvars import copy_context
15from queue import Empty, Queue
16from threading import Event, Lock, Thread

19def failed_result(error):
20 """The one result a call records when its calculation raised instead of returning."""
21 return {"status": "failed", "output": None, "usage": {}, "error": f"{type(error).__name__}: {error}"}

24class ModelCalls:
25 def __init__(self, *, on_returned=None, on_observation_error=None):
26 self.events = Queue()
27 self.futures = {}
28 self.threads = []
29 self.guard = Lock()
30 self.pumping = False
31 self.collecting = False
32 self.judging = False
33 self.stopping = False
34 self.on_returned = on_returned
35 self.on_observation_error = on_observation_error
36 # Set by anything the owner must handle: a callback request, an
37 # observation or a returned call. The owner waits on it, not a spin.
38 self.wake = Event()

40 def _started(self, function):
41 def call(*args):
42 response = Future()
43 with self.guard:
44 if self.stopping:
45 raise RuntimeError("executor is shutting down")
46 self.events.put((function, args, response))
47 self.wake.set()
48 from .operation_trace import phase
49 with phase('callback_wait', callback=getattr(function, '__name__', 'callback')):
50 return response.result()

53 def _observed(self, function, future):
54 def call(*args):
55 with self.guard:
56 if self.stopping:
57 raise RuntimeError("executor is shutting down")
58 future.observations.append((function, args))

62 def start_call(self, generator, function, arguments, *, execution_id=None):
63 arguments = dict(arguments)
64 if execution_id is not None:
65 if not isinstance(execution_id, str) or not execution_id:
66 raise ValueError("execution_id must be a nonempty string")
67 if any(future.execution_id == execution_id for future in self.futures):
68 raise ValueError("an execution already has an active call")
69 future = Future()
70 future.execution_id = execution_id
71 future.stop = Event()
72 future.judge = arguments.get("should_stop")
73 future.observations = deque()
74 future.observation_failed = False
75 future.judge_error = None
76 if "should_stop" in arguments:
77 arguments["should_stop"] = future.stop.is_set
78 if "on_started" in arguments:
79 arguments["on_started"] = self._started(arguments["on_started"])
80 if "on_observation" in arguments:
81 arguments["on_observation"] = self._observed(arguments["on_observation"], future)
82 if self.stopping:
83 future.stop.set()
84 elif future.judge is not None:
85 self._judge(future) # The state at start may already require a stop.
86 self.futures[future] = generator
87 def run():
88 try:
89 future.set_result(function(**arguments))
90 except BaseException as error:
91 future.set_exception(error)
92 finally:
93 self.wake.set()
94 context = copy_context()
95 thread = Thread(target=lambda: context.run(run), name="do-math-call")
96 self.threads.append(thread)
97 thread.start()
98 return future

100 def wait(self, timeout):
101 """Owner thread: sleep until a call needs the owner, at most ``timeout`` seconds.
102
103 A wake set after this returns stays set for the next wait; the pass that
104 follows handles everything set before the clear.
105 """
106 self.wake.wait(timeout)
107 self.wake.clear()

109 def _judge(self, future):
110 try:
111 stop = future.judge()
112 except Exception as error:
113 # A failing judgment fails its own call, not the owner.
114 future.judge_error, stop = error, True
115 if stop:
116 future.stop.set()

118 def judge(self):
119 """Owner thread: run the stop judgment of each live call whose flag is still clear."""
120 if self.judging:
121 return
122 self.judging = True
123 try:
124 for future in list(self.futures):
125 # A returned calculation no longer reads its flag; its continuation judges the result.
126 if future.judge is not None and not future.stop.is_set() and not future.done():
127 self._judge(future)
128 finally:
129 self.judging = False

131 def _apply_observations(self, future):
132 # Apply only the rows queued on entry: a running call keeps appending, and
133 # an unbounded drain would starve the owner loop (dispatch, live reviews,
134 # controls). A returned call appends nothing more, so collect applies all.
135 for _ in range(len(future.observations)):
136 function, args = future.observations.popleft()
137 if future.observation_failed:
138 continue # Later rows depend on the one that could not be applied.
139 try:
140 function(*args)
141 except Exception as error:
142 # The call has not been collected yet: stop it and record why.
143 future.observation_failed = True
144 future.stop.set()
145 if self.on_observation_error is not None:
146 self.on_observation_error(future.execution_id, error)
147
148 def pump(self):
149 if self.pumping or self.collecting:
150 return
151 self.pumping = True
152 try:
153 for _ in range(self.events.qsize()):
154 try:
155 function, args, response = self.events.get_nowait()
156 except Empty:
157 break
158 try:
159 response.set_result(function(*args))
160 except BaseException as error:
161 response.set_exception(error)
162 for future in list(self.futures):
163 self._apply_observations(future)
164 finally:
165 self.pumping = False
166
167 def collect(self):
168 if self.collecting:
169 return 0
170 self.collecting = True
171 finished = 0
172 try:
173 for future in list(self.futures):
174 if not future.done():
175 continue
176 generator = self.futures[future]
177 # The calculation queued its observations before it returned.
178 self._apply_observations(future)
179 try:
180 result = future.result()
181 except Exception as error:
182 result = failed_result(error)
183 error = future.judge_error
184 if error is not None and isinstance(result, dict):
185 result = {**result, "status": "failed",
186 "error": f"stop judgment failed: {type(error).__name__}: {error}"}
187 if self.on_returned is not None:
188 self.on_returned(future.execution_id, result)
189 try:
190 generator.send(result)
191 except StopIteration:
192 pass
193 else:
194 raise RuntimeError("an execution yielded more than one call")
195 self.futures.pop(future)
196 finished += 1
197 finally:
198 self.collecting = False
199 self.pump()
200 self.threads[:] = [thread for thread in self.threads if thread.is_alive()]
201 return finished

203 def close(self):
204 # The engine normally drains all calls first. On an unexpected owner
205 # error, stop every calculation and unblock start callbacks so their
206 # local termination runs.
207 with self.guard:
208 self.stopping = True
209 for future in self.futures:
210 future.stop.set()
211 future.observations.clear()
212 while True:
213 try:
214 _, _, response = self.events.get_nowait()
215 except Empty:
216 break
217 response.set_exception(RuntimeError("executor is shutting down"))
218 for thread in self.threads:
219 thread.join()
