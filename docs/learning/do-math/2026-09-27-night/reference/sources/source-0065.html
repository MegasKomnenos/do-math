24def assessment_tier(assessment: dict) -> str:

99def selection(config: dict, action: dict, tasks: dict, knowledge: dict, *, instance: dict | None = None) -> dict:
100 """Bind a requirement to a model without inspecting current slot availability.
101
102 With ``instance``, also select one of its registered profiles. Default
103 assignments use its adopted ``role_profile_refs``, the same editions the
104 engine publishes, never a profile rebuilt from the configuration.
105 """
106 from .execution_profiles import RESEARCH_PRODUCER_ROLES
107 role, target = action["role"], action.get("target")
108 requirement = action.get("model_assessment")
109 task = tasks.get(target, {})
110 if not task and target in knowledge:
111 task = tasks.get(knowledge[target].get("task_id"), {})
112 requirement = requirement or task.get("model_assessment")
113 floor = config["role_tiers"].get(role, ROLE_TIERS.get(role, "low"))
114 if requirement:
115 tier = max((assessment_tier(requirement), floor), key=TIERS.index)
116 reason = requirement["reason"]
117 elif role in RESEARCH_PRODUCER_ROLES:
118 tier, reason = "high", "No preserved difficulty/importance assessment; use high until explicitly assessed."
119 else:
120 tier, reason = floor, "Configured role requirement; raise when task difficulty or importance requires it."
121 result = {"tier": tier, "model": config["tier_models"][tier],
122 "assessment": deepcopy(requirement), "reason": reason}
123 if instance is not None:
124 from .execution_profiles import select_profile, profile_ref, match_profile_ref
125 profiles = instance["execution_profiles"]
126 # A task's allowlist governs its work attempts. Reviewers and other
127 # decision roles use their own adopted role profiles, while still
128 # inheriting the task's difficulty/importance requirement above.
129 allowed = (task.get("profile_refs")
130 if role in RESEARCH_PRODUCER_ROLES
131 and not action.get("review_cycle_id") else None)
132 if action.get("profile_ref"):
133 explicit = match_profile_ref(profiles, action["profile_ref"])
134 if explicit["actor_kind"] == "ai":
135 pinned = explicit["ai"]
136 if (
137 pinned.get("tier") not in TIERS or TIERS.index(pinned["tier"]) < TIERS.index(tier)):
138 raise ValueError("explicit profile does not meet the role/task model requirement")
139 result.update(model=pinned["model"], tier=pinned.get("tier"),
140 reason="Explicit immutable profile meeting the role/task requirement.")
141 if allowed is None and not action.get("profile_ref"):
142 # The registry retains old editions for running executions. New
143 # default assignments use only the currently adopted editions.
144 allowed = instance["role_profile_refs"].get(role, [])
145 selected = select_profile(profiles, action, result, allowed_refs=allowed)
146 result["profile_ref"] = profile_ref(selected)
147 return result
