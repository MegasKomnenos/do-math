63 def test_unassessed_new_results_and_protected_scope_are_not_withdrawable(self):
64 # Each change is rejected by its own withdrawal rule, not by the stored-record schema.
65 reasons = {'new_result': 'current item dispositions', 'stale_assessment': 'current item dispositions',
66 'user_required': 'protected obligations', 'transferred_obligations': 'protected obligations'}

141class TransferReferenceTests(unittest.TestCase):
142 def test_nonready_recipient_uses_separate_fixed_reference_population(self):
143 from record_fixtures import transfer_assessment, apply_assessment
144 from do_math.task_scope import apply_task_assessments
145 instance, runtime, assessment = transfer_assessment()
146 selected_population = {'TK-1': deepcopy(runtime['tasks']['TK-1'])}
147 original = deepcopy(runtime['tasks'])
148 _, changed = apply_task_assessments(instance, runtime, [assessment], execution_id='EX-plan',
149 input_tasks=selected_population, reference_tasks=original)
150 self.assertEqual(changed['tasks']['TK-2']['completion_obligations'][-1]['id'], 'OB-inherited')
151 self.assertEqual(set(selected_population), {'TK-1'})
152 for mutate in (lambda r: r['tasks']['TK-2'].update(revision=2),
153 lambda r: r['tasks']['TK-2'].update(status='closed'),
154 lambda r: r['executions'].update({'EX-peer': {'status': 'running', 'action': {'target': 'TK-2'}}})):
155 candidate = deepcopy(runtime)
156 mutate(candidate)
157 with self.assertRaises(ValueError):
158 apply_task_assessments(instance, candidate, [assessment], execution_id='EX-plan',
159 input_tasks=selected_population, reference_tasks=original)
160 with self.assertRaisesRegex(ValueError, 'recipient'):
161 apply_task_assessments(instance, runtime, [assessment], execution_id='EX-plan',
