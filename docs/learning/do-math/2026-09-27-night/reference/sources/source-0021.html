50def current_solution_basis(instance, runtime):
51 """The current goal judgments' solution basis, or None when the goal is not currently solved.
52
53 A pure function of state: the engine, the status report and the state checks share it.
54 governance and the knowledge modules import this module, so they are read on first use."""
55 if instance.get("goal_status") != "mathematically_solved":
56 return None
57 group = instance.get("goal_assessment_basis")
58 judgments = [j for j in instance.get("goal_judgments", []) if j.get("basis") == group]
59 if not judgments or any(not j.get("current", True) for j in judgments):
60 return None
61 from . import knowledge
62 if any(not knowledge.validate_basis(j.get("record_basis", []), instance, runtime)["valid"] for j in judgments):
63 return None
64 ids = sorted({ref for j in judgments for ref in j.get("dependencies", []) if ref in instance["knowledge"]})
65 from .knowledge_records import usable_links, all_records
66 if not ids or any(not usable_links(instance, kid) for kid in ids):
67 return None
68 support = all_records(instance)
69 pending, visited, support_ids = list(ids), set(), set()
70 while pending:
71 kid = pending.pop()
72 if kid in visited:
73 continue
74 visited.add(kid)
75 record = instance["knowledge"][kid]
76 support_ids.add(record["kind_definition_id"])
77 pending.extend(record.get("dependencies", []))
78 for link in usable_links(instance, kid):
79 support_ids.update(link.values())
80 for ref in list(support_ids):
81 env = support.get(ref, {}).get("trust_environment_id")
82 if env:
83 support_ids.add(env)
84 records = [support[ref] for ref in sorted(visited | support_ids)]
85 records += [{"id": j["execution_id"], **j} for j in judgments]
86 from .governance import governance_view
87 charter = governance_view(instance).get("charter")
88 goal_ref = (exact_content_hash({"content": {key: value for key, value in charter["content"].items()
89 if key != "academic_requirements"}, "document": charter.get("document")})
90 if charter else str(instance.get("goal_revision", 1)))
91 # Academic requirements have their own edition binding; changing them
92 # alone cannot retroactively change the mathematical solution evidence.
93 return solution_basis(goal_ref + ":" + str(group), records)

136def assess(requirement_id: str, basis: dict, requirements: dict, decision: str, evidence: list[dict]) -> dict:
137 if decision not in {"satisfied", "not_satisfied", "unresolved"} or not isinstance(evidence, list) or not evidence:
138 raise ValueError("assessment needs decision and evidence")
139 requirement = next((item for item in requirements.get("items", []) if item["id"] == requirement_id), None)
140 if requirement is None:
141 raise ValueError("unknown requirement")
142 if any(not isinstance(item, dict) or not isinstance(item.get("criterion"), str) or not isinstance(item.get("satisfied"), bool) or not isinstance(item.get("evidence"), str) for item in evidence):
143 raise ValueError("invalid criteria evidence")
144 criteria = {item["criterion"]: item for item in evidence}
145 if len(criteria) != len(evidence) or set(criteria) != set(requirement["criteria"]):
146 raise ValueError("assessment must cover each exact criterion once")
147 if decision == "satisfied" and any(not item["satisfied"] or not item["evidence"].strip() for item in evidence):
148 raise ValueError("satisfied assessment lacks exact criteria evidence")
149 return {"requirement_id": requirement_id, "solution_basis": deepcopy(basis),
150 "requirements_sha256": requirements["sha256"], "decision": decision,
151 "evidence": deepcopy(evidence), "assessor": "research_plan"}

154def academic_status(instance: dict, runtime: dict, current_basis: dict | None, *, resolve_evidence=None) -> dict:
155 if instance.get("goal_status") != "mathematically_solved":
156 return {"status": "not_applicable", "reason": "unsolved or no requirements"}
157 if current_basis is None:
158 return {"status": "incomplete", "reason": "solution basis unreadable"}
159 requirements = current_requirements(instance)
160 if unresolved_rejections(instance, runtime, current_basis, resolve_evidence=resolve_evidence):
161 return {"status": "externally_rejected", "reason": "unresolved external rejection"}
162 if not requirements.get("items"): return {"status":"not_applicable", "reason":"no academic requirements"}
163 assessments = runtime.get("academic_assessments", [])
164 for requirement in requirements["items"]:
165 matching = [assessment for assessment in assessments if assessment.get("requirement_id") == requirement["id"]
166 and assessment.get("solution_basis") == current_basis
167 and assessment.get("requirements_sha256") == requirements["sha256"]
168 and assessment_current(instance, runtime, assessment)
169 and assessment.get("assessor") == "research_plan" and assessment.get("decision") == "satisfied"]
170 if not matching:
171 break
172 if not any(effect_receipts(requirement, task_id=assessment.get("task_id"), requirements_sha256=requirements["sha256"],
173 solution_basis_value=current_basis, runtime=runtime)[0] for assessment in matching):
174 break
175 else:
176 return {"status": "completed", "reason": "all current requirements satisfied"}
177 tasks = [task for task in runtime.get("tasks", {}).values() if task.get("academic_requirement_id") in {item["id"] for item in requirements["items"]}
178 and task.get("solution_basis") == current_basis and task.get("requirements_sha256") == requirements["sha256"]]
179 started = any(task.get("selected_by") or task.get("results") for task in tasks)
180 return {"status": "active" if started else "pending", "reason": "requirements remain"}

220def record_resolution(instance, runtime, basis, resolution, *, execution_id, resolve_evidence=None):
221 """Seal an actual planner assessment and its currently resolvable evidence."""
222 rejection = next((r for r in runtime.get("external_rejections", []) if r.get("id") == resolution.get("id")), None)
223 execution = runtime.get("executions", {}).get(execution_id, {})
224 if basis is None or rejection is None or rejection.get("solution_basis") != basis: raise ValueError("rejection basis is not current")
225 if execution.get("role") != "research_plan" or execution.get("input_solution_basis") != basis: raise ValueError("planner execution lacks current solution input")
226 if resolution.get("decision") not in {"resolved", "unresolved"} or not isinstance(resolution.get("evidence"), str) or not resolution["evidence"].strip():
227 raise ValueError("resolution requires verdict and actual assessment")
228 refs = distinct_texts(resolution.get("evidence_refs"), "resolution evidence refs")
229 if not refs: raise ValueError("resolution needs resolvable evidence")
230 record = {"rejection_id": rejection["id"], "rejection_sha256": exact_content_hash(rejection), "solution_basis": deepcopy(basis),
231 "requirements_sha256": current_requirements(instance).get("sha256"), "execution_id": execution_id,
232 "decision": resolution["decision"], "evidence": resolution["evidence"],
233 "evidence_refs": refs, "evidence_hashes": {ref:_evidence_hash(evidence_record(instance, runtime, ref, resolve_evidence)) for ref in refs}}
234 record["id"] = "AR-" + exact_content_hash(record)[:24]
235 record["sha256"] = exact_content_hash(record)
236 records = runtime.setdefault("academic_resolution_records", [])
237 if not any(r.get("id") == record["id"] for r in records): records.append(record)
238 return deepcopy(record)

241def unresolved_rejections(instance, runtime, basis, *, resolve_evidence=None):
242 unresolved = []
243 for rejection in runtime.get("external_rejections", []):
244 if rejection.get("solution_basis") != basis: continue
245 matches = [r for r in runtime.get("academic_resolution_records", []) if r.get("rejection_id") == rejection.get("id")]
246 record = matches[-1] if matches else {}
247 try:
248 valid = (bool(record) and record.get("decision") == "resolved" and record.get("solution_basis") == basis
249 and record.get("rejection_sha256") == exact_content_hash(rejection)
250 and record.get("requirements_sha256") == current_requirements(instance).get("sha256")
251 and record.get("sha256") == exact_content_hash({k:v for k,v in record.items() if k != "sha256"})
252 and runtime.get("executions", {}).get(record.get("execution_id"), {}).get("input_solution_basis") == basis
253 and bool(record.get("evidence_refs"))
254 and all(_evidence_hash(evidence_record(instance, runtime, ref, resolve_evidence)) == record["evidence_hashes"][ref] for ref in record["evidence_refs"]))
255 except (ValueError, KeyError): valid = False
256 if not valid: unresolved.append(deepcopy(rejection))
257 return unresolved

284def apply_goal_assurance(instance, *, decision, basis, judgments, runtime=None):
285 """Only mathematical proof judgments can establish the v5 final goal fact."""
286 if decision not in {"unsolved", "mathematically_solved"}: raise ValueError("nonformal final goal assurance is not supported")
287 if decision == "mathematically_solved" and (not basis or not judgments or any(j.get("decision") != "satisfied" or j.get("current") is not True or not j.get("formal_result_refs") for j in judgments)):
288 raise ValueError("mathematical assurance needs current formal-result judgments")
289 if decision == "mathematically_solved":
290 from .knowledge_records import formal_usable
291 from .governance import validate_goal_additional_audits
292 validate_goal_additional_audits(instance, judgments, runtime=runtime)
293 current = [j for j in instance.get("goal_judgments",[]) if j.get("basis") == basis]
294 if current != judgments or any(not j.get("scope_complete") for j in judgments):
295 raise ValueError("final status must consume the complete recorded judgment group")
296 required = {j.get("required") for j in judgments}
297 if required != {len(judgments)} or len({j.get("execution_id") for j in judgments}) != len(judgments) or len({j.get("slot") for j in judgments}) != len(judgments):
298 raise ValueError("final status lacks the required independent judgment slots")
299 for ref in {ref for judgment in judgments for ref in judgment["formal_result_refs"]}:
300 formal = instance.get("formal_results",{}).get(ref)
301 if not formal or not formal_usable(instance,formal): raise ValueError("final goal cites unavailable formal evidence")
302 result = deepcopy(instance)
303 result.update(goal_status=decision, goal_assessment_basis=deepcopy(basis))
304 return result
