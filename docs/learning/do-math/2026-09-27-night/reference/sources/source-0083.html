12def route_failed_action(engine, action, execution_id, *, reason, front=False):
13 """Choose a diagnosis allowed in the current lifecycle, without recursion."""
14 wait = engine.runtime['waiting_actions'][action['id']]
15 if engine.instance.get('lifecycle', {}).get('state') == 'initializing':
16 wait['next_owner'] = 'maintenance'
17 if action['role'] == 'intent_interpretation' and not action.get('diagnosis_attempt'):
18 engine._enqueue('maintenance', execution_id, initial_submission_diagnosis=True, reason=reason)
19 return
20 wait['next_owner'] = 'operational_review'
21 if action['role'] != 'operational_review':
22 engine._enqueue('operational_review', execution_id, front=front, reason=reason)

25def route_submission_return(engine, action, received, execution_id, output, *, rejected_submission=None):
26 """Keep a bounded producer repair, then diagnose the preserved actual error."""
27 retry = {**deepcopy(action), 'continuation_of': execution_id,
28 'repair_attempt': action.get('repair_attempt', 0) + 1,
29 'repair': {'execution_id': execution_id, 'previous_output': output,
30 'retry_bundle_ids': received['retry_bundle_ids'],
31 'bundle_results': received['bundles'], 'issues': received['issues']}}
32 if rejected_submission:
33 retry['repair'].update(deepcopy(rejected_submission))
34 if retry['repair_attempt'] <= engine.instance['config']['max_output_repairs'] and not action.get('diagnosis_attempt'):
35 engine.runtime.get('waiting_actions', {}).pop(action['id'], None)
36 engine.runtime['queue'].append(retry)
37 return
38 engine.runtime.setdefault('waiting_actions', {})[action['id']] = {
39 'action': retry, 'kind': 'output_validation', 'execution_id': execution_id,
40 'reason': 'Producer repair was rejected again; diagnose the preserved contract and receiver errors',
41 'detail': {'issues': deepcopy(received['issues'])}}
42 route_failed_action(engine, action, execution_id,
43 reason='Bounded submission repair exhausted; distinguish producer, input, contract and installation defects')

46def initial_diagnosis_input(engine, action):
47 if engine.instance['lifecycle']['state'] != 'initializing':
48 raise ValueError('initial submission diagnosis requires initialization')
49 execution = engine.runtime['executions'].get(action.get('target'), {})
50 if execution.get('role') != 'intent_interpretation' or execution.get('status') != 'ended':
51 raise ValueError('initial diagnosis must identify an ended interpretation')
52 wait = engine.runtime.get('waiting_actions', {}).get(execution['action']['id'])
53 if not wait or wait.get('kind') not in {'output_validation', 'execution_failure', 'submission_failure'} or wait.get('execution_id') != execution['id']:
54 raise ValueError('initial diagnosis no longer owns this failure')
55 engine._validate_initial_retry(wait, diagnostic_action_id=action.get('id'))
56 return execution, wait

75def continue_initial_retries(engine):
76 if (engine.instance.get('lifecycle', {}).get('state') != 'initializing'
77 or engine.runtime.get('status') in {'paused', 'closed', 'completed'}
78 or engine.runtime.get('user_pause')):
79 return
80 for identifier in list(engine.runtime.get('waiting_actions', {})):
81 wait = engine.runtime['waiting_actions'].get(identifier)
82 if not wait:
83 continue
84 diagnosis = wait.get('diagnosis', {})
85 reviewer = engine.runtime['executions'].get(diagnosis.get('execution_id'), {})
86 if (diagnosis.get('decision') != 'retry' or diagnosis.get('retry_error')
87 or reviewer.get('role') != 'maintenance' or reviewer.get('status') != 'ended'
88 or reviewer.get('outcome') != 'output_accepted'
89 or reviewer.get('action', {}).get('target') != wait.get('execution_id')
90 or not reviewer.get('action', {}).get('initial_submission_diagnosis')):
91 continue
92 try:
93 engine._continue_initial_execution(wait, request_id=reviewer['id'], diagnosis=diagnosis)
94 except (ValueError, OSError, GitStateStoreError) as error:
95 current = engine.runtime.get('waiting_actions', {}).get(identifier)
96 if current:
97 current['diagnosis']['retry_error'] = str(error)
98 current.update(next_owner='maintenance', reason=str(error))
99 engine._save()

117def installation_failure(runtime, record):
118 """Resolve a published independent diagnosis; a caller label is not authority."""
119 current = runtime.get('operational_dispositions', {}).get(record.get('id'))
120 reviewer = runtime.get('executions', {}).get(record.get('execution_id'), {})
121 target = runtime.get('executions', {}).get(record.get('target_execution_id'), {})
122 disposition = record.get('disposition', {})
123 binding = record.get('result', {}).get('installation_repair', {})
124 if (current != record or record.get('status') not in {'installation_repair_pending', 'installation_repair_waiting'}
125 or reviewer.get('role') != 'operational_review' or reviewer.get('outcome') != 'output_accepted'
126 or reviewer.get('status') != 'ended' or reviewer.get('action', {}).get('target') != target.get('id')
127 or target.get('status') != 'ended' or not target.get('error')
128 or disposition.get('kind') != 'repair' or disposition.get('meaning_preserving') is not True
129 or not disposition.get('repair_target', '').startswith('installation:')
130 or not binding.get('installation_sha256') or not binding.get('evidence_refs')):
131 raise ValueError('installation repair requires a preserved accepted diagnosis of an ended failure')
132 return {'kind': 'installation_defect', 'diagnosis_id': record['id'],
133 'error': disposition['reason'], 'repair_target': disposition['repair_target'],
134 'verification': disposition['verification'], 'target_execution_id': target['id'],
135 'failure_sha256': exact_content_hash({'role': target.get('role'),
136 'lineage': target.get('action', {}).get('lineage_id') or target.get('action', {}).get('id'),
137 'target': target.get('action', {}).get('target'),
138 'contract_sha256': target.get('contract_sha256'),
139 'goal_revision': target.get('goal_revision'), 'policy_revision': target.get('policy_revision'),
140 'decision_basis': target.get('decision_basis'),
141 'issues': target.get('integration_result', {}).get('issues') or target['error']}),
142 **{key: deepcopy(binding[key]) for key in ('installation_sha256', 'evidence_refs')}}

145def validate_installation_target(disposition, target, root):
146 if disposition.get('meaning_preserving') is not True:
147 raise ValueError('installation repair cannot change research meaning or authority')
148 root = Path(root)
149 relative = disposition['repair_target'].removeprefix('installation:')
150 path = PurePosixPath(relative)
151 if (not relative or path.is_absolute() or path.as_posix() != relative
152 or any(p in {'.', '..'} or ':' in p or '\\' in p for p in path.parts)
153 or not (root / relative).is_file() or not (root / relative).resolve().is_relative_to(root.resolve())):
154 raise ValueError('installation repair target must name an existing file inside the installed skill')
155 if target.get('status') != 'ended' or not target.get('error'):
156 raise ValueError('installation repair requires a preserved ended failure')
157 refs = [target[key] for key in ('result_ref', 'submission_check_ref', 'submitted_output_ref') if target.get(key)]
158 if not refs:
159 raise ValueError('installation defect has no preserved failure artifacts')
160 return refs

176def resume_installation_repair(runtime, record, *, repaired_by, at):
177 """Resume the preserved rejected response, retaining its repair lineage."""
178 target_id = record['target_execution_id']
179 successor=activities.successor_state(runtime,target_id)
180 if successor is not None:
181 record['status']='applied'
182 record.setdefault('result',{}).setdefault('installation_repair',{}).update(
183 status='repaired',repaired_by=repaired_by,successor=successor)
184 runtime['executions'][target_id].pop('pending_operational_disposition',None)
185 runtime['status']='ready'
186 return None
187 target = runtime['executions'][target_id]
188 previous = runtime.get('waiting_actions', {}).get(target['action']['id'])
189 action = activities.continue_activity(runtime, target_id, at=at,
190 reason='verified_installation_repair:' + record['id'])
191 if previous and previous.get('execution_id') == target_id:
192 # The waiting request owns the latest returned bundles and error report.
193 action.update(deepcopy(previous['action']), continuation_of=target_id,
194 owner_activity_id=target['owner_activity_id'])
195 runtime['waiting_actions'].pop(target['action']['id'], None)
196 record['status'] = 'applied'
197 record.setdefault('result', {})['installation_repair'].update(
198 status='repaired', repaired_by=repaired_by, continuation_action_id=action['id'])
199 target.pop('pending_operational_disposition', None)
200 runtime['status'] = 'ready'
201 return action

204def complete_repair(engine, record):
205 """Hand the current installation diagnosis to the existing recovery owner."""
206 identifier = record["id"]
207 current = engine.runtime.get("operational_dispositions", {}).get(identifier)
208 if current is None or current != record:
209 raise ValueError("repair must consume the current authoritative OD record")
210 record = deepcopy(current)
211 disposition = operational_dispositions.validate_disposition(record["disposition"])
212 if disposition["kind"] != "repair":
213 raise ValueError("complete_repair only consumes repair dispositions")
214 if record['status'] in {'installation_repair_pending', 'installation_repair_waiting', 'applied'}:
215 return record
216 return request_installation_repair(engine, record)
