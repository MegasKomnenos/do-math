196 def test_question_mixin_publishes_advice_without_global_wait_and_resolves_atomically(self):
197 h=bare_engine();h.instance={"id":"I"}
198 h.runtime={"executions":{"E":{"id":"E","role":"research_plan","artifact_bundle":"B", "extensions":{"question_routes":[{"question_index":0,"owner":"user","target_ids":["I"],"blocks":False,"reason":"optional"}]}}}}
199 h._record_output_questions({"role":"research_plan"},{"questions":["Optional priority?"]},"E")
200 self.assertEqual(questions.user_waits({"role":"research_plan"}, h.instance, h.runtime), [])
201 one=questions.link(h.runtime,execution_id="A",field="questions/0",owner="research",targets=["I"])
202 from do_math.repository import ResearchRepository
203 evidence="runtime/artifacts/BA-answer/answer.md"
204 h.runtime["executions"]["E"]["extensions"]={"question_resolutions":[
205 {"question_id":one["id"],"disposition":"resolved","evidence_refs":[evidence],"reason":"answered"},
206 {"question_id":"missing","disposition":"resolved","evidence_refs":[evidence],"reason":"bad"}]}
207 # Artifact evidence resolves through the engine's store read, not a runtime family.
208 h.store=create_autospec(ResearchRepository, instance=True)
209 from do_math.git_store import GitStateStoreError
210 stored={evidence: b"evidence"}
211 def read_many(refs, head, *, missing="raise"):
212 # The store contract: "omit" leaves absent paths out; "raise" fails the batch.
213 if missing == "raise" and any(ref not in stored for ref in refs):
214 raise GitStateStoreError("batch read did not return a blob")
215 return {ref: stored[ref] for ref in refs if ref in stored}
216 h.store.read_many.side_effect=read_many
217 with self.assertRaisesRegex(ValueError, "only distinct open research or operations questions"):
218 h._apply_question_resolutions("E")
219 self.assertEqual(one["status"],"open")
