1"""Sequential shard-by-shard unittest runner with owned, bounded Lean/resource workers.
2
3Tests are selected by ID or module (positional), by shard (``--shard``), or all
4shards when neither is given. Every selected test runs; there is no result cache.
5Shards and the isolation labels live in ``test_categories.py`` beside this runner; a test
6module missing from ``SHARDS`` stops the run. When and what to run is in
7``tests/시험-지침.md``.
8
9Lean and resource tests run per class in an owned Windows Job with a per-phase
10deadline. Normal unittest discovery still works but imposes no outer deadlines.
11This runner never changes the product's Lean time limits.
12"""

56class TimedResult(unittest.TextTestResult):
57 def __init__(self, *args, directory, checkpoint=False, **kwargs):
58 super().__init__(*args, **kwargs)
59 self.directory, self.checkpoint = directory, checkpoint
60 self.rows, self.starts, self.sequence = {}, {}, 0

62 def _progress(self, identifier, phase):
63 if self.checkpoint:
64 self.sequence += 1
65 self.preserve(complete=False)
66 # Immutable publications avoid replacing a file the Windows parent
67 # may currently have open for reading.
68 _write_json(self.directory / f'progress-{self.sequence:06}.json', {
69 'sequence': self.sequence, 'test_id': identifier, 'phase': phase})

71 def startTest(self, test):
72 super().startTest(test)
73 self.starts[test.id()] = time.monotonic()
74 self.rows[test.id()] = {'id': test.id(), 'outcome': 'running', 'seconds': 0, 'details': []}
75 self._progress(test.id(), 'test')

77 def _outcome(self, test, outcome, detail=None):
78 row = self.rows.setdefault(test.id(), {'id': test.id(), 'outcome': outcome,
79 'seconds': 0, 'details': [], 'fixture': True})
80 row['outcome'] = outcome
81 if detail:
82 row['details'].append(detail)
83 if outcome in {'failed', 'error', 'unexpected_success'}:
84 # Publish this failure before the next possibly long test begins.
85 # It is diagnostic evidence, not a completed group result.
86 path = self.directory / 'failures' / (hashlib.sha256(test.id().encode()).hexdigest() + '.json')
87 _write_json(path, {'complete': False, 'event': 'test_failure', 'test': row})
88 self.stream.writeln(' failure evidence: ' + str(path))
89 self.stream.flush()

123 super().addSubTest(test, subtest, error)

128 def stopTest(self, test):
129 self.rows[test.id()]['seconds'] = time.monotonic() - self.starts.pop(test.id())
130 super().stopTest(test)
131 self._progress(None, 'fixture_or_shutdown')

133 def preserve(self, *, complete):
134 report = {'complete': complete, 'successful': complete and self.wasSuccessful(),
135 'tests_run': self.testsRun, 'rows': list(self.rows.values())}
136 _write_json(self.directory / 'result.json', report)

140def _run_suite(tests, directory, *, checkpoint=False):
141 directory.mkdir(parents=True, exist_ok=True)
142 result = unittest.TextTestRunner(verbosity=2, resultclass=lambda *a, **k: TimedResult(
143 *a, directory=directory, checkpoint=checkpoint, **k)).run(unittest.TestSuite(tests))
144 report = result.preserve(complete=True)
145 reported = {row['id'] for row in report['rows']}
146 fixtures = [row for row in report['rows'] if row.get('fixture')]
147 for test in tests:
148 if test.id() in reported:
149 continue
150 scopes = {f'setUpClass ({type(test).__module__}.{type(test).__qualname__})',
151 f'setUpModule ({type(test).__module__})'}
152 skipped = next((row for row in fixtures if row['id'] in scopes and row['outcome'] == 'skipped'), None)
153 report['rows'].append({'id': test.id(), 'outcome': 'skipped' if skipped else 'not_run', 'seconds': 0,
154 'details': skipped['details'] if skipped else ['A fixture failed before this test could run.']})
155 report['successful'] = report['successful'] and all(row['outcome'] in {
156 'passed', 'skipped', 'expected_failure'} for row in report['rows'])
157 _write_json(directory / 'result.json', report)
158 return report

166def _exit_code(job):
167 import ctypes
168 from ctypes import wintypes
169 function = ctypes.WinDLL('kernel32', use_last_error=True).GetExitCodeProcess
170 function.argtypes = [wintypes.HANDLE, ctypes.POINTER(wintypes.DWORD)]
171 function.restype = wintypes.BOOL
172 code = wintypes.DWORD()
173 if not function(job.process_handle, ctypes.byref(code)):
174 raise ctypes.WinError(ctypes.get_last_error())
175 return code.value

178def wait_for_worker(job, progress_path, timeout, *, clock=None, pause=None):
179 """Each test and the surrounding fixture/shutdown phase has its own limit."""
180 clock, pause = clock or time.monotonic, pause or time.sleep
181 phase_started = clock()
182 deadline, sequence, progress = phase_started + timeout, None, {'phase': 'startup_or_fixture', 'test_id': None}
183 while True:
184 published = sorted(progress_path.parent.glob(progress_path.stem + '-*.json'))
185 observed = _read_json(published[-1] if published else progress_path)
186 if observed and observed.get('sequence') != sequence:
187 sequence, progress = observed['sequence'], observed
188 phase_started = clock()
189 deadline = phase_started + timeout
190 if job.wait(0) and _empty(job):
191 return None
192 if clock() >= deadline:
193 elapsed = clock() - phase_started
194 job.terminate()
195 end = clock() + 10
196 while not _empty(job) and clock() < end:
197 pause(.02)
198 return {**progress, 'timeout_seconds': timeout, 'observed_seconds': elapsed,
199 'termination_confirmed': _empty(job)}
200 pause(.05)

203def _isolated(tests, directory, timeout):
204 from do_math.windows_job import WindowsJob
205 directory.mkdir(parents=True)
206 identifiers = [test.id() for test in tests]
207 _write_json(directory / 'requested.json', identifiers)
208 scratch = directory / 'scratch'
209 scratch.mkdir()
210 environment = {**os.environ, 'TMP': str(scratch), 'TEMP': str(scratch), 'TMPDIR': str(scratch)}
211 started = time.monotonic()
212 try:
213 with WindowsJob([sys.executable, '-B', str(Path(__file__).resolve()), '--worker', str(directory)],
214 cwd=directory, env=environment, name='do-math-tests-' + uuid.uuid4().hex) as job:
215 job.resume()
216 timeout_record = wait_for_worker(job, directory / 'progress.json', timeout)
217 returncode = _exit_code(job)
218 report = _read_json(directory / 'result.json') or {'complete': False, 'successful': False, 'rows': []}
219 report['returncode'] = returncode
220 report['successful'] = report['successful'] and returncode == 0
221 except Exception as error:
222 report = _read_json(directory / 'result.json') or {'rows': []}
223 report.update(complete=False, successful=False, worker_error=f'{type(error).__name__}: {error}')
224 timeout_record = None
225 report.update(worker_seconds=time.monotonic() - started, directory=str(directory), requested=identifiers)
226 if timeout_record is not None:
227 report.update(complete=False, successful=False, timeout=timeout_record)
228 # A class-level skip is different from unfinished tests after a worker error.
229 fixtures = [row for row in report['rows'] if row.get('fixture')]
230 skipped_fixture = next((row for row in fixtures if row['outcome'] == 'skipped'), None)
231 completed = {row['id'] for row in report['rows'] if row['outcome'] != 'running'}
232 report['rows'] = [row for row in report['rows'] if row['outcome'] != 'running']
233 for identifier in identifiers:
234 if identifier in completed:
235 continue
236 timed_out = timeout_record is not None and timeout_record.get('test_id') == identifier
237 skipped = report['complete'] and skipped_fixture is not None
238 report['rows'].append({'id': identifier, 'outcome': 'timeout' if timed_out else 'skipped' if skipped else 'not_run',
239 'seconds': timeout_record['observed_seconds'] if timed_out else 0,
240 'details': skipped_fixture['details'] if skipped else [
241 'Worker exceeded its limit.' if timed_out else 'Worker did not complete this test; inspect retained logs.']})
242 report['successful'] = bool(report['successful'] and all(row['outcome'] in {
243 'passed', 'skipped', 'expected_failure'} for row in report['rows']))
244 _write_json(directory / 'worker-report.json', report)
245 return report

248def _worker(directory):
249 sys.path.insert(0, str(TESTS))
250 with (directory / 'stdout.txt').open('w', encoding='utf-8', buffering=1) as out, \
251 (directory / 'stderr.txt').open('w', encoding='utf-8', buffering=1) as err, \
252 redirect_stdout(out), redirect_stderr(err):
253 loader = unittest.TestLoader()
254 suite = loader.loadTestsFromNames(_read_json(directory / 'requested.json'))
255 report = _run_suite(list(flatten(suite)), directory, checkpoint=True)
256 return 0 if report['successful'] else 1
