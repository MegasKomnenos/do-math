109class GitStateStore:
110 """Publish one parent-linked state chain at ``refs/heads/do-math/<id>``.
111
112 ``publish`` uses Git's expected-old-OID ref update as its concurrency
113 guarantee. ``writer_lock`` is a separate, optional local OS lock for
114 callers that need to serialize work before that final CAS; while a caller
115 holds it, no other do-math writer can move the ref.
116
117 Physical boundaries (no reparse ancestor, local fixed NTFS volume, final
118 path) are established when the store is opened and re-established when
119 Git reports an error, which is how a move or remount surfaces. They are
120 not re-proved before every Git process: racing a path swap is deliberate
121 tampering, outside the accidental-change threat model.
122 """
123
124 def __init__(self, repository: str | os.PathLike[str], instance_id: str) -> None:
125 if not isinstance(instance_id, str) or not is_new_id("RI", instance_id):
126 raise ValueError("instance_id must be the registry RI-prefixed canonical ULID")
127 self.repository = Path(repository).absolute()
128 self._repository_boundary = PhysicalBoundary(self.repository)
129 if not self.repository.is_dir():
130 raise ValueError("repository must be an existing Git work tree or bare repository")
131 self.instance_id = instance_id
132 self.ref = f"refs/heads/do-math/{instance_id}"
133 # Until Git reports its directory, the work tree stands in for it in
134 # _git (the disabled hooks path and the boundary re-proved on error).
135 self._git_dir = self.repository
136 self._git_boundary = self._repository_boundary
137 facts = self._git_text("rev-parse", "--absolute-git-dir", "--show-object-format",
138 "--git-path", "objects").splitlines()
139 if len(facts) != 3:
140 raise GitStateStoreError("git rev-parse did not report the Git directory, object format and objects")
141 self._git_dir = Path(facts[0]).absolute()
142 self._git_boundary = PhysicalBoundary(self._git_dir)
143 self.object_format = facts[1].strip()
144 if self.object_format not in {"sha1", "sha256"}:
145 raise GitStateStoreError(f"unsupported Git object format: {self.object_format!r}")
146 self.oid_length = 40 if self.object_format == "sha1" else 64
147 # --git-path is relative to the work tree unless Git reports it absolute
148 # (a linked worktree's common directory, GIT_OBJECT_DIRECTORY).
149 self._objects = self.repository / facts[2].strip()
150 # The trees this store last published and last read (see _tree).
151 self._published: _Tree | None = None
152 self._read: _Tree | None = None
153 # Set only while this store holds the writer lock and its own CAS moved the ref.
154 self._held_head: str | None = None
155 self._holds_writer_lock = False

157 def read_snapshot(self) -> StateSnapshot:
158 """Return the current state commit, checking that it holds research/runtime trees.
159
160 One batch query resolves the ref, its object type and both subtrees.
161 Only when the ref line is missing does ``rev-parse`` separate an absent
162 ref from a ref that names a missing object. Under the writer lock, after
163 this store's own compare-and-swap, no do-math writer can have moved the
164 ref from the commit that CAS installed, so that head is returned unread.
165 """
166 if self._held_head is not None:
167 return StateSnapshot(self._held_head)
168 names = ("research", "runtime")
169 queries = "".join(f"{self.ref}{suffix}\n" for suffix in ("", *(":" + name for name in names)))
170 result = self._git("cat-file", "--batch-check=%(objectname) %(objecttype)",
171 input=queries.encode("ascii"), check=False)
172 rows = result.stdout.decode("ascii").splitlines()
173 root = rows[0].split() if rows else []
174 if not root or root[-1] == "missing":
175 if self._ref_oid() is None:
176 return StateSnapshot(None)
177 detail = result.stderr.decode("utf-8", "replace").strip()
178 raise GitStateStoreError(f"state ref points to an unreadable object: {detail}")
179 if len(root) != 2:
180 raise GitStateStoreError("invalid state commit metadata")
181 head, object_type = root
182 self._validate_oid(head)
183 if object_type != "commit":
184 raise GitStateStoreError("state ref must point to a commit")
185 if result.returncode:
186 detail = result.stderr.decode("utf-8", "replace").strip()
187 raise GitStateStoreError(f"cannot read state subtree metadata: {detail}")
188 self._check_subtrees(rows[1:], names)
189 return StateSnapshot(head)

197 def read_many(self, commit: str, paths: list[str], *, missing: str = "raise") -> dict[str, bytes]:
198 """Read a fixed snapshot in one process, using binary lengths, not lines.
199
200 With ``missing="omit"`` a path absent from the snapshot is left out of
201 the result; with ``"raise"`` it fails the whole batch.
202 """
203 self._validate_oid(commit)
204 self._check_missing_mode(missing)
205 paths = list(dict.fromkeys(paths))
206 for path in paths:
207 self._validate_path(path)
208 if not paths:
209 return {}
210 raw = self._git("cat-file", "--batch", input="".join(
211 f"{commit}:{path}\n" for path in paths).encode("utf-8")).stdout
212 result, offset = {}, 0
213 for path in paths:
214 end = raw.find(b"\n", offset)
215 if missing == "omit" and end >= 0 and raw[offset:end] == f"{commit}:{path} missing".encode("utf-8"):
216 offset = end + 1
217 continue
218 header = raw[offset:end].split() if end >= 0 else []
219 if len(header) != 3 or header[1] != b"blob" or not header[2].isdigit():
220 raise GitStateStoreError("batch read did not return a blob: " + path)
221 self._validate_oid(header[0].decode("ascii"))
222 size = int(header[2])
223 start, offset = end + 1, end + 1 + size
224 if raw[offset:offset + 1] != b"\n":
225 raise GitStateStoreError("truncated batch blob: " + path)
226 result[path] = raw[start:offset]
227 offset += 1
228 if offset != len(raw):
229 raise GitStateStoreError("unexpected trailing batch output")
230 return result

232 def blob_sizes(self, commit: str, paths: list[str], *, missing: str = "raise") -> dict[str, int]:
233 """Read object lengths once so callers can bound batch-read memory.
234
235 ``missing`` has the meaning it has for :meth:`read_many`.
236 """
237 self._validate_oid(commit)
238 self._check_missing_mode(missing)
239 for path in paths:
240 self._validate_path(path)
241 if not paths:
242 return {}
243 output = self._git('cat-file', '--batch-check=%(objectsize) %(objecttype)',
244 input=''.join(f'{commit}:{path}\n' for path in paths).encode()).stdout.splitlines()
245 if len(output) != len(paths):
246 raise GitStateStoreError('batch size response count differs')
247 result = {}
248 for path, row in zip(paths, output):
249 if missing == "omit" and row == f"{commit}:{path} missing".encode("utf-8"):
250 continue
251 fields = row.split()
252 if len(fields) != 2 or not fields[0].isdigit() or fields[1] != b'blob':

262 def _write_object(self, kind: bytes, body: bytes) -> str:
263 """Store one loose object as installed Git does by default, returning its OID.
264
265 The OID hashes ``"<kind> <size>\\0"`` and the exact body, so writing is
266 idempotent: an existing loose object is kept, and a concurrent writer of
267 the same object only loses the final rename. Like Git for Windows
268 (``core.fsyncObjectFiles=true`` by default) each new object is flushed
269 before its final name appears; the directory is not, as Git does not.
270 """
271 data = b"%s %d\0" % (kind, len(body)) + body
272 oid = hashlib.new(self.object_format, data).hexdigest()
273 target = self._objects / oid[:2] / oid[2:]
274 if target.exists():
275 return oid
276 target.parent.mkdir(exist_ok=True)
277 # Git's prefix, so ``git prune`` removes a temporary a crash leaves behind.
278 descriptor, temporary = tempfile.mkstemp(prefix="tmp_obj_", dir=target.parent)
279 try:
280 with os.fdopen(descriptor, "wb") as stream:
281 stream.write(zlib.compress(data, 1)) # core.looseCompression default
282 stream.flush()
283 os.fsync(stream.fileno())
284 os.chmod(temporary, 0o444) # Git's mode for loose objects (read-only on Windows)
285 try:
286 os.rename(temporary, target)
287 except FileExistsError:
288 pass # Windows: another writer stored the identical object first
289 finally:
290 if os.path.lexists(temporary):
291 os.chmod(temporary, stat.S_IREAD | stat.S_IWRITE)
292 os.unlink(temporary)
293 return oid

295 def _write_tree(self, parent: Mapping[str, Mapping[str, tuple[str, str]]],
296 changes: Mapping[str, bytes | None]) -> tuple[str, dict[str, dict[str, tuple[str, str]]]]:
297 """Write each changed blob and the trees above it; other subtrees keep their OIDs.
298
299 Returns the root tree OID and the new directory map. A directory left
300 empty disappears from its parent, as it does from Git's index.
301 """
302 directories = dict(parent)
303 touched: set[str] = set()
304
305 def own(directory: str) -> dict[str, tuple[str, str]]:
306 if directory not in touched:
307 touched.add(directory)
308 directories[directory] = dict(directories.get(directory, {}))
309 return directories[directory]
310
311 for path, contents in changes.items():
312 directory, _, name = path.rpartition("/")
313 entries = own(directory)
314 if contents is not None:
315 entries[name] = (_FILE_MODE, self._write_object(b"blob", contents))
316 elif entries.get(name, ("",))[0] != _TREE_MODE:
317 # As in Git's index, deleting a directory's own path removes nothing.
318 entries.pop(name, None)
319 while directory:
320 directory = directory.rpartition("/")[0]
321 own(directory)
322 # Deepest first, so each parent sees its children already rewritten or emptied.
323 for directory in sorted(touched - {""}, key=lambda path: path.count("/"), reverse=True):
324 above, _, name = directory.rpartition("/")
325 if directories[directory]:
326 directories[above][name] = (_TREE_MODE, self._write_tree_object(directories[directory]))
327 else:
328 del directories[directory]
329 if directories[above].get(name, ("",))[0] == _TREE_MODE:
330 del directories[above][name]
331 return self._write_tree_object(directories[""]), directories

333 def _write_tree_object(self, entries: Mapping[str, tuple[str, str]]) -> str:
334 # Git orders entries by name bytes, a subtree's name as if it ended in "/".
335 rows = sorted((name.encode("utf-8") + (b"/" if mode == _TREE_MODE else b""), name, mode, oid)
336 for name, (mode, oid) in entries.items())
337 return self._write_object(b"tree", b"".join(
338 f"{mode} {name}\0".encode("utf-8") + bytes.fromhex(oid) for _, name, mode, oid in rows))

350 def _compare_and_swap(self, new: str, old: str | None, operation: str) -> None:
351 """Move the ref by Git's expected-old-OID update, the only ref move."""
352 updated = self._git("update-ref", self.ref, new, old or "0" * self.oid_length, check=False)
353 if updated.returncode:
354 self._held_head = None
355 raise GitStateStoreConflict(f"state ref changed before the {operation} compare-and-swap")
356 self._held_head = new if self._holds_writer_lock else None

358 @measured('git_publication')
359 def publish(self, expected_head: str | None, changes: Mapping[str, bytes | None]) -> StateSnapshot:
360 """Create and atomically publish one complete state commit.
361
362 ``expected_head`` is ``None`` only for the parentless initial commit.
363 Every later commit has exactly that expected commit as its only parent.
364 A failed expected-old-OID check raises :class:`GitStateStoreConflict`
365 and leaves the authoritative ref unchanged. The final update-ref CAS is
366 the only ref comparison here; callers that want to avoid work against a
367 stale head compare it first (``ResearchRepository`` does).
368 """
369 if expected_head is not None:
370 self._validate_oid(expected_head)
371 normalized_changes = self._validate_changes(changes)
372
373 parent = self._tree(expected_head) if expected_head is not None else None
374 existing_paths = parent.paths if parent is not None else frozenset()
375 self._validate_artifact_changes(existing_paths, normalized_changes)
376 candidate_paths = set(existing_paths)
377 for path, contents in normalized_changes.items():
378 if contents is None:
379 candidate_paths.discard(path)
380 else:
381 candidate_paths.add(path)
382 self._validate_candidate_paths(candidate_paths)
383 if expected_head is not None and not normalized_changes:
384 # A no-op still compares the authoritative ref at the publication
385 # boundary, but creates neither blobs nor an empty history commit.
386 self._compare_and_swap(expected_head, expected_head, "no-op")
387 return StateSnapshot(expected_head)
388 # The candidate path list decides both subtrees without reading the written tree.
389 if not any(path.startswith("research/") for path in candidate_paths):
390 raise ValueError("the state must contain a research subtree")
391 if not any(path.startswith("runtime/") for path in candidate_paths):
392 raise ValueError("the state must contain a runtime subtree")
393
394 # Objects are written in this process; only the ref move below runs Git.
395 tree, directories = self._write_tree(parent.directories if parent is not None else {"": {}},
396 normalized_changes)
397 commit = self._write_commit(tree, expected_head)
398 self._compare_and_swap(commit, expected_head, "publish")
399 # The new tree is known entry by entry, so it is never enumerated (ls-tree).
400 self._published = _Tree(commit, directories, frozenset(candidate_paths))
401 return StateSnapshot(commit)

405 """Acquire a non-stale, same-instance advisory OS lock in the Git dir."""

445 def _tree(self, commit: str) -> _Tree:
446 """The tree of ``commit``: kept from this store's last publication or read, else one ls-tree.
447
448 Two slots keep reading an older snapshot from displacing the head the
449 next publication builds on.
450 """
451 self._validate_oid(commit)
452 for known in (self._published, self._read):
453 if known is not None and known.commit == commit:
454 return known
455 directories: dict[str, dict[str, tuple[str, str]]] = {"": {}}
456 paths: set[str] = set()
457 for record in self._git("ls-tree", "-r", "-t", "-z", commit).stdout.split(b"\0"):
458 if not record:
459 continue
460 fields, _, raw_path = record.partition(b"\t")
461 mode, kind, oid = fields.decode("ascii").split(" ")
462 try:
463 path = raw_path.decode("utf-8")
464 except UnicodeDecodeError as error:
465 raise GitStateStoreError("existing state contains a non-UTF-8 path") from error
466 if kind == "tree":
467 directories.setdefault(path, {})
468 else:
469 self._validate_path(path)
470 paths.add(path)
471 above, _, name = path.rpartition("/")
472 # ls-tree prints a subtree's mode zero-padded (040000); the tree object stores 40000.
473 directories.setdefault(above, {})[name] = (mode.lstrip("0"), oid)
474 self._read = _Tree(commit, directories, frozenset(paths))
475 return self._read

477 def _validate_changes(self, changes: Mapping[str, bytes | None]) -> dict[str, bytes | None]:
478 if not isinstance(changes, Mapping):
479 raise TypeError("changes must be a mapping of paths to bytes or None")
480 normalized: dict[str, bytes | None] = {}
481 folded: dict[str, str] = {}
482 for path, contents in changes.items():
483 self._validate_path(path)
484 folded_path = path.casefold()
485 prior = folded.get(folded_path)
486 if prior is not None and prior != path:
487 raise ValueError(f"Windows case-colliding paths: {prior!r} and {path!r}")
488 folded[folded_path] = path
489 if contents is not None and not isinstance(contents, bytes):
490 raise TypeError("change contents must be bytes or None")
491 normalized[path] = contents
492 return normalized

494 def _validate_candidate_paths(self, paths: set[str]) -> None:
495 folded_prefixes: dict[str, str] = {}
496 directories: set[str] = set()
497 for path in paths:
498 self._validate_path(path)
499 parts = path.split("/")
500 for position in range(1, len(parts) + 1):
501 original = "/".join(parts[:position])
502 key = original.casefold()
503 prior = folded_prefixes.get(key)
504 if prior is not None and prior != original:
505 raise ValueError(f"Windows case-colliding path segments: {prior!r} and {original!r}")
506 folded_prefixes[key] = original
507 if position < len(parts):
508 directories.add(original)
509 # The tree writer would let the subtree replace the file entry of the same name, silently dropping the file.
510 conflicts = directories.intersection(paths)
511 if conflicts:
512 raise ValueError("state path is both a file and a directory: " + min(conflicts))

620class PhysicalBoundary:
621 """A root proved physical once, when the boundary is opened.
622
623 ``check`` proves the root: no reparse ancestor, the same local fixed NTFS
624 volume, and a directory handle whose final path is the lexical root.
625 Opening a file below the root only rejects reparse points between the root
626 and the file. Handles are not re-verified after opening: swapping a path
627 between check and use is deliberate tampering, not an accidental change.
628 Does not assert an unspecified ACL policy.
629 """
630 def __init__(self, root):
631 self.root = Path(os.path.abspath(root))
632 no_reparse(self.root)
633 if not self.root.is_dir():
634 raise ValueError("physical boundary root must be an existing directory")
635 self.device = self.root.stat().st_dev
636 self.check()
637
638 def check(self):
639 no_reparse(self.root)
640 if self.root.stat().st_dev != self.device:
641 raise ValueError("physical root volume changed")
642 if os.name == "nt":
643 import ctypes
644 from ctypes import wintypes
645 kernel = ctypes.WinDLL("kernel32", use_last_error=True)
646 create = kernel.CreateFileW
647 create.argtypes = [wintypes.LPCWSTR, wintypes.DWORD, wintypes.DWORD, ctypes.c_void_p,
648 wintypes.DWORD, wintypes.DWORD, wintypes.HANDLE]
649 create.restype = wintypes.HANDLE
650 close = kernel.CloseHandle
651 close.argtypes = [wintypes.HANDLE]
652 handle = create(str(self.root), 0, 3, None, 3, 0x02000000 | 0x00200000, None)
653 if handle == wintypes.HANDLE(-1).value:
654 raise ctypes.WinError(ctypes.get_last_error())
655 try:
656 final, volume = _windows_handle_facts(handle)
657 if os.path.normcase(str(final)) != os.path.normcase(str(self.root)):
658 raise ValueError("physical directory handle resolves elsewhere")
659 if hasattr(self, "volume") and volume != self.volume:
660 raise ValueError("physical volume identity changed")
661 self.volume = volume
662 finally:
663 close(handle)

665 @contextmanager
666 def open(self, path, mode):
667 if mode not in {"rb", "xb"}:
668 raise ValueError("physical boundary supports read or exclusive creation")
669 path = Path(os.path.abspath(path))
670 if not path.is_relative_to(self.root):
671 raise ValueError("file path is outside the physical boundary")
672 # The root and its ancestors were proved when this boundary opened.
673 no_reparse(path, below=self.root)
674 with path.open(mode) as stream:
675 yield stream
