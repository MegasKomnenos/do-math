100 def test_combined_preparation_preserves_user_terms_and_publishes_fresh_review(self):
101 engine = self.prepare()
102 before = deepcopy(engine.runtime['tasks'])
103 action = next(a for a in engine.runtime['queue'] if a.get('observation_rule_targets'))
104 self.assertEqual(set(action['assessment_targets']), set(action['observation_rule_targets']))
105 failed_id = 'AC-old-rules'
106 failed_action = {'id': failed_id, 'role': 'research_work', 'target': next(iter(before))}
107 activity = engine._activity_for_action(failed_action)
108 from do_math.activities import wait_for
109 wait_for(engine.runtime, activity['id'], reason='missing rules', condition={'kind': 'observation_rules'},
110 next_owner='research_preparation', at=engine._lifecycle_now().isoformat())
111 engine.runtime['waiting_actions'][failed_id] = {'action': failed_action,
112 'kind': 'observation_rules', 'reason': 'missing rules'}
113 engine._save()
114 engine.runner = PurposeRunner()
115 engine.run(max_steps=1)
116 self.assertNotIn(failed_id, engine.runtime['waiting_actions'])
117 self.assertEqual(engine.runtime['activities'][activity['id']]['status'], 'finished')
118 for tid, task in engine.runtime['tasks'].items():
119 for key in ('question', 'scope', 'expected', 'completion_obligations', 'academic_requirement_id', 'authority_refs'):
120 self.assertEqual(task[key], before[tid][key])
121 self.assertEqual(task['revision'], before[tid]['revision'] + 1)
122 self.assertTrue(task['initial_observation_rules'])
123 self.assertTrue(task['model_assessment'])
124 reviews = [a for a in engine.runtime['queue'] if a['role'] == 'task_contract_review' and a.get('target') == tid]
125 self.assertEqual(len(reviews), 1)
126 self.assertEqual(reviews[0]['contract_basis']['revision'], task['revision'])
127 self.assertEqual(engine.runtime['cycles'], 0)
128 self.assertTrue(any(h['kind'] == 'observation_rules_resolved' for h in engine.runtime['history']))
