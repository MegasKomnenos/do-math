21function Task-Arguments([string]$Mode, [string]$Bootstrap) {
22 $bootstrapMode = if ($Mode -eq 'engine') { 'run-engine' } else { $Mode }
23 return '"' + $Bootstrap + '" ' + $bootstrapMode + ' --repository "' + $Repository + '" --instance "' + $Instance + '" --workspace "' + $Workspace + '"'

26function Task-View([string]$Name) {
27 $task = Get-ScheduledTask -TaskName $Name -ErrorAction SilentlyContinue
28 if ($null -eq $task) { return $null }
29 $info = Get-ScheduledTaskInfo -TaskName $Name
30 $action = @($task.Actions)[0]
31 $trigger = @($task.Triggers)[0]
32 return [ordered]@{
33 name = $Name; state = [string]$task.State; action = [string]$action.Execute + ' ' + [string]$action.Arguments
34 interval = if ($null -eq $trigger) { $null } else { [string]$trigger.Repetition.Interval }; multiple_instances = [string]$task.Settings.MultipleInstances
35 execution_limit = [string]$task.Settings.ExecutionTimeLimit
36 next_run = if ($null -eq $info.NextRunTime) { $null } else { $info.NextRunTime.ToUniversalTime().ToString('o') }; last_result = $info.LastTaskResult; description = [string]$task.Description
37 }
