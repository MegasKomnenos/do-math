11# Acceptance receipt kinds. A control submission's receipt keeps its request
12# kind (control_transport.SUBMIT_INTENT); the others name their own receiver.
13NATIVE_FINAL_RECEIPT = "native_final"
14MACHINE_OUTPUT_RECEIPT = "machine_output"
15RECOVERY_REPORT_RECEIPT = "recovery_report"
16# Receipts that accept an AI role's sealed submission.
17SUBMISSION_RECEIPT_KINDS = frozenset({SUBMIT_INTENT, NATIVE_FINAL_RECEIPT})
18EXECUTION_STATUSES = frozenset({"queued", "starting", "running", "stopping", "ended"})
19STOP_PHASES = frozenset({"terminating"})
20OUTCOMES = frozenset({"output_accepted", "execution_incomplete", "cancelled", "start_failed"})
21_EDGES = {"queued": {"starting", "ended"}, "starting": {"running", "stopping"},
22 "running": {"stopping"}, "stopping": {"ended"}, "ended": set()}

25def create_execution(*, execution_id: str, owner_activity_id: str, actor_kind: str,
26 input_manifest: Any, profile_ref: Any, created_at: str,
27 continuation_of: str | None = None,
28 context_origin: list[str] | None = None) -> dict:
29 """Create a queued attempt. Continuation is explicit, never inferred."""
30 require_text(execution_id, "execution_id")
31 require_text(owner_activity_id, "owner_activity_id")
32 require_text(created_at, "created_at")
33 if actor_kind not in {"ai", "machine"}:
34 raise ValueError("actor_kind must be ai or machine")
35 if input_manifest is None or profile_ref is None:
36 raise ValueError("input_manifest and profile_ref are required")
37 if continuation_of is not None:
38 require_text(continuation_of, "continuation_of")
39 if continuation_of == execution_id:
40 raise ValueError("an execution cannot continue itself")
41 if context_origin is not None and (not isinstance(context_origin, list) or
42 any(not isinstance(item, str) or not item for item in context_origin)):
43 raise ValueError("context_origin must be a list of execution IDs")
44 return {"id": execution_id, "owner_activity_id": owner_activity_id, "actor_kind": actor_kind,
45 "input_manifest": deepcopy(input_manifest), "execution_profile_ref": deepcopy(profile_ref),
46 "created_at": created_at, "status": "queued", "continuation_of": continuation_of,
47 "context_origin": deepcopy(context_origin or [])}

50def transition_execution(execution: dict, target: str, *, at: str, reason: str | None = None,
51 stop_phase: str | None = None, outcome: str | None = None,
52 disposition: Any = None) -> dict:
53 """Apply one legal transition; terminal outcomes do not encode activity meaning."""
54 current = execution.get("status")
55 if current not in EXECUTION_STATUSES or target not in EXECUTION_STATUSES:
56 raise ValueError("execution status is invalid")
57 require_text(at, "at")
58 if target != current and target not in _EDGES[current]:
59 raise ValueError(f"illegal execution transition {current} -> {target}")
60 if target == "stopping":
61 if stop_phase not in STOP_PHASES:
62 raise ValueError("stopping requires the terminating stop phase")
63 elif stop_phase is not None:
64 raise ValueError("stop_phase exists only while stopping")
65 if target == "ended":
66 if outcome not in OUTCOMES:
67 raise ValueError("ended requires a closed outcome")
68 if outcome == "output_accepted" and not execution.get("submission_receipt"):
69 raise ValueError("accepted output requires a recorded submission receipt")
70 if current == "queued" and outcome not in {"cancelled", "start_failed"}:
71 raise ValueError("a queued execution can end only as cancelled or start_failed")
72 elif outcome is not None:
73 raise ValueError("outcome exists only at ended")
74 if target == current:
75 return execution
76 execution["status"] = target
77 execution.setdefault("transitions", []).append({"from": current, "to": target, "at": at,
78 "reason": reason, "stop_phase": stop_phase})
79 if target == "stopping":
80 execution["stop_phase"] = stop_phase
81 elif target == "ended":
82 execution.pop("stop_phase", None)
83 # A recovery disposition governs an unfinished execution; the outcome replaces it.
84 execution.pop("recovery_disposition", None)
85 execution.update(ended_at=at, outcome=outcome, result_disposition=deepcopy(disposition))
86 return execution

89def record_submission_intent(execution: dict, intent: dict, receipt: dict, *, at: str | None = None) -> dict:
90 """Stage receipt, termination phase and execution authority in one candidate.
91
92 The central caller saves this candidate and its control receipt in one CAS
93 before transport acceptance. Collection of the declared output remains
94 possible; the reusable grant is not revoked by this execution-local fact.
95 """
96 if (not isinstance(intent, dict) or not isinstance(intent.get("candidate_id"), str)
97 or not isinstance(intent.get("output_ref"), str)):
98 raise ValueError("submission intent requires a centrally sealed candidate")
99 if not isinstance(receipt, dict) or receipt.get("kind") not in SUBMISSION_RECEIPT_KINDS or receipt.get("state") != "accepted":
100 raise ValueError("submission intent requires an accepted control receipt")
101 if receipt.get("execution_id") != execution.get("id"):
102 raise ValueError("submission receipt belongs to another execution")
103 previous = execution.get("submission_intent")
104 if previous is not None:
105 if previous == intent and execution.get("submission_receipt") == receipt:
106 return execution
107 raise ValueError("a different submission intent or receipt was already recorded")
108 recovery = (receipt.get('kind') == NATIVE_FINAL_RECEIPT and receipt.get('recovery_request_id')
109 and execution.get('status') == 'ended' and execution.get('outcome') == 'execution_incomplete'
110 and local_termination_confirmed(execution))
111 if execution.get("status") != "running" and not recovery:
112 raise ValueError("submission intent requires a running execution or explicit terminated recovery")
113 at = require_text(at or receipt.get("received_at"), "submission receipt time")
114 receipt_id = require_text(receipt.get("id") or receipt.get("request_id"), "receipt identity")
115 candidate = deepcopy(execution)
116 candidate["submission_intent"] = deepcopy(intent)
117 candidate["submission_receipt"] = deepcopy(receipt)
118 if not recovery:
119 transition_execution(candidate, "stopping", at=at, reason="submission_intent", stop_phase="terminating")
120 candidate["authority_revocation"] = {"scope": "execution_research", "reason": "submission_intent",
121 "at": at, "receipt_id": receipt_id}
122 execution.clear()
123 execution.update(candidate)
124 return execution

127def begin_preparation(execution: dict, *, at: str, stage: str = "context") -> dict:
128 """Record the reached preparation stage while this attempt is still queued.
129
130 ENGINE registers and saves the queued record before preparing private input.
131 This function does not publish files or infer that any input is complete.
132 """
133 if execution.get("status") != "queued":
134 raise ValueError("input preparation requires a queued execution")
135 require_text(at, "at")
136 require_text(stage, "stage")
137 previous = execution.get("input_preparation", {})
138 execution["input_preparation"] = {"stage": stage, "started_at": previous.get("started_at", at),
139 "updated_at": at}
140 return execution

143def end_preparation(execution: dict, *, at: str, stage: str, reason: str, cancelled: bool) -> dict:
144 """Preserve failure/interruption facts before private input is discarded."""
145 if type(cancelled) is not bool:
146 raise ValueError("cancelled must be boolean")
147 require_text(reason, "reason")
148 candidate = deepcopy(execution)
149 begin_preparation(candidate, at=at, stage=stage)
150 if candidate.get("capacity"):
151 raise ValueError("admitted preparation must reconcile reservations and process before ending")
152 candidate["input_preparation"].update(finished_at=at, reason=reason)
153 transition_execution(candidate, "ended", at=at, reason=reason,
154 outcome="cancelled" if cancelled else "start_failed",
155 disposition={"kind": "input_preparation", "stage": stage, "reason": reason})
156 execution.clear()
157 execution.update(candidate)
158 return execution

161def drive_to_termination(execution: dict, *, at: str, reason: str) -> dict:
162 """Move a starting or running execution to the terminating barrier.
163
164 Queued, ended and already stopping executions are unchanged.
165 """
166 if execution.get("status") in {"starting", "running"}:
167 transition_execution(execution, "stopping", at=at, reason=reason, stop_phase="terminating")
168 return execution

171def release_or_hold(execution: dict, slots, *, at: str) -> bool:
172 """Return the execution's local capacity, or hold it at the termination barrier.
173
174 Without confirmed local termination the execution stays stopping and keeps
175 its capacity and output for recovery; an error the call already reported is
176 kept. The caller preserves outputs and saves.
177 """
178 if not execution.get("capacity") or slots.release(execution["id"]):
179 return True
180 transition_execution(execution, "stopping", at=at, stop_phase="terminating",
181 reason="local Job termination remains unconfirmed")
182 if not execution.get("error"):
183 execution["error"] = "Local Job termination is unconfirmed; capacity and output remain held for recovery."
184 return False

187def recover_execution(execution: dict, *, at: str) -> dict:
188 """Move a started execution that a new executor found unfinished to its barrier.
189
190 Its state alone decides the disposition. This function never probes a
191 provider or guesses remote completion; ``end_recovered`` ends it once the
192 caller has confirmed the local termination barrier.
193 """
194 require_text(at, "at")
195 state = execution.get("status")
196 if state == "starting":
197 drive_to_termination(execution, at=at, reason="recovery cannot establish start")
198 execution["recovery_disposition"] = "preserve_space_and_reservations"
199 elif state == "running":
200 drive_to_termination(execution, at=at, reason="previous engine lost")
201 execution["recovery_disposition"] = "reconcile_local_and_remote"
202 elif state == "stopping":
203 execution["recovery_disposition"] = "preserve_space_and_reservations"
204 else:
205 raise ValueError(f"a {state} execution has no recovery disposition")
206 return execution

209def end_recovered(execution: dict, *, at: str, outcome: str, disposition: Any) -> dict:
210 """End an unfinished execution whose local termination barrier recovery confirmed."""
211 drive_to_termination(execution, at=at, reason="recovery local termination reconciliation")
212 return transition_execution(execution, "ended", at=at, reason="recovery confirmed termination barrier",
213 outcome=outcome, disposition=disposition)
