87 def test_all_managed_subprocess_calls_explicitly_prevent_console_creation(self):
88 calls = []
89 for path in (SCRIPTS / 'do_math').glob('*.py'):
90 for node in ast.walk(ast.parse(path.read_text(encoding='utf-8'))):
91 if not isinstance(node, ast.Call):
92 continue
93 function = node.func
94 direct = (isinstance(function, ast.Attribute)
95 and isinstance(function.value, ast.Name)
96 and function.value.id == 'subprocess'
97 and function.attr in {'run', 'Popen'})
98 injected = (path.name == 'scheduled_operator.py'
99 and isinstance(function, ast.Name) and function.id == 'runner')
100 if direct or injected:
101 calls.append((path.name, node.lineno))
102 self.assertIn('creationflags', {k.arg for k in node.keywords}, calls[-1])
103 # The scan must reach every launching module. Both providers share one
104 # capability probe (codex_adapter) and one worker launch (cli_worker.run_cli),
105 # so the product has ten call sites: cli 2, codex_adapter 1, cli_worker 1,
106 # formal 2, git_store 1, scheduled_operator 3.
107 self.assertGreaterEqual(len(calls), 10)
