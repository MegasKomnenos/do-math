20def compete(root, iid, ready, release, output, name):
21 repository = ResearchRepository(Path(root), iid)
22 acquired = False
23 try:
24 if not ready.wait(120):
25 raise AssertionError("probe race was not released")
26 deadline = time.monotonic() + 120
27 while True:
28 try:
29 acquired = transact(repository, lambda slots, instance, runtime:
30 admit(slots, instance, runtime, name, current=500))
31 break
32 except GitStateStoreLockError:
33 if time.monotonic() >= deadline:
34 raise
35 time.sleep(.02)
