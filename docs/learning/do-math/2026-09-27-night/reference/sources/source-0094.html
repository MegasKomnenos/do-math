14def _validate(value: dict[str, Any], *, allow_relative_paths: bool) -> dict[str, Any]:
15 if not isinstance(value, dict) or set(value) != {"command", "args"}:
16 raise ValueError("scholarly server must contain exactly command and args")
17 command, args = value["command"], value["args"]
18 if not isinstance(command, str) or not command or "\x00" in command:
19 raise ValueError("scholarly command must be a nonempty NUL-free string")
20 if not isinstance(args, list) or any(not isinstance(arg, str) or "\x00" in arg for arg in args):
21 raise ValueError("scholarly args must be NUL-free strings")
22 if any(arg == "--data-dir" or arg.startswith("--data-dir=") for arg in args):
23 raise ValueError("scholarly args must not override the execution data directory")
24 if not allow_relative_paths:
25 if _pathlike(command) and not Path(command).is_absolute():
26 raise ValueError("relative scholarly command needs a configuration-file base")
27 if args and _pathlike(args[0]) and not Path(args[0]).is_absolute():
28 raise ValueError("relative scholarly server file needs a configuration-file base")
29 return {"command": command, "args": list(args)}

47def _from_config(path: Path) -> dict[str, Any] | None:
48 try:
49 with path.open("rb") as stream:
50 parsed = tomllib.load(stream)
51 except tomllib.TOMLDecodeError as error:
52 raise ValueError(f"invalid Codex configuration at {path}") from error
53 table = parsed.get("mcp_servers")
54 entry = table.get("scholarly") if isinstance(table, dict) else None
55 if entry is None:
56 return None
57 if not isinstance(entry, dict):
58 raise ValueError("scholarly server configuration must be a table")
59 # Codex's server table may also carry client timeout settings. They remain
60 # owned by the parent configuration and are deliberately not inherited.
61 server = _validate({key: entry.get(key) for key in ("command", "args")}, allow_relative_paths=True)
62 if _pathlike(server["command"]) and not Path(server["command"]).is_absolute():
63 server["command"] = _absolute_from_config(server["command"], path.parent, label="command")
64 if server["args"] and _pathlike(server["args"][0]) and not Path(server["args"][0]).is_absolute():
65 server["args"][0] = _absolute_from_config(server["args"][0], path.parent, label="server file")
66 return validate_scholarly_server(server)
