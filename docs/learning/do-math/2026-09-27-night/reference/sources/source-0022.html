125 def _academic_control(self, request):
126 change = request.get("config", {}).get("change", {})
127 if request.get("action") != "decision" or change.get("kind") != "academic_rejection": return False
128 basis = self._solution_basis()
129 if basis is None: raise ValueError("external rejection requires a current solved mathematical goal")
130 if not all(isinstance(change.get(key), str) and change[key].strip() for key in ("actor", "text")):
131 raise ValueError("external rejection needs its actor and preserved original text")
132 entry = {"id": new_id("RJ"), "actor": change["actor"], "text": change["text"], "source_ref": request["id"],
133 "created_at": request["created_at"], "solution_basis": basis, "resolved": False}
134 self.runtime.setdefault("external_rejections", []).append(entry)
135 if self.runtime["status"] == "completed": self.runtime["status"] = "ready"
136 self._enqueue("research_plan", reason="Examine the external rejection against the preserved solution evidence", rejection_id=entry["id"])
137 return True

139 def _record_academic_assessment(self, task, assessment, execution_id):
140 if not task.get("academic_requirement_id"): return

146 actor = self.runtime.get("executions", {}).get(execution_id, {})
147 if (actor.get("role") != "research_plan" or actor.get("input_solution_basis") != task["solution_basis"]
148 or applied.get("execution_id") != execution_id
149 or any(applied.get(key) != value for key, value in assessment.items())
150 or not assessment_current(task, self.instance, self.runtime)):

153 requirement = next(item for item in academic.current_requirements(self.instance)["items"]
154 if item["id"] == task["academic_requirement_id"])
155 expected = academic.criterion_obligations(task["id"], requirement)
156 actual = {row["id"]: row for row in task["completion_obligations"]}
157 if any(actual.get(row["id"]) != row for row in expected):
158 raise ValueError("academic criteria differ from the approved requirement obligations")
159 dispositions = {row["obligation_id"]: row for row in assessment["item_dispositions"]}
160 evidence = [{"criterion": row["claim"],
161 "satisfied": dispositions.get(row["id"], {}).get("kind") == "fulfilled",
162 "evidence": ", ".join(dispositions.get(row["id"], {}).get("evidence_refs", []))}
163 for row in expected]

165 else "unresolved" if assessment.get("unresolved") else "not_satisfied")

167 requirements_sha256=task["requirements_sha256"], solution_basis_value=task["solution_basis"], runtime=self.runtime)
168 if decision == "satisfied" and not receipts_ok:
169 raise ValueError("satisfied academic assessment lacks confirmed successful required external effects")
170 result = academic.assess(task["academic_requirement_id"], task["solution_basis"], academic.current_requirements(self.instance),
171 decision, evidence)
172 result.update(execution_id=execution_id, task_id=task["id"], external_effect_receipts=receipts,
173 task_assessment_sha256=exact_content_hash(applied))
