24_EFFORTS = frozenset({"minimal", "low", "medium", "high", "xhigh", "max", "ultra"})
25_SANDBOXES = frozenset({"read-only", "workspace-write", "danger-full-access"})
26_WINDOWS_SANDBOXES = frozenset({"unelevated", "elevated"})

29def sanitized_environment(source: Mapping[str, str] | None = None) -> dict[str, str]:
30 """Keep ChatGPT login state while preventing a CLI API-key route."""
31 return allowed_environment(source, frozenset({"CODEX_HOME"}))

34def load_request(path: Path) -> dict[str, Any]:
35 value = check_common_request(strict_json_loads(path.read_text(encoding="utf-8")),
36 required=frozenset({"codex_path", "approval_policy", "sandbox", "final_path"}),
37 optional=frozenset({"windows_sandbox"}))
38 if value["effort"] not in _EFFORTS or value["approval_policy"] != "never":
39 raise ValueError("unsupported effort or approval policy")
40 if value["sandbox"] not in _SANDBOXES:
41 raise ValueError("a supported sandbox is required")
42 if "windows_sandbox" in value and value["windows_sandbox"] not in _WINDOWS_SANDBOXES:
43 raise ValueError("windows_sandbox must be elevated or unelevated when present")
44 return value

47def build_argv(request: Mapping[str, Any]) -> list[str]:
48 """Use only documented non-interactive flags and a stdin prompt sentinel; ``load_request`` checked the request."""
49 argv = [request["codex_path"], "--ask-for-approval", request["approval_policy"], "exec",
50 "--json", "--color", "never", "--output-last-message", request["final_path"],
51 "--model", request["model"], "--cd", request["cwd"],
52 "--strict-config", "--config", 'model_reasoning_effort="' + request["effort"] + '"',
53 "--config", "project_doc_max_bytes=0",
54 "--ephemeral", "--ignore-user-config", "--ignore-rules", "--skip-git-repo-check"]
55 if "output_schema_path" in request:
56 argv.extend(["--output-schema", request["output_schema_path"]])
57 if "windows_sandbox" in request:
58 argv.extend(["--config", 'windows.sandbox="' + request["windows_sandbox"] + '"'])
59 if "control_binding" in request:
60 control = ControlTransport.mcp_server_config(request["control_binding"])
61 argv.extend(["--config", f"mcp_servers.{CONTROL_SERVER}.command=" + json.dumps(control["command"], ensure_ascii=False),
62 "--config", f"mcp_servers.{CONTROL_SERVER}.args=" + json.dumps(control["args"], ensure_ascii=False)])
63 server = request.get("scholarly_server")
64 if server is not None:
65 data_dir = str((Path(request["cwd"]).resolve() / "scholarly-data"))
66 args = [*server["args"], "--data-dir", data_dir]
67 argv.extend(["--config", "mcp_servers.scholarly.command=" + json.dumps(server["command"], ensure_ascii=False),
68 "--config", "mcp_servers.scholarly.args=" + json.dumps(args, ensure_ascii=False)])
69 argv.extend(["--sandbox", request["sandbox"]])
70 return [*argv, "-"]

73def run(request_path: Path) -> int:
74 request = load_request(request_path)
75 return run_cli(request_path, request, build_argv(request), sanitized_environment())

78def main(argv: list[str] | None = None) -> int:
79 arguments = sys.argv[1:] if argv is None else argv
80 if len(arguments) != 1:
81 raise SystemExit("usage: codex_worker.py REQUEST.json")
82 return run(Path(arguments[0]))
