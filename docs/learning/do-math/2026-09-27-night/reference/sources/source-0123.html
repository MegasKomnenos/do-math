26class BudgetTests(unittest.TestCase):
27 def constraints(self, *, work_slots: int = 1, max_jobs: int = 1, memory_bytes: int = 100,
28 reserve: dict | None = None) -> dict:
29 return {
30 "exec": {"id": "exec", "kind": "concurrency", "accounting_unit": "execution",
31 "work_slots": work_slots},
32 "host": {"id": "host", "kind": "host", "max_jobs": max_jobs, "memory_bytes": memory_bytes,
33 "cpu_rate_basis_points": 10000, "control_reserve": reserve or dict(RESERVE)},
34 }

36 def profile(self, *, host_memory: int = 50) -> dict:
37 return {
38 "id": "EP-1", "revision": 1, "sha256": "0" * 64, "actor_kind": "ai",
39 "ai": {"capacity_refs": ["exec"]},
40 "host_capacity_refs": ["host"],
41 "host_resources": {"constraint_id": "host", "demands": {
42 "jobs": 1, "memory_bytes": host_memory, "cpu_rate_basis_points": 5000}},
43 }

45 def test_one_profile_reserves_its_work_slot_and_host_demands_atomically(self) -> None:
46 constraints = self.constraints(max_jobs=2, memory_bytes=200)
47 before = deepcopy(constraints)
48 work = plan_reservations([], constraints, self.profile(), owner_id="EX-1")
49 self.assertEqual(sorted((r["constraint_id"], r["unit"], r["reserved"]) for r in work),
50 [("exec", "execution", 1), ("host", "cpu_rate_basis_points", 5000),
51 ("host", "jobs", 1), ("host", "memory_bytes", 50)])
52 self.assertTrue(all(r["state"] == "held" and r["owner_id"] == "EX-1"
53 and r["authority_refs"] == [{"id": "EP-1", "revision": 1, "sha256": "0" * 64}]
54 for r in work))
55 self.assertEqual(constraints, before)
56 with self.assertRaisesRegex(BudgetConflict, "concurrency capacity unavailable: exec"):
57 plan_reservations(work, constraints, self.profile(), owner_id="EX-2")

70 def test_returned_slot_is_reusable_but_held_slot_remains_occupied(self) -> None:
71 returned = {"id": "RR-returned", "owner_id": "EX-0", "constraint_id": "exec", "unit": "execution",

83 def test_settlement_is_exactly_once_and_requires_full_accounting(self) -> None:
84 record = next(r for r in plan_reservations([], self.constraints(), self.profile(), owner_id="EX-1")
85 if r["constraint_id"] == "exec")

87 settle_reservation(record, confirmed_consumed=0, returned=0, evidence_refs=[{"receipt": "x"}])
