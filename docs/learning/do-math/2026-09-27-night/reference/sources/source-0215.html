57 def test_scope_then_initial_organization_and_independent_judgment_use_only_sealed_stages(self):
58 cycle = start_review({"revision": "r1"}, reviewers=2, policy=policy())
59 scope = next_actions(cycle)[0]
60 cycle = accept_result(cycle, scope, {
61 "proposal_digest": cycle["proposal_digest"], "summary": "Two perspectives are required.",
62 "used_sources": ["proposal"], "questions": [], "scope_complete": True,
63 "scope_assessment": assessment(),
64 }, "scope-run")
65 for action in list(next_actions(cycle)):
66 finding = {"id": action["slot"] + "-finding", "statement": "The proposal is incomplete.",
67 "evidence": ["proposal"]}
68 cycle = accept_result(cycle, action, {
69 "proposal_digest": cycle["proposal_digest"], "summary": "A material mismatch exists.",
70 "used_sources": ["proposal"], "questions": [], "decision": "mismatch",
71 "scope_complete": True, "findings": [finding],
72 }, action["slot"] + "-run")
73 organize = next_actions(cycle)[0]
74 self.assertEqual(organize["stage"], "organize")
75 finding_refs = [entry["ref"] for entry in cycle["findings"].values()]
76 cycle = accept_result(cycle, organize, {
77 "proposal_digest": cycle["proposal_digest"], "summary": "Each finding remains a separate issue.",
78 "used_sources": ["proposal"], "questions": [],
79 "issues": [{"id": "issue-" + ref["slot"], "finding_refs": [ref],
80 "answer_role_slot_id": ref["slot"]} for ref in finding_refs],
81 }, "organize-run")
82 judges = next_actions(cycle)
83 self.assertEqual({action["stage"] for action in judges}, {"judge"})
84 self.assertTrue(all(context_for(cycle, action)["completed_results"] for action in judges))
85 for action in judges:
86 cycle = accept_result(cycle, action, {
87 "proposal_digest": cycle["proposal_digest"], "summary": "The objections are established.",
88 "used_sources": ["proposal"], "questions": [], "decision": "mismatch",

91 "reason": "The evidence identifies the mismatch.", "evidence": ["proposal"]}

93 }, action["slot"] + "-run")

213 def test_scope_execution_minimum_in_the_review_contract_matches_the_policy(self):
214 """The documented U+N+4 and U+2N+7 lower bounds are the function the checks run."""
215 import re
216 from do_math.review_policy import minimum_scope_executions
217 text = (Path(__file__).resolve().parents[2] / '계약' / '동일성-검토.md').read_text(encoding='utf-8')
218 formulas = {}
219 for label, rounds in (('한 회차의 하한은', (1,)), ('두 회차 이상의 하한은', (2, 3, 5))):
220 match = re.search(re.escape(label) + r' U\+(\d*)N\+(\d+)', text)
221 self.assertIsNotNone(match, label)
222 formulas[rounds] = (int(match.group(1) or 1), int(match.group(2)))
223 for rounds, (per_slot, fixed) in formulas.items():
224 for round_limit in rounds:
225 for used in (0, 2):
226 for slots in (1, 3):
227 self.assertEqual(minimum_scope_executions(slots, round_limit, used),
228 used + per_slot * slots + fixed, (round_limit, used, slots))
