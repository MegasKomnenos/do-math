13def validate_disposition(disposition: dict[str, Any]) -> dict[str, Any]:
14 """Return a detached v5 disposition or raise before any state transition.
15
16 The producer receives contracts.disposition_schema; this check reads the
17 variant of the same schema that the disposition's kind names.
18 """
19 import jsonschema
20 from .contracts import disposition_schema
21 variants = {row["properties"]["kind"]["const"]: row for row in disposition_schema()["anyOf"]}
22 if not isinstance(disposition, dict) or disposition.get("kind") not in variants:
23 raise ValueError("operational disposition kind is unsupported")
24 error = jsonschema.exceptions.best_match(
25 jsonschema.Draft202012Validator(variants[disposition["kind"]]).iter_errors(disposition))
26 if error is not None:
27 raise ValueError("operational disposition " + disposition["kind"]
28 + "".join(f"/{part}" for part in error.absolute_path) + ": " + error.message)
29 return deepcopy(disposition)

32def admissible_basis_refs(instance, runtime, action):
33 """Canonical operational evidence, shared by input production and application."""
34 from .revisions import action_repetition_waits
35 target = runtime.get('executions', {}).get(action.get('target'))
36 if target is None:
37 raise ValueError('operational disposition has no execution target')
38 refs = {target['id'], *action.get('observation_refs', []), *action.get('obligation_ids', [])}
39 refs.update(hold['event_ref'] for hold in action_repetition_waits(instance, runtime, target.get('action', {})))
40 refs.update(ref for ref in (target.get('result_ref'),
41 runtime.get('checkpoints', {}).get(target['id'], {}).get('artifact_ref')) if ref)
42 from .activities import successor_state
43 successor=successor_state(runtime,target['id'])
44 for identifier in successor['execution_ids'] if successor else []:
45 refs.add(identifier)
46 result=runtime['executions'][identifier].get('result_ref')
47 if result:refs.add(result)
48 return sorted(refs)

68def reconcile_recovery(runtime: dict, record: dict) -> bool:
69 """Settle an obsolete recovery proposal without duplicating its successor."""
70 if record.get('status') == 'superseded':
71 return True
72 if record['disposition']['kind'] not in {'retry', 'resume'}:
73 return False
74 # A successfully applied retry already owns the successor it created.
75 if record.get('result', {}).get('continuation_action_id'):
76 return False
77 from .activities import successor_state
78 successor = successor_state(runtime, record['target_execution_id'])
79 if successor is None:
80 return False
81 record['status'] = 'superseded'
82 record.setdefault('result', {})['successor'] = successor
83 target = runtime['executions'][record['target_execution_id']]
84 if target.get('pending_operational_disposition') == record['id']:
85 target.pop('pending_operational_disposition')
86 return True
