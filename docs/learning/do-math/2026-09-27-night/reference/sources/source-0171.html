122 def test_mathematical_relation_requires_match_in_addition_to_active_use(self):
123 from do_math.knowledge import assess_use
124 from knowledge_fixtures import formal_evidence
125 for decision in ("mismatch", "unresolved"):
126 with self.subTest(decision=decision):
127 ins, run, ids = knowledge_graph()
128 ins = activate_kind(ins, "formally_implies")
129 claim = dict(kind="formally_implies", source=ids["a"], target=ids["b"],
130 confirmed_scope="True implies True")
131 ins, run, proof, _ = add_knowledge(ins, run, verified=False, relation_claim=claim)
132 ins = kr.record_formal_result(ins, proof, formal_evidence(ins, proof), "TE-old", runtime=run)
133 ins, negative = kr.record_identity_assessment(ins, proof,
134 {"decision": decision, "basis": kr.identity_basis(ins, proof)},
135 "policy:fixture", runtime=run)
136 relation = {**claim, "evidence": "proof",
137 "formal_result_refs": [ins["knowledge"][proof]["formal_result_id"]],
138 "assessment_refs": [negative]}
139 self.assertTrue(assess_use(ins["identity_assessments"][negative], ins)["valid"])
140 checked = io.validate_relation(ins, relation)
141 self.assertFalse(checked["valid"])
142 self.assertTrue(any(r.startswith("formal_result_does_not_establish_this_relation:")
143 for r in checked["reasons"]))
144 ins, positive = kr.record_identity_assessment(ins, proof,
145 {"decision": "match", "basis": kr.identity_basis(ins, proof)},
146 "policy:fixture", runtime=run)
147 checked = io.validate_relation(ins, relation)
148 self.assertFalse(checked["valid"])
149 self.assertIn("assessment_not_effective:" + negative, checked["reasons"])
150 relation["assessment_refs"] = [positive]
151 self.assertTrue(io.validate_relation(ins, relation)["valid"])
