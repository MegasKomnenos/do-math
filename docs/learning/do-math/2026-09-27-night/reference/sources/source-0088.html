253 # Admission checked this attempt; its result is checked again when it is accepted.

255 raise ValueError('review execution admission identity changed')

260 changes = review_input_basis_changes(cycle.get('review_input_basis', {}), cycle)
261 if changes:
262 raise ValueError('review inputs changed before admission: ' + json.dumps(changes, ensure_ascii=False))

264 raise ValueError('review admission requires its actual owner activity')

293 activity['execution_limit'] = limit

295 accounting['admissions'] = admissions
296 accounting['executions_used'].append(eid)

532 def _prepare_review_context(self, action, workspace: Path, *, sources=None):
533 """Return a runnable context or an activity-owned recovery disposition.

537 should escape and be retried as though nothing had been recorded.
538 """
539 cycle = self._review_cycle(action)
540 key = slot_key(action)
541 if key in cycle.get('blocked_actions', {}) and key not in cycle.get('replacement_actions', {}):
