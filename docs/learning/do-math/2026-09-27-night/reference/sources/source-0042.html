45def seal_profile(body: dict) -> dict:
46 """Create a new immutable edition; never edit an existing registered row."""
47 value = deepcopy(body)
48 value.pop("id", None)
49 value.pop("sha256", None)
50 value.setdefault("revision", 1)
51 digest = exact_content_hash(value)
52 value.update(id="EP-" + digest, sha256=digest)
53 validate_profile(value)
54 return value

62def validate_profile(profile: dict) -> None:
63 body = {key: value for key, value in profile.items() if key not in {"id", "sha256"}}
64 digest = exact_content_hash(body)
65 if profile.get("sha256") != digest or profile.get("id") != "EP-" + digest:
66 raise ValueError("execution profile content differs from its immutable identity")
67 if type(profile.get("revision")) is not int or profile["revision"] < 1:
68 raise ValueError("unsupported profile edition")
69 role, actor = profile.get("role"), profile.get("actor_kind")
70 if actor not in {"ai", "machine"}:
71 raise ValueError("profile requires an actor")
72 if actor == "machine":
73 if role not in MACHINE_ROLES or "ai" in profile:
74 raise ValueError("machine profile cannot contain AI fields")
75 # Lean roles pin only the command; the approved trust environment fixes its identity.
76 # Activation reads the toolchain record and needs no machine settings.
77 machine = profile.get("machine")
78 if (role in LEAN_ROLES and (not isinstance(machine, dict) or set(machine) != {"lean_command"})
79 or role not in LEAN_ROLES and machine is not None):
80 raise ValueError("machine profile settings differ from the role contract")
81 elif not isinstance(profile.get("ai"), dict) or not profile["ai"].get("model"):
82 raise ValueError("AI profile requires a model")
83 elif bool(profile["ai"].get("capacity_refs")) != (role in WORK_ROLES):
84 raise ValueError("only work roles reserve a model slot")
85 expected_mode = ("deadline_only" if actor == "machine" or role not in WORK_ROLES
86 else None if role == "research_work" else "periodic")
87 mode = profile.get("observation", {}).get("mode")
88 if mode not in {"event_driven", "periodic", "deadline_only"} or expected_mode and mode != expected_mode:
89 raise ValueError("observation mode differs from role responsibility")
90 bindings = profile.get("observation", {}).get("rule_bindings")
91 if role == "research_work":
92 from .observations import validate_observation_rule_contract
93 validate_observation_rule_contract(bindings)
94 elif bindings != []:
95 raise ValueError("profile observation bindings differ from the supported role contract")
96 if profile.get("boundary") != BOUNDARY:
97 raise ValueError("this implementation supports only the declared instruction boundary")
98 host = profile.get("host_resources")
99 if host is not None:
100 if (set(host) != {"constraint_id", "scope", "host_id", "demands"}
101 or host["scope"] != "instance_on_host" or not isinstance(host["host_id"], str) or not host["host_id"]
102 or profile.get("host_capacity_refs") != [host["constraint_id"]]
103 or profile.get("host_capacity_status") != "adopted_local_job_limits"):
104 raise ValueError("host profile requires one exact local instance constraint")
105 _host_amounts(host["demands"], demand=True)
106 elif profile.get("host_capacity_refs"):
107 raise ValueError("host references require pinned execution demands")

110def _host_amounts(value, *, demand=False):
111 units = {"jobs", "memory_bytes", "cpu_rate_basis_points"}
112 if not isinstance(value, dict) or set(value) != units:
113 raise ValueError("host amounts require explicit jobs, memory_bytes and cpu_rate_basis_points")
114 if any(type(v) is not int or v < (1 if demand else 0) for v in value.values()):
115 raise ValueError("host amounts must be explicit nonnegative integers (positive for a demand)")
116 if value["cpu_rate_basis_points"] > 10000 or demand and value["jobs"] != 1:
117 raise ValueError("CPU cap must be at most 10000 and one execution owns one simultaneous Job tree")

120def host_constraint(*, host_id: str, limits: dict, control_reserve: dict, authority_refs: list) -> dict:
121 """Author an explicit instance-local approval; no host-wide coordinator exists.
122
123 The controller's reserve cannot be borrowed by any admitted child Job.
124 CPU is a hard cap relative to an enclosing rate-controlled Job, not a
125 processing guarantee. Memory is committed virtual memory, not RSS.
126 """
127 _host_amounts(limits)
128 _host_amounts(control_reserve)
129 if (not isinstance(host_id, str) or not host_id.strip() or not isinstance(authority_refs, list)
130 or not authority_refs or any(not isinstance(ref, str) or not ref.strip() for ref in authority_refs)
131 or any(control_reserve[k] > limits[k] for k in limits)):
132 raise ValueError("host constraint needs host, approval and reserve within the adopted limits")
133 body = {"kind": "host", "scope": "instance_on_host", "host_id": host_id,
134 "max_jobs": limits["jobs"], "memory_bytes": limits["memory_bytes"],
135 "cpu_rate_basis_points": limits["cpu_rate_basis_points"],
136 "control_reserve": deepcopy(control_reserve), "authority_refs": deepcopy(authority_refs)}
137 return {"id": "CP-" + exact_content_hash(body), **body}

173 """The configuration's host section: one approved constraint and a demand for exactly the capped roles.

175 Before binding, the constraint may still carry its request authority alias.

177 if not isinstance(host, dict) or set(host) != {"constraint", "demands_by_role"}:

190def build_profiles(config: dict) -> dict:
191 """Freeze the profile editions of every role from adopted values.
192
193 No new capacity, permission or measured limit is invented.
194 """
195 limits = config["session_limits"]
196 constraints = {}
197 adopted_host = config.get("host_resources")
198 if adopted_host is not None:
199 validate_host_resources(adopted_host)
200 constraints[adopted_host["constraint"]["id"]] = deepcopy(adopted_host["constraint"])
201 for model, work in sorted(limits["work"].items()):
202 identifier = "CP-" + exact_content_hash({"model": model, "work": work})
203 constraints[identifier] = {"id": identifier, "kind": "concurrency", "model": model,
204 "accounting_unit": "execution", "work_slots": work, "scope": "local_execution_not_provider_operations"}
205 profiles, refs = {}, {}
206 lean = _lean_contract(config)
207 for role in sorted(PROFILE_ROLES):
208 machine = role in MACHINE_ROLES
209 mode = (config["observation_mode"] if role == "research_work" else
210 "periodic" if role in WORK_ROLES else "deadline_only")
211 for tier in ((None,) if machine else TIERS):
212 body = {"role": role, "actor_kind": "machine" if machine else "ai",
213 "boundary": deepcopy(BOUNDARY), "host_capacity_refs": [],
214 "host_capacity_status": "unmeasured_not_enforced",
215 "observation": {"mode": mode, "settings": {k: deepcopy(config[k]) for k in OBSERVATION_KEYS if k in config},
216 "rule_bindings": deepcopy(_research_observation_rule_contract() if role == "research_work" else [])},
217 "window": {k: deepcopy(config[k]) for k in WINDOW_KEYS if k in config},
218 "input": {"max_input_bytes": config["max_input_bytes"], "unit": "utf8_bytes",
219 "provider_context_capacity": "unknown"},
220 "required_capabilities": [] if machine else ["jsonl", "model", "structured_output"],
221 "capability_evidence_refs": []}
222 if machine and role in LEAN_ROLES:
223 body["machine"] = deepcopy(lean)
224 if not machine:
225 model = config["tier_models"][tier]
226 capacity = next((c["id"] for c in constraints.values() if c.get("model") == model), None)
227 if capacity is None:
228 raise ValueError("tier model has no adopted capacity")
229 provider = provider_for(model)
230 body["ai"] = {"model": model, "tier": tier, "provider": provider,
231 "effort": config["operator_effort"] if role == "recovery" else config["effort"],
232 "capacity_refs": [capacity] if role in WORK_ROLES else [],
233 "provider_operations": "unknown", "remote_termination": "unknown",
234 "maximum_output_enforced": False,
235 "claude_only_assigned_model": config["claude_only_assigned_model"],
236 "scholarly_server": deepcopy(config["scholarly_server"]) if config["scholarly_enabled"] else None}
237 if role == "recovery":
238 body["window"]["call_timeout_seconds"] = config["operator_timeout_seconds"]
239 profile = seal_profile(body)
240 if adopted_host is not None and role in HOST_ROLES:
241 profile = with_host_resources(profile, adopted_host["constraint"], adopted_host["demands_by_role"][role])
242 profiles[profile["id"]] = profile
243 refs.setdefault(role, []).append(profile_ref(profile))
244 result = {"execution_profiles": profiles, "capacity_constraints": constraints, "role_profile_refs": refs}
245 if adopted_host is not None:
246 result["host_capacity_ref"] = adopted_host["constraint"]["id"]
247 return result

271def match_profile_ref(profiles: dict, reference: dict) -> dict:
272 """The stored profile a reference names, compared by id, revision and sha256.
273
274 Stored profiles passed validate_profile when saved or sealed, so the profile
275 is neither hashed again nor copied. A profile that has not passed through
276 storage or ``seal_profile`` must be checked with validate_profile first.
277 """
278 validate_profile_ref(reference)
279 profile = profiles.get(reference["id"])
280 if profile is None or any(profile.get(key) != reference[key] for key in ("id", "revision", "sha256")):
281 raise ValueError("profile reference is absent or stale")

290def select_profile(profiles: dict, action: dict, selection: dict, allowed_refs=None) -> dict:
291 """The one stored profile the action may use; like ``match_profile_ref``, not a copy."""
292 requested = action.get("profile_ref") or selection.get("profile_ref")
293 allowed = None if allowed_refs is None else [profile_ref(match_profile_ref(profiles, r)) for r in allowed_refs]
294 candidates = ([match_profile_ref(profiles, requested)] if requested else
295 [profiles[ref["id"]] for ref in {ref["id"]: ref for ref in allowed}.values()]
296 if allowed is not None else list(profiles.values()))
297 candidates = [p for p in candidates if p["role"] == action["role"]
298 and (p["actor_kind"] == "machine" or p["ai"]["model"] == selection["model"])
299 and (requested or p.get("ai", {}).get("tier") is None or not selection.get("tier")
300 or p["ai"]["tier"] == selection["tier"])
301 and (allowed is None or profile_ref(p) in allowed)]
302 if len(candidates) != 1:
303 raise ValueError("profile selection must resolve to exactly one permitted role/model edition")
304 return candidates[0]

387def capability_evidence(profile: dict, description: dict, *, observed_at: str) -> dict:
388 """Bind command-surface facts to one profile, host and exact implementation.
389
390 Help text is not a process-lifetime, isolation, or provider-behavior test.
391 Those claims need independently supplied evidence from their actual probes.
392 """
393 reference = profile_ref(profile)
394 if not description.get("identity") or not observed_at:
395 raise ValueError("capability evidence requires current identity and observation time")
396 names = set(description.get("supported", {})) | set(description.get("unknown", []))
397 surface = {"jsonl", "model", "structured_output", "effort_via_config", "sandbox"}
398 facts = {}
399 for name in sorted(names):
400 supported = description.get("supported", {}).get(name)
401 status = ("supported" if supported is True and description.get("verified") is True else
402 "unsupported" if supported is False and description.get("verified") is True else "unknown")
403 if description.get("scope") == "cli_command_surface_only" and name not in surface:
404 status = "unknown"
405 facts[name] = {"status": status, "scope": description.get("scope", "unknown")}
406 record = {"execution_profile_ref": reference, "identity": deepcopy(description["identity"]),
407 "observed_at": observed_at, "capabilities": facts,
408 "version_sha256": description.get("version_sha256"),
409 "help_sha256": description.get("exec_help_sha256"),
410 "boundary": "instructions_not_os_isolation"}
411 record["id"] = "CE-" + exact_content_hash(record)
412 return record
