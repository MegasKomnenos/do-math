22def _zone(name):
23 try:
24 return ZoneInfo(name)
25 except ZoneInfoNotFoundError:
26 if name == "Asia/Seoul":
27 return timezone(timedelta(hours=9), name="Asia/Seoul")
28 if name == "UTC":
29 return timezone.utc
30 raise ValueError("timezone database has no entry for " + name)

50def task_status(instance, runtime, *, evaluation_context=None):
51 """Per-task rows; readiness uses ``evaluation_context`` so a current view judges as dispatch does."""
52 from . import planning, questions, task_contracts
53 rows = []
54 pending_assessments = set(task_contracts.returned_task_ids(runtime.get('tasks', {})))
55 live = [e for e in runtime.get("executions", {}).values() if e.get("status") != "ended"]
56 for identifier, task in runtime.get("tasks", {}).items():
57 queued = [action for action in runtime.get("queue", []) if action.get("target") == identifier]
58 executions = [e for e in live if e.get("action", {}).get("target") == identifier]
59 running_ids = {e.get("action", {}).get("id") for e in executions}
60 obligations = [aid for aid, entry in runtime.get("waiting_actions", {}).items()
61 if entry.get("action", {}).get("target") == identifier or identifier in entry.get("action", {}).get("revision_targets", {})
62 or identifier in entry.get("delegated_targets", {}).values()]
63 waiting = []
64 for action in queued:
65 if action.get('id') in running_ids:
66 continue
67 waiting.extend(questions.user_waits(action, instance, runtime))
68 waiting.extend(planning.revision_waits(action, instance, runtime))
69 if action.get('dispatch_wait', {}).get('kind') in ADMISSION_WAITS:
70 waiting.append(action['dispatch_wait'])
71 readiness = None
72 if not executions and task.get('status') not in planning.TERMINAL_TASK_STATUSES:
73 readiness = planning.task_ready(task, instance, runtime, include_scheduling_waits=False,
74 evaluation_context=evaluation_context)
75 if readiness['state'] != 'ready':
76 waiting.append({'kind': 'contract_review' if 'current revision lacks approved contract review' in readiness['errors'] else 'required_input',
77 'errors': readiness['errors']})
78 waiting = list({json.dumps(row, sort_keys=True): row for row in waiting}.values())
79 next_owner = ('user' if any(row.get('kind') == 'user_decision' or row.get('next_owner') == 'user' for row in waiting) else
80 'task_contract_review' if any(row.get('kind') == 'contract_review' for row in waiting) else
81 next((row['next_owner'] for row in waiting if row.get('next_owner')),
82 queued[0]['role'] if queued else None))
83 assessment = task.get('assessment', {})
84 followups = [e['id'] for e in runtime.get('events', [])
85 if e['kind'] == 'task_assessment_invalidated' and e.get('target') == identifier]
86 if assessment.get('current') is False and task['status'] in planning.TERMINAL_TASK_STATUSES:
87 next_owner = next_owner or 'research_plan'
88 rows.append({"id": identifier, "title": task.get("title", task.get('question')), "status": task["status"], "revision": task.get("revision"),
89 "selected_actions": [a.get("id") for a in queued if a.get("id") not in running_ids],
90 "active_execution_ids": [e["id"] for e in executions],
91 "assessment_pending": identifier in pending_assessments,
92 'recorded_decision': assessment.get('decision'),
93 'assessment_current': assessment.get('current'),
94 'assessment_invalidated_reason': assessment.get('invalidated_reason'),
95 'assessment_followup_event_refs': followups,
96 "last_accepted_result": task.get("results", [])[-1].get("execution_id") if task.get("results") else None,
97 "unresolved_action_ids": obligations, "waits": waiting,
98 "readiness": readiness, "next_owner": next_owner,
99 "next_observation_at": next((row.get("retry_at") for row in waiting if row.get("retry_at")), None)})
100 return rows

103def operating_summary(instance, runtime):
104 """One persisted-state handoff view; scheduler/process observations stay separate."""
105 from .scheduled_operator import supervision_block, supervision_facts
106 lifecycle=instance.get('lifecycle',{})
107 facts=supervision_facts(instance,runtime)
108 code=supervision_block({key:value for key,value in facts.items() if key!='operator_enabled'})
109 repair=facts['installation_repair']
110 if code=='user_pause':
111 blocker={'kind':'user_pause','next_owner':'user','reason':runtime['user_pause'].get('reason')}
112 elif code=='maintenance':
113 blocker={'kind':'maintenance_fence','next_owner':'operator','reason':runtime['maintenance_fence'].get('state')}
114 elif code=='model_policy':
115 blocker={'kind':'model_policy','next_owner':'user'}
116 elif code=='paused':
117 blocker={'kind':'paused','next_owner':'operator','reason':runtime.get('stop_requested'),
118 'questions':deepcopy(runtime.get('questions',[]))}
119 elif code=='installation_repair':
120 blocker={'kind':'installation_repair','next_owner':'installation_recovery',
121 'record_id':repair['id'],'reason':repair['disposition']['reason']}
122 elif lifecycle.get('state')=='initializing':
123 blocker={'kind':'initialization','next_owner':'operator',
124 'missing':deepcopy(lifecycle.get('activation',{}).get('missing',[]))}
125 elif runtime.get('status') in {'waiting','capacity_waiting'}:
126 waits=runtime.get('waiting_actions',{})
127 blocker={'kind':runtime['status'],'next_owner':None,
128 'owners':sorted({row.get('next_owner','engine') for row in waits.values()})}
129 else:
130 blocker=None
131 config=instance.get('config',{})
132 return {'scope':'persisted_state_not_host_process_observation','lifecycle':lifecycle.get('state'),
133 'initialization':deepcopy(lifecycle.get('activation',{})),
134 'execution_status':runtime.get('status'),'goal_status':instance.get('goal_status'),
135 'primary_blocker':blocker,'workspace':instance.get('workspace'),
136 'conversation_loss_policy':instance.get('conversation_loss_policy'),
137 'active_execution_ids':sorted(e['id'] for e in runtime.get('executions',{}).values() if e['status']!='ended'),
138 'supervision':{'configured_enabled':config.get('operator_enabled'),
139 'interval_seconds':config.get('operator_interval_seconds'),
140 'actual_schedule':'operator status'},
141 'grants':[{'id':g['id'],'status':g.get('status'),'expires_at':g.get('expires_at')}
142 for g in instance.get('grants',{}).values()]}

152def status_report(store, head, instance, runtime, *, observation=None, requested_areas=None,
153 evaluation_context=None) -> dict:
154 """The one status projection: every field of the persisted snapshot at ``head``.
155
156 ``observation`` holds the OBSERVED_FIELDS the caller observed now; each
157 field it omits is None. ``store`` reads the snapshot's artifacts.
158 ``evaluation_context`` is the engine's current evaluation context, so task
159 readiness sees this host's facts (a discovered Lean, the clock) exactly as
160 dispatch does. A snapshot projection omits it.
161 """
162 from .academic import academic_status, stored_evidence
163 from .academic import current_solution_basis
164 from .execution_profiles import WORK_ROLES
165 from .initialization import governance_or_initial_view
166 from .knowledge_records import knowledge_view
167 from .question_engine import question_view, user_questions
168 from .retention import pending_cleanup_failures
169 from .scheduled_operator import supervision_facts
170 from .session_slots import SLOT_WAITS
171 observed = dict.fromkeys(OBSERVED_FIELDS)
172 unknown = set(observation or {}) - set(OBSERVED_FIELDS)
173 if unknown:
174 raise ValueError("status observation has unknown fields: " + ", ".join(sorted(unknown)))
175 observed.update(observation or {})
176 governance_view = governance_or_initial_view(instance, runtime)
177 executions = list(runtime.get("executions", {}).values())
178 live = [e["id"] for e in executions if e["status"] != "ended"]
179 waits = dispatch_waits(runtime)
180 questions = question_view(instance, runtime, lambda refs: store.read_many(refs, head))
181 return {"instance_id": instance["id"], "snapshot": head, "title": instance.get("title"),
182 "operating_summary": operating_summary(instance, runtime),
183 # Deriving check diagnostics reads every sealed check artifact; ``check`` does it.
184 "submission_checks": {"status": "not_evaluated",
185 "reason": "status reports persisted records; check derives submission check diagnostics"},
186 "goal": governance_view["goal"], "goal_status": instance.get("goal_status"),
187 "status": runtime.get("status"), "active_execution": live[0] if live else None,
188 "pending_actions": len(runtime.get("queue", [])), "cycles": runtime.get("cycles"),
189 "tasks": task_status(instance, runtime, evaluation_context=evaluation_context), "dispatch_waits": waits,
190 "knowledge": {"K": sum(row["usable"] for row in knowledge_view(instance)),
191 "formal_results": len(instance.get("formal_results", {})),
192 "expositions": len(instance.get("expositions", {})),
193 "identity_assessments": len(instance.get("identity_assessments", {}))},
194 "inspection": inspect_state(instance, runtime, snapshot=head, requested_areas=requested_areas,
195 read_artifact=lambda path: store.read(path, head),
196 read_artifacts=lambda paths: store.read_many(paths, head, missing="omit"),
197 artifact_sizes=lambda paths: store.blob_sizes(paths, head, missing="omit")),
198 "budget_unit": "concurrent_sessions",
199 "active_executions": live,
200 "execution_attempts": {"work": sum(e.get("role") in WORK_ROLES for e in executions),
201 "operations": sum(e.get("role") not in WORK_ROLES for e in executions)},
202 "capacity_waiting": [{k: a[k] for k in ('role', 'target', 'model_selection')}
203 for a in waits if a['wait']['kind'] in SLOT_WAITS],
204 "usage": [e.get("usage", {}) for e in executions if e.get("usage")],
205 "latest_results": list(runtime.get("activities", {}).values())[-5:],
206 "questions": user_questions(questions),
207 "question_links": questions,
208 "operational_notices": deepcopy(runtime.get("questions", [])),
209 # user_pause, conversation_lost, model_policy_violation, maintenance_fence,
210 # installation_repair and operator_enabled: the facts supervision decides on.
211 **supervision_facts(instance, runtime),
212 "waiting_actions": {aid: {"role": entry["action"].get("role"),
213 "target": entry["action"].get("target"), "kind": entry.get("kind"),
214 "reason": entry.get("reason"), "execution_id": entry.get("execution_id"),
215 "failure_kind": entry.get("failure_kind")}
216 for aid, entry in runtime.get("waiting_actions", {}).items()},
217 "grants": [{"id": g["id"], "kind": g.get("kind"), "status": g.get("status"), "expires_at": g.get("expires_at")}
218 for g in instance.get("grants", {}).values()],
219 "pending_events": runtime.get("pending_events", []),
220 "observations": runtime.get("observations", {}),
221 "checkpoints": runtime.get("checkpoints", {}),
222 "external_operations": list(runtime.get("external_operations", {}).values()),
223 "active_trust_environment": instance.get("active_trust_environment"),
224 "academic_validation": ({"status": "initialization_incomplete"}
225 if governance_view.get("status") == "unadopted_input" else
226 academic_status(instance, runtime, current_solution_basis(instance, runtime),
227 resolve_evidence=lambda ref: stored_evidence(store, head, ref))),
228 "cleanup_failures": pending_cleanup_failures(runtime),
229 "assurance": deepcopy(ASSURANCE),
230 **observed}

247def render_human_report(report: dict) -> str:
248 """Render a status report without reopening state or deriving new facts."""
249 inspection = report["inspection"]
250 tasks = report["tasks"]
251 budget = report.get("budget")
252 lines = [f"goal: {report.get('goal') or 'unknown'}",
253 f"goal_status: {report.get('goal_status') or 'unknown'}",
254 f"status: {report.get('status')} snapshot: {report['snapshot']}",
255 "work sessions: " + (", ".join(f"{model} {row['used']}/{row['limit']}" for model, row in budget.items())
256 if budget is not None else "not observed"),
257 "tasks: " + (", ".join(f"{task['id']}={task['status']}" for task in tasks) or "none"),
258 "observed (Asia/Seoul): " + str(display_time(report['research_observed_at'])
259 if report.get('research_observed_at') else 'not observed'),
260 "unknown or bounded scope: " + ", ".join(f"{key}={value}" for key, value in report["assurance"].items()),
261 f"inspection: {inspection['status']} ({inspection['checked_scope']})"]
262 summary = report.get('operating_summary', {})
263 blocker = summary.get('primary_blocker')
264 if blocker:
265 lines.append('blocking: '+str(blocker['kind'])+' next='+str(blocker.get('next_owner')))
266 if summary.get('lifecycle') == 'initializing':
267 lines.append('initialization missing: '+', '.join(summary.get('initialization', {}).get('missing', [])))
268 for finding in inspection["findings"]:
269 lines.append(f" {finding['area']}: {finding['status']} — {finding['message']} (next: {finding['next_owner']})")
270 for task in tasks:
271 waits = ', '.join(row.get('kind', 'unknown') for row in task.get('waits', [])) or 'none'
272 lines.append(f" {task['id']} revision={task.get('revision', 'unknown')} stage={task['status']} "
273 f"assessment_pending={str(task.get('assessment_pending', False)).lower()} "
274 f"active_records={','.join(task.get('active_execution_ids', [])) or 'none'} "
275 f"queued={','.join(str(value) for value in task.get('selected_actions', [])) or 'none'} "
276 f"waits={waits} next={task.get('next_owner') or 'none'}")
277 for name, gate in (report.get('provider_waits') or {}).get('providers', {}).items():
278 lines.append(f"provider {name}: {gate['state']} retry={display_time(gate.get('retry_at')) or 'unknown'} "
279 f"generation={gate['generation']}")
280 lines.extend(render_dispatch_waits(report))
281 return "\n".join(lines)

292def timeline(repository: Path, instance_id: str, *, snapshot=None, timezone_name="Asia/Seoul", include_operations=False):
293 store = ResearchRepository(repository, instance_id)
294 head, instance, runtime = store.load(snapshot)
295 zone = _zone(timezone_name)
296 rows = []
297 for execution in runtime.get("executions", {}).values():
298 action = execution.get("action", {})
299 created, started, ended = (execution.get(key) for key in ("created_at", "started_at", "ended_at"))
300 rows.append({"unit": "internal_execution", "id": execution["id"], "snapshot": head,
301 "input_snapshot": execution.get("input_snapshot"), "action_id": action.get("id"), "lineage_id": action.get("lineage_id"),
302 "continuation_of": execution.get("continuation_of") or action.get("continuation_of"),
303 "role": execution.get("role"), "target": action.get("target"),
304 "assigned_model": execution.get("model"), "reported_models": execution.get("reported_models"),
305 "created_at_utc": created, "started_at_utc": started, "ended_at_utc": ended,
306 "created_at_local": _display(created, zone), "started_at_local": _display(started, zone), "ended_at_local": _display(ended, zone),
307 "effective_deadline_at": execution.get("execution_window", {}).get("effective_deadline_at"),
308 "status": execution.get("status"), "outcome": execution.get("outcome"),
309 "failure_kind": execution.get("error_kind"), "error": execution.get("error"),
310 "currentness_changes": execution.get("currentness_changes"),
311 "output_files": execution.get("output_files", []), "artifact_bundle": execution.get("artifact_bundle"),
312 "usage": execution.get("usage"), "usage_segments": execution.get("usage_segments"),
313 "model_usage_reported": execution.get("model_usage_reported"), "usage_scope": execution.get("usage_scope"),
314 "next_actions": [a for a in runtime.get("queue", [])
315 if a.get("continuation_of") == execution["id"] or any(
316 a.get(key, {}).get("execution_id") == execution["id"] for key in ("repair", "refresh", "recovery"))],
317 "evidence_source": {"kind": "research_snapshot", "snapshot": head}})

327 "counts_by_unit": {unit: sum(row["unit"] == unit for row in rows) for unit in sorted({row["unit"] for row in rows})},
328 "observation": "Local operator evidence is separately observed; unknown historical timestamps are null."}
