13def original_input_bytes(record):
14 """Return and verify the one exact UTF-8 representation of immutable input."""
15 text = record.get('text')
16 if not isinstance(text, str):
17 raise ValueError('immutable input text is missing')
18 if any(record.get(key) != value for key, value in source_record(text).items()):
19 raise ValueError('immutable input bytes or hash differs')
20 return text.encode('utf-8')

28def capability_requirements(instance, reference, *, current_identity=None, at=None):
29 """Use exact profile/host/implementation evidence, never profile booleans."""
30 missing = []
31 try:
32 profile = match_profile_ref(instance.get("execution_profiles", {}), reference)
33 missing = sorted(set(profile.get("required_capabilities", [])))
34 if profile["actor_kind"] == "machine":
35 return {"satisfied": not missing, "missing": missing, "profile_ref": reference, "evidence_refs": []}
36 identity = current_identity if current_identity is not None else adapter_for(profile).capability_identity()
37 current_time = timestamp(at)
38 evidence = [row for row in instance.get("capability_evidence", {}).values()
39 if row.get("execution_profile_ref") == reference and row.get("identity") == identity
40 and timestamp(row["observed_at"]) <= current_time]
41 # The most recent observation supersedes older contradictory claims.
42 evidence.sort(key=lambda row: (timestamp(row["observed_at"]), row["id"]))
43 facts = {}
44 structural = None
45 for row in evidence:
46 facts.update(row.get("capabilities", {}))
47 job = row.get('capabilities', {}).get('atomic_local_job', {})
48 if job.get('status') == 'unsupported' and job.get('scope') == 'actual_local_start_failure':
49 structural = row['id']
50 # Help/version output cannot clear a concrete failed local start.
51 missing = [name for name in missing if facts.get(name, {}).get("status") != "supported"]
52 if structural and 'atomic_local_job' not in missing:
53 missing.append('atomic_local_job')
54 return {"satisfied": bool(evidence) and not missing, "missing": missing,
55 "profile_ref": reference, "evidence_refs": [row["id"] for row in evidence]}
56 except (ValueError, KeyError, TypeError, OSError) as exc:
57 return {"satisfied": False, "missing": missing, "profile_ref": reference, "reason": str(exc)}

60def _capability_batch_requirements(instance, references, *, at=None):
61 """Observe each provider once for this check; validate every exact profile."""
62 identities = {}
63 for reference in references:
64 try:
65 profile = match_profile_ref(instance.get("execution_profiles", {}), reference)
66 identity = None
67 if profile["actor_kind"] == "ai":
68 adapter = adapter_for(profile)
69 key = (profile['ai']['provider'], adapter.observation_key())
70 if key not in identities:
71 identities[key] = adapter.capability_identity()
72 identity = identities[key]
73 yield capability_requirements(instance, reference, current_identity=identity, at=at)
74 except (ValueError, KeyError, TypeError, OSError) as exc:
75 yield {"satisfied": False, "profile_ref": reference, "reason": str(exc)}

105def initial_intent_evidence(instance, runtime, *, owner_handlers=None):
106 """Resolve the sealed, independently checked and actually applied original."""
107 from .governance import _current_import, validate_independent_execution
108 lifecycle = instance['lifecycle']
109 original = lifecycle['immutable_input']
110 original_input_bytes(original)
111 original_sha256 = original['sha256']
112 for row in reversed(list(runtime.get('intent_imports', {}).values())):
113 if row.get('input_ref') != lifecycle.get('input_id') or row.get('superseded_by'):
114 continue
115 bundle = _current_import(instance, runtime, row['id'], owner_handlers=owner_handlers)
116 check, application = bundle.get('check', {}), bundle.get('application', {})
117 expected_units = {u['unit_id'] for u in bundle['output']['import_units'] if u['payload']}
118 if (bundle['source']['sha256'] != original_sha256 or bundle['source']['text'] != original['text']
119 or any(q.get('blocks') for q in bundle['output'].get('question_details', [])) or bundle['status'] != 'approved'
120 or check.get('decision') != 'approved' or check.get('scope_complete') is not True
121 or check.get('bundle_sha256') != bundle['sha256']
122 or check.get('sha256') != exact_content_hash({k:v for k,v in check.items() if k != 'sha256'})
123 or application.get('status') != 'applied' or application.get('import_id') != bundle['id']
124 or set(application.get('applied_units', [])) != expected_units):
125 raise ValueError('initial input requires a sealed, fully checked applied import')
126 validate_independent_execution(runtime, check['execution_id'], [bundle['execution_id']])
127 checker = runtime['executions'][check['execution_id']]
128 if (checker.get('role') != 'intent_interpretation' or checker.get('intent_import_sha256') != bundle['sha256']
129 or bundle['execution_id'] not in runtime['executions']):
130 raise ValueError('initial interpretation execution evidence does not resolve')
131 return {'input_sha256': original_sha256, 'interpretation_execution_id': bundle['execution_id'],
132 'check_execution_id': check['execution_id']}
133 raise ValueError('initial input has no applied independent import')

136def activation_requirements(instance, runtime, *, at=None, owner_handlers=None):
137 """Evaluate current evidence; none of these checks creates authority."""
138 lifecycle = instance.get("lifecycle", {})
139 adoption = lifecycle.get("adoption", {})
140 missing = []
141 source = lifecycle.get("immutable_input", {})
142 try:
143 if not original_input_bytes(source):
144 raise ValueError('empty original')
145 except (ValueError, KeyError, TypeError):
146 missing.append("immutable_input")
147 if adoption.get("governance_sha256") != governance_basis(instance):
148 missing.append("governance")
149 try:
150 from .governance import validate_governance_state
151 validate_governance_state(instance)
152 governance_working_view(instance)
153 except (ValueError, KeyError, TypeError):
154 missing.append('governance')
155 if adoption.get("config_sha256") != exact_content_hash(instance.get("config", {})):
156 missing.append("configuration")
157 if instance.get("conversation_loss_policy") not in {"pause", "continue_authorized"}:
158 missing.append("conversation_policy")
159 if "external_policy" not in adoption:
160 missing.append("external_policy")
161 try:
162 if lifecycle.get('intent_check') != initial_intent_evidence(instance, runtime, owner_handlers=owner_handlers):
163 raise ValueError('initial intent receipt differs from applied import')
164 except (ValueError, KeyError, TypeError):
165 missing.append("independent_intent")
166 policy = instance.get("review_policy")
167 try:
168 from .review_policy import build_review_policy
169 build_review_policy(policy)
170 if adoption.get("review_policy_sha256") != exact_content_hash(policy):
171 raise ValueError("review policy not adopted")
172 except (ValueError, TypeError):
173 missing.append("review_policy")
174 from .trust_validation import environment_errors
175 env = instance.get("trust_environments", {}).get(instance.get("active_trust_environment"))
176 if (not env or env.get("status") != "active"
177 or environment_errors(env, runtime.get("trust_bindings", {}).get(instance.get("active_trust_environment")))):
178 missing.append("trust_environment")
179 references = adoption.get("profile_refs", [])
180 if not references:
181 missing.append("profiles")
182 from .observations import observation_settings_errors
183 observation_errors = []
184 # Dispatch uses role_profile_refs after activation; a narrow bootstrap
185 # adoption must not hide incomplete profiles that will then be selected.
186 selected = [*references, *(ref for refs in instance.get('role_profile_refs', {}).values() for ref in refs)]
187 checked_profiles = set()
188 for ref in selected:
189 try:
190 profile = match_profile_ref(instance['execution_profiles'], ref)
191 except (ValueError, KeyError, TypeError):
192 missing.append('profiles')
193 continue
194 if profile['id'] in checked_profiles:
195 continue
196 checked_profiles.add(profile['id'])
197 observation_errors.extend(observation_settings_errors(profile))
198 if observation_errors:
199 missing.append('observation_configuration')
200 for capability in _capability_batch_requirements(instance, references, at=at):
201 if not capability["satisfied"]:
202 missing.append("capability_evidence")
203 break
204 grants = adoption.get("research_grant_ids", [])
205 now = timestamp(at)
206 if not grants or any(not grant_current(instance, gid, now, initial=False) for gid in grants):
207 missing.append("research_authority")
208 if not runtime.get("recovery_checked"):
209 missing.append("recovery_check")
210 if any(row.get("status") in {"checking", "ready"} for row in runtime.get("trust_transitions", {}).values()):
211 missing.append("trust_transition")
212 return {"satisfied": not missing, "missing": list(dict.fromkeys(missing)),
213 "observation_errors": observation_errors,
214 "trust_environment_id": instance.get("active_trust_environment")}

217def grant_current(instance, gid, at, *, initial):
218 grant = instance.get("grants", {}).get(gid, {})
219 if grant.get("kind") != "local" or grant.get("status") != "active" or not grant.get("source_ref"):
220 return False
221 if initial and (not grant.get("expires_at") or not set(grant.get("roles", [])).issubset(INITIAL_ROLES)
222 or grant.get('reuse') != 'reusable'):
223 return False
224 if not initial and (grant.get("initialization_only") or gid in
225 instance.get('lifecycle', {}).get('adoption', {}).get('initial_grant_ids', [])):
226 return False
227 return bool(grant.get("roles")) and grant_time_reason(grant, at) is None
