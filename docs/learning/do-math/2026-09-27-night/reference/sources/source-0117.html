1"""Scripted model runners for engine drives, and the role outputs they write.
2
3``ScriptedRunner`` answers every role with the smallest output the role's
4contract accepts, built from its actual input: the preparation proposes one
5candidate (``research_fixtures.candidate``) or the ``preparation_candidates``
6given, the plan selects every ready task and assesses every returned one, the
7work returns one theorem (formal when ``formal=True``), reviews match and the
8goal audit is satisfied when eligible knowledge exists. Its output passes
9through the producer path a real worker takes (``bind_candidates``,
10``seal_response``, schema validation, ``submit_intent``) inside a real local
11job (``local_job_runner``). ``invalid_plan`` and ``fail_role`` inject the two
12failures tests ask for; ``prompts`` records every (role, prompt).
13
14- ``common(role)``: the neutral fields every role output carries, plus the
15 empty shape of that role's own fields.
16- ``context_from(prompt, workspace=None)``: the canonical input context in a
17 prompt, with its source references read from ``workspace`` when given.
18- ``contract_checks(candidates)``: the smallest preparation self-check per candidate.
19- ``rating(**identity)``: one model assessment row for ``identity``.
20- ``formal_ok``: a compiler double that verifies the two fixture sources
21 (``theorem t : True := by trivial`` and ``add_zero_fixture``) and nothing else.
22"""

158def formal_ok(*, source, declarations, workspace, lean_command,
159 allowed_axioms=(), should_stop=None, expected_toolchain=None, on_started=None, launch_ticket=None):
160 if launch_ticket is not None:
161 from do_math.launch import validate_start
162 validate_start(launch_ticket)
163 if source == "theorem t : True := by trivial" and declarations == ["t"]:
164 signature, dependencies, value = "True", ["True", "True.intro"], "True.intro"
165 elif (source == "theorem add_zero_fixture (n : Nat) : n + 0 = n := Nat.add_zero n"
166 and declarations == ["add_zero_fixture"]):
167 signature, dependencies, value = "(n : Nat) -> n + 0 = n", ["Nat", "Nat.add_zero"], "Nat.add_zero"
168 else:
169 raise AssertionError("No explicit compiler fixture covers this source and declaration set")
170 signatures = {name: f"{name} : {signature}" for name in declarations}
171 return {"status": "verified", "launch_not_created": True,
172 "source_sha256": hashlib.sha256(source.encode()).hexdigest(),
173 "declarations": declarations, "axioms": {name: [] for name in declarations},
174 "declaration_signatures": signatures,
175 "signature_hash": hashlib.sha256(json.dumps(signatures, sort_keys=True).encode()).hexdigest(),
176 "allowed_axioms": list(allowed_axioms),
177 "dependency_closure": {name: dependencies for name in declarations},
178 "dependency_evidence": {name: {ref: "fixture-" + ref for ref in dependencies} for name in declarations},
179 "declaration_evidence": {name: {"type": signature, "value": value} for name in declarations},
180 "toolchain": expected_toolchain or {"executable": "test-lean", "sha256": "f" * 64,
181 "version": "Lean test 4.32.0", "version_exit": 0, "configured_command": list(lean_command)}}
