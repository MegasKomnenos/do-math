14class CliSurfaceTests(unittest.TestCase):
15 def test_file_query_and_read_include_resolved_snapshot_provenance(self):
16 # CLI imports ENGINE, so keep this boundary test independent of any
17 # concurrent engine/observation implementation work.
18 from do_math.cli import build_parser, execute
19 with tempfile.TemporaryDirectory() as directory:
20 root = Path(directory)
21 git_init(root)
22 identifier = new_id("RI")
23 store = ResearchRepository(root, identifier)
24 head = store.initialize(*current_state({"id": identifier, "sources": [], "knowledge": {}, "trust_environments": {}},
25 {"tasks": {}, "executions": {}, "queue": [], "status": "ready", "question_links": []}))
26 parser = build_parser()
27 query, code = execute(parser.parse_args(["query", "--repository", str(root), "--instance", identifier,
28 "--files", "--prefix", "research/", "--limit", "1"]))
29 self.assertEqual(code, 0)
30 self.assertEqual(query["provenance"]["snapshot"], head)
31 self.assertEqual(query["provenance"]["filters"]["prefix"], "research/")
32 path = store.list_files("research/", head)[0]
33 read, code = execute(parser.parse_args(["read", "--repository", str(root), "--instance", identifier,
34 "--path", path]))
35 self.assertEqual(code, 0)
36 self.assertEqual(read["snapshot"], head)
37 self.assertEqual(len(read["sha256"]), 64)
