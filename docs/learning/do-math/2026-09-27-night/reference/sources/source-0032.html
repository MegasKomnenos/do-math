40def validate_config(config: dict) -> None:
41 """The configuration section fixes keys, types and ranges; only cross-field rules remain here."""
42 schema = config_schema()
43 unknown = set(config) - set(schema["properties"])
44 if unknown:
45 raise ValueError("configuration has fields outside the stored configuration section: "
46 + ", ".join(sorted(unknown)))
47 config = {**DEFAULT_CONFIG, **config}
48 # A requested host constraint may still carry its request authority alias,
49 # which the stored schema refuses; the host section is checked below instead.
50 error = next(jsonschema.Draft202012Validator(schema).iter_errors(
51 {key: value for key, value in config.items() if key != "host_resources"}), None)
52 if error is not None:
53 raise ValueError("config/" + "/".join(str(part) for part in error.absolute_path) + ": " + error.message)
54 model_policy.validate_policy(config)
55 if config["scholarly_server"] is not None:
56 validate_scholarly_server(config["scholarly_server"])
57 if config["provider_retry_max_seconds"] < config["provider_retry_initial_seconds"]:
58 raise ValueError("provider retry maximum must be at least the initial interval")
59 if config["lean_command"] is not None and any("\0" in part for part in config["lean_command"]):
60 raise ValueError("lean_command must be a nonempty argv list")
61 if config.get('host_resources') is not None:
62 execution_profiles.validate_host_resources(config['host_resources'])
63 if config['host_resources']['constraint']['host_id'] != platform.node():
64 raise ValueError('host_resources must name this local host')
