75class PayloadEncodingTests(unittest.TestCase):
76 def test_payloads_are_exact_shared_and_immutable(self):
77 text = '원문 코드포인트\n' * 20000
78 value = payloads.seal_large({'title':text, 'source':{'text':text}})
79 envelope={'contract_id':'test','family':'test','record':value}
80 encoded, blobs=payloads.encode(envelope)
81 self.assertEqual(len(blobs), 1)
82 self.assertLess(len(canonical_json(encoded)), 1500)
83 def read(path): return blobs[path]
84 read.decoded={}
85 restored=payloads.decode(encoded, read)
86 self.assertEqual(restored, value)
87 self.assertTrue(restored['title'] is restored['source']['text'], 'one immutable value should be shared')
88 bad = deepcopy(encoded)
89 bad['payloads'][0]['sha256']='0'*64
90 with self.assertRaises(ValueError): payloads.decode(bad, read)
