29def validate_resource_limits(value):
30 if value is None:
31 return None
32 if (not isinstance(value, Mapping) or set(value) != {"memory_bytes", "cpu_rate_basis_points"}
33 or any(type(v) is not int or v <= 0 for v in value.values())
34 or value["memory_bytes"] > (1 << (8 * ctypes.sizeof(ctypes.c_size_t))) - 1
35 or value["cpu_rate_basis_points"] > 10000):
36 raise ValueError("Job limits need positive commit bytes and CPU hard-cap basis points in 1..10000")
37 return dict(value)

238def _environment_block(env: Mapping[str, str] | None):
239 if env is None:
240 return None
241 entries: list[str] = []
242 for key, value in env.items():
243 if not isinstance(key, str) or not isinstance(value, str):
244 raise TypeError("environment names and values must be strings")
245 if not key or "=" in key or "\x00" in key or "\x00" in value:
246 raise ValueError("environment names and values must be NUL-free; names cannot contain '='")
247 entries.append(f"{key}={value}")
248 entries.sort(key=str.upper)
249 return ctypes.create_unicode_buffer("\x00".join(entries) + "\x00\x00")

461 def resume(self) -> None:
462 """Start the initially suspended root exactly once."""
463 if self._resumed:
464 raise RuntimeError("root thread has already been resumed")
465 thread = self._require_open_handle(self._thread, "root thread")
466 if _kernel32.ResumeThread(thread) == _WAIT_FAILED:
467 _raise_last_error("ResumeThread")
468 self._resumed = True
469 if not _kernel32.CloseHandle(thread):
470 self._thread = None
471 _raise_last_error("CloseHandle(root thread)")

474 def wait(self, timeout: float | None = None) -> bool:
475 """Wait for the root; return ``False`` on timeout without dropping ownership."""
476 if timeout is None:
477 milliseconds = _INFINITE
478 else:
479 if timeout < 0:
480 raise ValueError("timeout must be non-negative")
481 milliseconds = min(int(timeout * 1000), _INFINITE - 1)
482 result = _kernel32.WaitForSingleObject(self.process_handle, milliseconds)
483 if result == _WAIT_OBJECT_0:
484 return True
485 if result == _WAIT_TIMEOUT:
486 return False
