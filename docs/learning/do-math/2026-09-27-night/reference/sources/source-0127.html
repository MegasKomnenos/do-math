32def response_body(response):
33 envelope = json.loads(response.stdout)
34 assert envelope['schema_version'] == 5
35 assert envelope['exit_code'] == response.returncode
36 assert ('result' in envelope) != ('error' in envelope)
37 return envelope.get('result', envelope.get('error'))

40def cli_instance(root, shared):
41 """``restore_template`` build: one instance the CLI's ``init`` created; returns its result.
42
43 The repository is ``root / 'repository'``, the workspace ``root / 'workspace'``,
44 and ``root / 'operator-disabled.json'`` the configuration file ``init`` read.
45 """
46 repository = git_init(root / "repository")
47 configuration = root / "operator-disabled.json"
48 configuration.write_text(json.dumps({"operator_enabled": False}), encoding="utf-8")
49 initialized = run_cli(
50 "init", "--repository", str(repository), "--workspace", str(root / "workspace"),
51 "--goal", "Prove a small theorem", "--title", "Small theorem", "--config-file", str(configuration),
52 cwd=root)
53 if initialized.returncode != 0:
54 raise AssertionError(initialized.stdout + initialized.stderr)
55 return response_body(initialized)

506 def test_normalized_control_identity_at_public_boundary(self):
507 # The CLI passes the command's identity inputs to ENGINE's control, which
508 # derives the request ID (control_contracts.control_request_id), and reports
509 # the ID control returns: the same command repeats one ID, a new key makes another.
510 from do_math import cli
511 from do_math.control_contracts import control_request_id
512 forwarded, ids = [], []
513 def admit(repository, instance, action, **kwargs):
514 forwarded.append(kwargs)
515 ids.append(kwargs['request_id'] or control_request_id(instance, action, text=kwargs['text'],
516 config=kwargs['config'], origin=kwargs['origin'], idempotency_key=kwargs['idempotency_key']))
517 return {'request_id': ids[-1], 'accepted': True}
518 reported = []
519 with patch.object(cli,'control',side_effect=admit), patch.object(cli,'ResearchEngine') as engine:
520 engine.return_value.process_controls.return_value={'status':'paused','snapshot':'a'*40}
521 engine.return_value.control_result.return_value=None
522 engine.return_value.runtime={'history':[]}
523 for extra in ([], [], ['--idempotency-key','again']):
524 output = io.StringIO()
525 with redirect_stdout(output):
526 code=cli.main(['pause','--repository','.','--instance','RI-test','--reason','P',*extra])
527 self.assertEqual(code,0)
528 reported.append(json.loads(output.getvalue())['request_id'])
529 self.assertEqual([row['request_id'] for row in forwarded], [None, None, None])
530 self.assertEqual([row['idempotency_key'] for row in forwarded], [None, None, 'again'])
531 self.assertEqual(reported, ids)
532 self.assertEqual(ids[0],ids[1])
533 self.assertNotEqual(ids[0],ids[2])
