82@contextmanager
83def owner_calls(engine):
84 """Run the calls started inside the block on ModelCalls, as the scheduler does.
85
86 The engine has no synchronous call path: machine steps and model calls
87 start on ``engine._calls``. Leaving the block answers their callbacks,
88 receives their control transport requests (a submission intent waits for
89 its receipt), collects every call and resumes its continuation once.
90 """
91 from do_math.parallel import ModelCalls
92 calls = engine._calls = ModelCalls(on_returned=engine._local_call_returned,
93 on_observation_error=engine._observation_failed)
94 try:
95 yield calls
96 while calls.futures:
97 calls.wait(0.05)
98 calls.pump()
99 # The owner loop pumps the transport every pass (engine._controls).
100 if engine._transport is not None:
101 engine._transport.pump(engine._publish_control_request)
102 calls.collect()
103 finally:
104 calls.close()
105 engine._calls = None

108def start_model_call(engine, action):
109 """Begin one AI action's call as the scheduler does; use inside ``owner_calls``.
110
111 The scheduler passes the action's lifecycle gate (``_dispatchable``) in the
112 pass that starts it, so this passes the same gate first.
113 """
114 engine._action_lifecycle_gate(action)
115 generator = engine._call_steps(action)
116 try:
117 arguments = next(generator)
118 except StopIteration as done:
119 return done.value
120 engine._calls.start_call(generator, engine.runner.run, arguments, execution_id=arguments.get('execution_id'))
121 return False

124def complete_academic_task(engine, task, basis):
125 """Supply a returned report and apply the real independent assessment rules."""
126 from do_math.execution_lifecycle import transition_execution
127 from do_math.task_scope import apply_task_assessments
128 from research_fixtures import assessment
129
130 executions = []
131 for role in ("research_work", "research_plan"):
132 action = {"id": new_id("AC"), "role": role, "target": task["id"]}
133 execution = engine._register_execution(action, {
134 "id": new_id("EX"), "role": role, "status": "running",
135 "input_solution_basis": deepcopy(basis)})
136 at = engine._lifecycle_now().isoformat()
137 transition_execution(execution, "stopping", at=at, stop_phase="terminating")
138 transition_execution(execution, "ended", at=at, outcome="execution_incomplete")
139 executions.append(execution["id"])
140 task["results"] = [{"execution_id": executions[0], "report": "Synthetic returned academic account."}]
141 completed = assessment(task)
142 engine.instance, engine.runtime = apply_task_assessments(
143 engine.instance, engine.runtime, [completed], execution_id=executions[1],
144 input_tasks=deepcopy(engine.runtime["tasks"]))
145 engine._record_academic_assessment(engine.runtime["tasks"][task["id"]], completed, executions[1])
146 engine._ensure_academic_followup()
147 engine._save()
