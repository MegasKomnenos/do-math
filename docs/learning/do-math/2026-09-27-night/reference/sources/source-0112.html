1"""Doubles at the provider, host-process and external-service boundaries.
2
3Nothing here starts a model, a Lean compiler or a network service of its own.
4Each double stands where the product reaches something outside the process:
5
6- Model runner: ``Adapter`` is a scripted provider collector for a real
7 ``ModelRunner`` (``runner_with``); ``mock_adapter`` is a bare Mock with the
8 provider's real worker options. ``Job`` and ``Execution`` are the process
9 handles ``Adapter.start`` returns. ``FakeJob`` answers ``inspect`` with a
10 fixed ``JobInspection``.
11- Claude stream: ``claude_stream(*events)`` encodes JSONL rows;
12 ``CLAUDE_INIT`` and ``CLAUDE_RESULT`` are the first and last rows of a
13 successful run.
14- Provider limits over session slots: ``provider_repository(root)`` initializes
15 one ready instance in the Git repository at ``root`` whose three tiers all use
16 ``PROVIDER_MODEL`` (three work slots). ``admit`` registers and admits one
17 execution, ``report`` applies a synthetic runner result (a
18 ``PROVIDER_LIMIT_FAILURE`` or a completed call), ``finish`` releases and ends
19 it, and ``transact`` runs one such operation under the writer lock and saves.
20 ``instant`` turns epoch seconds into the ISO time the records use.
21- Execution profiles: ``PROFILE_CONFIG`` is a complete profile configuration
22 for ``build_profiles``; ``host_capacity_fixture()`` adds an approved host
23 constraint and host-bounded work, recovery and formal-check profiles.
24- Lean: ``actual_command()`` is the installed compiler command
25 (``DO_MATH_KNOWLEDGE_TEST_LEAN`` overrides it); only ``lean``-labelled tests call it.
26- HTTP: ``HTTPFixtureHandler`` answers POST with 201 and GET with a redirect.
27"""

137def admit(slots, instance, runtime, identifier, role="research_work", current=100):

143 actor_kind="ai", input_manifest=[], profile_ref=profile_ref(profile), created_at=instant(current))
144 execution["role"] = role
145 if not slots.acquire(identifier, profile=profile, execution=execution, current_time=current):
146 return False

161def finish(slots, runtime, identifier, current, *, close_activity=True):
162 if not slots.release(identifier):
163 raise AssertionError("test execution without an acknowledged process did not settle")
164 execution = runtime["executions"][identifier]
165 lifecycle.transition_execution(execution, "stopping", at=instant(current), stop_phase="terminating")
166 lifecycle.transition_execution(execution, "ended", at=instant(current), outcome="start_failed",
167 disposition={"kind": "synthetic_provider_observation"})
168 if not close_activity:
169 return
170 activities.finish_activity(runtime, execution["owner_activity_id"], result_refs=[],
171 unresolved=[{"reason": "No model process was started in this allocator fixture"}],
172 followups=[], at=instant(current))

175def transact(repository, operation):
176 with repository.writer_lock():
177 head, instance, runtime = repository.load()
178 slots = SessionSlots(runtime, instance)
179 result = operation(slots, instance, runtime)
180 repository.save_with_state(head, instance, runtime)[0]
181 return result
