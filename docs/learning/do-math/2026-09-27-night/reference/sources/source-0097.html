52 def register_family(self, root, name, *, shape="map", schema=None,
53 validator=None, references=None, identity=True):
54 if root not in {"research", "runtime"} or not re.fullmatch(r"[a-z][a-z0-9_]*", name):
55 raise ValueError("invalid family root/name")
56 if shape not in {"map", "list", "singleton", "metadata"}:
57 raise ValueError("invalid family shape")
58 if shape == "metadata" and identity:
59 raise ValueError("metadata fields cannot allocate independent record identities")
60 if self.family(root, name) is not None:
61 raise ValueError("record family already registered: " + name)
62 if schema is None:
63 raise ValueError("explicit record schema is required: " + name)
64 schema = deepcopy(schema)
65 if (shape in {"map", "list"} or identity) and (schema.get("type") != "object" or schema.get("additionalProperties") is not False):
66 raise ValueError("record schema must reject unknown fields: " + name)
67 self._families[root, name] = RegisteredFamily(root, name, shape, schema, validator, references, identity)
68 self._manifest = None
69
70 def family(self, root, name):
71 return self._families.get((root, name))
72
73 def families(self):
74 return tuple(self._families[key] for key in sorted(self._families))

76 def manifest(self):
77 """Describe the registered schemas; its digest identifies the writer.
78
79 The digest is computed once; only register_family resets it.
80 """
81 if self._manifest is None:
82 families = [{"root": f.root, "name": f.name, "shape": f.shape, "identity": f.identity,
83 "schema": f.schema} for f in self.families()]
84 self._manifest = json_digest({"contract_id": STATE_CONTRACT_ID, "families": families})
85 return {"contract_id": STATE_CONTRACT_ID, "sha256": self._manifest}

87 def validate(self, instance, runtime, parent=None):
88 """Pure schema and relationship validation; no ID/time allocation.
89
90 ``parent`` is an (instance, runtime) pair already validated under this
91 contract. A record equal to the parent's record at the same family and
92 ID skips the record-local checks (schema and family validator), whose
93 result depends on the record alone. Relations, identity uniqueness and
94 semantic rules are still checked over the whole state.
95
96 Equality is Python ``==``, under which ``1 == 1.0 == True``. A float
97 cannot pass: payloads.seal_large rejects every float of a candidate
98 before this comparison. A bool swapped for an equal integer (or back)
99 in a record otherwise equal to the parent's is an accepted gap: it
100 skips that record's schema check and keeps the parent's bytes
101 (repository._encode_families). A type-strict comparison would walk
102 every node in Python, about 27 times the cost of ``==`` on a 3.6 MB
103 state, twice per save.
104 """
105 rows, identifiers = [], {}
106 for index, (root, state) in enumerate((("research", instance), ("runtime", runtime))):
107 prior_state = parent[index] if parent is not None else {}
108 if not isinstance(state, dict):
109 raise ValueError(root + " must be an object")
110 if root == "research" and state.get("object_format") not in {"sha1", "sha256"}:
111 raise ValueError("research object_format is missing or unsupported")
112 for name, value in state.items():
113 family = self.family(root, name)
114 if family is None:
115 raise ValueError("unregistered family: " + root + "/" + name)
116 if family.shape == "map":
117 if not isinstance(value, dict):
118 raise ValueError(name + " must be a record map")
119 entries = list(value.items())
120 elif family.shape == "list":
121 if not isinstance(value, list):
122 raise ValueError(name + " must be a record list")
123 entries = [(None, row) for row in value]
124 else:
125 entries = [(None, value)]
126 prior = prior_state.get(name, _ABSENT)
127 if family.shape == "list" and isinstance(prior, list):
128 prior = {row.get("id"): row for row in prior if isinstance(row, dict)}
129 for key, record in entries:
130 if family.shape in {"map", "list"}:
131 identifier = key if key is not None else record.get("id") if isinstance(record, dict) else None
132 before = prior.get(identifier, _ABSENT) if isinstance(prior, dict) else _ABSENT
133 else:
134 before = prior
135 if before is _ABSENT or before != record:
136 validate_record(family.checker, record, f"{root}/{name}/{key}")
137 if family.validator:
138 family.validator(deepcopy(record))
139 if family.identity:
140 # The family schema holds the RECORD_ID pattern; the
141 # path segment rule adds what a pattern cannot say.
142 identifier = record.get("id") if isinstance(record, dict) else None
143 if not isinstance(identifier, str):
144 raise ValueError(name + " requires a path-safe record id")
145 try:
146 check_path_segment(identifier)
147 except ValueError as error:
148 raise ValueError(name + " requires a path-safe record id: " + str(error)) from error
149 if key is not None and key != identifier:
150 raise ValueError(name + " map key and id differ")
151 if identifier in identifiers:
152 raise ValueError("record IDs must be globally unique: " + identifier)
153 identifiers[identifier] = (root, name, record)
154 rows.append((family, record))
155 if instance.get("id"):
156 if instance["id"] in identifiers:
157 raise ValueError("record IDs must be globally unique: " + instance["id"])
158 identifiers[instance["id"]] = ("research", "id", instance)
159 for family, record in rows:
160 for ref in family.references(record) if family.references else ():
161 target, expected_family = ref if isinstance(ref, tuple) else (ref, None)
162 if target is None:
163 continue
164 if not isinstance(target, str) or target not in identifiers:
165 raise ValueError(f"{family.name} references unknown record: {target}")
166 if expected_family and identifiers[target][1] not in expected_family.split("|"):
167 raise ValueError(f"{family.name} reference has wrong family: {target}")
168 from .record_schemas import validate_semantics
169 validate_semantics(instance, runtime, identifiers, parent, contract=self)
170 return identifiers

186def validate_record(checker, record, label):
187 """Raise ValueError naming ``label`` and the first schema violation of ``record``."""
188 errors = sorted(checker.iter_errors(record), key=lambda e: str(e.path))
189 if errors:
190 location = "/".join(str(part) for part in errors[0].absolute_path)
