36def reviewed_task(root, shared, *, config):
37 """``restore_template`` build: one task from a real preparation, ready after a real
38 contract review, with the plan it opened queued. Returns the instance ID."""
39 instance_id = active_instance(root, shared, config=config)
40 engine = purpose_engine(root, instance_id)
41 for _ in range(45):
42 engine.run(max_steps=1)
43 if (any(row['state'] == 'ready' for row in engine._task_readiness.values())
44 and any(a['role'] == 'research_plan' for a in engine.runtime['queue'])):
45 return instance_id
46 if engine.runtime['status'] in {'paused', 'closed', 'waiting'}:
47 break
48 raise AssertionError('No reviewed task and plan: ' + json.dumps({
49 'status': engine.runtime['status'], 'queue': engine.runtime['queue']}))
