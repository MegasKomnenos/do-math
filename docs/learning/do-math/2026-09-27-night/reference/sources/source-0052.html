16def bind_host_configuration(instance, config, *, origin):
17 """Resolve the explicit request-source alias to the preserved user input."""
18 config = deepcopy(config)
19 host = config.get('host_resources')
20 if host is None or 'configuration:request' not in host['constraint']['authority_refs']:
21 return config
22 if origin != 'user':
23 raise ValueError('host configuration request requires user authority')
24 raw = canonical_json({'host_resources': host})
25 identifier = 'GS-host-' + exact_content_hash({'host_resources': host})
26 source = governance.preserve_source(identifier, raw.decode('utf-8'), origin='user')
27 sources = instance['governance_sources']
28 if identifier in sources and sources[identifier] != source:
29 raise ValueError('host configuration source identity changed')
30 sources[identifier] = source
31 constraint = host['constraint']
32 host['constraint'] = execution_profiles.host_constraint(host_id=constraint['host_id'],
33 limits={'jobs': constraint['max_jobs'], 'memory_bytes': constraint['memory_bytes'],
34 'cpu_rate_basis_points': constraint['cpu_rate_basis_points']},
35 control_reserve=constraint['control_reserve'],
36 authority_refs=[identifier if ref == 'configuration:request' else ref for ref in constraint['authority_refs']])
37 return config
