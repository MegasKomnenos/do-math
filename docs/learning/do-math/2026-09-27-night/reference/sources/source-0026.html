21def usage_total(segments):
22 """Sum reported segment counters only, never cumulative modelUsage/cost."""
23 keys = ("input_tokens", "output_tokens", "cache_read_input_tokens", "cache_creation_input_tokens")
24 return {key: sum(row["usage"][key] for row in segments if _number(row["usage"].get(key)))

28def provider_limit(parsed):
29 """Classify the last native admission event, not tool errors or overage flags."""
30 events = parsed.get("rate_limits", [])
31 if not events or events[-1]["info"].get("status") != "rejected" or parsed.get("turn_completed"):
32 return None
33 record = events[-1]
34 info = record["info"]
35 windows = [{"type": info.get("rateLimitType"), "resetsAt": info.get("resetsAt")}]
36 windows_info = info.get("unifiedWindows")
37 for name, window in (windows_info if isinstance(windows_info, dict) else {}).items():
38 if isinstance(window, dict) and (window.get("status") == "rejected" or
39 (_number(window.get("utilization")) and window["utilization"] >= 1)):
40 windows.append({"type": name, **window})
41 times = [w["resetsAt"] for w in windows if _number(w.get("resetsAt"))]
42 retry_at = None
43 if times:
44 try:
45 retry_at = datetime.fromtimestamp(max(times), timezone.utc).isoformat()
46 except (ValueError, OverflowError, OSError):
47 pass
48 return {"kind": "provider_limit", "provider": "claude", "retry_at": retry_at,
49 "limit_windows": windows, "evidence_line": record["line"], "native_info": info}
