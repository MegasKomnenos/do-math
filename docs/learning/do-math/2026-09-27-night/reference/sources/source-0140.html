14 def test_current_resolution_commits_only_with_its_complete_bundle(self):
15 owner = bare_engine()

18 'dispositions': [], 'task_assessments': [], 'preparation_requests': [], 'goal_audit': False}
19 context = {'action': action, 'execution_id': 'EX-current', 'snapshot': 'a' * 40}
20 claim = {'action_id': 'AC-failed', 'evidence_execution_id': 'current',
21 'reason': 'The current accepted judgment covers the failed request'}
22 runtime = {'history': [], 'integrations': {}, 'waiting_actions': {'AC-failed': {
23 'kind': 'execution_failure', 'execution_id': 'EX-failed',
24 'action': {'role': 'research_plan', 'target': 'A'}}},
25 'executions': {'EX-current': {'action': action, 'extensions': {'action_resolutions': [claim]}}}}
26 def apply(instance, candidate, bundle):

28 owner.instance = instance
29 owner._apply_action_resolutions('EX-current', accepting_current=True)
30 def run(state, validate, publish=lambda *_: None):
31 return integration.apply_bundles({}, state, output, {}, context, execution_id='EX-current',
32 apply_bundle=apply, validate_current=lambda *_: None,
33 validate_candidate=validate, publish=publish)
34 before = deepcopy(runtime)
35 failed = run(runtime, lambda *_: (_ for _ in ()).throw(ValueError('later candidate failure')))
36 self.assertEqual(failed['status'], 'partial')
37 self.assertEqual(runtime, before)
