51class DigestCharacterizationTests(unittest.TestCase):
52 def test_state_value_digests_stay_fixed(self):
53 for value, digest in STATE_DIGESTS:
54 with self.subTest(value=value):
55 self.assertEqual(exact_content_hash(value), digest)
56
57 def test_json_digests_stay_fixed(self):
58 for value, digest in JSON_DIGESTS:
59 with self.subTest(value=value):
60 self.assertEqual(json_digest(value), digest)
61
62 def test_observation_rule_digests_stay_fixed(self):
63 for value, digest in RULE_DIGESTS:
64 with self.subTest(value=value):
65 self.assertEqual(hashlib.sha256(_raw_json(value).encode()).hexdigest(), digest)
66
67 def test_state_value_identity_refuses_values_outside_the_storage_domain(self):
68 for value in ({"limit": 0.5}, [1.0], {"value": float("nan")}):
69 with self.subTest(value=value), self.assertRaises(ValueError):
70 exact_content_hash(value)

72 def test_stored_academic_and_external_identities_use_the_state_value_digest(self):
73 requirements = academic.requirement_record([{"id": "R", "scope": "S", "assessor": "research_plan",
74 "criteria": ["C"]}])
