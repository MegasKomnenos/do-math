249def validate_rules(rules: list[dict], *, allow_empty: bool = False) -> list[dict]:
250 if not isinstance(rules, list) or len(rules) > 16 or (not rules and not allow_empty):
251 raise ValueError("observation rules must be a nonempty list of at most 16 rules")
252 seen = set()
253 for rule in rules:
254 expected = _RULE_FIELDS if isinstance(rule, dict) and "repeat_interval" in rule else _RULE_FIELDS - {"repeat_interval"}
255 if not isinstance(rule, dict) or set(rule) != expected:
256 raise ValueError("invalid observation rule fields")
257 if not isinstance(rule.get("id"), str) or not rule["id"] or rule["id"] in seen:
258 raise ValueError("observation rule IDs must be distinct and nonempty")
259 seen.add(rule["id"])
260 operator = rule.get("operator")
261 if not isinstance(operator, str) or operator not in OPERATORS:
262 raise ValueError("unsupported observation operator")
263 spec = OPERATORS[operator]
264 subject, basis = rule.get("subject"), rule.get("basis")
265 if not isinstance(subject, dict) or set(subject) != {"kind", "id"}:
266 raise ValueError("rule subject must contain exactly kind and id")
267 if subject.get("kind") != spec["subject_kind"] or not isinstance(subject.get("id"), str) or not subject["id"]:
268 raise ValueError("rule subject is incompatible with its operator")
269 if basis != {"source": spec["source"], "unit": spec["unit"]}:
270 raise ValueError("rule basis of " + operator + " must be source " + spec["source"] + " in " + spec["unit"])
271 evidence = rule.get("required_evidence")
272 if (not isinstance(evidence, list) or not evidence or
273 any(not isinstance(ref, str) or not ref for ref in evidence) or len(evidence) != len(set(evidence))):
274 raise ValueError("rule required_evidence must be a nonempty unique string list")
275 if operator == "wall_time_reached":
276 _utc(rule.get("threshold"), "wall_time_reached threshold")
277 if "repeat_interval" in rule:
278 raise ValueError("wall_time_reached cannot repeat")
279 continue
280 threshold = rule.get("threshold")
281 if isinstance(threshold, bool) or not isinstance(threshold, int) or threshold < 1:
282 raise ValueError("rule threshold must be a positive integer")
283 interval = rule.get("repeat_interval")
284 if "repeat_interval" in rule and (isinstance(interval, bool) or not isinstance(interval, int) or interval < 1):
285 raise ValueError("repeat_interval must be a positive integer")
286 if operator in _ONE_TIME_OPERATORS and (threshold != 1 or interval is not None):
287 raise ValueError(operator + " is a one-time recorded event")
288 return deepcopy(rules)

319 """Admit only signals registered in the immutable profile's observation contract.

340def evaluate_rules(rules: list[dict], snapshot: dict, *, consumed_boundaries: dict[str, int] | None = None) -> list[dict]:
341 """Evaluate already validated rules against one accumulated observation snapshot.
342
343 The tracker validates its rules when it is created and when they are replaced.
344 """
345 consumed = consumed_boundaries or {}
346 calls = set(snapshot.get("completed_call_ids", []))
347 boundaries = set(snapshot.get("safe_boundary_ids", []))
348 counters = snapshot.get("counters", {})
349 iterations = snapshot.get("iterations", {})
350 checkpoints = snapshot.get("checkpoint_counts", {})
351 due = []
352 for rule in rules:
353 operator, subject = rule["operator"], rule["subject"]["id"]
354 if operator == "counter_gte": value = counters.get(subject)
355 elif operator == "iteration_gte": value = iterations.get(subject)
356 elif operator == "checkpoint_count_gte": value = checkpoints.get(subject)
357 elif operator == "wall_time_reached":
358 observed = snapshot.get("observed_at")
359 value = 1 if observed is not None and _utc(observed, "observed_at") >= _utc(rule["threshold"], "threshold") else None
360 elif operator == "tool_call_completed": value = 1 if subject in calls else None
361 else: value = 1 if subject in boundaries else None
362 threshold = 1 if operator == "wall_time_reached" else rule["threshold"]
363 if value is not None and (type(value) is not int or value < 0):
364 raise ValueError("invalid observation value for " + rule["id"])
365 if value is None or value < threshold:
366 continue
367 key = _boundary_key(rule)
368 previous, interval = consumed.get(key, 0), rule.get("repeat_interval")
369 boundary = threshold if previous < threshold else threshold + ((previous - threshold) // interval + 1) * interval if interval is not None else None
370 if boundary is None or value < boundary:
371 continue
372 crossed = boundary if interval is None else threshold + ((value - threshold) // interval) * interval
373 due.append({"rule_id": rule["id"], "operator": operator, "subject": subject,
374 "basis": _json_safe(rule["basis"]), "required_evidence": list(rule["required_evidence"]),
375 "value": value, "boundary_key": key, "first_boundary": boundary, "boundary": crossed,
376 "next_boundary": None if interval is None else crossed + interval})
377 return due

391def read_checkpoint(workspace: Path) -> dict:
392 path = workspace / "checkpoint.json"
393 if not path.exists():
394 return {"status": "missing"}
395 try:
396 resolved = path.resolve(strict=True)
397 if not resolved.is_relative_to(workspace.resolve()) or path.is_symlink():
398 raise ValueError("checkpoint must stay within the execution workspace")
399 before = file_fingerprint(path)
400 if before[0] > CHECKPOINT_MAX_BYTES:
401 raise ValueError("checkpoint exceeds the 4 MiB record limit")
402 raw = read_bytes_shared(path)
403 if file_fingerprint(path) != before:
404 return {"status": "being_written"}
405 digest = hashlib.sha256(raw).hexdigest()
406 value = checkpoint_value(raw)
407 for reference in value["evidence_paths"]:
408 evidence = safe_path(workspace, reference)
409 relative = Path(reference)
410 if (relative.parts[0] != "input" and is_excluded(relative)) or not evidence.is_file():
411 raise ValueError("checkpoint evidence is unavailable or excluded from preservation: " + reference)
412 return {"status": "valid", "value": value, "sha256": digest, "raw": raw, "observed_at": utc_now()}
413 except Exception as error:
414 return {"status": "invalid", "error": f"{type(error).__name__}: {error}"}

594 def _observe_control_calls(self, event: dict) -> None:
595 """Match provider-observed identity and arguments before action accounting."""
596 calls = []
597 item = event.get("item", {})
598 if (event.get("type") in {"item.started", "item.completed"} and isinstance(item, dict)
599 and item.get("type") == "mcp_tool_call" and item.get("server") == CONTROL_SERVER):
600 calls.append((item.get("id"), item.get("tool"), item.get("arguments")))
601 if event.get("type") == "assistant":
602 message = event.get("message", {})
603 for block in message.get("content", []) if isinstance(message, dict) else []:
604 if isinstance(block, dict) and block.get("type") == "tool_use":
605 prefix = "mcp__" + CONTROL_SERVER + "__"
606 name = block.get("name", "")
607 if isinstance(name, str) and name.startswith(prefix):
608 calls.append((block.get("id"), name[len(prefix):], block.get("input")))
609 for call_id, tool, arguments in calls:
610 if not isinstance(call_id, str) or not isinstance(arguments, dict): continue
611 if tool == SUBMIT_TOOL and arguments.get("kind") == SAFETY_BOUNDARY:
612 payload = arguments.get("payload")
613 request_id = arguments.get("request_id")
614 if (isinstance(request_id, str) and request_id and isinstance(payload, dict)
615 and payload.get("inflight_calls") is False
616 and isinstance(payload.get("call_ids", []), list)
617 and all(isinstance(cid, str) for cid in payload.get("call_ids", []))):
618 # Submitting the boundary is not an inflight research operation.
619 # Its receipt still owns validation and making obligations available.
620 self.boundary_call_ids.add(call_id)
621 self.boundary_request_ids.add(request_id)
622 continue
623 if tool == RECEIPT_TOOL:
624 request_id = arguments.get("request_id")
625 if isinstance(request_id, str):
626 if request_id in self.control_requests:
627 self.self_check_call_ids.add(call_id)
628 elif request_id in self.boundary_request_ids:
629 self.boundary_call_ids.add(call_id)
630 continue
631 if tool == SELF_CHECK_TOOL:
632 oid = arguments.get("self_check_id")
633 payload = {"self_check_id": oid, "response": arguments.get("response")}
634 request_id = self_check_request_id(oid) if isinstance(oid, str) else None
635 elif tool == SUBMIT_TOOL and arguments.get("kind") == SELF_CHECK_RESPONSE:
636 payload = arguments.get("payload")
637 oid = payload.get("self_check_id") if isinstance(payload, dict) else None
638 request_id = arguments.get("request_id")
639 else:
640 continue
641 if not isinstance(oid, str) or oid not in self.obligations or not isinstance(request_id, str): continue
642 if payload.get("response") is None and tool == SELF_CHECK_TOOL:
643 # This installed tool reads the already-issued self-check only.
644 self.self_check_call_ids.add(call_id)
645 if self.self_check_polls.get(call_id, oid) != oid:
646 raise ValueError('self-check poll ID reused for another obligation')
647 self.self_check_polls[call_id] = oid
648 continue
649 fingerprint = hashlib.sha256(_raw_json(payload).encode()).hexdigest()
650 row = {"request_id": request_id, "obligation_id": oid, "payload_sha256": fingerprint}
651 if call_id in self.control_calls and self.control_calls[call_id] != row:
652 raise ValueError("control call ID reused with different request")
653 known = self.control_requests.get(request_id)
654 if known and known["payload_sha256"] != fingerprint:
655 raise ValueError("observed control arguments differ from received request")
656 self.control_calls[call_id] = row
657 if known: self.self_check_call_ids.add(call_id)
658 # A poll attempt or a generic safe boundary does not acknowledge this
659 # request. Match the successful provider-observed returned envelope.
660 if event.get('type') == 'item.completed' and isinstance(item, dict) and item.get('status') != 'failed' and not item.get('error'):
661 self._acknowledge_poll(item.get('id'), item.get('result'))
662 message = event.get('message')
663 if event.get('type') == 'user' and isinstance(message, dict) and isinstance(message.get('content'), list):
664 for block in message['content']:
665 if isinstance(block, dict) and block.get('type') == 'tool_result' and not block.get('is_error'):
666 self._acknowledge_poll(block.get('tool_use_id'), block.get('content'))

846 def record_self_check_response(self, response: dict, *, available_evidence_refs: set[str] | None = None,
847 receipt_id: str | None = None) -> dict:

849 obligation_id = response.get("obligation_id") if isinstance(response, dict) else None
850 obligation = self.obligations.get(obligation_id) if isinstance(obligation_id, str) else None
851 errors = []
852 try:
853 # The one response check the worker's sender also runs (이음새-정본.md §6).
854 if self_check_response_errors(response):
855 raise ValueError("self-check response does not match its schema")
856 _json_safe({key: value for key, value in response.items() if key != "rule_proposal"})
857 except (ValueError, TypeError):
858 errors.append("invalid_response_schema")
859 if obligation is None: errors.append("unknown_obligation")
860 if self.execution_status != "running": errors.append("execution_not_running")
861 if obligation is not None:
862 if obligation["status"] != "open": errors.append("obligation_not_open")
863 if response.get("execution_id") != self.execution_id: errors.append("execution_mismatch")
864 if response.get("rule_revision") != obligation["rule_revision"]: errors.append("revision_mismatch")
865 if obligation["response_availability"] != "available": errors.append("response_boundary_unconfirmed")
866 if obligation["deadline_offset_ms"] is not None and now_offset > obligation["deadline_offset_ms"]:
867 errors.append("response_deadline_passed")
868 if obligation.get("missed_reason"): errors.append(obligation["missed_reason"])
869 calls = [cid for cid, row in self.control_calls.items() if row["obligation_id"] == obligation_id]
870 if calls and not any(receipt_id is not None and self.control_requests.get(self.control_calls[cid]["request_id"], {}).get("receipt_id") == receipt_id
871 for cid in calls):
872 errors.append("control_request_unconfirmed")
873 bound = [row for row in self.control_requests.values() if receipt_id is not None and row["receipt_id"] == receipt_id]
874 if bound:
875 fingerprint = hashlib.sha256(_raw_json({"self_check_id": obligation_id, "response": response}).encode()).hexdigest()
876 if not any(row["payload_sha256"] == fingerprint for row in bound):
877 errors.append("control_response_mismatch")
878 if not errors:
879 # The schema has already fixed each row's fields, status and reference lists.
880 results = response["rule_results"]
881 expected = {row["rule_id"]: row for row in obligation["due"]}
882 received = [row["rule_id"] for row in results]
883 if len(received) != len(set(received)) or set(received) != set(expected): errors.append("rule_coverage_mismatch")
884 for row in results:
885 required = set(expected.get(row["rule_id"], {}).get("required_evidence", []))
886 supplied, declared_missing = set(row["evidence_refs"]), set(row["missing_evidence_refs"])
887 actual_missing = required - supplied
888 if row["status"] == "indeterminate":
889 if declared_missing != actual_missing: errors.append("indeterminate_evidence_mismatch")
890 elif actual_missing or declared_missing: errors.append("required_evidence_missing")
891 if available_evidence_refs is None: errors.append("evidence_currentness_unverified")
892 elif not supplied.issubset(available_evidence_refs): errors.append("evidence_not_current")
893 record = {"kind": "self_check_rejected" if errors else "self_check_received",
894 "id": f"SR-{self.execution_id}-{self.response_count + 1}", "receipt_ref": receipt_id,
895 "execution_id": self.execution_id,
896 "obligation_id": obligation_id if isinstance(obligation_id, str) else None,
897 "response_json": _raw_json(response),
898 "received_offset_ms": now_offset, "observed_at": utc_now()}
899 record["control_call_ids"] = [cid for cid, row in self.control_calls.items() if
900 row["obligation_id"] == obligation_id and receipt_id is not None and self.control_requests.get(row["request_id"], {}).get("receipt_id") == receipt_id]
901 if errors:
