82def inspect_state(instance: dict, runtime: dict, *, requested_areas: list[str] | None = None,
83 snapshot: str | None = None, read_artifact=None, artifact_errors=None,
84 read_artifacts=None, artifact_sizes=None) -> dict[str, Any]:
85 """Inspect declared persisted-state areas without inventing semantic facts.
86
87 Each area lists its actual checked invariants and missing evidence.
88 No result certifies mathematical truth, live providers or OS isolation.
89 The state must be one the repository loaded: loading validated it under the
90 state contract, and the areas add only conditions loading does not check.
91 ``artifact_sizes(paths)`` and ``read_artifacts(paths)`` return the stored
92 lengths or bytes of the paths present in the snapshot and omit absent ones.
93 ``artifact_sizes`` replaces byte-hash checks of referenced artifacts with
94 presence and stored-length checks; ``check`` omits it to compare bytes.
95 """
96 if requested_areas is None:
97 requested_areas = list(CHECK_AREAS)
98 if not isinstance(requested_areas, list) or any(not isinstance(name, str) for name in requested_areas):
99 raise ValueError("requested_areas must be a list of area names")
100 requested = list(dict.fromkeys(requested_areas))
101 findings: list[dict[str, Any]] = []
102 areas: list[dict[str, Any]] = []
103 not_checked: list[str] = []
104 for area in requested:
105 if area not in CHECK_AREAS:
106 not_checked.append(area)
107 findings.append(_finding(area, status="incomplete", message="unknown inspection area",
108 next_owner="operator"))
109 areas.append({"name": area, "status": "incomplete", "checked": False, "findings": 1})
110 continue
111 try:
112 local, checked, missing = _inspect_area(area, instance, runtime, read_artifact, read_artifacts, artifact_sizes)
113 except (ValueError, TypeError, KeyError, AttributeError) as error:
114 local = [_finding(area, status='blocked', message='malformed domain records: ' + str(error),
115 blocked_scope=[area], next_owner='maintenance')]
116 checked, missing = [], ['remaining invariants after malformed records']
117 if area == 'record_storage':
118 local.extend(_finding(area, status='blocked', message=str(error), blocked_scope=['artifacts'], next_owner='maintenance') for error in artifact_errors or [])
119 findings.extend(local)
120 if missing:
121 not_checked.append(area)
122 statuses = {row['status'] for row in local}
123 status = 'blocked' if 'blocked' in statuses else 'incomplete' if missing else 'degraded' if 'degraded' in statuses else 'healthy'
124 areas.append({'name': area, 'status': status, 'checked': not missing,
125 'checked_invariants': checked, 'not_checked': missing, 'findings': len(local)})
126 statuses = {finding["status"] for finding in findings}
127 if "blocked" in statuses:
128 overall = "blocked"
129 elif not_checked or "incomplete" in statuses:
130 overall = "incomplete"
131 elif "degraded" in statuses:
132 overall = "degraded"
133 else:
134 overall = "healthy"
135 return {"status": overall, "snapshot": snapshot, "requested_areas": requested, "areas": areas,
136 "not_checked": not_checked, "findings": findings,
137 "checked_scope": "persisted records only; no new mathematical, provider, or host verification"}
