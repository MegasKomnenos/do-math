66 def test_observation_burst_preserves_order_and_split_utf8_tail(self):
67 import json
68 from unittest.mock import patch
69 expected = [{"type": "item.updated", "index": index, "text": "x" * 960}
70 for index in range(2000)]
71 last = {"type": "item.updated", "index": 2000, "text": "가"}
72 encoded = json.dumps(last, ensure_ascii=False).encode("utf-8")
73 cut = encoded.index("가".encode("utf-8")) + 1
74 first = b"".join(json.dumps(row).encode("utf-8") + b"\r\n" for row in expected) + b"\r\n" + encoded[:cut]
75 second = encoded[cut:] + b"\r\nnot-json\r\n" + b'{"type":'

77 class BurstAdapter(Adapter):
78 polls = 0
79 def wait_for_termination(self, *args):
80 self.polls += 1
81 path = Path(self.request["stdout_path"])
82 if self.polls == 1:
83 path.write_bytes(first)
84 elif self.polls == 2:
85 with path.open("ab") as stream:
86 stream.write(second)
87 return self.polls >= 4

89 with tempfile.TemporaryDirectory() as directory:
90 root = Path(directory)
91 worker = runner_with(BurstAdapter({"result": "output_collected", "final": {"raw": b"{}"}, "jsonl": {}}))
92 observed = []
93 with patch.object(runner.time, "sleep"):
94 result = worker.run(prompt="fixture", output_schema={}, workspace=root, evidence_dir=root / "evidence",
95 model="fixture", effort="low", timeout_seconds=None, on_observation=observed.append)
96 self.assertEqual(result["status"], "completed")
97 events = [row for batch in observed for row in batch.events]
98 self.assertEqual([row for row in events if row.get("type") == "item.updated"], [*expected, last])
99 # After the Job ends, the runner also delivers the unterminated last row.
100 self.assertEqual([row for row in events if row.get("type") == "unparsed_observation"],
