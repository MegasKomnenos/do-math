33def build_review_policy(policy=None, *, reviewer_count=None, max_rounds=None, judge_slots=None):
34 """Validate the adopted review policy or construct the configured default."""
35 if policy is None:
36 count = _positive(reviewer_count, "reviewer count")
37 rounds = _positive(max_rounds, "round limit")
38 judges = list(judge_slots or ["judge-0", "judge-1", "judge-2"])
39 policy = {
40 "id": "review-policy-v5-default",
41 "required_perspectives": [],
42 "minimum_initial_slots": count,
43 "judge_slots": judges,
44 "round_limit": rounds,
45 # This is a derived finite upper bound for the existing finite cycle,
46 # not an asserted host or provider capacity.
47 "activity_execution_limit": count + rounds * (count + len(judges) + 1),
48 "overflow_disposition": "unresolved",
49 }
50 if not isinstance(policy, dict):
51 raise ValueError("review policy must be an object")
52 required = {"id", "required_perspectives", "minimum_initial_slots", "judge_slots",
53 "round_limit", "activity_execution_limit", "overflow_disposition"}
54 if set(policy) != required:
55 raise ValueError("review policy fields differ from the policy contract")
56 if not isinstance(policy["id"], str) or not policy["id"]:
57 raise ValueError("review policy ID is required")
58 distinct_texts(policy["required_perspectives"], "required perspectives")
59 _positive(policy["minimum_initial_slots"], "minimum initial slots")
60 judges = distinct_texts(policy["judge_slots"], "judge slots", nonempty=True)
61 if len(judges) != 3:
62 raise ValueError("review policy requires exactly three judge slots")
63 _positive(policy["round_limit"], "round limit")
64 _positive(policy["activity_execution_limit"], "activity execution limit")
65 if policy["overflow_disposition"] not in OVERFLOW_DISPOSITIONS:
66 raise ValueError("unsupported review overflow disposition")
67 return deepcopy(policy)

72 return executions_used + initial_slots + 1 + (3 if round_limit == 1 else initial_slots + 6)

75def validate_scope_assessment(assessment, policy, *, executions_used=0):
76 """Validate an assessment without letting it split the proposal's review scope.
77
78 The schema fixes its shape; the checks here are the meaning it has under the policy.
79 """
80 policy = build_review_policy(policy)
81 validate_record(_SCOPE_CHECKER, assessment, "scope_assessment")
82 if assessment["policy_id"] != policy["id"]:
83 raise ValueError("scope assessment belongs to another review policy")
84 perspectives = assessment["required_perspectives"]
85 if not set(policy["required_perspectives"]).issubset(perspectives):
86 raise ValueError("scope assessment omits a required perspective")
87 slots = assessment["initial_slots"]
88 if len(slots) < policy["minimum_initial_slots"]:
89 raise ValueError("scope assessment has too few initial slots")
90 identifiers = [slot["id"] for slot in slots]
91 if len(set(identifiers)) != len(identifiers):
92 raise ValueError("duplicate scope assessment slot")
93 if set(identifiers).intersection([*policy['judge_slots'], 'scope', 'organize']):
94 raise ValueError('scope assessment reuses an independent role slot')
95 if not set(perspectives).issubset({slot["perspective"] for slot in slots}):
96 raise ValueError("scope assessment has no slot for a required perspective")
97 limit = assessment["activity_execution_limit"]
98 if limit > policy["activity_execution_limit"]:
99 raise ValueError("scope assessment exceeds the policy execution limit")
100 minimum = minimum_scope_executions(len(slots), policy['round_limit'], executions_used)
101 if limit < minimum:
102 raise ValueError(f"scope activity execution limit requires at least {minimum} attempts including {executions_used} already used")
103 if assessment["overflow_disposition"] != policy["overflow_disposition"]:
104 raise ValueError("scope assessment changes the policy overflow disposition")
105 return deepcopy(assessment)
