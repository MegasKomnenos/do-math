124 def _plan_citing_task_while(self, change):
125 self.root, instance_id = restore_template(plan_ready_instance, config=CURRENTNESS_CONFIG)
126 engine, runner = self._engine(instance_id, PurposeRunner())
127 engine._refresh_readiness()
128 self.assertEqual(engine.runtime['queue'][0]['role'], 'research_plan')
129 task_id = next(tid for tid, row in engine._task_readiness.items() if row['state'] == 'ready')
130 answer = runner.answer
131 def citing_answer(role, context, workspace):
132 output = answer(role, context, workspace)
133 if role == 'research_plan':
134 output['used_sources'] = [*output.get('used_sources', []), task_id]
135 return output
136 runner.answer = citing_answer
137 # The concurrent change is an owner-published state change made after the
138 # plan's input was fixed and before its submission is received. Applying it
139 # on the owner thread keeps the test free of a worker/owner race.
140 receive = engine._receive_submission
141 def changed_then_received(action, *args, **kwargs):
142 if action['role'] == 'research_plan':
143 change(engine.runtime['tasks'][task_id])
144 engine._save()
145 return receive(action, *args, **kwargs)
146 with patch.object(engine, '_receive_submission', changed_then_received):
147 engine.run(max_steps=1)
148 return engine, task_id, next(e for e in reversed(list(engine.runtime['executions'].values())) if e['role'] == 'research_plan')

303 def test_goal_dependency_detects_changed_relation_endpoint_even_when_proof_record_is_unchanged(self):
304 from do_math.knowledge_io import import_candidate
305 from purpose_fixtures import solved_instance
306 # The public activation driven to mathematically_solved, built once per process.
307 self.root, solved = restore_template(solved_instance, criteria=None)

309 proof_id = next(iter(engine.instance['knowledge']))
310 endpoint_ids = []

313 'statement': 'Natural zero identity', 'explanation': 'Specify the exact natural-number endpoint',
314 'summary': label, 'lean_code': '', 'declaration_names': [], 'dependencies': []},
315 source_ref=engine.instance['id'], actor_ref=engine.instance['id'])
316 endpoint_ids.append(endpoint_id)
317 proof_basis = capture_basis(engine.instance, engine.runtime, [proof_id])
318 endpoint_refs = [*endpoint_ids, *[engine.instance['knowledge'][kid]['exposition_id'] for kid in endpoint_ids]]
319 endpoint_basis = capture_basis(engine.instance, engine.runtime, endpoint_refs)
320 # The goal assessor explicitly used both relation endpoints as premises.
321 judgment = engine.instance['goal_judgments'][-1]

327 self.assertFalse(validate_basis(endpoint_basis, engine.instance, engine.runtime)['valid'])
