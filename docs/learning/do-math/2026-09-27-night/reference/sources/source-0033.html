26def relative_path(name: str) -> PurePosixPath:

35def is_reparse(path: Path) -> bool:
36 """Whether ``path`` itself (not an ancestor) is a symlink or reparse point."""
37 path = path.absolute()
38 try:
39 no_reparse(path, below=path.parent)

45def safe_path(root: Path, name: str, *, require_input: bool = False) -> Path:
46 relative = relative_path(name)
47 if require_input and (not relative.parts or relative.parts[0] != "input"):
48 raise ValueError("source reference must stay below input/")
49 root = _physical_root(root)
50 candidate = root.joinpath(*relative.parts)
51 no_reparse(candidate, below=root)
52 resolved = candidate.resolve(strict=False)
53 if not resolved.is_relative_to(root):
54 raise ValueError(f"path leaves workspace: {name}")
55 return candidate

64def read_stable(path: Path, *, boundary: PhysicalBoundary | None = None) -> bytes:
65 # The boundary proves its root once (a caller reading many files passes
66 # one). Reading compares the same descriptor's metadata before and after,
67 # so a file written while it is read is not taken as one version.
68 path = Path(os.path.abspath(path))
69 try:
70 boundary = boundary or PhysicalBoundary(path.parent)
71 with boundary.open(path, "rb") as stream:
72 before = stat_fingerprint(os.fstat(stream.fileno()))
73 data = stream.read()
74 after = stat_fingerprint(os.fstat(stream.fileno()))
75 except ValueError as error:
76 raise _PhysicalBoundaryError(f"physical boundary rejected read: {error}") from error
77 if before != after or len(data) != after[0]:
78 raise _ChangedDuringRead(f"file changed while being read: {path}")
79 return data

84 if not isinstance(documents, dict):
85 raise TypeError("documents must be a path-to-bytes object")

88 if is_reparse(workspace):
89 raise ValueError("workspace must not be a reparse point")

105 stream.write(data)

115def collect_outputs(workspace: Path, *, required_paths: list[str] = [],
116 max_bytes: int = 268435456, captured_files: dict[str, bytes] | None = None) -> dict:
117 if isinstance(max_bytes, bool) or not isinstance(max_bytes, int) or max_bytes < 0:
118 raise ValueError("max_bytes must be a nonnegative integer")
119 if not isinstance(required_paths, list):
120 raise TypeError("required_paths must be a list")
121 required = {relative_path(name).as_posix() for name in required_paths}
122 workspace = _physical_root(workspace)
123 files: dict[str, bytes] = {}
124 manifest: list[dict] = []
125 omitted: list[dict] = []
126 seen: set[str] = set()
127 total_bytes = 0
128 if not workspace.is_dir():
129 return {"files": files, "manifest": manifest,
130 "omitted": [{"path": name, "reason": "workspace_missing"} for name in sorted(required)],
131 "complete": False}
132 captured_files = {} if captured_files is None else captured_files
133 if not isinstance(captured_files, dict):
134 raise TypeError("captured_files must be a path-to-bytes object")
135 for name, data in sorted(captured_files.items()):
136 relative = relative_path(name)
137 if is_excluded(relative):
138 raise ValueError("captured research file is excluded: " + name)
139 if not isinstance(data, bytes):
140 raise TypeError("captured research file must be bytes: " + name)
141 safe_path(workspace, name)
142 if total_bytes + len(data) > max_bytes:
143 omitted.append({"path": name, "reason": "max_bytes_exceeded", "bytes": len(data),
144 "required": name in required})
145 continue
146 files[name] = data
147 manifest.append({"path": name, "sha256": hashlib.sha256(data).hexdigest(), "bytes": len(data)})
148 total_bytes += len(data)
149 seen.add(name)
150 boundary = PhysicalBoundary(workspace)
151 def traversal_error(error):
152 path = Path(error.filename) if error.filename else workspace
153 name = path.relative_to(workspace).as_posix()
154 omitted.append({"path": name, "reason": "read_error", "error_type": type(error).__name__,
155 "error": str(error), "required": any(
156 name == "." or ref == name or ref.startswith(name + "/") for ref in required)})
157 for directory, dirnames, filenames in os.walk(workspace, topdown=True, followlinks=False,
158 onerror=traversal_error):
159 base = Path(directory)
160 kept = []
161 for dirname in sorted(dirnames):
162 path = base / dirname
163 relative = PurePosixPath(path.relative_to(workspace).as_posix())
164 if is_excluded(relative):
165 continue
166 if is_reparse(path):
167 reason = "outside_symlink" if not path.resolve(strict=False).is_relative_to(workspace) else "reparse_directory"
168 omitted.append({"path": relative.as_posix(), "reason": reason})
169 continue
170 kept.append(dirname)
171 dirnames[:] = kept
172 for filename in sorted(filenames):
173 path = base / filename
174 relative = PurePosixPath(path.relative_to(workspace).as_posix())
175 name = relative.as_posix()
176 if is_excluded(relative):
177 continue
178 if name in captured_files:
179 continue
180 seen.add(name)
181 if is_reparse(path):
182 reason = "outside_symlink" if not path.resolve(strict=False).is_relative_to(workspace) else "reparse_file"
183 omitted.append({"path": name, "reason": reason, "required": name in required})
184 continue
185 try:
186 data = read_stable(path, boundary=boundary)
187 except _PhysicalBoundaryError as error:
188 omitted.append({"path": name, "reason": "physical_boundary", "error": str(error),
189 "error_type": type(error).__name__, "required": name in required})
190 continue
191 except FileNotFoundError:
192 omitted.append({"path": name, "reason": "disappeared", "required": name in required})
193 continue
194 except _ChangedDuringRead as error:
195 omitted.append({"path": name, "reason": "changed_during_read", "error": str(error), "required": name in required})
196 continue
197 except OSError as error:
198 omitted.append({"path": name, "reason": "read_error", "error_type": type(error).__name__,
199 "error": str(error), "required": name in required})
200 continue
201 if total_bytes + len(data) > max_bytes:
202 omitted.append({"path": name, "reason": "max_bytes_exceeded", "bytes": len(data), "required": name in required})
203 continue
204 files[name] = data
205 manifest.append({"path": name, "sha256": hashlib.sha256(data).hexdigest(), "bytes": len(data)})
206 total_bytes += len(data)
207 for name in sorted(required - set(files)):
208 if not any(row["path"] == name for row in omitted):
209 omitted.append({"path": name, "reason": "required_missing" if name not in seen else "required_not_collected", "required": True})
210 complete = not omitted and required <= set(files)
211 return {"files": files, "manifest": sorted(manifest, key=lambda row: row["path"]),
212 "omitted": sorted(omitted, key=lambda row: (row["path"], row["reason"])), "complete": complete}

221def write_source_catalog(context: dict, workspace: Path) -> dict:
222 if not isinstance(context, dict):
223 raise TypeError("context must be an object")
224 workspace = _physical_root(workspace)
225 workspace.mkdir(parents=True, exist_ok=True)
226 if is_reparse(workspace):
227 raise ValueError("workspace must not be a reparse point")
228 entries: dict[str, dict] = {}
229 manifest: list[dict] = []
230 files: dict[str, bytes] = {}
231 boundary = None
232 for key in sorted(context):
233 logical = "input/catalog/" + _catalog_name(key)
234 target = safe_path(workspace, logical, require_input=True)
235 data = canonical_json(context[key])
236 if target.exists():
237 boundary = boundary or PhysicalBoundary(workspace) # one root proof per catalog
238 if read_stable(target, boundary=boundary) != data:
239 raise FileExistsError(f"catalog path contains different bytes: {logical}")
240 else:
241 target.parent.mkdir(parents=True, exist_ok=True)
242 with target.open("xb") as stream:
243 stream.write(data)
244 # The written bytes are these memory bytes; hash them once, never a reread.
245 ref = {"path": logical, "sha256": hashlib.sha256(data).hexdigest(), "bytes": len(data),
246 "format": "canonical-json"}
247 entries[key] = ref
248 manifest.append({"key": key, **ref})
249 files[logical] = data
250 return {"entries": entries, "manifest": manifest, "total": len(manifest), "files": files}

253def build_role_manifest(context: dict, *, workspace: Path, role: str,
254 direct_keys: list[str], reference_keys: list[str],
255 catalog: dict | None = None) -> dict:
256 """Declare direct role inputs separately from exact source references.
257
258 A caller that already wrote this exact context's catalog passes it as catalog.
259 """
260 if not isinstance(context, dict):
261 raise TypeError("context must be an object")
262 if not isinstance(role, str) or not role:
263 raise ValueError("role must be a non-empty string")
264 for label, keys in (("direct_keys", direct_keys), ("reference_keys", reference_keys)):
265 if not isinstance(keys, list) or any(not isinstance(key, str) for key in keys) or len(keys) != len(set(keys)):
266 raise ValueError(label + " must contain distinct string keys")
267 overlap = set(direct_keys).intersection(reference_keys)
268 if overlap:
269 raise ValueError("a role input cannot be both direct and referenced: " + ", ".join(sorted(overlap)))
270 missing = [key for key in [*direct_keys, *reference_keys] if key not in context]
271 if missing:
272 raise ValueError("role manifest keys are missing: " + ", ".join(missing))
273 if catalog is None:
274 catalog = write_source_catalog(context, workspace)
275 # Catalog entries already measure each key's exact canonical bytes.
276 direct = [{"key": key, "sha256": catalog["entries"][key]["sha256"],
277 "bytes": catalog["entries"][key]["bytes"]} for key in direct_keys]
278 references = []
279 for key in reference_keys:
280 source_ref = catalog["entries"].get(key)
281 if source_ref is None:
282 raise ValueError("reference key has no exact catalog entry: " + key)
283 references.append({"key": key, "source_ref": source_ref})
284 return {"role": role, "direct": direct, "references": references,
285 "catalog": catalog["manifest"]}

288def _member_size(key: str, raw: bytes) -> int:
289 """Bytes the member ``key`` adds to the canonical_json of an object holding it.
290
291 ``raw`` is the canonical_json of the member's value. One level deeper,
292 each interior line break of that encoding gains two spaces; string
293 contents never hold a raw line break because JSON escapes it.
294 """
295 interior_breaks = raw.count(b"\n") - 1
296 return 4 + len(json.dumps(key, ensure_ascii=False).encode("utf-8")) + len(raw) - 1 + 2 * interior_breaks

733def _candidate_order(ids, snapshot, role, action_id, replay=None):
734 algorithm = 'sha256-seed-id-v1'
735 basis = {'snapshot': snapshot, 'role': role, 'action_id': action_id}
736 seed = replay.get('seed') if isinstance(replay, dict) else os.urandom(32).hex()
737 if not is_sha256(seed):
738 raise ValueError('candidate order requires a 256-bit seed')
739 ordered = sorted(ids, key=lambda identifier: (hashlib.sha256(bytes.fromhex(seed) + b'\0' + identifier.encode()).hexdigest(), identifier))
740 result = {'algorithm': algorithm, 'seed': seed, 'basis': basis, 'ids': ordered}
741 if replay is not None and replay != result:
742 raise ValueError('candidate order replay differs from population or input basis')
743 return result

785def prefetch_artifacts(paths, read_artifact, read_artifacts=None):
786 """Read one fixed snapshot in bulk, retaining per-file failure handling.
787
788 ``read_artifacts(paths)`` omits paths absent from the snapshot, so reading
789 one of them raises its own ValueError. The returned reader belongs only to
790 this inspection. A batch that fails as a whole is read individually by the
791 caller's existing path, so error order and scope stay local.
792 """
793 if read_artifacts is None or not paths:
794 return read_artifact
795 from .git_store import GitStateStoreError
796 wanted = list(dict.fromkeys(paths))
797 try:
798 files = read_artifacts(wanted)
799 except (GitStateStoreError, OSError, ValueError, TypeError, KeyError):
800 return read_artifact
801 wanted = set(wanted)
802
803 def read(path):
804 if path in files:
805 return files[path]
806 if path in wanted:
807 raise ValueError('artifact is missing: ' + path)
808 return read_artifact(path)
809 return read

883def fit_context(context: dict, *, max_bytes: int, workspace: Path,
884 mandatory_keys: list[str], role: str | None = None,
885 profile_id: str | None = None, activity_id: str | None = None) -> dict:
886 """Pack optional source references without changing mandatory direct bytes."""
887 if not isinstance(context, dict):
888 raise TypeError("context must be an object")
889 if isinstance(max_bytes, bool) or not isinstance(max_bytes, int) or max_bytes < 1:
890 raise ValueError("max_bytes must be a positive integer")
891 if not isinstance(mandatory_keys, list) or any(not isinstance(k, str) for k in mandatory_keys) or len(mandatory_keys) != len(set(mandatory_keys)):
892 raise ValueError("mandatory_keys must contain distinct string keys")
893 mandatory_keys = list(dict.fromkeys([*mandatory_keys, *[key for key in role_direct_keys(role, context)
894 if key in context and key not in {'goal', 'policies', 'implementation_documents'}]]))
895 missing = set(mandatory_keys) - context.keys()
896 if missing:
897 raise ValueError("mandatory context keys are missing: " + ", ".join(sorted(missing)))
898 catalog = write_source_catalog(context, workspace)
899 fitted = deepcopy(context)
900 # The catalog holds each member's canonical bytes; object sizes are summed
901 # from them instead of encoding the whole context again for every member.
902 raw = {key: catalog["files"][entry["path"]] for key, entry in catalog["entries"].items()}
903 sizes = {key: _member_size(key, raw[key]) for key in context}
904 original_bytes = _object_size(sizes.values())
905 mandatory_bytes = _object_size(sizes[key] for key in mandatory_keys)
906 reductions = []
907 # Catalogs that define a comparison universe are never shortened. Their
908 # inclusion as required direct input is enforced by prepare_role_input.
909 candidates = sorted((key for key in catalog["entries"] if key not in mandatory_keys),
910 key=lambda key: len(raw[key]), reverse=True)
911 reference_only = {'execution_profiles', 'role_profile_refs'} if role in _ROLE_DIRECT else set()
912 if role in {'research_work', 'supplemental_research', 'task_contract_review'}:
913 reference_only.add('tasks')
914 for key in candidates:
915 if mandatory_bytes > max_bytes:
916 break
917 if key not in reference_only and _object_size(sizes.values()) <= max_bytes:
918 continue
919 ref = catalog["entries"][key]
920 projected = {"source_ref": ref, "total": len(context[key]) if isinstance(context[key], (dict, list)) else None}
921 projected_raw = canonical_json(projected)
922 if len(projected_raw) < len(raw[key]):
923 fitted[key] = projected
924 sizes[key] = _member_size(key, projected_raw)
925 reductions.append({"key": key, "kind": "source_reference", "source_ref": ref})
926 actual = _object_size(sizes.values())
927 overflow = None
928 if actual > max_bytes:
929 overflow = _input_overflow(
930 "mandatory_context_exceeds_limit" if mandatory_bytes > max_bytes else "reference_index_exceeds_limit",
931 actual_bytes=original_bytes, mandatory_bytes=mandatory_bytes, max_bytes=max_bytes,
932 mandatory_keys=mandatory_keys, role=role, profile_id=profile_id, activity_id=activity_id)
933 overflow.update(packed_bytes=actual, original_bytes=original_bytes,
934 component_bytes={key: len(raw[key]) for key in context},
