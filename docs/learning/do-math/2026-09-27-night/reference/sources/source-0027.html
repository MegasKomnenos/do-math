22def sanitized_environment(source=None):
23 return allowed_environment(source, frozenset({"CLAUDE_CONFIG_DIR", "CLAUDE_CODE_GIT_BASH_PATH"}))

26def load_request(path: Path) -> dict:
27 request = check_common_request(strict_json_loads(path.read_text(encoding="utf-8")),
28 required=frozenset({"claude_path"}),
29 optional=frozenset({"claude_only_assigned_model"}))
30 if "output_schema_path" not in request and "control_binding" not in request:
31 raise ValueError("an unbound Claude worker requires its structured final schema")
32 if request["effort"] not in {"low", "medium", "high", "xhigh", "max"}:
33 raise ValueError("Claude effort must be low, medium, high, xhigh or max")
34 if not isinstance(request.get("claude_only_assigned_model", False), bool):
35 raise ValueError("claude_only_assigned_model must be a boolean")
36 if request.get("claude_only_assigned_model") and not request["model"].startswith("claude-"):
37 raise ValueError("claude_only_assigned_model requires a full Claude model ID")
38 return request

41def build_argv(request: dict) -> list[str]:
42 """The Claude print command for a request ``load_request`` has checked."""
43 settings = {"disableAllHooks": True, "autoMemoryEnabled": False}
44 if request.get("claude_only_assigned_model", False):
45 model = request["model"]
46 settings.update(availableModels=[model], fallbackModel=[], switchModelsOnFlag=False,
47 env={"ANTHROPIC_DEFAULT_HAIKU_MODEL": model,
48 "CLAUDE_CODE_SUBAGENT_MODEL": model,
49 "CLAUDE_CODE_SUBAGENT_MODEL_FORCE": "1"})
50 servers = {}
51 if "control_binding" in request:
52 servers[CONTROL_SERVER] = ControlTransport.mcp_server_config(request["control_binding"])
53 server = request.get("scholarly_server")
54 if server is not None:
55 servers["scholarly"] = {**server, "args": [*server["args"], "--data-dir",
56 str(Path(request["cwd"]).resolve() / "scholarly-data")]}
57 # Keep subscription login; --bare deliberately disables OAuth login.
58 argv = [request["claude_path"], "--print", "--output-format", "stream-json", "--verbose",
59 "--model", request["model"], "--effort", request["effort"],
60 "--no-session-persistence", "--setting-sources", "",
61 "--settings", json.dumps(settings),
62 "--disable-slash-commands", "--disallowedTools", "Agent,Task",
63 "--strict-mcp-config", "--mcp-config", json.dumps({"mcpServers": servers}, ensure_ascii=False),
64 "--permission-mode", "bypassPermissions"]
65 if "output_schema_path" in request:
66 schema = strict_json_loads(Path(request["output_schema_path"]).read_text(encoding="utf-8"))
67 if schema.get("$schema") == "https://json-schema.org/draft/2020-12/schema":
68 schema.pop("$schema")
69 schema.pop("x-contract-id", None)
70 argv.extend(["--json-schema", json.dumps(schema, ensure_ascii=False)])
71 return argv
