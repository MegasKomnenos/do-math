339class ReassessmentHarness(unittest.TestCase):
340 """A ``ReviewFakeEngine`` whose ``K1`` review completed, ready for a reassessment of that assessment.
341
342 Sets ``directory``, ``engine``, ``original`` (the first review cycle ID) and
343 ``assessment``. ``request(evidence)`` is the public reassessment request for
344 ``assessment``; ``defect()`` makes one initial reviewer's context include
345 the author's and returns that evidence; ``use()`` is the assessment's use
346 record; ``complete(decision='match')`` answers every queued review action
347 toward ``decision``. Subclass it, or call ``setUp`` on an instance and
348 ``doCleanups`` when done.
349 """
350 def setUp(self):
351 self.directory = tempfile.TemporaryDirectory()
352 self.addCleanup(self.directory.cleanup)
353 self.engine = ReviewFakeEngine(self.directory.name)
354 self.original = self.engine._start_identity_review('K1')
355 while any(a['role'] == 'identity_review' for a in self.engine.runtime['queue']):
356 self.engine.execute()
357 self.assessment = self.engine.assessment()

359 def request(self, evidence):
360 return {'assessment_id': self.assessment['id'], 'assessment_sha256': exact_content_hash(self.assessment),
361 'evidence': evidence}
362
363 def defect(self):
364 cycle = self.engine.runtime['review_cycles'][self.original]
365 execution_id = cycle['results']['initial:1:initial-0']['execution_id']
366 self.engine.runtime['executions'][execution_id]['context_origin'] = ['author']
367 self.engine.runtime['executions']['author'] = {'id': 'author', 'status': 'ended'}
368 return {'kind': 'author_context', 'execution_id': execution_id}
369
370 def use(self):
371 return next(row for row in self.engine.instance['assessment_uses'].values()
372 if row['assessment_id'] == self.assessment['id'])
373
374 def complete(self, decision='match'):
375 while any(a['role'] == 'identity_review' for a in self.engine.runtime['queue']):
376 action = self.engine.pending()
377 cycle = self.engine.runtime['review_cycles'][action['review_cycle_id']]

382 changes = {'decision': 'mismatch', 'findings': [{'id': 'defect', 'statement': 'The stated domain differs', 'evidence': ['proposal']}]}

384 changes = {'decision': 'mismatch', 'issue_verdicts': [dict(issue_id=i, decision='approved', reason='Exact mismatch confirmed', evidence=['proposal']) for i in cycle['issues']]}
385 self.engine.execute(changes)
