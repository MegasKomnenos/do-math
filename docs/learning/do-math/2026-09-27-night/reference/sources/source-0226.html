103 # states and activity ownership, so the contract is where they are rejected.
104 from do_math import activities, execution_profiles, knowledge_io, model_policy
105 from do_math.state_contract import default_state_contract
106 profiles = execution_profiles.build_profiles({
107 "session_limits": model_policy.LIMITS, "tier_models": model_policy.MODELS,
108 "observation_mode": "event_driven", "max_input_bytes": 500000,
109 "effort": "medium", "operator_effort": "high", "operator_timeout_seconds": 600,
110 "scholarly_enabled": True, "scholarly_server": None, "claude_only_assigned_model": False})
111 instance, runtime = current_state({"object_format": "sha1", "knowledge": {}, "expositions": {},
112 **{k: v for k, v in profiles.items() if k != "role_profile_refs"}}, {"executions": {}})
113 instance = knowledge_io.ensure_ontology(instance)
114 activity = activities.open_activity(runtime, activity_id="AT-1", activity_kind="research_work",
115 owner_ref="task", input_refs=[], opened_at="2026-09-20T00:00:00Z")
116 profile = next(p for p in profiles["execution_profiles"].values() if p["role"] == "research_work")
117 execution = {"id": "EX-1", "role": "research_work", "actor_kind": "ai", "owner_activity_id": "AT-1",
118 "created_at": "2026-09-20T00:00:00Z", "status": "running", "input_manifest": [], "context_origin": []}
119 execution_profiles.bind_execution(execution, profile)
120 runtime["executions"]["EX-1"] = execution
121 activity["assigned_execution_ids"] = ["EX-1"]
122 contract = default_state_contract()
123 contract.validate(instance, runtime)
124 invalid = deepcopy(runtime)
125 invalid["executions"]["EX-1"]["status"] = "teleporting"
126 with self.assertRaisesRegex(ValueError, "executions/EX-1/status"):
127 contract.validate(instance, invalid)
128 orphan = deepcopy(runtime)
