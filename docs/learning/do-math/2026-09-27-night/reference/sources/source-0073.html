81def action_read_ids(action: dict, instance: dict, runtime: dict, *, records=None) -> set[str]:
82 """records may carry record_index() of this same state, shared by one evaluation."""
83 from .relation_refs import relation_references
84 records = record_index(instance, runtime) if records is None else records
85 seeds = {action.get("target"), *action_write_ids(action)} - {None}
86 if action.get("role") == "operational_review":
87 parent = runtime.get("executions", {}).get(action.get("target"), {})
88 seeds.update(row["id"] for row in parent.get("decision_basis", {}).get("records", []))
89 parent_target = parent.get("action", {}).get("target")
90 if parent_target is not None and not parent.get("action", {}).get("review_cycle_id"):
91 seeds.add(parent_target)
92 if action.get("role") == "research_plan":
93 seeds.update(records)
94 visited = set()
95 while seeds:
96 identifier = seeds.pop()
97 if identifier in visited:
98 continue
99 visited.add(identifier)
100 if identifier not in records:
101 continue
102 family, record = records[identifier]
103 for key in READ_REFERENCE_FIELDS:
104 if isinstance(record.get(key), list):
105 seeds.update(ref if isinstance(ref, str) else ref['id'] for ref in record[key]
106 if isinstance(ref, str) or isinstance(ref, dict) and isinstance(ref.get('id'), str))
107 # A reader also follows the provenance source that a record names.
108 seeds.update(record[key] for key in (*READ_REFERENCE_ID_FIELDS, 'source_ref') if isinstance(record.get(key), str))
109 if record.get('formal_result_id'):
110 seeds.update(r['id'] for r in instance.get('identity_assessments',{}).values()
111 if r.get('formal_result_id')==record['formal_result_id'] and r.get('exposition_id')==record.get('exposition_id'))
112 seeds.update(r['id'] for r in instance.get('assessment_uses',{}).values() if r.get('assessment_id')==identifier)
113 seeds.update(condition_refs(record.get("occurrence_condition")))
114 for review in record.get("reviews", []):
115 seeds.update(row["id"] for row in review.get("decision_basis", {}).get("records", []))
116 seeds.update(relation_references(record.get('relation_claim') or {}))
117 if family == 'relation':
118 seeds.update(relation_references(record))
119 return visited

146def revision_waits(action: dict, instance: dict, runtime: dict, *, records=None) -> list[dict]:
147 """Declared writers wait for existing readers and older writers; new dependent readers wait."""
148 records = record_index(instance, runtime) if records is None else records
149 reads = action_read_ids(action, instance, runtime, records=records)
150 writes = action_write_ids(action)
151 role = action.get("role")
152 if role in {"operational_review", "intent_interpretation", "maintenance"}:
153 return []
154 own_request = runtime.get("tasks", {}).get(action.get("target"), {}).get("revision_request_action_id")
155 owners = {own_request, *action.get("revision_parent_ids", [])} - {None}
156 owned_targets = writes | set(runtime.get("tasks", {}).get(action.get("target"), {}).get("knowledge_revision_targets", []))
157 live = [e for e in runtime.get("executions", {}).values() if e.get("status") != "ended"
158 and e.get("action", {}).get("id") != action.get("id")]
159 reasons = []
160 for execution in live:
161 overlap = writes & action_read_ids(execution.get("action", {}), instance, runtime, records=records)
162 if overlap:
163 reasons.append({"kind": "contract_revision", "execution_id": execution["id"],
164 "record_ids": sorted(overlap), "reason": "existing reader must finish before revision"})
165 for pending in pending_revisions(runtime):
166 other = pending["action"]
167 if other.get("id") == action.get("id"):
168 continue
169 # Diagnosis and planning can resolve a failed revision. Only live writers
170 # block a new global plan; their actual readers already drained above.
171 if role == "research_plan" and pending["state"] != "running":
172 continue
173 # Unstarted writers drain oldest first. A declared writer waits only for
174 # older ones, so writers reading each other's targets cannot deadlock.
175 if writes and pending["state"] != "running" and other.get("id", "") > action.get("id", ""):
176 continue
177 overlap = reads & action_write_ids(other)
178 if other.get("id") in owners:
179 overlap -= owned_targets
180 if overlap:
181 reasons.append({"kind": "contract_revision", "action_id": other.get("id"),
182 "record_ids": sorted(overlap), "reason": "required record has an unresolved revision"})
183 return reasons

317def validate_condition(condition: dict) -> None:
318 """Check the expression's grammar even when its referenced record is absent."""
319 fields = CONDITION_FIELDS
320 if not isinstance(condition, dict) or condition.get("op") not in fields:
321 raise ValueError("unsupported condition op")
322 op = condition["op"]
323 allowed = {"op", *fields[op]}
324 if set(condition) != allowed:
325 raise ValueError("condition fields do not match " + op)
326 if op in {"all", "any"}:
327 if not isinstance(condition["args"], list) or not condition["args"]:
328 raise ValueError(op + " requires nonempty args")
329 for child in condition["args"]:
330 validate_condition(child)
331 elif op == "not":
332 validate_condition(condition["arg"])
333 elif op == "artifact_valid":
334 ref = condition["artifact_ref"]
335 if not isinstance(ref, dict) or set(ref) != {"snapshot", "path", "sha256"}:
336 raise ValueError("artifact_ref requires snapshot, path, sha256")
337 if not all(isinstance(value, str) and value for value in ref.values()):
338 raise ValueError("artifact reference values must be text")
339 if not is_sha256(ref["sha256"]):
340 raise ValueError("artifact hash must be lowercase SHA256")
341 # The reader accepts only archived artifacts; each segment follows the state path rule.
342 if not ref["path"].startswith(ARTIFACT_ROOT):
343 raise ValueError("artifact path must name an archived artifact under " + ARTIFACT_ROOT)
344 for part in ref["path"].split("/"):
345 check_path_segment(part)
346 else:
347 for name in fields[op] - {"value"}:
348 if not isinstance(condition[name], str) or not condition[name].strip():
349 raise ValueError("condition " + name + " must be nonempty text")
350 if op == "task_disposition_is" and condition["value"] not in {"completed", "closed"}:
351 raise ValueError("task disposition must be completed or closed")
352 if op == "record_field_equals" and (not condition["pointer"].startswith("/") or isinstance(condition["value"], (dict, list))):
353 raise ValueError("record comparison needs a pointer and scalar value")
354 if "space" in condition and condition["space"] not in {"research", "runtime"}:
355 raise ValueError("unknown condition space")
356 if "family" in condition and condition["family"] not in RECORD_FAMILIES:
357 raise ValueError("unknown condition family")
358 if "family" in condition and RECORD_FAMILIES[condition["family"]][0] != condition["space"]:
359 raise ValueError("condition family is outside the requested space")
360 if op == "record_lifecycle_is":
361 values = lifecycle_values().get(condition['family'],set())
362 if condition["value"] not in values:
363 raise ValueError("value is not a lifecycle value of this family")
364 if op=='record_field_equals':
365 family,pointer=condition['family'],condition['pointer']
366 if pointer not in EXPOSED_FIELDS.get(family,set()):
367 raise ValueError('pointer is not an exposed machine field')
368 if pointer == '/usable_as_k' and type(condition['value']) is not bool:
369 raise ValueError('usable_as_k comparison requires a boolean')
370 if pointer in {'/status','/lifecycle'} and condition['value'] not in lifecycle_values().get(family,set()):
371 raise ValueError('value is not a lifecycle value of this family')

420def record_index(instance: dict, runtime: dict, *, families=None, source_aliases=False,
421 question_links=False) -> dict[str, tuple[str, dict]]:
422 """Map record IDs to (family, record): the one population definition of RECORD_FAMILIES.
423
424 families narrows the index to the named RECORD_FAMILIES keys. A duplicate ID
425 raises ValueError. question_links adds question rows as family "question";
426 source_aliases then adds each source alias that no entry already claims as
427 family "source". Neither extra is a condition family or a read-closure seed.
428 """
429 found: dict[str, tuple[str, dict]] = {}
430 for family, (space, key) in RECORD_FAMILIES.items():
431 if families is not None and family not in families:
432 continue
433 if key is None:
434 container = [instance_record(instance)] if instance.get("id") else []
435 else:
436 container = (instance if space == "research" else runtime).get(key, {})
437 values = container.values() if isinstance(container, dict) else container if isinstance(container, list) else []
438 for record in values:
439 if isinstance(record, dict) and isinstance(record.get("id"), str):
440 if record["id"] in found:
441 raise ValueError("globally duplicate record ID: " + record["id"])
442 found[record["id"]] = (family, record)
443 if question_links:
444 for row in runtime.get("question_links", []):
445 if row["id"] in found:
446 raise ValueError("globally duplicate record ID: " + row["id"])
447 found[row["id"]] = ("question", row)
448 if source_aliases:
449 from .input_basis import aliases
450 for source in instance.get("sources", []):
451 for alias in aliases(source):
452 found.setdefault(alias, ("source", source))
453 return found

475def _pointer(record: dict, family: str, pointer: str) -> Any:
476 if not isinstance(pointer, str) or not pointer.startswith("/"): raise ValueError("JSON Pointer must start with /")
477 if pointer not in EXPOSED_FIELDS.get(family,set()): raise ValueError("pointer is not an exposed machine field")
478 value: Any = record
479 for token in pointer.split("/")[1:]:
480 token = token.replace("~1", "/").replace("~0", "~")
481 if not isinstance(value, dict) or token not in value: raise KeyError(pointer)
482 value = value[token]
483 if isinstance(value, (dict, list)): raise ValueError("pointer must name a scalar machine field")
484 if isinstance(value,bool): return value
485 if not isinstance(value,(str,int,float)) or isinstance(value,float) and not value.is_integer(): raise ValueError("pointer must name a scalar machine field")
486 return value

489def evaluate_condition(condition: dict, instance: dict, runtime: dict, *, evaluation_context=None, records=None) -> dict:
490 """Evaluate a closed readiness AST without treating malformed evidence as true.
491
492 records may carry record_index() of this same state, shared by one evaluation."""
493 basis, errors = [], []
494 try:
495 records = record_index(instance, runtime) if records is None else records
496 validate_condition(condition)
497 except (ValueError, TypeError) as error:
498 return {"state": "evaluation_error", "basis": [], "errors": [str(error)]}
499 def visit(node: dict) -> str:
500 try:
501 if not isinstance(node, dict) or not isinstance(node.get("op"), str): raise ValueError("condition op is required")
502 op = node["op"]
503 if op not in CONDITION_FIELDS: raise ValueError(f"unsupported condition op: {op}")
504 if op == "always": return "ready"
505 if op in {"all", "any"}:
506 args = node.get("args")
507 if not isinstance(args, list) or not args: raise ValueError(f"{op} requires nonempty args")
508 results = [visit(x) for x in args]
509 if "evaluation_error" in results: return "evaluation_error"
510 return "ready" if (all(x == "ready" for x in results) if op == "all" else any(x == "ready" for x in results)) else "not_ready"
511 if op == "not":
512 result = visit(node.get("arg"))
513 return "evaluation_error" if result == "evaluation_error" else "not_ready" if result == "ready" else "ready"
514 if op == "relation_exists":
515 from .knowledge_io import _RELATION_FIELDS, validate_relation
516 candidates = [r for r in instance.get("relations", []) if isinstance(r, dict) and r.get("kind") == node.get("subkind") and r.get("source") == node.get("from_id") and r.get("target") == node.get("to_id")]
517 matches = []
518 for relation in candidates:
519 basis.append(_basis(relation))
520 if relation.get("status", "active") in {"suspended", "withdrawn", "invalid"} or relation.get("use_status", "active") in {"suspended", "withdrawn"}:
521 continue
522 value = {key: value for key, value in relation.items() if key in _RELATION_FIELDS}
523 result = validate_relation(instance, value)
524 unavailable = ("formal_result_not_usable:", "assessment_not_effective:")
525 structural = [reason for reason in result["reasons"] if not reason.startswith(unavailable)]
526 if structural:
527 raise ValueError("invalid relation: " + ", ".join(structural))
528 from .relation_refs import relation_references
529 for record_id in sorted(relation_references(relation)):
530 record = records.get(record_id)
531 if record is not None:
532 basis.append(_basis(record[1]))
533 if result["valid"]:
534 matches.append(relation)
535 if matches:
536 return "ready"
537 basis.append({"absencequery": {"op": op, "subkind": node.get("subkind"), "from_id": node.get("from_id"), "to_id": node.get("to_id")}})
538 return "not_ready"
539 if op == "task_disposition_is":
540 record = _record_target(records, node.get("task_id"), "task")
541 if record is None: basis.append({"absencequery": {"op": op, "task_id": node.get("task_id")}}); return "not_ready"
542 basis.append(_basis(record[1])); return "ready" if record[1].get("status") == node.get("value") else "not_ready"
543 if op == "decision_value_is":
544 record = _record_target(records, node.get("decision_id"), "decision")
545 if record is None:
546 basis.append({"absencequery": {"op": op, "decision_id": node.get("decision_id")}})
547 return "not_ready"
548 decision = record[1]
549 basis.append(_basis(decision))
550 from .task_scope import decision_is_current
551 effective = decision_is_current(decision, instance, runtime, evaluation_context=evaluation_context)
552 scope=decision.get('change',{}).get('task_id')
553 if scope and scope!=(evaluation_context or {}).get('task_id'):
554 effective=False
555 value = decision.get("change", {}).get("value", decision.get("change"))
556 return "ready" if effective and type(value) is type(node["value"]) and value == node["value"] else "not_ready"
557 if op == "artifact_valid":
558 context, ref = evaluation_context or {}, node["artifact_ref"]
559 if not context.get("snapshot") or not callable(context.get("read_artifact")) or not callable(context.get("validate_artifact")):
560 raise ValueError("artifact requires a fixed-snapshot reader and format validator")
561 # Sealed bytes never change: this repository matches them once
562 # (이음새-정본.md §3) and a known verdict needs no bytes at all.
563 git_dir, verdict = context.get("git_dir"), (context["validate_artifact"], ref["path"], ref["sha256"])
564 valid = _VALID_ARTIFACTS.get(verdict)
565 data = sealed_bytes(git_dir, ref["path"], ref["sha256"], None,
566 lambda: context["read_artifact"](ref["path"], ref["snapshot"]),
567 needed=valid is None, error="artifact bytes differ from sealed hash")
568 if valid is None:
569 valid = _VALID_ARTIFACTS[verdict] = context["validate_artifact"](ref, data) is True
570 if not valid:
571 raise ValueError("artifact format validation failed")
572 size = len(data) if data is not None else VERIFIED_ARTIFACTS[artifact_key(git_dir, ref["path"], ref["sha256"])]
573 basis.append({"artifact_ref": deepcopy(ref), "bytes": size})
574 return "ready"
575 record = _record_target(records, node.get("record_id"), node.get("family"))
576 if op == "record_exists":
577 if record is None: basis.append({"absencequery": {"op": op, "record_id": node.get("record_id"), "family": node.get("family")}}); return "not_ready"
578 basis.append(_basis(record[1])); return "ready"
579 if record is None: basis.append({"absencequery": {"op": op, "record_id": node.get("record_id")}}); return "not_ready"
580 basis.append(_basis(record[1]))
581 if op == "record_lifecycle_is": return "ready" if record[1].get("lifecycle", record[1].get("status")) == node.get("value") else "not_ready"
582 if op == "record_field_equals":
583 if record[0] == 'knowledge' and node['pointer'] == '/usable_as_k':
584 from .knowledge import assess_use
585 actual = assess_use(record[1], instance)['valid']
586 else:
587 actual = _pointer(record[1], record[0], node.get("pointer"))
588 expected = node.get("value")
589 return "ready" if type(actual) is type(expected) and actual==expected else "not_ready"
590 raise ValueError(f"unsupported condition op: {op}")
591 except (KeyError, TypeError, ValueError, OSError) as error:
592 errors.append(str(error)); return "evaluation_error"
593 state = visit(condition)
594 return {"state": state, "basis": basis, "errors": errors}

848def _event(kind: str, when: str, target: str | None = None, *, key: str | None = None) -> dict:
849 """A due event whose ID is derived from key (default: when), so one cause is one event."""
850 seed = f"{kind}|{when if key is None else key}|{target or ''}"
851 return {"id": hashlib.sha256(seed.encode()).hexdigest()[:24], "kind": kind, "at": when, "target": target}
