68def collect_repair_verification(output, workspace):
69 """Preserve actual regression evidence; a completed model call is insufficient."""
70 if not output or output.get('repair_outcome') != 'repaired' or not output.get('verification_evidence'):
71 raise ValueError('diagnosed repair requires repaired outcome and regression evidence files')
72 rows = []

112def schedule(
113 action: str,
114 *,
115 repository: Path,
116 instance: str,
117 workspace: Path,
118 interval_seconds: int | None = None,
119 operator_timeout_seconds: int | None = None,
120 wake_at: str | None = None,
121 wake_key: str | None = None,
122 runner: Runner = subprocess.run,
123) -> dict[str, Any]:
124 """Call the narrow PowerShell scheduler adapter and decode its JSON result.
125
126 Only ``install`` uses the configured interval and operator timeout;
127 configuration validation already checked their ranges.
128 """
129 if action == "install" and (interval_seconds is None or operator_timeout_seconds is None):
130 raise ValueError("install requires the configured operator interval and timeout")
131 names = task_names(repository, instance)
132 pythonw = Path(sys.executable).with_name("pythonw.exe")
133 executable = str(pythonw) if pythonw.is_file() else sys.executable
134 powershell = shutil.which("powershell.exe") or shutil.which("powershell") or "powershell.exe"
135 command = [
136 powershell, "-NoProfile", "-ExecutionPolicy", "Bypass", "-File", str(SCHEDULE_SCRIPT),
137 "-Action", action,
138 "-Repository", str(repository.resolve()),
139 "-Instance", instance,
140 "-Workspace", str(workspace.resolve()),
141 "-Python", executable,
142 "-OperatorTask", names["operator"],
143 "-EngineTask", names["engine"],
144 "-WakeTask", names["wake"],
145 ]
146 if action == "install":
147 command.extend(["-IntervalSeconds", str(interval_seconds),
148 "-OperatorTimeoutSeconds", str(operator_timeout_seconds)])
149 if action == "schedule-wake":
150 if epoch_seconds(wake_at) is None or not wake_key:
151 raise ValueError("wake requires a timezone-aware retry time and generation key")
152 command.extend(["-WakeAt", wake_at, "-WakeKey", wake_key])
153 completed = runner(command, capture_output=True, text=True, encoding="utf-8", check=False, creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0))
154 if completed.returncode:
155 raise RuntimeError(completed.stderr.strip() or completed.stdout.strip() or "scheduled-task operation failed")
156 try:
157 value = json.loads(completed.stdout)
158 except json.JSONDecodeError as error:
159 raise RuntimeError("scheduled-task adapter returned invalid JSON") from error
160 if not isinstance(value, dict):
161 raise RuntimeError("scheduled-task adapter result must be an object")
162 return value

189def supervision_block(facts: dict[str, Any]) -> str | None:
190 """The first supervision blocker present in ``facts``, or None.
191

193 a persisted-state view or an explicit engine start omits them.
194 """

196 'disabled': 'operator_enabled' in facts and not facts['operator_enabled'],
197 'user_pause': facts.get('user_pause'), 'maintenance': facts.get('maintenance_fence'),

297def engine_start_guard(repository: Path, instance: str, workspace: Path, *, status=None) -> dict[str, Any]:
298 """Read the conditions that must hold before either task starts an engine.
299
300 An explicit engine start does not depend on automatic supervision being enabled.
301 """
302 if _held(workspace):
303 return {"outcome": "held"}
304 if status is None:
305 status = read_status(repository, instance)
306 block = supervision_block({key: value for key, value in status.items() if key != 'operator_enabled'})
307 if block == 'maintenance':
308 return {'outcome':'maintenance_waiting', 'before':status}
309 if block == 'model_policy':
310 return {"outcome": "model_policy_waiting", "status": status, "reason": "Reported model violated the instance lock; no further operator model was called."}
311 if block in {'user_pause', 'paused'}:
312 return {"outcome": "paused", "before": status}
313 if block in {'closed', 'completed'}:
314 return {"outcome": "not_started", "before": status}
315 if block == 'conversation_lost':
316 return {'outcome': 'user_paused', 'before': status}
317 if block == 'installation_repair':
318 return {'outcome': 'installation_repair_waiting', 'before': status}
319 providers = status.get("provider_waits", {}).get("providers", {})
320 # A queued request waits on a provider whose gate is open again.
321 provider_open = any(row['wait'].get('kind') == PROVIDER_WAIT
322 and providers.get(row['wait'].get('provider'), {}).get('open')
323 for row in status.get("dispatch_waits", []))
324 if status.get("status") == "waiting" and provider_open:
325 return {"outcome": "startable", "before": status, "reason": "provider re-observation due"}
326 if status.get("status") == "waiting" and status.get("research_wake_reasons"):
327 return {"outcome": "startable", "before": status, "reason": "research reconsideration due",
328 "research_wake_reasons": status["research_wake_reasons"]}
329 if status.get("status") not in {"ready", "running", "capacity_waiting"}:
330 return {"outcome": "not_started", "before": status}
331 return {"outcome": "startable", "before": status}

334def reconcile_wake(repository: Path, instance: str, workspace: Path, *, facts: dict, providers: dict) -> dict:
335 """Repair a missing/stale one-shot reservation before invoking any model.
336
337 ``facts`` are the caller's supervision facts and ``providers`` its provider
338 snapshot, both from the state it already holds. A queried task status is
339 returned as ``observed`` so the caller need not query it again.
340 """
341 marker = _operator_dir(workspace) / "wake-registration.json"
342 previous = _load_json(marker)
343 block = supervision_block({**facts, 'held': _held(workspace)})
344 held = block not in {None, 'disabled', 'installation_repair'}
345 waits = [row for row in providers.values() if row["state"] == "waiting"]
346 if held or block == 'disabled' or not waits:
347 if previous:
348 cancel_wake(repository, instance, workspace)
349 return {"status": "held" if held else "not_needed"}
350 row = min(waits, key=lambda value: epoch_seconds(value["retry_at"]))
351 key = row["provider"] + ":" + str(row["generation"])
352 wanted = {"key": key, "retry_at": row["retry_at"]}
353 observed = schedule("status", repository=repository, instance=instance, workspace=workspace)
354 actual = observed.get("wake")
355 future_trigger = actual and (epoch_seconds(actual.get("next_run")) or 0) > datetime.now(timezone.utc).timestamp()
356 running = (observed.get("engine") or {}).get("state") == "Running" or (actual or {}).get("state") == "Running"
357 if previous == wanted and actual and actual.get("description") == key and (future_trigger or running):
358 return {"status": "scheduled", **wanted, "observed": observed}
359 schedule("schedule-wake", repository=repository, instance=instance, workspace=workspace,
360 wake_at=row["retry_at"], wake_key=key)
361 _json(marker, wanted)
362 return {"status": "scheduled", **wanted, "observed": observed}

433def recovery(repository: Path, instance: str, workspace: Path, *, failure: dict | None = None) -> dict:
434 """Failure-only installation repair under the engine's writer lock.
435
436 Without ``failure``, repair the pending installation diagnosis read under
437 the lock. Repair retains the already adopted operator authority.
438 Instruction scope is explicit; it is not an OS sandbox. Research and user
439 decisions are unchanged.
440 """

455 return {"outcome": "held"}

457 try:

461 block = supervision_block({**supervision_facts(state, runtime), 'held': _held(workspace)})
462 if block == 'disabled':
463 return {"outcome": "recovery_disabled", "failure": failure}
464 if block not in {None, 'installation_repair'}:
465 return {"outcome": "held"}

467 pending_installation_repair, resume_installation_repair)
468 if failure is None:
469 pending = pending_installation_repair(runtime)
470 if pending is None:

475 diagnosis = runtime.get('operational_dispositions', {}).get(failure.get('diagnosis_id'))

477 raise ValueError('installation diagnosis is missing or differs from its published record')
478 from .launch import termination_confirmed
479 for execution in runtime.get("executions", {}).values():
480 if execution.get("status") == "ended":
481 continue
482 if not execution.get('launch_ticket') or not termination_confirmed(execution['launch_ticket'], recovering=True):
483 return {"outcome": "recovery_waiting", "reason": "existing execution termination is unconfirmed",
484 "execution_id": execution["id"]}
485 maintenance = ExitStack()
486 try:

497 failure_identity = failure['failure_sha256']
498 # One automatic attempt per actual failure and installation edition.
499 if any(e.get('role') == 'recovery'
500 and e.get('input_manifest', {}).get('repair_identity') == {
501 'failure_sha256': failure_identity, 'installation_sha256': installation_sha256}
502 for e in runtime.get('executions', {}).values()):
503 return {'outcome': 'recovery_waiting', 'reason': 'same failure and installation already attempted'}
