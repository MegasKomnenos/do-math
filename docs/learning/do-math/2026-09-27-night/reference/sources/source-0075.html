33def inspect_bundle(skill_root: Path | None = None) -> dict:
34 """Inspect only the specified skill directory, or this installation."""
35 root = (skill_root or Path(__file__).resolve().parents[2]).resolve()
36 checks = []
37 try:
38 roles = all_prompt_documents(skill_root=root)
39 documents = {document["path"]: document["sha256"]
40 for assigned in roles.values() for document in assigned}
41 for name in INSTALLED_DOCUMENTS:
42 path = root / name
43 if not path.resolve().is_relative_to(root):
44 raise ValueError(f"document leaves the skill: {name}")
45 raw = path.read_bytes()
