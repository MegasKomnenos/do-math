28 import uuid
29 reads = []
30 # A nonce keeps these bytes apart from memo entries other tests left in the process.
31 nonce = uuid.uuid4().hex
32 def read(path, snapshot):
33 reads.append(path)
34 # Every bundle holds the same bytes for the same file name.
35 return canonical_json({'file': path.rsplit('/', 1)[-1], 'nonce': nonce})
36 execution = {'input_bundle': 'BA-' + uuid.uuid4().hex}
37 with patch.object(submission, 'load_json', wraps=submission.load_json) as parse:
38 first, _ = submission.issued_input(execution, 'head', read)
39 first['changed'] = True
40 second, schema = submission.issued_input(execution, 'head', read)
41 # The bytes are read on every call; each content is parsed once.
42 self.assertEqual(reads, ['runtime/artifacts/' + execution['input_bundle'] + '/' + name
43 for name in ('input/submission/context.json', 'schema.json')] * 2)
44 self.assertEqual(parse.call_count, 2)
45 self.assertNotIn('changed', second)
46 self.assertEqual(schema['file'], 'schema.json')
47 schema['changed'] = True
48 # The same bytes under another bundle and snapshot reuse the parse, as copies.
49 other, other_schema = submission.issued_input({'input_bundle': 'BA-' + uuid.uuid4().hex}, 'other', read)
50 self.assertEqual(parse.call_count, 2)
51 self.assertEqual(other, second)
52 self.assertNotIn('changed', other_schema)

54 nonce = uuid.uuid4().hex
55 third, _ = submission.issued_input(execution, 'head', read)
