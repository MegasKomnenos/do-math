14def job_native_result(case, execution, context, raw, workspace):
15 """Real Windows Job and adapter collector; only CLI response generation is scripted."""
16 import os, uuid
17 from do_math.windows_job import WindowsJob, process_identity
18 from do_math.codex_adapter import CLIExecution, CodexExecAdapter
19 result = native_result(execution, context, raw, workspace)
20 collected = result['execution']['collect']
21 evidence = workspace.parent / 'native-evidence'; evidence.mkdir()
22 payload = {'stdout.jsonl': collected['stdout']['raw'].decode(), 'final.txt': raw.decode(),
23 'stderr.txt': '', 'status.json': json.dumps(collected['worker_status'])}
24 # The suspended process itself writes all terminal evidence after resume.
25 script = "from pathlib import Path; import json,sys; p=Path(sys.argv[1]); d=json.loads(sys.argv[2]); [(p/k).write_bytes(v.encode('utf-8')) for k,v in d.items()]"
26 argv = (sys.executable, '-B', '-c', script, str(evidence), json.dumps(payload, ensure_ascii=False))
27 execution['launch_ticket']['phase'] = 'dispatched'
28 execution['capacity']['local_termination'] = 'unknown'
29 job = WindowsJob(argv, cwd=evidence, env=os.environ, name=execution['launch_ticket']['job_name'])
30 native = CLIExecution(collected['source_request']['sha256'], collected['source_prompt']['sha256'], argv,
31 evidence/'stdout.jsonl', evidence/'stderr.txt', evidence/'final.txt', evidence/'status.json',
32 job, None, collected['source_request']['raw'], collected['source_prompt']['raw'])
33 adapter = CodexExecAdapter('fixture-cli')
34 try:
35 early = adapter.collect(native)
36 case.assertFalse(early['artifacts_read'])
37 case.assertFalse((evidence/'final.txt').exists())
38 started = {**process_identity(job), 'job_name': job.name, 'request_sha256': native.request_sha256,
39 'execution_id': execution['id'], 'profile_ref': execution['execution_profile_ref']}
40 execution['capacity']['process'] = deepcopy(started)
41 result['execution']['started'] = started
42 adapter.resume(native)
43 case.assertTrue(adapter.wait_for_termination(native, 10))
44 result['execution']['collect'] = adapter.collect(native)
45 case.assertEqual(result['execution']['collect']['result'], 'output_collected')
46 case.assertTrue(result['execution']['collect']['termination_barrier']['confirmed'])
47 finally:
48 job.close()
49 execution['launch_ticket']['phase'] = 'finished'
50 execution['capacity']['local_termination'] = 'confirmed'
51 return result
