53def _replace_once(source,destination):
54 if os.name!='nt':
55 os.replace(source,destination)
56 return
57 import ctypes
58 from ctypes import wintypes
59 create,close,invalid,rename=_windows_files()
60 # FileRenameInfoEx (22), REPLACE_IF_EXISTS | POSIX_SEMANTICS (3).
61 # Sharing DELETE alone does not make MoveFileEx replace an open target.
62 # https://learn.microsoft.com/windows/win32/api/winbase/ns-winbase-file_rename_info
63 class RenameInfo(ctypes.Structure):
64 _fields_=[('Flags',wintypes.DWORD),('RootDirectory',wintypes.HANDLE),
65 ('FileNameLength',wintypes.DWORD),('FileName',wintypes.WCHAR*1)]
66 name=_windows_path(destination).encode('utf-16-le')
67 size=max(ctypes.sizeof(RenameInfo),RenameInfo.FileName.offset+len(name)+ctypes.sizeof(wintypes.WCHAR))
68 buffer=ctypes.create_string_buffer(size)
69 info=ctypes.cast(buffer,ctypes.POINTER(RenameInfo)).contents
70 info.Flags=3;info.RootDirectory=None;info.FileNameLength=len(name)
71 ctypes.memmove(ctypes.addressof(buffer)+RenameInfo.FileName.offset,name,len(name))
72 handle=create(_windows_path(source),0x10000,7,None,3,0x00200000,None)
73 if handle==invalid:raise ctypes.WinError(ctypes.get_last_error())
74 try:
75 if not rename(handle,22,buffer,size):raise ctypes.WinError(ctypes.get_last_error())
76 finally:
77 close(handle)

119@contextmanager
120def os_lock(path, *, blocking):
121 """Hold an advisory OS lock on byte 0 of ``path``, creating the file if absent.
122
123 ``blocking=False`` raises ``OSError`` at once when another holder has it.
124 ``blocking=True`` waits: on POSIX without bound, on Windows ``LK_LOCK``
125 retries ten times at one-second intervals and then raises ``OSError``.
126 """
127 descriptor=os.open(path,os.O_RDWR|os.O_CREAT,0o600)
128 try:
129 if os.fstat(descriptor).st_size==0:
130 os.write(descriptor,b'\0')
131 os.lseek(descriptor,0,os.SEEK_SET)
132 if os.name=='nt':
133 import msvcrt
134 msvcrt.locking(descriptor,msvcrt.LK_LOCK if blocking else msvcrt.LK_NBLCK,1)
135 else:
136 import fcntl
137 fcntl.flock(descriptor,fcntl.LOCK_EX if blocking else fcntl.LOCK_EX|fcntl.LOCK_NB)
138 try:
139 yield
140 finally:
141 os.lseek(descriptor,0,os.SEEK_SET)
142 if os.name=='nt':
143 msvcrt.locking(descriptor,msvcrt.LK_UNLCK,1)
144 else:
145 fcntl.flock(descriptor,fcntl.LOCK_UN)
146 finally:
147 os.close(descriptor)

150def stat_fingerprint(st):
151 """Metadata fingerprint (지문) of one file: size, modification and birth time.
152
153 Accepts ``os.stat`` results and ``os.DirEntry.stat(follow_symlinks=False)``.
154 A rewrite that preserves all three values is not detected; only an explicit
155 full-byte check catches it (accidental-change threat model). NTFS tunneling
156 keeps the birth time when a name is deleted and recreated within ~15 s, so a
157 same-size replacement that also restores the modification time is missed.
158 """
159 return (st.st_size, st.st_mtime_ns, getattr(st, 'st_birthtime_ns', st.st_ctime_ns))

291def cached_sha256(path):
292 """Full-byte SHA-256, recomputed only when the file's fingerprint changes."""
293 key=os.path.normcase(os.path.abspath(path))
294 before=file_fingerprint(key)
295 hit=_hashed.get(key)
296 if hit and hit[0]==before:
297 return hit[1]
298 value=sha256_file(key)
299 # A file that changed while being read is hashed again on the next call.
300 if file_fingerprint(key)==before and fingerprint_settled(before):
301 _hashed[key]=(before,value)
302 return value
