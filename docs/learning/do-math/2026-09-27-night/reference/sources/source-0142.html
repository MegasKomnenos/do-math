39def respond(cycle, **changes):
40 action = review.next_actions(cycle)[0]
41 stage = action['stage']
42 output = {'proposal_digest': cycle['proposal_digest'], 'summary': 'Fixture full-source response',
43 'used_sources': ['proposal:1'], 'questions': []}
44 if stage == 'initial': output.update(decision='match', scope_complete=True, findings=[])
45 elif stage == 'organize':
46 output['issues'] = [{'id': f'i{index}', 'finding_refs': [f['ref']],
47 'answer_role_slot_id': f['ref']['slot'] if f['ref']['stage'] in {'initial', 'answer'} else 'initial-0'}
48 for index, f in enumerate(review.context_for(cycle, action)['pending_findings'], start=len(cycle['issues']))]
49 elif stage == 'answer':
50 output.update(answers=[{'issue_id': i, 'disposition': 'maintain', 'statement': 'Still applies',
51 'evidence': ['proposal:1']} for i in review.context_for(cycle, action)['answer_requests']], new_findings=[])
52 elif stage == 'judge':
53 output.update(decision='mismatch', scope_complete=True, issue_verdicts=[{'issue_id': i,
54 'decision': 'approved', 'reason': 'Source comparison', 'evidence': ['proposal:1']}
55 for i in cycle['issues']], new_findings=[])
56 output.update(changes)
57 return review.accept_result(cycle, action, output, new_id('EX'))
