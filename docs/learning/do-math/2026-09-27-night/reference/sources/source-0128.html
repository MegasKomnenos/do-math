16def created_instance(root, shared):
17 """``restore_template`` build: ``init --create-repository`` through the CLI; returns the instance ID.
18
19 The repository is ``root / 'repo'`` and the workspace ``root / 'workspace'``.
20 """
21 configuration = root / "config.json"
22 configuration.write_text(json.dumps({"operator_enabled": False}), encoding="utf-8")
23 init = run_cli("init", "--create-repository", "--repository", str(root / "repo"),
24 "--workspace", str(root / "workspace"), "--goal", "Original goal",
25 "--config-file", str(configuration), cwd=root)
26 if init.returncode != 0:
27 raise AssertionError(init.stdout + init.stderr)
28 return json.loads(init.stdout)["result"]["instance_id"]

36 def cli(self, *arguments):
37 result = run_cli(*arguments, cwd=self.root)
38 text = result.stdout.strip()
39 envelope = json.loads(text) if text else None
40 if envelope is not None:
41 self.assertEqual(envelope['schema_version'], 5)
42 self.assertEqual(envelope['exit_code'], result.returncode)
43 self.assertNotEqual('result' in envelope, 'error' in envelope)
