16def run_case(destination, case, *, provider, model, effort, timeout, only_assigned, scholarly=False):
17 root = destination / case
18 workspace = root / "workspace"
19 workspace.mkdir(parents=True)
20 nonce = uuid.uuid4().hex
21 schema = {"type": "object", "additionalProperties": False,
22 "properties": {"status": {"const": "CLI_OK"}, "nonce": {"const": nonce},
23 "phase": {"enum": ["finished", "pending"]}}, "required": ["status", "nonce", "phase"]}
24 prompt = f'This is an isolated CLI integration probe. Work only in {workspace}. Return the provided JSON schema with status CLI_OK and nonce {nonce}. '
25 if case == "structured":
26 prompt += 'Use no tools. Return phase finished.'
27 elif case == "files":
28 checkpoint = {"current_claim": "The local probe file exists.", "evidence_paths": ["probe-evidence.txt"],
29 "attempts": ["Write and read the probe file."], "remaining_work": [],
30 "next_step": "Return the structured probe result.", "progress_state": "ready_to_submit"}
31 prompt += ('Write probe-evidence.txt containing exactly ' + nonce + ', read it back, then write checkpoint.json as ' +
32 json.dumps(checkpoint) + '. Return phase finished after verifying both files. ')
33 if scholarly:
34 prompt += 'Also call the provided scholarly list_sources tool once. Do not search or fetch literature.'
35 elif case == "background":
36 prompt += ('Use Bash to start exactly one background command (run_in_background=true) which waits 5 seconds '
37 'and writes completed.txt containing exactly ' + nonce + '. Write a local phase-pending.json checkpoint while it runs. '
38 'Use the actual TaskOutput tool to wait for that background task, then read completed.txt and submit phase finished. '
39 'Keep the CLI invocation active until the job finishes. Do not submit the final JSON while it is still pending; '
40 'print mode may terminate outstanding background work when the final result is returned. Do not start agents or another model.')
41 else:
42 prompt += 'Start a Bash command that waits 120 seconds. The local runner will deliberately stop this probe while that tool is active. Do not start background agents.'
43 stop = False
44 native_tool_names = []
45 def observe_event(event):
46 nonlocal stop
47 content = event.get("message", {}).get("content", []) if isinstance(event.get("message"), dict) else []
48 for block in content if isinstance(content, list) else []:
49 if isinstance(block, dict) and block.get("type") == "tool_use":
50 native_tool_names.append(block.get("name"))
51 if case == "stop":
52 stop = True
53 item = event.get("item", {})
54 if event.get("type") == "item.started" and isinstance(item, dict):
55 if item.get("type") == "mcp_tool_call":
56 server, tool = item.get("server"), item.get("tool")
57 if isinstance(server, str) and isinstance(tool, str):
58 native_tool_names.append(f"mcp__{server}__{tool}")
59 if case == "stop" and item.get("type") == "command_execution":
60 stop = True
61 def observe(batch):
62 for event in batch.events:
63 observe_event(event)
64 server = resolve_scholarly_server(None, start=Path.cwd()) if scholarly else None
65 runner = ModelRunner()
66 routed_model = model if provider == "codex" or model.startswith("claude-") else "claude:" + model
67 result = runner.run(prompt=prompt, output_schema=schema, workspace=workspace, evidence_dir=root / "evidence",
68 model=routed_model, effort=effort, timeout_seconds=timeout, claude_only_assigned_model=only_assigned,
69 scholarly_server=server, on_observation=observe, should_stop=lambda: stop)
70 collected = result.get("execution", {}).get("collect", {})
71 parsed = collected.get("jsonl", {})
72 checks = {"local_termination": collected.get("termination_barrier", {}).get("confirmed") is True}
73 if case == "stop":
74 checks["requested_stop"] = stop and result["status"] == "stopped"
75 else:
76 checks["structured_result"] = result["status"] == "completed" and result["output"] == {"status": "CLI_OK", "nonce": nonce, "phase": "finished"}
77 if case == "files":
78 checks["preserved_file"] = (workspace / "probe-evidence.txt").is_file() and (workspace / "probe-evidence.txt").read_text(encoding="utf-8").strip() == nonce
79 checks["checkpoint"] = read_checkpoint(workspace)["status"] == "valid"
80 if scholarly:
81 checks["scholarly_actual_tool"] = any(name and "scholarly" in name and "list_sources" in name for name in native_tool_names)
82 if case == "background":
83 checks["native_background_start"] = any(row["event"].get("subtype") == "task_started" for row in parsed.get("background_events", []))
84 checks["actual_wait_tool"] = "TaskOutput" in native_tool_names
85 checks["completed_file"] = (workspace / "completed.txt").is_file() and (workspace / "completed.txt").read_text(encoding="utf-8").strip() == nonce
86 if provider == "claude" and only_assigned and case != "stop":
87 checks["reported_models"] = bool(parsed.get("reported_models")) and parsed["reported_models"] == [model]
88 report = {"case": case, "passed": all(checks.values()), "checks": checks, "result": json_safe(result),
89 "actual_tool_names": native_tool_names}
90 (root / "report.json").write_text(json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8")
91 return {"case": case, "passed": report["passed"], "checks": checks, "report": str(root / "report.json")}

94def main(argv=None):
95 parser = argparse.ArgumentParser(description=__doc__)
96 parser.add_argument("--skill-only", action="store_true")
97 parser.add_argument("--provider", choices=("claude", "codex"))
98 parser.add_argument("--model")
99 parser.add_argument("--effort", default="low")
100 parser.add_argument("--only-assigned-model", action="store_true")
101 parser.add_argument("--scholarly", action="store_true")
102 parser.add_argument("--timeout-seconds", type=int, default=180)
103 parser.add_argument("--cases", nargs="+", choices=("structured", "files", "background", "stop"), default=["structured", "files", "stop"])
104 parser.add_argument("--output-dir", type=Path)
105 args = parser.parse_args(argv)
106 if args.skill_only:
107 result = inspect_bundle()
108 print(json.dumps(result, ensure_ascii=False))
109 return 0 if result["status"] == "verified" else 1
110 if not args.provider or not args.model or not args.output_dir or args.timeout_seconds < 1:
111 parser.error("live probes require explicit --provider, --model, --output-dir and a positive timeout")
112 if args.provider == "claude" and args.only_assigned_model and not args.model.startswith("claude-"):
113 parser.error("model locking requires the full Claude model ID")
114 if args.provider == "codex" and args.model.startswith(("claude:", "claude-")):
115 parser.error("provider and model disagree")
116 if args.provider == "codex" and ("background" in args.cases or args.only_assigned_model):
117 parser.error("background and --only-assigned-model require --provider claude")
118 if len(set(args.cases)) != len(args.cases):
119 parser.error("cases must be distinct")
120 args.output_dir.mkdir(parents=True, exist_ok=False)
121 results = []
122 for case in args.cases:
123 print("probe " + case, flush=True)
124 try:
125 results.append(run_case(args.output_dir, case, provider=args.provider, model=args.model, effort=args.effort,
126 timeout=args.timeout_seconds, only_assigned=args.only_assigned_model, scholarly=args.scholarly and case == "files"))
127 except Exception as error:
128 results.append({"case": case, "passed": False, "error": type(error).__name__ + ": " + str(error)})
129 root = Path(__file__).parent
130 report = {"at": datetime.now(timezone.utc).isoformat(), "provider": args.provider, "model": args.model,
131 "passed": all(row["passed"] for row in results), "cases": results,
132 "sources": {str(path.relative_to(root)): hashlib.sha256(path.read_bytes()).hexdigest() for path in (root / "do_math").glob("*.py")},
133 "scope": "representative local CLI protocol, files, tool and stop boundaries; no research efficacy claim"}
134 (args.output_dir / "report.json").write_text(json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8")
135 print(json.dumps(report, ensure_ascii=False))
136 return 0 if report["passed"] else 1
