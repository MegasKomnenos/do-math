62 def test_status_reports_artifact_presence_and_length_in_one_batch_without_bytes(self):
63 import hashlib
64 from unittest.mock import create_autospec, patch
65 from do_math.reporting import snapshot_report
66 from do_math.git_store import GitStateStoreError
67 instance, runtime = surface_state()
68 raw = b'preserved result'
69 paths = ['runtime/artifacts/BA-1/' + str(index) for index in range(12)]
70 runtime['executions']['EX-1']['output_files'] = [
71 {'path': str(index), 'artifact_ref': path,
72 'sha256': hashlib.sha256(raw).hexdigest(), 'bytes': len(raw)}
73 for index, path in enumerate(paths)]
74 head = 'a' * 40
75 stored = dict.fromkeys(paths, len(raw))
76
77 def sizes(requested, snapshot, *, missing='raise'):
78 self.assertEqual(snapshot, head)
79 if missing != 'omit' and any(path not in stored for path in requested):
80 raise GitStateStoreError('batch size did not return a blob')
81 return {path: stored[path] for path in requested if path in stored}
82
83 store = create_autospec(ResearchRepository, instance=True)
84 store.load.return_value = (head, instance, runtime)
85 store.read.side_effect = store.read_many.side_effect = AssertionError('status reads no artifact bytes')
86 store.blob_sizes.side_effect = sizes
87 with patch('do_math.reporting.ResearchRepository', return_value=store):
88 report = snapshot_report(Path('.'), 'RI-test', requested_areas=['record_storage'])
89 self.assertEqual(report['inspection']['status'], 'healthy')
90 store.blob_sizes.assert_called_once_with(paths, head, missing='omit')
91 # A changed length or a missing artifact blocks; same-length byte changes are for check.
92 stored[paths[0]] = len(raw) + 1
93 del stored[paths[1]]
94 report = snapshot_report(Path('.'), 'RI-test', requested_areas=['record_storage'])
95 self.assertEqual(report['inspection']['status'], 'blocked')
96 blocked = {evidence for row in report['inspection']['findings'] for evidence in row['evidence']}
97 self.assertEqual(blocked & set(paths), {paths[0], paths[1]})
