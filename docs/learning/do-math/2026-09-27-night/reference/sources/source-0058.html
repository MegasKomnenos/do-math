41def capture_basis(instance: dict, runtime: dict, record_ids: list[str]) -> list[dict]:
42 if not isinstance(record_ids, list) or len(record_ids) != len(set(record_ids)):
43 raise ValueError("record_ids must be a list of distinct IDs")

55def validate_basis(basis: list[dict], instance: dict, runtime: dict) -> dict:
56 reasons: list[str] = []
57 if not isinstance(basis, list):
58 return {"valid": False, "reasons": ["basis_not_a_list"]}
59 seen: set[str] = set()
60 records = _basis_records(instance, runtime)
61 for row in basis:
62 if not isinstance(row, dict) or set(row) != {"id", "sha256", "status"}:
63 reasons.append("invalid_basis_row")
64 continue
65 record_id = row["id"]
66 if not isinstance(record_id, str) or record_id in seen:
67 reasons.append(f"duplicate_or_invalid_id:{record_id}")
68 continue
69 seen.add(record_id)
70 record = records.get(record_id)
71 if record is None:
72 reasons.append(f"missing:{record_id}")
73 continue
74 if _hash(record) != row["sha256"]:
75 reasons.append(f"hash_changed:{record_id}")
76 if _status(record) != row["status"]:
77 reasons.append(f"status_changed:{record_id}")
78 return {"valid": not reasons, "reasons": reasons}

123def _field_refs(value: Any, *, strict: bool, known_ids: set[str]) -> set[str]:
124 refs: set[str] = set()
125 if isinstance(value, str):
126 if strict or value in known_ids or _KNOWLEDGE_ID.fullmatch(value):
127 refs.add(value)
128 elif isinstance(value, list):
129 for item in value:
130 refs.update(_field_refs(item, strict=strict, known_ids=known_ids))
131 if isinstance(value, dict):
132 identifiers = [item for key, item in value.items()
133 if key in _ID_KEYS and isinstance(item, str)]
134 for item in identifiers:
135 if strict or item in known_ids or _KNOWLEDGE_ID.fullmatch(item):
136 refs.add(item)
137 # A source-ref object is a locator. Its path/hash/range strings are
138 # validated by context_store, not interpreted as knowledge IDs here.
139 if not identifiers and strict:
140 for item in value.values():
141 refs.update(_field_refs(item, strict=True, known_ids=known_ids))
142 return refs

145def _explicit_refs(value: Any, known_ids: set[str]) -> set[str]:
146 refs: set[str] = set()
147 if isinstance(value, dict):
148 for key, item in value.items():
149 if key in _HISTORY_KEYS:
150 continue
151 if key in _STRICT_REFERENCE_KEYS:
152 refs.update(_field_refs(item, strict=True, known_ids=known_ids))
153 elif key in _REFERENCE_KEYS:
154 refs.update(_field_refs(item, strict=False, known_ids=known_ids))
155 elif key == "basis" and isinstance(item, list):
156 refs.update(_field_refs(item, strict=False, known_ids=known_ids))
157 else:
158 refs.update(_explicit_refs(item, known_ids))
159 elif isinstance(value, list):
160 for item in value:
161 refs.update(_explicit_refs(item, known_ids))
162 return refs

165def assess_research_use(record: dict, instance: dict, _visiting=None) -> dict:
166 """Use a current research record at its stated evidence level, without certifying it.
167
168 The shared record premises are knowledge_records.premise_failures; research use
169 adds use suspension, an active exposition and research-level dependencies."""
170 from .knowledge_records import premise_failures
171 identifier = record.get('id')
172 visiting = set(_visiting or ())
173 reasons = []
174 if not identifier or identifier in visiting:
175 reasons.append('missing_or_cyclic_research_dependency')
176 reasons.extend('research_' + failure for failure in premise_failures(instance, record))
177 if record.get('use_suspended') and 'research_record_not_active' not in reasons:
178 reasons.append('research_record_not_active')
179 exposition = instance.get('expositions', {}).get(record.get('exposition_id'), {})
180 if exposition.get('lifecycle') != 'active':
181 reasons.append('research_exposition_not_active')
182 if reasons:
183 return {'valid': False, 'reasons': reasons}
184 visiting.add(identifier)
185 for dependency in record.get('dependencies', []):
186 prior = instance.get('knowledge', {}).get(dependency)
187 if prior is None or not assess_research_use(prior, instance, visiting)['valid']:
188 reasons.append('research_dependency_not_current:' + dependency)
189 return {'valid': not reasons, 'reasons': reasons}

192def assess_use(record: dict, instance: dict) -> dict:
193 if not isinstance(record, dict):
194 raise TypeError("record must be an object")
195 from .knowledge_records import usable_links, formal_usable, assessment_current
196 family = record.get("family", "knowledge")
197 if family == "knowledge":
198 links = usable_links(instance, record.get("id"))
199 valid = bool(links)
200 return {"valid": valid, "reasons": [] if valid else ["no_current_verified_exposition"], "basis": links}

202 valid = formal_usable(instance, record)

209 "basis": [{"assessment_id": record['id'], "assessment_use_id": use['id']} for use in uses] if valid else []}
210 return {"valid": False, "reasons": ["record_is_not_a_verified_knowledge_authority"], "basis": []}

227def plan_impact(instance, runtime, roots, declared_impact, *, sensitive=False, excluded_dependents=()):
228 from .knowledge_records import usable_links
229 records = _basis_records(instance, runtime)
230 roots, declared = set(roots), set(declared_impact)
231 excluded = set(excluded_dependents)
232 if not roots or roots - set(records):
233 raise ValueError("impact roots must exist")
234 if declared - set(records):
235 raise ValueError("declared impact refers to unknown records")
236 if excluded - set(records):
237 raise ValueError("excluded impact dependents must exist")
238 known = set(records)
239 # Record references and K support links stay fixed while the closure grows.
240 refs = {rid: _impact_refs(record, known) for rid, record in records.items()}
241 links = {kid: usable_links(instance, kid) for kid in instance.get("knowledge", {})}
242 affected = set(roots)
243 while True:
244 found = {rid for rid, targets in refs.items() if rid not in excluded and targets & affected}
245 # K depends on the available positive links, not on every historical vote.
246 # An unaffected independent link continues to support the same K.
247 for kid, support in links.items():
248 if support and all({link['assessment_id'], link['assessment_use_id']} & affected for link in support):
249 found.add(kid)
250 if found <= affected:
251 break
252 affected |= found
253 computed = affected - roots
254 unknown = sorted({ref for rid in affected for ref in refs[rid] if ref not in records})
255 plan = {"roots": sorted(roots), "computed_impact": sorted(computed), "declared_impact": sorted(declared),
256 "missing_declarations": sorted(computed - declared), "extra_declarations": sorted(declared - computed),
257 "unresolved_refs": unknown, "sensitive": sensitive,
258 "requires_independent_review": bool(sensitive or computed != declared or unknown),
259 "basis": {rid: _hash(records[rid]) for rid in sorted(affected | declared)}}
260 plan["digest"] = _hash(plan)
261 return plan

264def _confirmed_impact(plan, confirmation):
265 if plan.get("digest") != _hash({key: value for key, value in plan.items() if key != "digest"}):
266 raise ValueError("impact plan digest is invalid")
267 affected = set(plan["computed_impact"])
268 if plan["requires_independent_review"]:
269 confirmation = confirmation or {}
270 if (confirmation.get("plan_digest") != plan["digest"] or confirmation.get("decision") != "approve"
271 or not confirmation.get("review_ref") or not confirmation.get("reviewer_ref")
272 or not confirmation.get("producer_ref") or confirmation["reviewer_ref"] == confirmation["producer_ref"]
273 or set(confirmation.get("affected_ids", [])) != affected | set(plan["declared_impact"])):
274 raise ValueError("impact requires independent scope and disposition confirmation")
275 affected |= set(plan["declared_impact"])
276 return affected

279def _apply_impact_effects(instance, runtime, affected, roots, *, reason, impact_digest, in_place=False):
280 # in_place is for callers that already hold their own copies of both views.
281 output, new_runtime = (instance, runtime) if in_place else (deepcopy(instance), deepcopy(runtime))
282 records = _basis_records(output, new_runtime)
283 semantic_only = all(records[rid].get('family') in {'identity_assessments', 'assessment_uses'} for rid in roots)
284 for rid in sorted(affected):
285 record = records[rid]
286 family = record.get("family")
287 if family == "formal_results":
288 if semantic_only:
289 continue # A semantic judgment defect does not refute checked Lean code.
290 record["formal_use"] = "suspended"
291 elif family == "assessment_uses":
292 if record.get("status") != "withdrawn":
293 record["status"] = "suspended"
294 record.pop("environment_revalidation", None)
295 elif family == "identity_assessments":
296 for use in output.get("assessment_uses", {}).values():
297 if use["assessment_id"] == rid and use["status"] != "withdrawn":
298 use["status"] = "suspended"
299 use.pop("environment_revalidation", None)
300 continue # Sealed judgment is immutable.
301 elif rid in new_runtime.get("tasks", {}):
302 if isinstance(record.get("assessment"), dict):
303 record["assessment"]["current"] = False
304 # Task lifecycle belongs to TASKS; its readiness consumes this currentness.
305 record["input_current"] = False
306 elif record in output.get("relations", []):
307 record["status"] = "suspended"
308 elif record in output.get("goal_judgments", []):
309 record["current"] = False
310 elif family in {"knowledge", "expositions"}:
311 # Currentness is derived from the actual premise/use, not a duplicate lifecycle.
312 continue
313 elif rid in new_runtime.get('executions', {}) or rid in new_runtime.get('activities', {}):
314 # Preserve historical attempts and activity results. Their input
315 # snapshots and execution currentness checks own validity; the
316 # affected IDs remain in the reassessment event below.
317 continue
318 else:
319 record["input_current"] = False
320 record.setdefault("use_history", []).append({"reason": reason, "impact_digest": impact_digest})
321 new_runtime.setdefault("history", []).append({"kind": "knowledge_reassessment_required",
322 "affected_ids": sorted(affected), "reason": reason, "impact_digest": impact_digest})
323 if affected:
324 action = {"role": "research_plan", "reason": "knowledge currentness changed", "affected_ids": sorted(affected)}
325 if not any(all(queued.get(key) == value for key, value in action.items())
326 for queued in new_runtime.setdefault("queue", [])):
327 action["id"] = action["lineage_id"] = new_id("AC")
328 new_runtime["queue"].append(action)
329 return output, new_runtime

393def _assessment_disposition(instance, runtime, assessment_id, status, *, reason, actor_ref,
394 reason_refs, reassessment_ref=None):
395 from .knowledge_records import set_assessment_use
396 output, run = deepcopy(instance), deepcopy(runtime)
397 use = next(u for u in output['assessment_uses'].values() if u['assessment_id'] == assessment_id)
398 if reassessment_ref and any(h.get('reassessment_ref') == reassessment_ref for h in use['history']):
399 return output, run
400 old_status = use['status']
401 # Calculate fallout while the original support is still visible.
402 if status != 'active' and old_status == 'active':
403 _propagate(output, run, [assessment_id], reason)
404 # Restore the root's state for the transition primitive: propagation can also
405 # reach its AU, but its own explicit disposition must retain the actual prior state.
406 use = next(u for u in output['assessment_uses'].values() if u['assessment_id'] == assessment_id)
407 use['status'] = old_status
408 output = set_assessment_use(output, assessment_id, status, reason, actor_ref)
409 use = next(u for u in output['assessment_uses'].values() if u['assessment_id'] == assessment_id)
410 evidence = {'reason_refs': list(reason_refs)}
411 if reassessment_ref:
412 evidence['reassessment_ref'] = reassessment_ref
413 evidence['assessment_sha256'] = _hash(output['identity_assessments'][assessment_id])
414 evidence['proposal_digest'] = run['review_cycles'][reassessment_ref]['proposal_digest']
415 if old_status == status:
416 use['history'].append({'from': old_status, 'to': status, 'reason': reason, 'actor_ref': actor_ref})
417 use['history'][-1].update(evidence)
418 return output, run

433def apply_assessment_reassessment(instance, runtime, assessment_id, review_cycle_id, *, reason, actor_ref):
434 """Consume a sealed independent review of the exact original judgment.
435
436 Replay uses the existing review state machine, including scope, sealing and
437 finite-round decisions. Reviewer semantics are evidence, not a machine proof.
438 """
439 from . import review_cycle as reviews
440 assessment = instance['identity_assessments'][assessment_id]
441 cycle = runtime.get('review_cycles', {}).get(review_cycle_id, {})
442 expected = assessment_reassessment_proposal(instance, assessment_id, runtime=runtime)

444 or cycle.get('policy_ref') != assessment['policy_ref']):
445 raise ValueError('reassessment must identify the exact original judgment and policy')
446 original = runtime.get('review_cycles', {}).get(assessment.get('review_cycle_id'), {})

448 raise ValueError('reassessment must retain the original review policy')

462 actor_ref=actor_ref, reason_refs=[entry['execution_id'] for entry in replay['results'].values()],
463 reassessment_ref=review_cycle_id)

491def replace_knowledge(instance: dict, runtime: dict, knowledge_id: str, changes: dict,
492 *, reason: str, actor_ref: str, replacement_id=None,
493 reserved_dependency_ids=()) -> tuple[dict, dict]:
494 """Create one replacement after its bundle's impact disposition is complete."""
495 from . import knowledge_records as kr
496 require_text(reason, "reason"); require_text(actor_ref, "actor_ref")
497 validate_changes(changes)
498 old = instance["knowledge"][knowledge_id]
499 view = kr.proposal_view(instance, knowledge_id)
500 candidate = {key: deepcopy(value) for key, value in view.items() if key in AUTHORED_FIELDS}
501 candidate.update(deepcopy(changes))
502 if replacement_id is not None:
503 candidate["id"] = replacement_id
504 from .knowledge_io import import_candidate
505 output, new_runtime, actual_replacement_id = import_candidate(instance, runtime, candidate,
506 source_ref=(old.get("origin_refs") or [actor_ref])[0], actor_ref=actor_ref,
507 reserved_dependency_ids=reserved_dependency_ids)
508 replacement = output["knowledge"][actual_replacement_id]
509 replacement["supersedes"] = knowledge_id
510 replacement["revision"] = old.get("revision", 1) + 1
511 replacement["revision_reason"] = reason
512 output["knowledge"][knowledge_id]["lifecycle"] = "superseded"
513 output["expositions"][old["exposition_id"]]["lifecycle"] = "superseded"
514 old_fid = old.get("formal_result_id")
515 # Reuse unchanged formal evidence only while it is currently usable.
516 formal_changed = any(candidate.get(key) != view.get(key) for key in ("lean_code", "declaration_names", "dependencies"))
517 reuse_formal = not formal_changed and kr.formal_usable(
518 output, output.get("formal_results", {}).get(old_fid, {}))
519 if reuse_formal:
520 unused = replacement.get("formal_result_id")
521 if unused:
522 del output["formal_results"][unused]
523 replacement["formal_result_id"] = old_fid
524 output["expositions"][replacement["exposition_id"]]["formal_result_id"] = old_fid
525 # Superseding the authored record retires its pending checks at the domain
526 # boundary, including callers that apply a revision through integration.
527 # Completed executions and unrelated work remain available as history.
528 new_runtime["queue"] = [action for action in new_runtime.get("queue", [])
529 if action.get("target") != actual_replacement_id and not (
530 action.get("target") == knowledge_id and (
531 action.get("role") in {"formal_check", "identity_review"}
532 or (action.get("role") == "supplemental_research" and action.get("review_cycle_id"))))]
533 next_role = "identity_review" if reuse_formal else "formal_check"
534 if not replacement.get("formal_result_id"):
535 next_role = "research_plan"
536 action_id = new_id("AC")
537 new_runtime["queue"].append({"id": action_id, "lineage_id": action_id, "role": next_role,
538 "target": actual_replacement_id, "reason": reason})
539 return output, new_runtime

602 "previous_environment_ids": [formal.get("trust_environment_id")]})

654def _validated_staged_identity_review(instance, transition, knowledge_id, receipt, runtime):
655 from . import knowledge_records as kr
656 from .review_cycle import validate_sealed_review, identity_assessment
657 stored = runtime.get("trust_transitions", {}).get(transition.get("id"))
658 if stored is None or _transition_without_identity_reviews(stored) != _transition_without_identity_reviews(transition):
659 raise ValueError("staged identity review transition is stale or belongs to another transition")
660 cycle = runtime.get("review_cycles", {}).get(receipt.get("review_cycle_id"))
661 if cycle is None or cycle.get("transition_id") != transition["id"] or cycle.get("knowledge_id") != knowledge_id:

672 "identity_basis": basis["live_identity_basis"],
673 "active_trust_environment": environment_id,
674 "trust_environment": deepcopy(instance.get("trust_environments", {}).get(environment_id)),
675 "review_policy": deepcopy(instance.get("review_policy")),
676 "governance": deepcopy(instance.get("governance", {})),

687 "transition_basis": basis["transition_basis"],
688 }

690 raise ValueError("staged identity review receipt is stale, altered, or not a complete match")

694def activate_transition(instance: dict, transition: dict, *, runtime=None) -> dict:
695 from . import knowledge_records as kr
696 from .formal import compare_formal_results
697 output = deepcopy(instance)
698 environments = output.get("trust_environments", {})
699 target_env_id = transition["to_environment"]
700 environment = environments.get(target_env_id)
701 if not environment or environment.get("status") == "invalid":
702 raise ValueError("transition environment is unavailable")
703 if output.get("active_trust_environment") != transition.get("from_environment"):
704 raise ValueError("active trust environment changed during transition")
705 targets = {target["id"]: target for target in transition["targets"]}
706 results = transition.get("results", {})
707 if set(targets) != set(results) or any(result.get("status") != "verified" for result in results.values()):
708 raise ValueError("all transition targets must be verified")
709 comparisons, reviewed = {}, {}
710 for kid, target in targets.items():
711 record = output["knowledge"].get(kid)
712 formal = output.get("formal_results", {}).get(target["formal_result_id"], {})
713 exposition = output["expositions"].get((record or {}).get("exposition_id"), {})
714 result = results[kid]
715 if (not record or record.get("exposition_id") != target["exposition_id"]
716 or _hash(record) != target["basis"]["sha256"] or kr.record_digest(formal) != target["formal_basis"]
717 or kr.record_digest(exposition) != target["exposition_basis"]):
718 raise ValueError("transition basis changed: " + kid)
719 # This checks source, toolchain, declarations and axioms. formal is unchanged
720 # since begin_transition (record_digest above covers lean_code), so its
721 # source hash is target["source_sha256"].
722 kr.validate_formal_evidence(formal, result, environment)
723 comparison = compare_formal_results(target, result)
724 comparisons[kid] = comparison
725 if comparison["requires_identity_review"]:
726 if runtime is None:
727 raise ValueError("transition requires its sealed staged identity review runtime: " + kid)
728 reviewed[kid] = _validated_staged_identity_review(
729 instance, transition, kid, result.get("identity_review", {}), runtime)
730 old_env_id = output.get("active_trust_environment")
731 if old_env_id and environments[old_env_id].get("status") != "invalid":
732 environments[old_env_id]["status"] = "superseded"
733 environment["status"] = "active"
734 output["active_trust_environment"] = target_env_id
735 for kid, target in targets.items():
736 record = output["knowledge"][kid]
737 old_formal = output["formal_results"][target["formal_result_id"]]
738 old_xid = record["exposition_id"]
739 fid, xid = new_id("FR"), new_id("XP")
740 result = {key: deepcopy(value) for key, value in results[kid].items() if key not in {"id", "identity_review"}}
741 output["formal_results"][fid] = {**deepcopy(old_formal), "id": fid, "verification": result,
742 "trust_environment_id": target_env_id, "formal_use": "active", "transition_id": transition["id"],
743 "supersedes": old_formal["id"]}
744 output["expositions"][xid] = {**deepcopy(output["expositions"][old_xid]), "id": xid,
745 "formal_result_id": fid, "supersedes": old_xid}
746 record.update(formal_result_id=fid, exposition_id=xid)
747 if kid in reviewed:
748 receipt = reviewed[kid]
749 previous_cycle = record.get("review_cycle_id")
750 if previous_cycle:
751 record["previous_review_cycle_id"] = previous_cycle
752 record["review_cycle_id"] = receipt["review_cycle_id"]
753 assessment = {key: deepcopy(receipt[key]) for key in (
754 "decision", "scope_complete", "execution_ids", "review_cycle_id",
755 "summary", "proposal_digest", "transition_id", "transition_basis")}
756 assessment.update(
757 basis=kr.identity_basis(output, kid),
758 reviewed_identity_basis=deepcopy(receipt["identity_basis"]),
759 )
760 # output is this activation's own copy; seal in place, not per target.
761 kr.seal_identity_assessment(output, kid, assessment, receipt["policy_ref"], runtime={})
762 continue
763 # Preserve the old sealed judgment. A new continuity judgment names both
764 # exact new inputs and the transition comparison instead of mutating it.
765 for assessment in list(output.get("identity_assessments", {}).values()):
766 if assessment.get("exposition_id") != old_xid or assessment.get("decision") != "match":
767 continue
768 uses = [use for use in output.get("assessment_uses", {}).values() if use["assessment_id"] == assessment["id"]]
769 recoverable = bool(uses and uses[0].get("environment_revalidation", {}).get("environment_id")
770 in target["previous_environment_ids"])
771 if not uses or (uses[0]["status"] != "active" and not recoverable):
772 continue
773 inherited = {key: deepcopy(value) for key, value in assessment.items() if key not in {"id", "formal_result_id", "exposition_id"}}
774 inherited.update(basis=kr.identity_basis(output, kid), previous_assessment_id=assessment["id"], transition_id=transition["id"],
775 continuity_comparison=comparisons[kid])
776 kr.seal_identity_assessment(output, kid, inherited, assessment["policy_ref"], runtime={})
777 return output

780def invalidate_environment(instance, runtime, env_id, reason, *, affected_constants=None,
781 scope_evidence=None) -> tuple[dict, dict]:
782 require_text(reason, "reason")
783 output, new_runtime = deepcopy(instance), deepcopy(runtime)
784 environment = output.get("trust_environments", {}).get(env_id)
785 if environment is None:
786 raise ValueError("unknown trust environment: " + env_id)
787 scoped = bool(affected_constants and scope_evidence)
788 candidates = [formal for formal in output.get("formal_results", {}).values()
789 if formal.get("trust_environment_id") == env_id]
790 if any(not formal.get("verification", {}).get("dependency_closure") for formal in candidates):
791 scoped = False
792 if not scoped:
793 environment.update(status="invalid", invalidated_reason=reason)
794 if output.get("active_trust_environment") == env_id:
795 output["active_trust_environment"] = None
796 roots = set()
797 for formal in candidates:
798 closure = {name for names in formal.get("verification", {}).get("dependency_closure", {}).values() for name in names}
799 closure.update(formal.get("declaration_names", []))
800 if not scoped or closure & set(affected_constants):
801 formal["formal_use"] = "suspended"
802 formal.setdefault("use_history", []).append({"reason": reason, "environment_id": env_id,
803 "affected_constants": sorted(affected_constants or []), "scope_evidence": scope_evidence})
804 roots.add(formal["id"])
805 previously_active = {uid for uid, use in output.get("assessment_uses", {}).items() if use.get("status") == "active"}
806 if roots:
807 _propagate(output, new_runtime, roots, reason)
808 for uid in previously_active:
809 use = output["assessment_uses"][uid]
810 if use["status"] == "suspended":
811 use["environment_revalidation"] = {"environment_id": env_id, "reason": reason}
812 return output, new_runtime
