20def record_digest(record):

23 'expositions': ('items', 'relation_claim')}.get(record.get('family'), ())

40def exposition_items(candidate):
41 """Stable local coordinates participate in the identity review basis."""
42 items = []
43 for field in EXPOSITION_FIELDS:
44 value = candidate.get(field)
45 if not value:
46 continue
47 values = value if isinstance(value, list) else [value]
48 for index, text in enumerate(values):
49 if not isinstance(text, str) or not text.strip():
50 raise ValueError("exposition text must be nonempty")
51 items.append({"id": field + ":" + str(index + 1), "kind": field,
52 "text": text, "declaration_refs": list(candidate.get("declaration_names", []))})
53 return items

56def create_candidate(output, candidate, *, source_ref, actor_ref):
57 """Write one checked candidate into the caller's own copy; import_candidate owns the checks."""
58 for family in FAMILIES:
59 output.setdefault(family, {})
60 kid = candidate.get("id") or new_id("KN")
61 fid = candidate.get("formal_result_ref") or (new_id("FR") if candidate.get("lean_code") else None)
62 xid = new_id("XP")
63 if fid and not candidate.get("formal_result_ref"):
64 output["formal_results"][fid] = {
65 "id": fid, "family": "formal_results", "lean_code": candidate["lean_code"],
66 "declaration_names": list(candidate["declaration_names"]),
67 "dependencies": list(candidate.get("dependencies", [])),
68 "publication_state": "candidate", "formal_use": "suspended",
69 "verification": {}, "producer_execution": actor_ref,
70 "origin_refs": [source_ref],
71 }
72 output["expositions"][xid] = {
73 "id": xid, "family": "expositions", "formal_result_id": fid,
74 "items": exposition_items(candidate), "lifecycle": "active",
75 "producer_execution": actor_ref, "origin_refs": [source_ref],
76 }
77 if candidate.get("relation_claim"):
78 output["expositions"][xid]["relation_claim"] = deepcopy(candidate["relation_claim"])
79 excluded = {"id", "lean_code", "declaration_names", "formal_result_ref", *EXPOSITION_FIELDS}
80 record = {key: deepcopy(value) for key, value in candidate.items() if key not in excluded}
81 record.update(id=kid, family="knowledge", lifecycle="active", formal_result_id=fid,
82 exposition_id=xid, producer_execution=actor_ref,
83 origin_refs=list(dict.fromkeys([*candidate.get("origin_refs", []), source_ref])))
84 output["knowledge"][kid] = record
85 return output, kid

88def proposal_view(instance, knowledge_id):
89 record = deepcopy(instance["knowledge"][knowledge_id])
90 exposition = instance["expositions"][record["exposition_id"]]
91 formal = instance.get("formal_results", {}).get(record.get("formal_result_id"), {})
92 for field in EXPOSITION_FIELDS:
93 values = [item["text"] for item in exposition["items"] if item["kind"] == field]
94 record[field] = values if field in {"definitions", "assumptions"} else "\n".join(values)
95 record.update(lean_code=formal.get("lean_code", ""), declaration_names=formal.get("declaration_names", []),
96 formal=deepcopy(formal.get("verification", {})))
97 if formal.get("trust_environment_id"):
98 record["formal"]["trust_environment_id"] = formal["trust_environment_id"]
99 record["status"] = "verified" if usable_links(instance, knowledge_id) else "candidate"
100 return record

151 """K premises of one knowledge record: shared premises, usable dependencies, current relation endpoints.
152
153 visiting holds the records whose premises are being judged; a dependency
154 among them is a cycle and is not usable."""
155 if next(premise_failures(instance, record), None) is not None:
156 return False
157 inner = {*visiting, record["id"]}
158 if any(not usable_links(instance, dependency, inner) for dependency in record.get("dependencies", [])):

166def usable_links(instance, knowledge_id, _visiting=None):
167 record = instance.get("knowledge", {}).get(knowledge_id, {})
168 if knowledge_id in (_visiting or ()) or not record_premises_current(instance, record, _visiting or ()):
169 return []
170 fid = record.get("formal_result_id")
171 formal = instance.get("formal_results", {}).get(fid, {})
172 if not formal_usable(instance, formal):
173 return []
174 links = []
175 for assessment in instance.get("identity_assessments", {}).values():
176 if (assessment.get("formal_result_id") != fid or assessment.get("decision") != "match"
177 or assessment.get("exposition_id") != record.get("exposition_id")):
178 continue
179 exposition = instance.get("expositions", {}).get(assessment.get("exposition_id"), {})
180 if (exposition.get("lifecycle") != "active" or exposition.get("formal_result_id") != fid
181 or assessment.get("basis") != _assessment_basis(formal, exposition)):
182 continue
183 uses = [use for use in instance.get("assessment_uses", {}).values()
184 if use.get("assessment_id") == assessment["id"] and use.get("status") == "active"]
185 if uses:
186 links.append({"formal_result_id": fid, "exposition_id": exposition["id"],
187 "assessment_id": assessment["id"], "assessment_use_id": uses[0]["id"]})
188 return sorted(links, key=lambda link: link["assessment_id"])

200def validate_formal_evidence(formal, verification, env):
201 if verification.get("toolchain") != env.get("toolchain"):
202 raise ValueError("verification toolchain differs from approved environment")
203 if verification.get("source_sha256") != lean_source_sha256(formal["lean_code"]):
204 raise ValueError("verification source differs from formal result")
205 if verification.get("declarations") != formal["declaration_names"]:
206 raise ValueError("verification declarations differ from formal result")
207 for field in FORMAL_EVIDENCE_FIELDS:
208 if set(verification.get(field, {})) != set(formal["declaration_names"]):
209 raise ValueError("verification evidence incomplete: " + field)
210 for declaration in formal["declaration_names"]:
211 if set(verification["dependency_evidence"][declaration]) != set(verification["dependency_closure"][declaration]):
212 raise ValueError("dependency body evidence differs from the closure")
213 if verification.get("status") == "verified" and forbidden_axioms(verification["axioms"], env.get("allowed_axioms", [])):
214 raise ValueError("verification uses unapproved axioms")

241def seal_identity_assessment(output, knowledge_id, assessment, policy_ref, *, runtime):
242 """Seal into caller-owned state in place; the caller discards it if this raises."""
243 if not isinstance(policy_ref, str) or not policy_ref.strip():
244 raise ValueError("identity assessment requires a fixed policy reference")
245 for family in FAMILIES:
246 output.setdefault(family, {})
247 record = output["knowledge"][knowledge_id]
248 fid, xid = record["formal_result_id"], record["exposition_id"]
249 formal, exposition = output["formal_results"][fid], output["expositions"][xid]
250 decision = assessment.get("decision")
251 if decision == "match" and any(t.get("status") in {"checking", "ready"} for t in runtime.get("trust_transitions", {}).values()):
252 raise ValueError("identity publication waits for the trust transition")
253 if decision not in {"match", "mismatch", "unresolved"}:
254 raise ValueError("invalid identity decision")
255 if decision == "match" and not formal_usable(output, formal):
256 raise ValueError("match cannot activate without current formal evidence")
257 expected_basis = _assessment_basis(formal, exposition)
258 if assessment.get("basis") != expected_basis:
259 raise ValueError("identity assessment must name the exact current formal and exposition basis")
260 aid = assessment.get("id") or new_id("IA")
261 sealed = {key: deepcopy(value) for key, value in assessment.items() if key not in {"effective", "status"}}
262 sealed.update(id=aid, family="identity_assessments", formal_result_id=fid,
263 exposition_id=xid, policy_ref=policy_ref, basis=expected_basis)
264 if exposition.get("relation_claim"):
265 sealed["relation_claim"] = deepcopy(exposition["relation_claim"])
266 if aid in all_records(output):
267 if output["identity_assessments"].get(aid) == sealed:
268 return output, aid
269 raise ValueError("immutable assessment ID already exists")
270 output["identity_assessments"][aid] = sealed
271 uid = new_id("AU")
272 output["assessment_uses"][uid] = {"id": uid, "family": "assessment_uses", "assessment_id": aid,
273 "status": "active", "history": []}
274 return output, aid

302def set_assessment_use(instance, assessment_id, status, reason, actor_ref):
303 """State primitive; production restoration is authorized by the reassessment consumer."""
304 if status not in {"active", "suspended", "withdrawn"} or not reason or not actor_ref:
305 raise ValueError("assessment use requires status, reason and actor")
306 output = deepcopy(instance)
307 assessment = output["identity_assessments"][assessment_id]
308 if status == "active":
309 if not assessment_current(output, assessment_id):
310 raise ValueError("cannot restore an obsolete assessment")
311 uses = [use for use in output["assessment_uses"].values() if use["assessment_id"] == assessment_id]
312 if len(uses) != 1:
313 raise ValueError("assessment must have exactly one use record")
314 use = uses[0]
315 # A later semantic disposition cannot be discharged by environment repair.
316 use.pop("environment_revalidation", None)
317 if use["status"] == "withdrawn" and status != "withdrawn":
318 raise ValueError("withdrawn assessment use cannot be restored")
319 if use["status"] != status:
320 use["history"].append({"from": use["status"], "to": status, "reason": reason, "actor_ref": actor_ref})
321 use["status"] = status
322 return output

325def validate_knowledge_records(instance):
326 """The knowledge-family rules no record schema or reference table states.
327
328 Field shapes, states and record references are the storage contract's
329 (record_schemas); what remains is uniqueness across records.
330 """
331 for exposition in instance.get("expositions", {}).values():
332 items = exposition.get("items", [])
333 if len({item["id"] for item in items}) != len(items):
334 raise ValueError("duplicate exposition item ID")
335 seen = set()
336 for use in instance.get("assessment_uses", {}).values():
337 if use["assessment_id"] in seen:
338 raise ValueError("duplicate assessment use")
339 seen.add(use["assessment_id"])
