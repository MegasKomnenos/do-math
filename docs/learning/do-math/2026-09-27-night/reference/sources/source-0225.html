29def archived_role_input(bundle, snapshot, files):
30 """The manifest of a maintenance role input archived once below ``runtime/artifacts/<bundle>``.
31
32 Each row names its archived bytes (prompt at ``input.txt``); ``files``
33 receives those bytes.
34 """
35 prefix = 'runtime/artifacts/' + bundle + '/'
36
37 # A document row carries its own ``path`` field, so the archive location has another name.
38 def row(archived, raw, **fields):
39 files[prefix + archived] = raw
40 return {**fields, **artifact(prefix + archived, raw)}
41 return {'role': 'maintenance', 'snapshot': snapshot,
42 'output_contract': {'id': 'do-math-cli/5'}, 'prompt': row('input.txt', b'exact prompt'),
43 'direct': [row('input/catalog/' + key + '.json', b'{}\n', key=key)
44 for key in ('snapshot', 'governance', 'execution_profile', 'action')],
45 'documents': [row('input/implementation/' + path, path.encode() + b'\n', path=path)
46 for path in ('SKILL.md', 'GLOSSARY.md')],
47 'references': []}

324 def test_engine_entry_checks_presence_and_length_while_check_compares_bytes(self):

326 output_path = 'runtime/artifacts/BA-output/result.bin'
327 input_path = 'runtime/artifacts/BA-input/input.txt'
