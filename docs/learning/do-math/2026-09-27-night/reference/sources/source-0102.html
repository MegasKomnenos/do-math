14AXIOMS = ('propext', 'Classical.choice', 'Quot.sound')

20# The runtime binding: the accepted baseline of the record and its digests.

22BINDING_SCHEMA = {'type': 'object', 'additionalProperties': False, 'required': list(BINDING_FIELDS),
23 'properties': {'baseline': {'type': 'integer', 'minimum': 0},
24 **{key: _HASH for key in BINDING_FIELDS[1:]}}}
25# The stored environment record: its state, the identity copied from one
26# toolchain record and that record's file. The accepted baseline is the

29ENVIRONMENT_STATUSES = frozenset({'registered', 'active', 'superseded', 'invalid'})
30ENVIRONMENT_PROPERTIES = {
31 'status': {'enum': sorted(ENVIRONMENT_STATUSES)}, 'toolchain': {'type': 'object'},
32 'allowed_axioms': {'type': 'array', 'items': {'type': 'string'}}, 'source_ref': {'type': 'string'},
33 'toolchain_record': {'type': 'object', 'additionalProperties': False, 'required': ['path', 'id'],

37ENVIRONMENT_REQUIRED = ('status', 'toolchain', 'allowed_axioms', 'source_ref', 'toolchain_record')

43def binding_errors(binding):
44 if (not isinstance(binding, dict) or set(binding) != set(BINDING_FIELDS)

61def environment_errors(environment, binding):
62 """Lexical contract of a stored environment and its runtime binding; no host IO."""
63 from .trust_inventory import identity_errors
64 if not isinstance(environment, dict):
65 return ['trust environment absent']
66 errors = []
67 if not _ENVIRONMENT_FIELDS <= set(environment) or set(environment) - _ENVIRONMENT_FIELDS - _OPTIONAL_FIELDS:
68 errors.append('trust environment fields differ from the current contract')
69 errors.extend(identity_errors(environment.get('toolchain')))
70 if not environment.get('source_ref'):
71 errors.append('trust environment has no explicit adoption')
72 axioms = environment.get('allowed_axioms')
73 if (not isinstance(axioms, list) or any(not isinstance(a, str) for a in axioms)
74 or len(axioms) != len(_AXIOM_SET) or set(axioms) != _AXIOM_SET):
75 errors.append('foundation axiom profile differs')
76 errors.extend(record_reference_errors(environment.get('toolchain_record')))
77 errors.extend(binding_errors(binding))
78 return errors

117def check_environment(environment, binding, *, identity=True):
118 """Live admission: record contract, identity and the olean fingerprint.
119
120 A different fingerprint is accepted only when it equals the record's current
121 baseline, the bound baseline is older and the identity still matches. The
122 caller then rebinds; this function never writes a baseline.
123 """
124 from .trust_inventory import read_record, usable_errors, observe_fingerprint, binding as baseline_binding, changed_files
125 from .formal import toolchain_identity_errors
126 from .repository import exact_content_hash
127 reference = environment['toolchain_record']
128 result = {'environment_id': environment.get('id'), 'toolchain_record': deepcopy(reference),
129 'record_state': record_state(environment), 'checked_at': now(),
130 'identity_sha256': exact_content_hash(environment['toolchain']), 'binding': deepcopy(binding)}
131 try:
132 record = read_record(reference['path'])
133 except (OSError, ValueError) as error:
134 return {**result, 'status': 'invalid', 'errors': ['toolchain record unavailable: ' + str(error)]}
135 errors = []
136 if record['id'] != reference['id']:
137 errors.append('toolchain record ID differs from the registered record')
138 if record['identity'] != environment['toolchain']:
139 errors.append('toolchain record identity differs from the registered environment')
140 errors.extend(usable_errors(record))
141 if not errors and identity:
142 errors.extend(toolchain_identity_errors(environment['toolchain']))
143 if errors:
144 return {**result, 'status': 'invalid', 'errors': errors}
145 try:
146 fingerprint = observe_fingerprint(record['identity'])
147 except OSError as error:
148 return {**result, 'status': 'invalid', 'errors': ['judgment files unavailable: ' + str(error)]}
149 result['fingerprint_sha256'] = fingerprint['sha256']
150 if fingerprint['sha256'] == binding['fingerprint_sha256']:
151 return {**result, 'status': 'verified', 'errors': []}
152 current = baseline_binding(record)
153 if fingerprint['sha256'] == current['fingerprint_sha256'] and current != binding:
154 if not identity:
155 errors.extend(toolchain_identity_errors(environment['toolchain']))
156 if not errors:
157 entry = record['baselines'][current['baseline']]
158 return {**result, 'status': 'verified', 'errors': [], 'binding': current,
159 'rebind': {'previous': deepcopy(binding), 'current': current,
160 'recorded_at': entry['recorded_at']}}
161 return {**result, 'status': 'invalid', 'errors': errors}
162 bound = record['baselines'][binding['baseline']] if binding['baseline'] < len(record['baselines']) else None
163 expected = (bound['fingerprint']['files'] if bound and bound['fingerprint']['sha256'] == binding['fingerprint_sha256']
164 else record['baselines'][-1]['fingerprint']['files'])
165 changed = changed_files(expected, fingerprint['files'])
166 return {**result, 'status': 'invalid', 'changed_files': changed[:50], 'changed_file_count': len(changed),
167 'errors': ['olean fingerprint differs from the bound toolchain baseline; '
168 'setup_lean.py --rerecord (user or operator) or a new record and transition is required']}

171def verify_formal(*, environment, binding, verifier, **arguments):
172 """Check the record and fingerprint before compiling; nothing is re-observed afterwards."""
173 check = check_environment(environment, binding, identity=False)
174 if check['status'] != 'verified':
175 return {'status': 'failed', 'reason': 'environment admission failed: ' + '; '.join(check['errors']),
176 'error_kind': 'trust_environment', 'environment_check': check, 'launch_not_created': True}
177 result = verifier(**arguments)
178 result['environment_check'] = check
179 return result
