38def allowed_environment(source: Mapping[str, str] | None, extra: frozenset[str]) -> dict[str, str]:
39 """Keep the allowed names, and a provider's ``extra`` names, of ``source`` (default: this process)."""
40 allowed = ENV_ALLOWLIST | extra
41 return {key: value for key, value in (os.environ if source is None else source).items()
42 if key.upper() in allowed}

45def validate_execution_binding(request):
46 fields = {"execution_id", "profile_ref", "control_binding", "resource_limits"}
47 if not fields.intersection(request):
48 return
49 ref = request.get("profile_ref")
50 if (not isinstance(request.get("execution_id"), str) or not request["execution_id"].strip()
51 or not isinstance(ref, dict) or set(ref) != {"id", "revision", "sha256"}
52 or not isinstance(ref["id"], str) or not ref["id"]
53 or type(ref["revision"]) is not int or ref["revision"] < 1
54 or not is_sha256(ref["sha256"])):
55 raise ValueError("worker execution binding requires exact execution_id and profile_ref")
56 if "resource_limits" in request:
57 try:
58 from do_math.windows_job import validate_resource_limits

69def validate_max_runtime(timeout: Any) -> None:
70 """A CLI runtime limit is a positive finite number of seconds, or none."""
71 if timeout is not None and (isinstance(timeout, bool) or not isinstance(timeout, (int, float))

73 raise ValueError("max_runtime_seconds must be positive and finite or null")

76def check_common_request(value: Any, *, required: frozenset, optional: frozenset) -> dict[str, Any]:
77 """Check the request fields both workers share, then the provider's extra names."""
78 if not isinstance(value, dict):
79 raise ValueError("worker request must be an object")
80 required, optional = COMMON_REQUIRED | required, COMMON_OPTIONAL | optional
81 missing = sorted(required - set(value))
82 if missing:
83 raise ValueError("worker request misses " + ", ".join(missing))
84 unknown = sorted(set(value) - required - optional)
85 if unknown:
86 raise ValueError("unknown worker request fields: " + ", ".join(unknown))
87 validate_execution_binding(value)
88 for key in (required - {"max_runtime_seconds"}) | ({"output_schema_path"} & set(value)):
89 if not isinstance(value[key], str) or not value[key].strip() or "\0" in value[key]:
90 raise ValueError(key + " must be a nonempty NUL-free string")
91 validate_max_runtime(value["max_runtime_seconds"])
92 if value.get("scholarly_server") is not None:
93 value["scholarly_server"] = validate_scholarly_server(value["scholarly_server"])
94 return value

97def run_cli(request_path: Path, request: Mapping[str, Any], command: list[str],
98 environment: dict[str, str]) -> int:
99 """Run one CLI with its stdin prompt and record the status the collector checks."""
100 prompt = Path(request["prompt_path"]).read_bytes()
101 schema = Path(request["output_schema_path"]).read_bytes() if "output_schema_path" in request else None
102 stdout_path, stderr_path = Path(request["stdout_path"]), Path(request["stderr_path"])
103 for path in (stdout_path, stderr_path):
104 path.parent.mkdir(parents=True, exist_ok=True)
105 status: dict[str, Any] = {"argv_sha256": hashlib.sha256("\0".join(command).encode("utf-8")).hexdigest(),
106 "state": "starting", "cli_exit_code": None,
107 "request_sha256": hashlib.sha256(request_path.read_bytes()).hexdigest(),
108 "stdin_sha256": hashlib.sha256(prompt).hexdigest(),
109 "schema_sha256": None if schema is None else hashlib.sha256(schema).hexdigest()}
110 status.update({key: request[key] for key in STATUS_BINDING_KEYS if key in request})
111 temporary = Path(request["cwd"]) / ".scratch" / ".execution-temp"
112 temporary.mkdir(parents=True, exist_ok=True)
113 environment = {**environment, "TEMP": str(temporary.resolve()), "TMP": str(temporary.resolve())}
114 try:
115 with stdout_path.open("wb") as stdout, stderr_path.open("wb") as stderr:
116 process = subprocess.Popen(command, stdin=subprocess.PIPE, stdout=stdout, stderr=stderr,
117 cwd=request["cwd"], env=environment, shell=False,
118 creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0))
119 try:
120 process.communicate(input=prompt, timeout=request["max_runtime_seconds"])
121 status.update(state="exited", cli_exit_code=process.returncode)
122 except subprocess.TimeoutExpired:
123 process.kill()
124 process.communicate()
125 status.update(state="timeout", cli_exit_code=process.returncode, runner_stop_required=True)
126 except Exception as exc:
127 status.update(state="worker_error", error_type=type(exc).__name__, error=str(exc))
128 write_bytes_atomic(Path(request["status_path"]), compact_json_bytes(status))
129 return 0
