22@unittest.skipUnless(os.name == 'nt', 'requires Windows Job host')
23class IsolatedHostTests(unittest.TestCase):
24 def until(self, predicate, timeout=10):
25 deadline = time.monotonic() + timeout

28 time.sleep(.02)

31 def test_resume_failure_never_runs_the_suspended_child(self):
32 with tempfile.TemporaryDirectory() as directory:
33 marker = Path(directory) / 'substantive-action'
34 code = f'from pathlib import Path; Path({str(marker)!r}).write_text("ran")'
35 with jobs.WindowsJob([sys.executable, '-c', code]) as job:
36 with patch.object(jobs._kernel32, 'ResumeThread', return_value=jobs._WAIT_FAILED):
37 with self.assertRaises(OSError): job.resume()
38 self.assertFalse(marker.exists())
39 job.terminate()
40 self.assertTrue(job.wait(5))
41 self.until(lambda: job.inspect().active_processes == 0)

43 def test_owned_process_identity_does_not_follow_a_changed_pid_field(self):
44 with jobs.WindowsJob([sys.executable, '-c', 'import time; time.sleep(30)']) as first:
45 with jobs.WindowsJob([sys.executable, '-c', 'import time; time.sleep(30)']) as other:
46 original_pid, original_time = first.pid, first.creation_filetime
47 self.assertGreater(original_time, 0)
48 first.pid = other.pid # synthetic recycled locator, not OS PID reuse
49 self.assertEqual(first._process_creation_filetime(), original_time)
50 first.terminate()
51 self.assertTrue(first.wait(5))
52 self.assertFalse(other.inspect().root_signaled)
53 self.assertNotEqual(original_pid, other.pid)

55 def test_nested_job_child_remains_owned_by_outer_job(self):
56 with tempfile.TemporaryDirectory() as directory:
57 marker = Path(directory) / 'inner-running'
58 nested = ('import sys,time; from pathlib import Path; '
59 f'sys.path.insert(0,{str(Path(__file__).resolve().parents[1])!r}); '
60 'from do_math.windows_job import WindowsJob; '

62 f'j.resume(); Path({str(marker)!r}).write_text(str(j.pid)); time.sleep(30)')
63 with jobs.WindowsJob([sys.executable, '-c', nested]) as outer:
