19def repository_double(root, instance, runtime, *, published):
20 """A ``ResearchRepository`` double (autospec) holding ``(instance, runtime)`` at ``head``.
21
22 Saving publishes ``published``. The recovery path takes its maintenance
23 lock in the repository's public ``git_dir``, which names ``root``.
24 """
25 from contextlib import nullcontext
26 from unittest.mock import create_autospec
27 from do_math.repository import ResearchRepository
28 repo = create_autospec(ResearchRepository, instance=True)
29 repo.writer_lock.return_value = nullcontext()
30 repo.load.return_value = ("head", instance, runtime)
31 repo.save_with_state.return_value = (published, {}, {})
32 repo.git_dir = root
33 return repo
