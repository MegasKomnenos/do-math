25 def assign_sequences_in_place(runtime, *, published_events=None):
26 """Validate event history and assign sequence fields on the supplied runtime.
27
28 ``published_events`` maps IDs to the parent's rows. The parent was read
29 or published under the same contract, so its numbering already follows
30 :func:`check_sequence` and is not checked again here. An error can
31 leave this runtime partially numbered; callers apply this to their own
32 candidate copy.
33 """
34 previous = published_events or {}
35 if "events" not in runtime and not previous and not runtime.get("pending_events"):
36 return
37 events = runtime.setdefault("events", [])
38 rows = event_rows(events)
39 pending = runtime.get("pending_events", [])
40 if not isinstance(pending, list):
41 raise ValueError("pending_events must be a list")
42 # This is the planner's unconsumed-event queue, not unpublished objects.
43 if any(not isinstance(ref, str) for ref in pending) or len(set(pending)) != len(pending):
44 raise ValueError("pending_events requires distinct event references")
45 ids = [row.get("id") for row in rows]
46 if any(not isinstance(i, str) or not i for i in ids) or len(set(ids)) != len(ids):
47 raise ValueError("event IDs must be distinct")
48 if set(previous) - set(ids):
49 raise ValueError("published events cannot be removed")
50 if set(pending) - set(ids):
51 raise ValueError("pending_events references unknown event")
52 next_sequence = len(previous)
53 for row in rows:
54 identifier = row["id"]
55 if "event_seq" in row and (type(row["event_seq"]) is not int or row["event_seq"] < 1):
56 raise ValueError("invalid event sequence")
57 if identifier in previous:
58 old = previous[identifier]
59 if {k: v for k, v in row.items() if k != "event_seq"} != {k: v for k, v in old.items() if k != "event_seq"}:
60 raise ValueError("published event content cannot be rewritten")
61 if row.get("event_seq", old["event_seq"]) != old["event_seq"]:
62 raise ValueError("published event sequence cannot be rewritten")
63 row["event_seq"] = old["event_seq"]
64 else:
65 next_sequence += 1
66 if row.get("event_seq", next_sequence) != next_sequence:
67 raise ValueError("new event sequence must be contiguous")
68 row["event_seq"] = next_sequence
69 # Published rows keep their numbers and new rows follow them, so the
70 # result is 1..n in list order exactly when no published row moved.
71 check_sequence(rows)
72 runtime["event_cursor"] = next_sequence
