129 def setUp(self):

137 @contextmanager
138 def running(self):
139 """One started operational-role execution whose calculation only reads its flag."""
140 from do_math.windows_job import WindowsJob, process_identity
141 e = self.engine
142 e._calls = ModelCalls(on_returned=e._local_call_returned, on_observation_error=e._observation_failed)
143 steps = e._call_steps(e.runtime['queue'][0])
144 arguments = next(steps)
145 job = WindowsJob([sys.executable, '-c', 'import time; time.sleep(600)'],
146 cwd=arguments['workspace'], name=arguments['launch_ticket']['job_name'])
147 release, reads = threading.Event(), []
148 def calculate(should_stop, **_):
149 while not release.is_set():
150 reads.append(should_stop())
151 if reads[-1]:
152 return {'status': 'stopped', 'output': None, 'usage': {}, 'error': 'flag observed'}
153 time.sleep(.01)
154 return {'status': 'stopped', 'output': None, 'usage': {}, 'error': 'released'}
155 try:
156 arguments['on_started']({**process_identity(job), 'job_name': job.name})
157 job.resume()
158 future = e._calls.start_call(steps, calculate, arguments, execution_id=arguments['execution_id'])
159 self.assertFalse(future.stop.is_set(), 'nothing requires a stop at start')
160 yield e, arguments['execution_id'], future, reads
161 finally:
162 release.set()
163 e._calls.close()
164 steps.close()
165 e._calls = None
166 job.terminate()
167 self.assertTrue(job.wait(10))
168 job.close()
169
170 def assertStopped(self, engine, execution_id, future, reads, reason=None):
171 self.assertTrue(future.stop.is_set())
172 self.assertIsNone(getattr(future, 'judge_error', None))
173 execution = engine.runtime['executions'][execution_id]
174 self.assertEqual(execution['status'], 'stopping')
175 if reason is not None:
176 self.assertEqual(execution['stop_reason'], reason)
177 until = time.monotonic() + 5
178 while (not reads or not reads[-1]) and time.monotonic() < until:
179 time.sleep(.01)
180 self.assertTrue(reads and reads[-1], 'the calculation observes the flag without an owner round trip')

239 workspace = Path(e.instance['workspace']) / 'runs' / eid / 'workspace'
240 def at(instant):
241 # Only the executor's deadline clock moves; the clock guard keeps real time.
242 import time as real
243 clock = SimpleNamespace(**{name: getattr(real, name) for name in dir(real) if not name.startswith('_')})
244 clock.monotonic = lambda: instant
245 return patch('do_math.engine.time', clock), patch('do_math.scheduler.time', clock)
246 e._settle_stops()
247 first, second = at(notice + .001)
248 with first, second:
249 e._settle_stops()
250 self.assertTrue((workspace / '.control' / 'finish-request.json').exists())
251 self.assertFalse(future.stop.is_set(), 'a finish notice does not revoke authority')
