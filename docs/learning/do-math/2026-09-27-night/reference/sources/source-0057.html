116def impact_review_response_schema(request):
117 """A producer can reject or leave unresolved every explicitly assigned scope."""
118 alternatives = []
119 for scope, plan in request["plans"].items():
120 alternatives.append(closed_object({"scope": {"const": scope}, "plan_digest": {"const": plan["digest"]},
121 "decision": {"enum": ["approve", "reject", "unresolved"]}, "affected_ids": REFS, "reason": TEXT}))
122 if not alternatives:
123 raise ValueError("impact review must name an actual plan")
124 schema = closed_object({**{key: {"const": request[key]} for key in
125 ("integration_id", "payload_sha256", "producer_execution_id", "input_snapshot")},
126 "plan_decisions": {"type": "array", "minItems": len(alternatives), "maxItems": len(alternatives),
127 "items": {"oneOf": alternatives}}})
128 schema["description"] = (
129 "Independently assess the held mutation using the original candidate and the supplied premise records. "
130 "For every assigned scope, explain discrepancies between declared and computed dependencies and judge "
131 "whether suspending/reassessing the computed-and-declared union is justified and sufficient. "
132 "Use unresolved when essential evidence or scope is uncertain, and reject an unjustified disposition. "
133 "Do not approve solely because hashes match. This judgment grants no new mathematical result, "
134 "user authority or external-effect permission; it only resolves this held bundle's impact gate.")
135 return json_copy(schema)

138def validate_impact_review_response(request, output):
139 response = output.get("impact_review")
140 errors = list(jsonschema.Draft202012Validator(impact_review_response_schema(request)).iter_errors(response))
141 if errors:

143 rows = response["plan_decisions"]
144 if {row["scope"] for row in rows} != set(request["plans"]) or len({row["scope"] for row in rows}) != len(rows):
145 raise ValueError("impact review must judge every scope exactly once")
146 for row in rows:
147 plan = request["plans"][row["scope"]]

261def _refs(value):
