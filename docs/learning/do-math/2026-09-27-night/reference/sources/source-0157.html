156 def test_writer_lock_leaves_the_final_cas_as_the_only_ref_comparison(self):
157 first = self.store.publish(None, self.initial_changes())
158 with self.store.writer_lock():
159 with patch.object(self.store, '_git', wraps=self.store._git) as calls:
160 later = self.store.publish(first.head, {'runtime/state.json': b'{"state": "running"}\n'})
161 self.assertEqual([c.args[0] for c in calls.call_args_list], ['update-ref'])
162 # Under the lock no do-math writer moves the ref, so the head this
163 # store just published is not read back.
164 with patch.object(self.store, '_git', wraps=self.store._git) as calls:
165 self.assertEqual(self.store.read_snapshot(), StateSnapshot(later.head))
166 self.assertEqual(calls.call_count, 0)
167 # A writer outside the lock is still refused by that final CAS.
168 outside = GitStateStore(self.repository, INSTANCE_ID)
169 moved = outside.publish(later.head, {'runtime/state.json': b'{"state": "moved"}\n'})
170 with self.assertRaisesRegex(GitStateStoreConflict, 'publish compare-and-swap'):
171 self.store.publish(later.head, {'runtime/state.json': b'{"state": "lost"}\n'})
172 # The failed CAS proves the remembered head stale; it is read again.
173 self.assertEqual(self.store.read_snapshot().head, moved.head)
174 with patch.object(self.store, '_git', wraps=self.store._git) as calls:
175 self.assertEqual(self.store.read_snapshot().head, moved.head)
176 self.assertEqual(calls.call_count, 1)

228 def test_simultaneous_stale_cas_has_one_winner_and_no_overwrite(self) -> None:
229 initial = self.store.publish(None, self.initial_changes())
230 barrier = threading.Barrier(2)
231 results: list[tuple[str, object]] = []
232 results_lock = threading.Lock()
233
234 def publish(label: str) -> None:
235 local = GitStateStore(self.repository, INSTANCE_ID)
236 barrier.wait()
237 try:
238 snapshot = local.publish(initial.head, {"runtime/state.json": label.encode()})
239 result: tuple[str, object] = ("published", snapshot)
240 except GitStateStoreConflict as error:
241 result = ("conflict", error)
242 with results_lock:
243 results.append(result)
244
245 left = threading.Thread(target=publish, args=("left",))
246 right = threading.Thread(target=publish, args=("right",))
247 left.start()
248 right.start()
249 left.join(20)
250 right.join(20)
251 self.assertFalse(left.is_alive())
252 self.assertFalse(right.is_alive())
253 self.assertEqual([kind for kind, _ in results].count("published"), 1)
254 self.assertEqual([kind for kind, _ in results].count("conflict"), 1)
255 self.assertIn(self.store.read_bytes(self.store.read_snapshot().head, "runtime/state.json"), (b"left", b"right"))

413 directories = {path[:end] for path in paths for end in range(len(path)) if path[end] == "/"}
414 return not directories & set(paths)

416 def random_changes(self, rng: random.Random, model: dict[str, bytes]) -> dict[str, bytes | None]:
417 """Adds, edits, deletes (some emptying a directory), directory-path deletes, file<->directory swaps."""
418 proposals: list[dict[str, bytes | None]] = []
419 files = sorted(path for path in model if not path.endswith("/keep"))
420 if files:
421 victim = rng.choice(files)
422 proposals.append({victim: None, victim + "/a": b"file became directory"})
423 directories = sorted({path.rpartition("/")[0] for path in files if path.count("/") >= 2})
424 if directories:
425 replaced = rng.choice(directories)
426 below = {path: None for path in model if path.startswith(replaced + "/")}
427 proposals.append({**below, replaced: b"directory became file"})
428 others = [path for path in directories if path != replaced and not path.startswith(replaced + "/")]
429 if others:
430 proposals.append({rng.choice(others): None}) # names a directory: no file to delete
431 for _ in range(10):
432 roll = rng.random()
433 if files and roll < 0.3:
434 proposals.append({rng.choice(files): None})
435 elif files and roll < 0.5:
436 proposals.append({rng.choice(files): rng.randbytes(rng.randint(0, 24))})
437 else:
438 parts = [rng.choice(("research", "runtime")), *(rng.choice(self.NAMES) for _ in range(rng.randint(1, 3)))]
439 proposals.append({"/".join(parts): rng.randbytes(rng.randint(0, 24))})
440 changes: dict[str, bytes | None] = {}
441 for proposal in proposals:
442 candidate = {**changes, **proposal}
443 paths = {path for path in model if candidate.get(path, model[path]) is not None}
444 paths |= {path for path, raw in candidate.items() if raw is not None}
445 if self.valid(paths):
446 changes = candidate
447 return changes

449 def oracle_tree(self, repository: Path, index: dict[str, str], changes: dict[str, bytes | None], width: int) -> str:
450 """Apply the same changes to a persistent index, deletions first, and let Git write the tree."""
451 with tempfile.TemporaryDirectory() as directory:
452 added = [(path, raw) for path, raw in changes.items() if raw is not None]
453 for number, (_, raw) in enumerate(added):
454 (Path(directory) / str(number)).write_bytes(raw)
455 names = "".join(f"{(Path(directory) / str(number)).as_posix()}\n" for number in range(len(added))).encode()
456 oids = git(repository, "hash-object", "--no-filters", "--stdin-paths", input=names).decode().split() if added else []
457 records = [f"0 {'0' * width}\t{path}" for path, raw in changes.items() if raw is None]
458 records += [f"100644 {oid}\t{path}" for (path, _), oid in zip(added, oids)]
459 git(repository, "update-index", "-z", "--index-info", input="".join(record + "\0" for record in records).encode(), env=index)
460 return git(repository, "write-tree", env=index).decode().strip()
