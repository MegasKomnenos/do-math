40# Immutable artifact rows whose stored bytes this process already matched,
41# keyed by (git directory, artifact path, sha256). Existing artifact bundles
42# never change (git_store bundle invariant), so one match holds for the process.
43# The value is the matched byte length, so a row that declares a size is
44# still compared against the actual length without reading the bytes again.
45VERIFIED_ARTIFACTS: dict[tuple[str, str, str], int] = {}

65def verify_artifact(git_dir, path, raw, sha256, size=None):
66 """Raise unless ``raw`` matches the sealed row; remember the match in ``git_dir``."""
67 if (not isinstance(raw, bytes) or hashlib.sha256(raw).hexdigest() != sha256
68 or (size is not None and len(raw) != size)):
69 raise ValueError("artifact evidence missing or hash differs: " + path)
70 if git_dir is not None:

74def sealed_bytes(git_dir, path, sha256, size, read, *, raw=None, needed=True, error=None):
75 """Stored artifact bytes, matched against their sealed row once per process (이음새-정본.md §3).
76
77 ``read()`` supplies the stored bytes when ``raw`` is not given. ``size`` is
78 the declared length, or None when the row declares none. A row this process
79 already matched is not hashed again; with ``needed=False`` it is not read
80 either, and the result is ``raw`` (possibly None). Without ``git_dir`` the
81 bytes are compared on every call and no match is remembered. A mismatch
82 raises ``ValueError(error)`` when ``error`` is given, else verify_artifact's.
83 """
84 if artifact_verified(git_dir, path, sha256, size):
85 if raw is None and needed:
86 raw = read()
87 return raw
88 raw = read() if raw is None else raw
89 try:
90 verify_artifact(git_dir, path, raw, sha256, size)
91 except ValueError as mismatch:
92 if error is None:
93 raise
94 raise ValueError(error) from mismatch
95 return raw

98# Validated snapshot reads shared by every repository object in this process,
99# keyed by (git directory, commit, contract). A commit's bytes never change
100# and a contract does not change after its first use (state_contract), so an
101# entry stays valid; the bound limits memory. Repositories share the installed
102# contract object and so its entries. Entries come from a full read of a head,
103# or from this process's own publication, whose candidate was validated
104# before the ref moved.
105_VALIDATED_READS: OrderedDict = OrderedDict()
106_VALIDATED_READ_LIMIT = 4

109# Heads kept beyond the limit, per git directory: the input snapshots of live
110# executions, which result acceptance reads again after later publications.
111_RETAINED_READS: dict[str, frozenset] = {}

114def retain_reads(git_dir, heads):

116 _RETAINED_READS[str(git_dir)] = frozenset(heads)

119def _remember_read(key, entry):
120 _VALIDATED_READS[key] = entry
121 _VALIDATED_READS.move_to_end(key)
122 unretained = [k for k in _VALIDATED_READS if k[1] not in _RETAINED_READS.get(k[0], ())]
123 for old in unretained[:max(0, len(unretained) - _VALIDATED_READ_LIMIT)]:
124 del _VALIDATED_READS[old]

127# Artifact facts this process wrote or read, keyed by (git directory, path):
128# [head where the path was known to exist, length, bytes or None]. Existing
129# artifact bundles never change and no publication removes them
130# (git_store._validate_artifact_changes), so a fact known at a head holds at
131# every later head this process published from it (_PUBLISHED_PARENTS).
132# Bytes are held up to _ARTIFACT_BYTES_LIMIT; lengths stay while rows fit.
133_ARTIFACT_FACTS: OrderedDict = OrderedDict()
134_PUBLISHED_PARENTS: OrderedDict = OrderedDict()
135_ARTIFACT_ROW_LIMIT = 4096
136_ARTIFACT_BYTES_LIMIT = 64 << 20
137_LINEAGE_LIMIT = 1024
138_held_bytes = 0

141def _descends(git_dir, snapshot, head):
142 """True when ``snapshot`` is ``head`` or a later head this process published from it."""
143 current = snapshot
144 for _ in range(_LINEAGE_LIMIT):
145 if current == head:
146 return True
147 current = _PUBLISHED_PARENTS.get((git_dir, current))
148 if current is None:
149 return False
150 return False

153def _known_artifact(git_dir, path, snapshot):
154 """The [head, length, bytes] row for ``path`` when it holds at ``snapshot``, else None."""
155 row = _ARTIFACT_FACTS.get((git_dir, path))
156 if row is None or not _descends(git_dir, snapshot, row[0]):
157 return None
158 _ARTIFACT_FACTS.move_to_end((git_dir, path))
159 return row

162def _learn_artifact(git_dir, path, head, size, raw=None):
163 global _held_bytes
164 old = _ARTIFACT_FACTS.pop((git_dir, path), None)
165 if old is not None and old[2] is not None:
166 _held_bytes -= len(old[2])
167 if raw is None and old[1] == size:
168 raw = old[2]
169 _ARTIFACT_FACTS[(git_dir, path)] = [head, size, raw]
170 if raw is not None:
171 _held_bytes += len(raw)
172 for key in list(_ARTIFACT_FACTS):
173 if len(_ARTIFACT_FACTS) <= _ARTIFACT_ROW_LIMIT and _held_bytes <= _ARTIFACT_BYTES_LIMIT:
174 break
175 if len(_ARTIFACT_FACTS) > _ARTIFACT_ROW_LIMIT:
176 dropped = _ARTIFACT_FACTS.pop(key)
177 else:
178 dropped = _ARTIFACT_FACTS[key]
179 _ARTIFACT_FACTS[key] = [dropped[0], dropped[1], None]
180 if dropped[2] is not None:
181 _held_bytes -= len(dropped[2])

184def _learn_publication(git_dir, parent, head, artifacts):
185 if parent is not None and parent != head: # a no-op publication keeps its head
186 _PUBLISHED_PARENTS[(git_dir, head)] = parent
187 while len(_PUBLISHED_PARENTS) > _LINEAGE_LIMIT:
188 _PUBLISHED_PARENTS.popitem(last=False)
189 for path, raw in artifacts.items():
190 _learn_artifact(git_dir, path, head, len(raw), raw)

202def new_id(prefix: str) -> str:
203 """Create a canonical uppercase ULID with the caller's registry prefix."""
204 if not isinstance(prefix, str) or not _PREFIX.fullmatch(prefix):
205 raise ValueError("prefix must be a canonical uppercase registry prefix")
206 value = ((time.time_ns() // 1_000_000) << 80) | int.from_bytes(secrets.token_bytes(10), "big")
207 encoded = ""
208 for _ in range(ULID_LENGTH):
209 encoded = ULID_ALPHABET[value & 31] + encoded
210 value >>= 5
211 return f"{prefix}-{encoded}"

214def canonical_json(value: Any) -> bytes:
215 """Encode a finite JSON value deterministically without changing Unicode."""

266 @measured('snapshot_read_validation')
267 def _read_documents(self, head):
268 """Validate one snapshot and retain its exact non-artifact file bytes.
269
270 The result is the shared cache entry. Callers that return or mutate
271 state copy it; a publication parent is only compared, never copied.
272 A head this process did not publish (startup, another writer after a
273 CAS conflict) is validated in full the first time it is read.
274 """
275 from .operation_trace import phase
276 key = self._read_key(head)
277 cached = _VALIDATED_READS.get(key)
278 if cached is not None:
279 _VALIDATED_READS.move_to_end(key)
280 return cached
281 paths = self._store._tree_paths(head)
282 # Sealed artifacts are not parsed or loaded just to reconstruct state.
283 files = self._store.read_many(head, sorted(p for p in paths if not p.startswith("runtime/artifacts/")))
284 payload_bytes = {}
285 def read_payload(path):
286 if path not in payload_bytes:
287 payload_bytes[path] = self._store.read_bytes(head, path)
288 return payload_bytes[path]
289 read_payload.decoded = {}
290 with phase('snapshot_decode', snapshot=head):
291 instance, runtime, _ = _decode_families(files, self.contract, self._store.object_format,
292 read_payload=read_payload, git_dir=self.git_dir)
293 self.contract.validate(instance, runtime)
294 # Stored bytes are outside the trust boundary; the published numbering is checked on read.
295 check_sequence(event_rows(runtime.get("events", [])), runtime.get("event_cursor"))
296 self._validate_artifact_evidence(runtime, head)
297 result = (instance, runtime, None, {**files, **payload_bytes})
298 _remember_read(key, result)
299 return result

341 @measured('candidate_validation')
342 def validate_candidate(self, instance, runtime, expected_head=None, *, artifacts=None):
343 """Pure candidate check, including parent event facts when supplied.
344
345 The returned views are detached. Callers reload after publication to
346 observe assigned events. No rejected/CAS-losing proposal mutates them.
347 :meth:`save_validated` publishes the result without validating it again.
348 """
349 if not isinstance(instance, dict) or not isinstance(runtime, dict):
350 raise ValueError("instance and runtime must be objects")
351 instance = deepcopy(instance)
352 runtime = deepcopy(runtime)
353 instance, runtime = seal_large(instance), seal_large(runtime)
354 if instance.get("object_format", self._store.object_format) != self._store.object_format:
355 raise ValueError("persisted object_format differs from Git")
356 instance["object_format"] = self._store.object_format
357 parent = None
358 parent_instance, parent_runtime = {}, {}
359 if expected_head is not None:
360 parent_instance, parent_runtime, _, parent_files = self._read_documents(expected_head)
361 parent = (parent_instance, parent_runtime, parent_files)
362 previous = {row["id"]: row for row in event_rows(parent_runtime.get("events", []))}
363 EventPublisher.assign_sequences_in_place(runtime, published_events=previous)
364 # Immutable judgments/profile editions cannot be rewritten under one ID.
365 for family in ("identity_assessments", "execution_profiles", "governance_sources"):
366 for identifier, row in parent_instance.get(family, {}).items():
367 if instance.get(family, {}).get(identifier) != row:
368 raise ValueError("immutable record changed or removed: " + identifier)
369 if parent_instance.get("lifecycle", {}).get("immutable_input") is not None:
370 if instance.get("lifecycle", {}).get("immutable_input") != parent_instance["lifecycle"]["immutable_input"]:
371 raise ValueError("immutable original input changed")
372 for identifier, row in parent_instance.get("formal_results", {}).items():
373 if row.get("verification"):
374 current = instance.get("formal_results", {}).get(identifier)
375 if current is None or any(current.get(k) != v for k, v in row.items() if k not in {"formal_use", "history", "use_history"}):
376 raise ValueError("sealed formal result changed: " + identifier)
377 # The parent was validated under this contract; records it already
378 # holds unchanged skip record-local checks (StateContract.validate).
379 self.contract.validate(instance, runtime, parent=parent[:2] if parent else None)
380 # Encoding checks the storage value domain of each record it
381 # serializes; a record equal to the parent's reuses the parent's bytes.
382 files = _encode_families(instance, runtime, self.contract, self._store.object_format, parent=parent)
383 supplied = self._snapshot_artifacts(artifacts) or {}
384 for path in files.keys() & supplied.keys():
385 if files[path] != supplied[path]:
386 raise ValueError('immutable artifact differs from encoded state: ' + path)
387 # Evidence validation and publication must see the same merged bytes.
388 self._validate_artifact_evidence(runtime, expected_head, {**files, **supplied})
389 self._store._validate_candidate_paths(set(files) | set(supplied))
390 return {"instance": instance, "runtime": runtime, "files": files, "artifacts": supplied,
391 "basis": {"head": expected_head}}

427 def _publish_candidate(self, expected_head, candidate):
428 old_files = self._read_documents(expected_head)[3]
429 files = candidate["files"]
430 # Unreferenced payloads remain immutable Git blobs. They are reusable
431 # even when the parent decoder did not need to read them into old_files.
432 missing = {p for p in files if p.startswith('runtime/artifacts/') and p not in old_files}
433 if missing:
434 existing = self._store.read_many(expected_head, sorted(missing & self._store._tree_paths(expected_head)))
435 for path, raw in existing.items():
436 if files[path] != raw:
437 raise ValueError('immutable artifact differs from encoded state: ' + path)
438 old_files = {**old_files, **existing}
439 changes = {p: None for p in old_files if p not in files and not p.startswith('runtime/artifacts/')}
440 changes.update({p: raw for p, raw in files.items() if old_files.get(p) != raw})
441 changes.update(candidate["artifacts"])
442 published = self._store.publish(expected_head, changes)
443 self._remember_published(published.head, candidate)
444 _learn_publication(str(self.git_dir), expected_head, published.head, candidate["artifacts"])
445 # The remembered candidate is the shared entry; callers mutate their views.
446 return published.head, deepcopy(candidate["instance"]), deepcopy(candidate["runtime"])

478 def _validate_artifact_evidence(self, runtime, head, artifacts=None):
479 """Resolve immutable byte evidence separately from record identities.
480
481 Existing artifact bundles are immutable, so a row this process already
482 matched (VERIFIED_ARTIFACTS) is not read again; only new rows and rows
483 whose bytes are supplied with this candidate are compared.
484 """
485 references = self._artifact_references(runtime)
486 if not references:
487 return

489 for ref, _, _ in references:
490 if not ref.startswith("runtime/artifacts/"):
491 raise ValueError("artifact evidence must use the immutable artifact namespace")
492 self._store._validate_path(ref)

494 if row[0] in supplied or not artifact_verified(self.git_dir, *row)]
495 wanted = sorted({ref for ref, _, _ in pending if ref not in supplied})
496 available = self._store._tree_paths(head) if head and wanted else set()
497 stored = self._store.read_many(head, [ref for ref in wanted if ref in available]) if available else {}
498 for ref, digest, size in pending:
499 raw = supplied.get(ref)
500 if raw is None:

572 def query(
573 self,
574 *,
575 snapshot: str | None = None,
576 prefix: str = "research/",
577 text: str = "",
578 limit: int = 50,
579 cursor: str | None = None,
580 ) -> dict[str, Any]:
581 """Return a page bound to one immutable snapshot and textual query.
582
583 Matching includes the POSIX path and UTF-8 query bytes in file contents.
584 A fixed-window-like guarantee is not inferred from the current head: a
585 cursor always carries the exact commit it was created from.
586 """
587 normalized_prefix = _validate_prefix(prefix)
588 if not isinstance(text, str) or not isinstance(limit, int) or isinstance(limit, bool) or limit <= 0:
589 raise ValueError("text must be a string and limit a positive integer")
590 last_path: str | None = None
591 if cursor is not None:
592 payload = _decode_cursor(cursor)
593 cursor_snapshot = payload["snapshot"]
594 if snapshot is not None and snapshot != cursor_snapshot:
595 raise RepositoryCursorError("cursor is bound to another snapshot or query")
596 if payload["prefix"] != normalized_prefix or payload["text"] != text:
597 raise RepositoryCursorError("cursor is bound to another snapshot or query")
598 snapshot = cursor_snapshot
599 last_path = payload["last_path"]
600 head = self._resolve_snapshot(snapshot)
601
602 needle = text.encode("utf-8")
603 matches: list[dict[str, Any]] = []
604 paths = self.list_files(normalized_prefix, head)
605 if text:
606 # Content matching requires scanning the snapshot.
607 for offset in range(0, len(paths), 16):
608 batch = paths[offset:offset + 16]
609 contents = self._store.read_many(head, batch)
610 for path in batch:
611 content = contents[path]
612 if text not in path and needle not in content:
613 continue
614 matches.append({"path": path, "size": len(content), "sha256": hashlib.sha256(content).hexdigest()})
615 else:
616 matches = [{"path": path} for path in paths]
617 total = len(matches)
618 start = 0
619 if last_path is not None:
620 matching_indexes = [index for index, item in enumerate(matches) if item["path"] == last_path]
621 if len(matching_indexes) != 1:
622 raise RepositoryCursorError("cursor path is not present in this query result")
623 start = matching_indexes[0] + 1
624 items = matches[start:start + limit]
625 if not text:
626 hydrated = []
627 for offset in range(0, len(items), 16):
628 batch = [item["path"] for item in items[offset:offset + 16]]
629 contents = self._store.read_many(head, batch)
630 hydrated.extend({"path": path, "size": len(contents[path]),
631 "sha256": hashlib.sha256(contents[path]).hexdigest()} for path in batch)
632 items = hydrated
633 next_cursor = None
634 if start + len(items) < total:
635 next_cursor = _encode_cursor({
636 "snapshot": head,
637 "prefix": normalized_prefix,
638 "text": text,
639 "last_path": items[-1]["path"],
640 })
641 return {
642 "snapshot": head,
643 "total": total,
644 "items": items,
645 "next_cursor": next_cursor,
646 "truncated": next_cursor is not None,
647 }

650 self,
651 destination: Path,
652 snapshot: str | None = None,
653 prefixes: tuple[str, ...] = ("research/",),
654 ) -> dict[str, Any]:
655 """Copy selected snapshot files into a new or empty non-reparse directory."""
656 head = self._resolve_snapshot(snapshot)

668 boundary = PhysicalBoundary(destination)
669
670 files = sorted({path for prefix in normalized_prefixes for path in self.list_files(prefix, head)})
671 sizes = self._store.blob_sizes(head, files)

719def validate_storage_value(value: Any) -> None:
720 """Validate the exact storage value domain without constructing JSON bytes.
721
722 Plain ``dict``, ``list`` and ``str`` values, nearly every node of a state,
723 take a direct path with the same rules; string leaves are checked in
724 place, and an ASCII string holds no surrogate. Other types (subclasses,
725 sealed payloads) fall through to the general rules.
726 """
727 kind = type(value)
728 if kind is dict:
729 for key, item in value.items():
730 if type(key) is not str:
731 if not isinstance(key, str):
732 raise ValueError("JSON object keys must be strings")
733 validate_storage_value(key)
734 elif not key.isascii() and _SURROGATE.search(key):
735 raise ValueError("JSON strings cannot contain surrogate code points")
736 if type(item) is str:
737 if not item.isascii() and _SURROGATE.search(item):
738 raise ValueError("JSON strings cannot contain surrogate code points")
739 else:
740 validate_storage_value(item)
741 return
742 if kind is list:
743 for item in value:
744 validate_storage_value(item)
745 return
746 if kind is str:
747 if not value.isascii() and _SURROGATE.search(value):
748 raise ValueError("JSON strings cannot contain surrogate code points")
749 return
750 if isinstance(value, FrozenText) and hasattr(value, '_payload_bytes'):
751 return # Sealing validated exact bytes and these values cannot be edited.
752 if value is None or isinstance(value, (bool, str)):
753 if isinstance(value, str) and _SURROGATE.search(value):
754 raise ValueError("JSON strings cannot contain surrogate code points")
755 return
756 if isinstance(value, int) and not isinstance(value, bool):
757 if abs(value) > _SAFE_INTEGER:
758 raise ValueError("JSON integers must be safe integers")
759 return
760 if isinstance(value, float):
761 raise ValueError("JSON floating-point numbers are not permitted")
762 if isinstance(value, list):
763 for item in value:
764 validate_storage_value(item)
765 return
766 if isinstance(value, dict):
767 for key, item in value.items():
768 if not isinstance(key, str):
769 raise ValueError("JSON object keys must be strings")
770 validate_storage_value(key)
771 validate_storage_value(item)
772 return
773 raise ValueError("value is not JSON-serializable")

828 """

830 installed = contract.manifest()

832 for index, (root, state) in enumerate((("research", instance), ("runtime", runtime))):
833 parent_state = parent[index] if parent is not None else {}

868 # when only runtime families or event membership change.

874def _decode_families(files, contract, object_format, *, read_payload=None, git_dir=None, describe=False):
875 """Decode one snapshot's record files and check their family layout.
876
877 Returns (research, runtime, descriptors). The descriptors, the exact
878 persisted record rows, are built only with ``describe``; otherwise the
879 third value is None.
880 """
881 if read_payload is None:
882 def read_payload(path):
883 raise ValueError('immutable payload reader required: ' + path)

885 installed = contract.manifest()

893 or not isinstance(declared, dict) or set(declared) != set(installed)

898 # with the installed schema; loading never rewrites saved values.

900 raise ValueError("snapshot object_format differs from Git")
901 if not isinstance(manifest["containers"], dict):
902 raise ValueError("invalid family manifest containers")
903 expected.add(manifest_path)
904 state = states[root] = {}
905 metadata_path = f"{root}/_state/metadata.json"
906 if metadata_path not in files:
907 raise ValueError("missing root metadata record: " + root)

911 if not isinstance(metadata, dict) or envelope.get('contract_id') != STATE_CONTRACT_ID or envelope.get('family') != 'metadata':
912 raise ValueError("invalid root metadata envelope: " + root)

914 if descriptor == {"shape": "metadata"}}

921 "record": {}, "json_pointer": "/record", "payload_refs": envelope.get('payloads', [])})
