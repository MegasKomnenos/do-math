22class OperationTrace:
23 """One operation's append-only trace and its throttled live progress snapshot.
24
25 Every row is appended at once. The live snapshot (active spans, read by
26 ``progress``) is rewritten at most once per ``LIVE_INTERVAL``; a later row
27 inside the interval is written by one pending timer, so the snapshot is
28 never more than ``LIVE_INTERVAL`` behind. ``close`` flushes and stops both;
29 a row that still arrives afterwards is appended but never rewrites the snapshot.
30 """
31 def __init__(self, directory, *, operation_id=None):
32 self.id = operation_id or uuid.uuid4().hex
33 self.path = Path(directory) / (self.id + '.jsonl')
34 self.path.parent.mkdir(parents=True, exist_ok=True)
35 self.lock = Lock()
36 self.live = self.path.parent / 'live' / (self.id + '.json')
37 self.live.parent.mkdir(parents=True, exist_ok=True)
38 self.errors = 0
39 self.failure = None
40 self.active = {}
41 self.parent_operation_id = os.environ.get('DO_MATH_PARENT_OPERATION_ID')
42 self.stream = None
43 self.pending = None
44 self.live_written = float('-inf')
45 self.timer = None
46 self.closed = False
47
48 def emit(self, row):
49 # No raw requests, prompts, argv, credentials, or control capabilities.
50 row = {'operation_id': self.id, 'parent_operation_id':self.parent_operation_id,
51 'observed_at': datetime.now(timezone.utc).isoformat(), **row}
52 raw = json.dumps(row, ensure_ascii=False, separators=(',', ':')) + '\n'
53 with self.lock:
54 if self.closed:
55 try:
56 with self.path.open('a', encoding='utf-8') as stream:
57 stream.write(raw)
58 except OSError as error:
59 self._failed('append_trace', error)
60 return
61 try:
62 if self.stream is None:
63 self.stream = self.path.open('a', encoding='utf-8')
64 self.stream.write(raw)
65 self.stream.flush()
66 except OSError as error:
67 self._close_stream()
68 self._failed('append_trace', error)
69 span = row.get('span_id')
70 if row.get('event') in {'started','progress'}:
71 self.active[span] = {**self.active.get(span, {}), **row}
72 elif span:
73 self.active.pop(span, None)
74 self.pending = {**row, 'active_spans':list(self.active.values())}
75 wait = self.live_written + LIVE_INTERVAL - time.monotonic()
76 if wait <= 0:
77 self._write_live()
78 elif self.timer is None:
79 self.timer = Timer(wait, self._write_pending)
80 self.timer.daemon = True
81 self.timer.start()
82
83 def close(self):
84 with self.lock:
85 self.closed, self.pending = True, None
86 if self.timer is not None:
87 self.timer.cancel()
88 self.timer = None
89 self._close_stream()
90
91 def _write_pending(self):
92 with self.lock:
93 self.timer = None
94 # A timer that fired while close() held the lock finds nothing to write.
95 if not self.closed and self.pending is not None:
96 self._write_live()
97
98 def _write_live(self):
99 snapshot, self.pending = self.pending, None
100 self.live_written = time.monotonic()
101 temporary = self.live.with_suffix('.tmp')
102 operation = 'write_live'
103 try:
104 temporary.write_text(json.dumps(snapshot, ensure_ascii=False), encoding='utf-8')
105 operation = 'replace_live'
106 atomic_replace(temporary, self.live)
107 except OSError as error:
108 self._failed(operation, error)
109
110 def _close_stream(self):
111 stream, self.stream = self.stream, None
112 if stream is not None:
113 try:
114 stream.close()
115 except OSError:
116 pass

118 def _failed(self, operation, error):
119 self.errors += 1
120 observed = datetime.now(timezone.utc).isoformat()
121 self.failure = {'operation_id':self.id,'operation':operation,
122 'exception_type':type(error).__name__,'errno':error.errno,
123 'winerror':getattr(error,'winerror',None),'count':self.errors,
124 'first_at':self.failure['first_at'] if self.failure else observed,'last_at':observed,
125 'relative_path':self.path.name if operation=='append_trace' else 'live/'+self.live.name}
126 # A separate bounded record survives failure of the live replace.
127 # No exception message, raw argv or request is copied into telemetry.
128 try:
129 directory=self.path.parent/'errors'
130 directory.mkdir(exist_ok=True)
131 target=directory/(self.id+'.json')
132 # Diagnostics must also survive failure of rename itself. A
133 # partial diagnostic is reported unreadable, never as authority.
134 target.write_text(json.dumps(self.failure,ensure_ascii=False),encoding='utf-8')
135 except OSError:
136 pass
137 if self.errors == 1:
138 warnings.warn(f'operation telemetry {operation} failed ({type(error).__name__}, '
139 f'errno={error.errno}, winerror={getattr(error,"winerror",None)}); '
140 'execution authority is unchanged', RuntimeWarning)

143@contextmanager
144def tracing(directory, *, operation_id=None):
145 try:
146 sink = OperationTrace(directory, operation_id=operation_id)
147 except OSError:
148 warnings.warn('operation telemetry unavailable', RuntimeWarning)

153 yield sink

157 try:
158 sink.live.unlink(missing_ok=True)

163@contextmanager
164def phase(name, **binding):
165 sink = _sink.get()
166 span_id, parent, started = uuid.uuid4().hex, _parent.get(), time.perf_counter_ns()
167 row = {'span_id': span_id, 'parent_span_id': parent, 'phase': name, **binding}
168 if sink:
169 sink.emit({**row, 'event': 'started'})
170 token = _parent.set(span_id)
171 outcome = 'completed'
172 class Counters(dict):
173 last = 0.
174 def update(self, *args, **kwargs):
175 super().update(*args, **kwargs)
176 current = time.monotonic()
177 if sink and current - self.last >= 1:
178 sink.emit({**row, 'event':'progress', 'elapsed_ns':time.perf_counter_ns() - started, **self})
179 self.last = current
180 counters = Counters()
181 try:
182 yield counters
183 except BaseException:
184 outcome = 'interrupted'
185 raise
186 finally:
187 _parent.reset(token)
188 if sink:
189 sink.emit({**row, 'event': outcome, 'elapsed_ns': time.perf_counter_ns() - started,
190 'trace_write_errors':sink.errors, **counters})

234def progress(directory):
235 rows = []
236 for path in sorted((Path(directory) / 'live').glob('*.json'))[:16]:
237 try:
238 rows.append(json.loads(read_bytes_shared(path).decode('utf-8')))
239 except (OSError, ValueError):
240 rows.append({'event':'unreadable_progress','operation_id':path.stem})
241 failures=[]
242 for path in sorted((Path(directory)/'errors').glob('*.json'))[-16:]:
243 try:failures.append(json.loads(read_bytes_shared(path).decode('utf-8')))
244 except (OSError,ValueError):failures.append({'operation_id':path.stem,'status':'unreadable_error_record'})
245 return {'source':'local_operation_sidecars','observed_at':datetime.now(timezone.utc).isoformat(),
246 'rows':rows,'telemetry_errors':failures,
247 'scope':'progress only; not process liveness or authoritative completion'}

250def trace_rows(directory):
251 import hashlib
252 rows = []
253 for path in sorted(Path(directory).glob('*.jsonl')):
254 try:
255 data = read_bytes_shared(path)
256 except OSError:
257 rows.append({'event':'unreadable_trace','file':path.name})
258 continue
259 source = {'path':str(path), 'sha256':hashlib.sha256(data).hexdigest(),
260 'observed_at':datetime.now(timezone.utc).isoformat()}
261 for raw in data.splitlines():
262 try:
263 value = json.loads(raw)
264 if not isinstance(value, dict):
265 raise ValueError('trace must be an object')
266 rows.append({**value, '_trace_source':source})
267 except (ValueError, UnicodeError):
268 # An interrupted append is unknown, not a fabricated end time.
269 rows.append({'event': 'incomplete_trace', 'file': path.name})
270 return rows

273def phase_rows(directory):
274 spans = {}
275 for event in trace_rows(directory):
276 if not all(isinstance(event.get(k), str) for k in ('span_id','operation_id','phase','event','observed_at')):
277 continue
278 key = (event['operation_id'], event['span_id'])
279 row = spans.setdefault(key, {'unit':'operation_phase', 'id':key[0] + ':' + key[1],
280 'operation_id':key[0], 'span_id':key[1], 'parent_span_id':event.get('parent_span_id'),
281 'parent_operation_id':event.get('parent_operation_id'),
282 'phase':event['phase'], 'created_at_utc':None, 'ended_at_utc':None, 'elapsed_ns':None})
283 row['evidence_source'] = {'kind':'local_operation_file', **event['_trace_source']}
284 if event['event'] == 'started': row['created_at_utc'] = event['observed_at']
285 if event['event'] in {'completed','interrupted'}:
286 row.update(ended_at_utc=event['observed_at'], elapsed_ns=event['elapsed_ns'], outcome=event['event'])
287 if 'trace_write_errors' in event: row['trace_write_errors'] = event['trace_write_errors']
288 return list(spans.values())
