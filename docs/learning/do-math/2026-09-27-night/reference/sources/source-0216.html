33 def answer(self, role, context, workspace):
34 action = context['action']
35 if role == 'identity_review' and action['stage'] == 'initial' and action['slot'] == 'initial-0':
36 review = context['role_review']
37 if not review['supplemental_results'] or self.repeat:
38 requests = [{'id': 'domain', 'question': 'Which binder domain does the exact exported declaration signature use?',
39 'source_refs': ['proposal:formal.declaration_signatures', 'proposal:lean_code']}]
40 if self.repeat:
41 requests[0]['question'] = 'Does this signature match the external textbook definition that has not been supplied?'
42 if self.batch:
43 requests.append({'id': 'export', 'question': 'Which declaration exports the supplied proof?',
44 'source_refs': ['proposal:declaration_names', 'proposal:lean_code']})
45 (workspace / 'activity-extensions.json').write_bytes(canonical_json({'research_requests': requests}))
46 return {**common('A fixed source comparison is needed before my complete review'),
47 'proposal_digest': review['proposal_digest'], 'decision': 'unresolved', 'scope_complete': False, 'findings': []}
48 self.resumed_contexts.append(deepcopy(review))
49 for research in review['supplemental_results']:
50 for source in research['result']['source_records']:
51 raw = (workspace / source['local_path']).read_bytes()
52 if hashlib.sha256(raw).hexdigest() != source['sha256']:
53 raise AssertionError('Returned supplemental source differs from its fixed hash')
54 return super().answer(role, context, workspace)
55 if role == 'supplemental_research':
56 research = context['review_research']
57 if context.get('task') is not None:
58 raise AssertionError('Review investigation incorrectly inherited a task execution contract')
59 for source in research['sources']:
60 raw = (workspace / source['local_path']).read_bytes()
61 if hashlib.sha256(raw).hexdigest() != source['sha256']:
62 raise AssertionError('Investigator source differs from its fixed hash')
63 if self.repeat:
64 output = {**common('The external textbook definition is not among the fixed inputs'),
65 'questions': ['The exact external definition remains unavailable'],
66 'research_request_id': research['id'], 'request_digest': research['request_digest'],
67 'outcome': 'unresolved', 'answer': 'The supplied signature cannot establish equivalence to an unavailable textbook definition.',
68 'scope_complete': False, 'evidence_refs': []}
69 else:
70 answer = 'The binder domain is Nat.' if 'binder domain' in research['question'] else 'The exported declaration is add_zero_fixture.'

73 'used_sources': [research['sources'][0]['ref']], 'research_request_id': research['id'],

75 'scope_complete': True, 'evidence_refs': [research['sources'][0]['ref'], 'workspace:comparison.txt']}
