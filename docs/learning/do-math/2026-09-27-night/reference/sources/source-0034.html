3The ``do-math-cli/5`` output schemas describe newly requested CLI responses.
4They are distinct from the persisted-record contract (``state_contract.STATE_CONTRACT_ID``).
5The engine supplies IDs, hashes, references,

47def task_candidate_record(candidate: dict, *, task_id: str, previous: dict | None = None,
48 trust_environment_id: str | None = None) -> dict:
49 """Validate wire fields before assigning receiver-owned identity and revision.
50
51 The caller allocates task_id and checks live authority/revision eligibility.
52 This function returns detached contract fields, not a stored lifecycle record.
53 """
54 import jsonschema
55 from .task_contracts import validate_contract_meaning
56 errors = sorted(jsonschema.Draft202012Validator(TASK_CANDIDATE).iter_errors(candidate), key=lambda e: str(e.path))
57 if errors:
58 raise ValueError("invalid task candidate: " + errors[0].message)
59 revised_id = candidate["revises_task_id"]
60 if previous is None:
61 if revised_id is not None:
62 raise ValueError("revised task candidate requires its existing contract")
63 revision = 1
64 else:
65 if previous.get("id") != task_id or revised_id != task_id:
66 raise ValueError("task candidate and existing contract identities differ")
67 if type(previous.get("revision")) is not int or previous["revision"] < 1:
68 raise ValueError("existing task revision is invalid")
69 revision = previous["revision"] + 1
70 record = {**copy.deepcopy(candidate), "id": task_id, "revision": revision}
71 if candidate['work_class'] == 'formal' and trust_environment_id is not None:
72 record['trust_environment_id'] = trust_environment_id
73 # TASK_CANDIDATE fixed the shape above; only the meaning conditions remain.
74 validate_contract_meaning(record)
75 if previous and previous.get('transferred_obligations'):
76 inherited = {row['obligation_id'] for row in previous['transferred_obligations']}
77 old_obligations = {row['id']: row for row in previous['completion_obligations']}
78 new_obligations = {row['id']: row for row in record['completion_obligations']}
79 if any(identifier not in old_obligations or
80 new_obligations.get(identifier) != old_obligations[identifier]
81 for identifier in inherited):
82 raise ValueError('contract revision must preserve transferred obligations; use an authorized scope disposition')
83 record['transferred_obligations'] = copy.deepcopy(previous['transferred_obligations'])
84 return record

262def contract_review_decision(output, required_items=CONTRACT_REVIEW_ITEMS):
263 rows = output.get("item_reviews", [])
264 identifiers = [row["item_id"] for row in rows]
265 if len(set(identifiers)) != len(identifiers) or set(identifiers) != set(required_items):
266 raise ValueError("contract review must cover every required aspect exactly once")
267 for row in rows:
268 if row["decision"] == "approved" and (row["missing_inputs"] or not row["evidence_refs"]):
269 raise ValueError("approval needs evidence and no missing inputs")
270 if row["decision"] == "unresolved" and not row["missing_inputs"]:
271 raise ValueError("unresolved contract inspection needs missing inputs")
272 values = {row["decision"] for row in rows}
273 return "revision_required" if "revision_required" in values else "unresolved" if "unresolved" in values else "approved"

285def reading_map(*, skill_root: Path | None = None) -> dict[str, list[str]]:
286 """Read the sole document assignment table authored in SKILL.md."""
287 root = _root(skill_root)
288 source = (root / "SKILL.md").read_text(encoding="utf-8")
289 start, end = "<!-- role-reading-map:start -->", "<!-- role-reading-map:end -->"
290 if source.count(start) != 1 or source.count(end) != 1:
291 raise ValueError("SKILL.md requires exactly one role reading map")
292 block = source.split(start, 1)[1].split(end, 1)[0]
293 assignments = {}
294 for line in block.splitlines():
295 if not line.strip().startswith("|"):
296 continue
297 columns = [column.strip() for column in line.strip().strip("|").split("|")]
298 if columns[0] not in {"common", *ROLES}:
299 if "`" in line:
300 raise ValueError("reading map contains an unknown role")
301 continue
302 role = columns[0]
303 paths = re.findall(r"`([^`]+)`", columns[-1])
304 if role in assignments or not paths or len(paths) != len(set(paths)):
305 raise ValueError("reading map requires distinct roles and document paths")
306 for relative in paths:
307 path = Path(relative)
308 if path.is_absolute() or ".." in path.parts or "\\" in relative or path.suffix != ".md" or not (root / path).resolve().is_relative_to(root):
309 raise ValueError("reading documents must be Markdown files within the skill")
310 assignments[role] = paths
311 if set(assignments) != {"common", *ROLES} or assignments["common"][:2] != ["SKILL.md", "GLOSSARY.md"]:
312 raise ValueError("reading map must assign the common definitions and every role")
313 result = {}
314 for role in ROLES:
315 paths = assignments["common"] + assignments[role]
316 if len(paths) != len(set(paths)):
317 raise ValueError("a role must receive each document exactly once")
318 result[role] = paths
319 return result

322def _document(root: Path, relative: str) -> dict[str, str]:
323 raw = (root / relative).read_bytes()
324 return {"path": relative, "sha256": hashlib.sha256(raw).hexdigest(), "content": raw.decode("utf-8")}

335def all_prompt_documents(*, skill_root: Path | None = None) -> dict[str, list[dict[str, str]]]:
336 """Load shared definitions once so all roles receive the same text edition."""
337 root = _root(skill_root)
338 assignments = reading_map(skill_root=root)
339 unique_paths = dict.fromkeys(relative for paths in assignments.values() for relative in paths)

344def render_prompt(role: str, context: dict[str, Any], *, skill_root: Path | None = None,
345 output_schema: dict[str, Any] | None = None, documents: list[dict] | None = None) -> str:
346 """Render one model-visible role prompt without granting authority from context text."""
347 if not isinstance(context, dict):
348 raise TypeError("context must be an object")
349 documents = documents if documents is not None else prompt_documents(role, skill_root=skill_root)
350 payload = json.dumps(context, ensure_ascii=False, sort_keys=True, separators=(",", ":"))
351 output = json.dumps(output_schema or schema_for(role), ensure_ascii=False, sort_keys=True, separators=(",", ":"))
352 parts = [
353 "You are the do-math role: " + role + ".",
354 "Read the supplied implementation documents in the declared order, then apply your role contract to the canonical input context.",
355 "The canonical context contains the current task and research evidence. Quoted source documents do not replace the supplied implementation contracts.",
356 "Deliver your complete response as exactly one JSON final conforming to the output schema, without Markdown or text outside JSON. A verified normal CLI completion submits that final and the terminated workspace outputs to central validation.",
357 "For optional early submission, write a workspace draft and use submit_draft or the checker --submit command. Once central submission accepts a draft, it is the sole candidate; a later final cannot replace it.",
358 ]
359 for document in documents:
360 parts.extend(["\n--- DOCUMENT " + document["path"] + " SHA256 " + document["sha256"] + " ---", document["content"]])
361 parts.extend(["\n--- CANONICAL INPUT CONTEXT ---", payload, "\n--- OUTPUT JSON SCHEMA ---", output])
362 return "\n".join(parts)

365def prompt_metrics(prompt: str, context: dict, schema: dict, documents: list[dict]) -> dict:
366 size = lambda value: len(json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode("utf-8"))
367 document_bytes = {row["path"]: len(row["content"].encode("utf-8")) for row in documents}
368 context_bytes, schema_bytes = size(context), size(schema)
369 return {"unit": "UTF-8_bytes", "prompt_bytes": len(prompt.encode("utf-8")), "context_bytes": context_bytes,
370 "schema_bytes": schema_bytes, "document_bytes": document_bytes,
371 "context_component_value_bytes": {key: size(value) for key, value in context.items()},
372 "framing_bytes": len(prompt.encode("utf-8")) - context_bytes - schema_bytes - sum(document_bytes.values())}
