29def prepared_tasks(root, shared, *, config, count):
30 """``restore_template`` build: validated active state with ``count`` nonformal tasks
31 ``"0"``..., each actually prepared, contract-reviewed and selected by the plan,
32 so that ``count`` ``research_work`` actions are queued. Returns the instance ID.
33
34 Preparation and approval are real applications, never injected ready flags.
35 """
36 runner = purpose.PurposeRunner()
37 original = runner.answer
38 def answer(role, context, workspace):
39 if role != "research_preparation":
40 return original(role, context, workspace)
41 candidates = [candidate(str(i), formal=False) for i in range(count)]
42 (workspace / "activity-extensions.json").write_bytes(canonical_json({
43 "contract_checks": contract_checks(candidates), "model_assessments": [
44 {"task_title": item["title"], "difficulty": "high", "importance": "medium", "reason": "Bounded scenario"}
45 for item in candidates]}))
46 return {**common(role), "candidates": candidates}
47 runner.answer = answer
48 instance_id = active_instance(root, shared, config=config)
49 engine = ResearchEngine(root / "repository", instance_id, runner=runner, formal_verifier=purpose.formal_fixture)
50 with patch("do_math.initialization.adapter_for", return_value=FixtureAdapter()):
51 for _ in range(45):
52 engine.run(max_steps=1)
53 if sum(a["role"] == "research_work" for a in engine.runtime["queue"]) == count:
54 break
55 else:
56 raise AssertionError("The prepared tasks were not queued for work: " + repr(engine.status()))
57 if any(e["status"] != "ended" for e in engine.runtime["executions"].values()):
58 raise AssertionError("A template must not end with unfinished executions")
59 return instance_id

62class BoundaryFixture(purpose.ActivatedPurposeHarness):
63 def engine(self, runner, count=1, **configuration):
64 # Public initialization and the preparation chain have separate coverage.
65 # Start these boundary cases from a copy of validated active state with
66 # ``count`` actually prepared and reviewed tasks (``prepared_tasks``).
67 # A rejected submission is not retried (max_output_repairs=0), as the
68 # currentness cases require; the other cases reach no rejected
69 # submission, so one value lets all cases share their start state.
70 config = self.fixture_config(**{"max_output_repairs": 0, **configuration})
71 if count:
72 self.root, instance_id = restore_template(prepared_tasks, config=config, count=count)
73 engine, _ = self._engine(instance_id, purpose.PurposeRunner())
74 else:
75 engine, _ = self.make_active(**config)
76 engine.runner = CheckedRunner(runner)
77 return engine
