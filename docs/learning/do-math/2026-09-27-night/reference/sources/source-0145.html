21def set_birthtime(path, nanoseconds):
22 """Change only the creation time (Windows SetFileTime)."""
23 from ctypes import wintypes
24 kernel = ctypes.WinDLL('kernel32', use_last_error=True)
25 kernel.CreateFileW.restype = wintypes.HANDLE
26 kernel.CreateFileW.argtypes = [wintypes.LPCWSTR, wintypes.DWORD, wintypes.DWORD, ctypes.c_void_p,
27 wintypes.DWORD, wintypes.DWORD, wintypes.HANDLE]
28 handle = kernel.CreateFileW(str(path), 0x100, 0x7, None, 3, 0x80, None)
29 if handle in (None, wintypes.HANDLE(-1).value):
30 raise ctypes.WinError(ctypes.get_last_error())
31 try:
32 ticks = nanoseconds // 100 + 116444736000000000
33 created = wintypes.FILETIME(ticks & 0xFFFFFFFF, ticks >> 32)
34 if not kernel.SetFileTime(handle, ctypes.byref(created), None, None):
35 raise ctypes.WinError(ctypes.get_last_error())
36 finally:
37 kernel.CloseHandle(handle)

147 def test_fingerprint_detects_size_modification_creation_and_recreation(self):

156 'size': lambda: self.olean.write_bytes(original + b'x'),

158 'creation time': lambda: set_birthtime(self.olean, info.st_birthtime_ns - 10**10),
159 'recreated with the same bytes': lambda: (self.olean.unlink(), self.olean.write_bytes(original),
160 os.utime(self.olean, ns=(info.st_atime_ns, info.st_mtime_ns + 10**9))),
161 'added module': lambda: self.olean.with_name('Extra.olean').write_bytes(b'new'),
162 }
163 for name, change in changes.items():
