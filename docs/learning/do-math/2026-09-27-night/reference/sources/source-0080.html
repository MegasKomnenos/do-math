195def _execution_rules(row):
196 if row["status"] == "ended" and (not row.get("ended_at") or not row.get("outcome")):
197 raise ValueError("ended execution requires termination evidence")
198 if (row["status"] == "stopping") != ("stop_phase" in row):
199 raise ValueError("stop_phase belongs exactly to stopping executions")
200 if row.get("outcome") == "output_accepted" and not row.get("submission_receipt"):
201 raise ValueError("accepted output lacks submission receipt")
202 if row.get("execution_window", {}).get("execution_id", row["id"]) != row["id"]:
203 raise ValueError("execution window belongs to a different execution")
204 if row.get("authority_revocation"):

207 or receipt.get("state") != "accepted" or receipt.get("kind") not in {"submit_intent", "native_final"}
208 or receipt.get("execution_id") != row["id"]
209 or row["authority_revocation"]["receipt_id"] != (receipt.get("id") or receipt.get("request_id"))):
210 raise ValueError("submission authority revocation differs from its actual receipt")

297def validate_lifecycle(instance, runtime):
298 """The lifecycle rules of one whole state; the schema fixes the state values.
299
300 A state has a lifecycle and a runtime status, a permanently closed
301 instance only the closed status, and a persisted UTC high-water mark.
302 """
303 lifecycle, status = instance.get("lifecycle"), runtime.get("status")
304 if lifecycle is None or status is None:
305 raise ValueError("current lifecycle contract: the lifecycle and the runtime status are required")
306 if lifecycle.get("permanently_closed") and status != "closed":
307 raise ValueError("current lifecycle contract: a permanently closed instance cannot have a live projection")
308 observed = runtime.get("last_observed_utc")
309 from ..file_io import parse_utc
310 try:
311 if observed is None or not observed.endswith(("Z", "+00:00")):
312 raise ValueError("UTC offset required")
313 parse_utc(observed)
314 except ValueError as error:
315 raise ValueError("current lifecycle contract: the persisted UTC high-water is absent or invalid") from error
