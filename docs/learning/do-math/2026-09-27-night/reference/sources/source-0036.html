211 def receive(self, request: dict, *, publish: Callable[[dict], dict] | None = None) -> dict:
212 """Validate and publish a request, or record its non-final receipt.
213
214 A publisher may return ``accepted`` only with ``published: true`` and a
215 nonempty authoritative snapshot. The engine's callback must have made
216 that CAS publication before returning.
217 """
218 if not isinstance(request, dict):
219 raise ValueError("control request must be an object")
220 request_id = _require_file_name(request.get("request_id"), "request_id")
221 request_sha256 = json_digest(request)
222 try:
223 sender = _require_file_name(request.get("transport_id"), "transport_id")
224 except ValueError:
225 sender = None
226 previous = self._receipt(request_id, sender)
227 if previous is not None:
228 if previous.get("request_sha256") != request_sha256:
229 raise ValueError("request_id was reused with different content")
230 if previous.get("state") in _FINAL_RECEIPTS:
231 return previous
232 try:
233 request = self._validate_request(request)
234 except ValueError as error:
235 return self._reject_unvalidated(request, str(error))
236 state = self._state(request)
237 if state["state"] != "open":
238 return self._store(request, {"state": "withdrawn" if state["state"] == "withdrawn" else "closed",
239 "reason": state.get("reason", "transport is not open")})
240 publisher = publish or self._publish
241 if publisher is None:
242 return self._store(request, {"state": "received"})
243 decision = publisher(deepcopy(request))
244 if request["kind"] == SUBMISSION_PREVIEW:
245 if (not isinstance(decision, dict) or decision.get("state") not in {"previewed", "rejected"}
246 or decision.get("published") is True):
247 raise ValueError("preview must return a check result without publication")
248 if decision["state"] == "previewed" and not isinstance(decision.get("result"), dict):
249 raise ValueError("previewed receipt requires a structured check result")
250 return self._store(request, decision)
251 if not isinstance(decision, dict) or decision.get("state") not in {"accepted", "rejected"}:
252 raise ValueError("transport publisher must return accepted or rejected")
253 if decision["state"] == "accepted":
254 if decision.get("published") is not True or not isinstance(decision.get("snapshot"), str) or not decision["snapshot"]:
255 raise ValueError("accepted receipt requires a published authoritative snapshot")
256 return self._store(request, decision)

258 def pump(self, publish: Callable[[dict], dict] | None = None, *, limit: int | None = None) -> list[dict]:
259 """Read delivery files and call the engine's validation-and-CAS hook."""
260 if self.root is None:
261 raise ValueError("a memory-only transport has no inbox to pump")
262 if limit is not None and (isinstance(limit, bool) or not isinstance(limit, int) or limit < 1):
263 raise ValueError("limit must be a positive integer or null")
264 if self._open is None:
265 # issue and close keep state.json current, so this scan sees every earlier issue and close.
266 self._open = ({path.parent.name for path in self.root.glob("*/binding.json") if _is_open(path.parent)}
267 if self.root.is_dir() else set())
268 paths = [path for transport_id in sorted(self._open)
269 for path in sorted((self.root / transport_id / "inbox").glob("*.json"))]
270 return self._pump_paths(paths[:limit], publish)

272 def _pump_paths(self, paths: list[Path], publish: Callable[[dict], dict] | None) -> list[dict]:
273 results = []
274 for path in paths:
275 try:
276 mark = file_fingerprint(path)
277 except FileNotFoundError:
278 continue
279 # A delivery with a final receipt is not read again while its settled
280 # fingerprint is unchanged; a rewritten delivery is read and received.
281 cached = self._pump_cache.get(path)
282 if cached is not None and cached[0] == mark:
283 results.append(deepcopy(cached[1]))
284 continue
285 self._pump_cache.pop(path, None)
286 try:
287 try:
288 raw = path.read_bytes()
289 except FileNotFoundError:
290 continue
291 receipt = self.receive(_decode_json(path, raw), publish=publish)
292 except ValueError as error:
293 receipt = self._reject_delivery_file(path, error)
294 results.append(receipt)
295 if (receipt.get('state') in _FINAL_RECEIPTS and fingerprint_settled(mark)
296 and file_fingerprint(path) == mark):
297 self._pump_cache[path] = (mark, deepcopy(receipt))
298 return results

300 def close(self, binding: dict, *, reason: str, withdrawn: bool = False) -> dict:
301 endpoint = self._endpoint(binding)
302 _require_text(reason, "reason")
303 state = {"state": "withdrawn" if withdrawn else "closed", "reason": reason}
304 _atomic_json(endpoint / "state.json", state)
305 # A delivery that raced the close gets its closed receipt now; later pumps skip this transport.
306 inbox = sorted((endpoint / "inbox").glob("*.json"))
307 unsettled = []
308 for path in inbox:
309 try:
310 self._pump_paths([path], None)
311 except ValueError as error:
312 # A changed redelivery keeps the receipt it already has; the closed state records why.
313 unsettled.append({"source": "inbox/" + path.name, "reason": str(error)})
314 if unsettled:
315 state["unsettled_deliveries"] = unsettled
316 _atomic_json(endpoint / "state.json", state)
317 if self._open is not None:
318 self._open.discard(endpoint.name)
319 for path in inbox:
320 self._pump_cache.pop(path, None)
321 return state

416 def _validate_request(self, request: dict) -> dict:
417 if not isinstance(request, dict):
418 raise ValueError("control request must be an object")
419 result = deepcopy(request)
420 for key in ("request_id", "transport_id"):
421 _require_file_name(result.get(key), key)
422 for key in ("capability", "execution_id", "grant_id"):
423 _require_text(result.get(key), key)
424 if (not isinstance(result.get("kind"), str) or result["kind"] not in REQUEST_KINDS
425 or not isinstance(result.get("payload"), dict)):
426 raise ValueError("unsupported control kind or payload")
427 binding = self._binding(result)
428 if any(result[key] != binding[key] for key in ("execution_id", "grant_id", "capability")):
429 raise ValueError("execution, grant, or capability differs from transport binding")
430 if result["kind"] == SAFETY_BOUNDARY and result["payload"].get("inflight_calls") is not False:
431 raise ValueError("boundary report has an in-flight call")
432 if result["kind"] == SELF_CHECK_RESPONSE:
433 sid = result["payload"].get("self_check_id")
434 _require_file_name(sid, "self_check_id")
435 if _json(self._endpoint(binding) / "worker-inbox" / (sid + ".json")) is None:
436 raise ValueError("self-check response has no issued request")

488 def _receipt(self, request_id: str, transport_id: str | None = None) -> dict | None:
489 """The receipt of ``request_id``: the sender's own first, then any transport's.
490
491 Request IDs are unique across the instance's transports, so another
492 transport's receipt for the ID rejects a reuse. The sender's own receipt
493 is read first: a rejection another transport earned for reusing this ID
494 must not stand in for the sender's receipt when the sender redelivers.
495 """
496 _require_file_name(request_id, "request_id")
497 name = request_id + ".json"
498 if transport_id is not None:
499 entry = self._receipts.get((transport_id, request_id))
500 if entry is not None:
501 return deepcopy(entry)
502 if self.root is not None and os.path.exists(self.root / transport_id / "outbox" / name):
503 return _json(self.root / transport_id / "outbox" / name)
504 for (_, known_id), entry in self._receipts.items():
505 if known_id == request_id:
506 return deepcopy(entry)
507 if self.root is None or not self.root.is_dir():
508 return None
509 # A new ID is checked against every outbox by one existence test per transport.
510 for entry in os.scandir(self.root):
511 path = Path(entry.path) / "outbox" / name
512 if os.path.exists(path):
513 return _json(path)
514 return None

516 def _binding(self, request: dict) -> dict:
517 transport_id = _require_file_name(request["transport_id"], "transport_id")
518 binding = self._capabilities.get(transport_id)
519 if binding is not None:
520 return binding
521 if self.root is None:
522 raise ValueError("unknown transport capability")
523 binding = _json(self.root / transport_id / "binding.json")
524 if binding is None:
525 raise ValueError("unknown transport capability")
526 self._capabilities[transport_id] = binding
527 return binding

530 binding = self._binding(request)
531 if not binding.get("endpoint"):
532 return {"state": "open"}
533 endpoint = self._endpoint(binding)
534 return _json(endpoint / "state.json") or {"state": "open"}
