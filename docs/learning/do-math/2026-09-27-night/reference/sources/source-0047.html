240def _worker(request_path: Path) -> int:
241 q=json.loads(request_path.read_text())
242 code=None
243 try:
244 # Files receive all bytes directly: no PIPE buffers, reader threads or output truncation.
245 with open(q['stdout'],'wb',buffering=0) as out, open(q['stderr'],'wb',buffering=0) as err:
246 with subprocess.Popen(q['command'],cwd=q['cwd'],stdout=out,stderr=err,
247 shell=False,creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0)) as process:
248 code=process.wait()
249 state='exited'
250 except Exception:
251 state='worker_error'
252 with open(q['stderr'],'ab') as err:
253 err.write(traceback.format_exc().encode('utf-8'))
254 Path(q['status']).write_text(json.dumps({'state':state,'exit':code}))
255 return 0

258def _job_run(command: list[str], cwd: Path, *, should_stop=None, on_started=None,
259 on_finished=None, raw_output=False, job_name=None):
260 """Wait for the whole Job; only explicit stop terminates it.
261
262 The two-second termination check applies only after cancellation, never to computation.
263 """
264 root=Path(tempfile.mkdtemp(dir=cwd)); out=root/'out'; err=root/'err'; status=root/'status'; request=root/'request.json'
265 request.write_text(json.dumps({'command':command,'cwd':str(cwd),'stdout':str(out),'stderr':str(err),'status':str(status)}))
266 worker = "import sys; from pathlib import Path; sys.path.insert(0,sys.argv[1]); from do_math.formal import _worker; raise SystemExit(_worker(Path(sys.argv[2])))"
267 environment={key:value for key,value in os.environ.items()
268 if not key.upper().startswith(('LEAN_', 'LAKE_', 'ELAN_'))}
269 job=WindowsJob([sys.executable,'-B','-c',worker,str(Path(__file__).resolve().parents[1]),str(request)],cwd=root,env=environment,
270 name=job_name or 'do-math-formal-'+uuid.uuid4().hex)
271 try:
272 started={**process_identity(job),'job_name':job.name}
273 if on_started: on_started(started)
274 job.resume(); stopped=False
275 while True:
276 if should_stop and should_stop():
277 job.terminate(); stopped=True; break
278 if job.wait(0):
279 observed=job.inspect()
280 if observed.root_signaled and not observed.active_processes and not observed.process_ids:
281 break
282 time.sleep(.02)
283 if stopped:
284 termination_deadline=time.monotonic()+2
285 while True:
286 observed=job.inspect()
287 if observed.root_signaled and not observed.active_processes and not observed.process_ids: break
288 if time.monotonic()>=termination_deadline:
289 raise RuntimeError('formal Job termination could not be confirmed')
290 time.sleep(.02)
291 if on_finished: on_finished(job.resource_usage())
292 def read_output(path):
293 if not path.exists(): return b'' if raw_output else ''
294 return path.read_bytes() if raw_output else path.read_text(encoding='utf-8',errors='replace')
295 stdout,stderr=read_output(out),read_output(err)
296 error_stderr=stderr.decode(errors='replace') if isinstance(stderr,bytes) else stderr
297 if stopped: return None,stdout,stderr,True
298 if not status.exists():
299 raise RuntimeError('formal worker completion status missing' + (': '+error_stderr if error_stderr else ''))
300 try:
301 data=json.loads(status.read_text())
302 if (not isinstance(data,dict) or data.get('state') not in {'exited','worker_error'}
303 or (data['state']=='exited' and type(data.get('exit')) is not int)):
304 raise ValueError('invalid completion status')
305 except ValueError as error:
306 raise RuntimeError('formal worker completion status invalid: '+str(error)) from error
307 if data['state']=='worker_error': raise RuntimeError('formal worker failed: '+error_stderr)
308 return data['exit'],stdout,stderr,False
309 finally: job.close()

311def toolchain_errors(toolchain):
312 """The structural contract shared by registration, inspection and execution."""
313 required = {'executable', 'sha256', 'configured_command', 'version', 'version_exit',
314 'verifier_sha256', 'verifier_path', 'executables', 'project'}
315 if not isinstance(toolchain, dict):
316 return ['toolchain must be an object']
317 errors = ['toolchain.' + key + ' is missing' for key in sorted(required - toolchain.keys())]
318 for key in ('sha256', 'verifier_sha256'):
319 value = toolchain.get(key)
320 if key in toolchain and not is_sha256(value):
321 errors.append('toolchain.' + key + ' must be a SHA-256')
322 for key in ('executable', 'version'):
323 if key in toolchain and (not isinstance(toolchain[key], str) or not toolchain[key].strip()):
324 errors.append('toolchain.' + key + ' must be nonempty')
325 if not isinstance(toolchain.get('verifier_path'),str) or not Path(toolchain['verifier_path']).is_absolute():
326 errors.append('toolchain.verifier_path must identify the absolute checker source')
327 command = toolchain.get('configured_command')
328 if not isinstance(command, list) or not command or any(not isinstance(p, str) or not p or '\x00' in p for p in command):
329 errors.append('toolchain.configured_command must be a nonempty argv')
330 if type(toolchain.get('version_exit')) is not int or toolchain['version_exit'] != 0:
331 errors.append('toolchain.version_exit must record a successful version observation')
332 executables = toolchain.get('executables')
333 if not isinstance(executables, dict) or any(not isinstance(p, str) or not p or
334 not is_sha256(h) for p, h in executables.items()):
335 errors.append('toolchain.executables must map paths to SHA-256 values')
336 project = toolchain.get('project')
337 if project is not None and (not isinstance(project, dict) or
338 set(project) != {'manifest_hashes', 'source_identity', 'search_paths'}
339 or not isinstance(project['source_identity'], dict)
340 or not isinstance(project['source_identity'].get('root'), str)
341 or not Path(project['source_identity']['root']).is_absolute()):
342 errors.append('toolchain.project must be null or the observed Lake project identity')
343 elif project is not None:
344 paths=project['search_paths']
345 if not isinstance(paths,list) or any(not isinstance(p,str) or not Path(p).is_absolute() for p in paths):
346 errors.append('toolchain project search paths must be observed absolute paths')
347 return errors

350def _project_identity(command, should_stop=None, *, workspace=None, expected=None):
351 if not Path(command[0]).name.lower().startswith('lake') or '--dir' not in command:
352 return None
353 _check_stop(should_stop)
354 index = command.index('--dir') + 1
355 if index >= len(command):
356 raise ValueError('Lake --dir requires a project path')
357 root = Path(command[index]).resolve(strict=True)
358 names = ('lean-toolchain', 'lake-manifest.json', 'lakefile.lean', 'lakefile.toml')
359 identity={'manifest_hashes': {name: cached_sha256(root/name) for name in names if (root/name).is_file()},
360 'source_identity': project_source_identity(root,should_stop=should_stop)}
361 # Never execute changed project code merely to discover its environment.
362 if expected is not None and any(identity[k]!=expected.get(k) for k in identity):
363 return {**identity,'search_paths':[]}
364 if workspace is None or 'env' not in command:
365 raise ValueError('Lake environment observation requires env and an observation workspace')
366 prefix=command[:command.index('env')+1]
367 # Lake derives the search paths from its executable and the inputs identified
368 # above; unchanged inputs give the same paths, so one observation per process.
369 lake=Path(shutil.which(prefix[0]) or prefix[0])
370 key=json_digest([prefix,identity,cached_sha256(lake) if lake.is_file() else None])
371 if key in _SEARCH_PATHS:
372 return {**identity,'search_paths':list(_SEARCH_PATHS[key])}
373 script=('import os,json; print(json.dumps({"cwd":os.getcwd(),"paths":'
374 '[p for k in ("LEAN_PATH","LEAN_SRC_PATH") for p in os.environ.get(k,"").split(os.pathsep) if p]}))')
375 code,out,err,stopped=_job_run([*prefix,sys.executable,'-c',script],Path(workspace),should_stop=should_stop)
376 if stopped:raise VerificationStopped('project environment observation stopped')
377 if code!=0:raise RuntimeError('project environment observation failed: '+err)
378 observed=json.loads(out)
379 if (not isinstance(observed,dict) or not isinstance(observed.get('cwd'),str)
380 or not isinstance(observed.get('paths'),list) or any(not isinstance(p,str) for p in observed['paths'])):
381 raise ValueError('invalid project environment observation')
382 identity['search_paths']=sorted({str((Path(observed['cwd'])/p).resolve()) for p in observed['paths']})
383 _SEARCH_PATHS[key]=tuple(identity['search_paths'])
384 return identity

387def inspect_toolchain(command, workspace, *, should_stop=None):
388 """Observe exact tool identity; this does not run environment validation."""
389 command = list(command)
390 if not command or any(not isinstance(p, str) or not p or '\x00' in p for p in command):

405 if errors:
406 raise ValueError('; '.join(errors))

410def toolchain_identity_errors(toolchain, *, should_stop=None):
411 """Recompute the local facts of an approved identity: the recorded executable
412 and checker bytes, the running checker and the Lake project inputs.
413
414 Formal checks, activation and publication gates share this one check. It does
415 not isolate processes from concurrent writers under the same OS identity.
416 """
417 errors = toolchain_errors(toolchain)
418 if errors:
419 return errors
420 if toolchain['verifier_sha256'] != _RUNNING_CHECKER_SHA256:
421 errors.append('approved checker is not the running checker')
422 # Pairs, not a mapping: the executable may also appear among the command's
423 # files, and each recorded hash must hold on its own.
424 files = [(toolchain['executable'], toolchain['sha256']), *toolchain['executables'].items(),
425 (toolchain['verifier_path'], toolchain['verifier_sha256'])]
426 for path, expected in files:
427 try:
428 current = cached_sha256(Path(path))
429 except OSError:
430 current = None
431 message = 'approved toolchain file changed: ' + path
432 if current != expected and message not in errors:
433 errors.append(message)
434 try:
435 with tempfile.TemporaryDirectory(prefix='do-math-identity-') as workspace:
436 project = _project_identity(toolchain['configured_command'], should_stop,
437 workspace=Path(workspace), expected=toolchain['project'])
438 except VerificationStopped:
439 raise
440 except (OSError, RuntimeError, ValueError) as error:
441 return errors + ['approved project inputs could not be observed: ' + str(error)]
442 if project != toolchain['project']:
443 errors.append('approved project inputs changed')
444 return errors
