54 def test_schema_builders_return_copies_that_share_no_node(self):
55 """A builder's result shares no container with a module global or within itself.
56
57 A caller may then change any node in place without changing another
58 field of the same schema or a schema built later in the process.
59 """
60 from do_math import governance, integration, record_schemas, review_cycle, scheduled_operator, submission
61
62 def containers(value):
63 found, pending = [], [value]
64 while pending:
65 node = pending.pop()
66 if isinstance(node, (dict, list)):
67 found.append(node)
68 pending.extend(node.values() if isinstance(node, dict) else node)
69 return found
70
71 request = {"integration_id": "IN-1", "payload_sha256": "a" * 64, "producer_execution_id": "EX-1",
72 "input_snapshot": "b" * 40, "plans": {"scope": {"digest": "c" * 64}}}
73 built = {
74 **{"schema_for " + role: schema_for(role) for role in ROLES},
75 **{"action_schema " + role: contracts.action_schema({"role": role})
76 for role in ("research_preparation", "research_work", "maintenance", "intent_interpretation")},
77 "disposition_schema": contracts.disposition_schema(),
78 "extension_schema": submission.extension_schema({"role": "research_work", "review_cycle_id": "RC-1"}),
79 "impact_review_response_schema": integration.impact_review_response_schema(request),
80 **{"stage_schema " + stage: review_cycle.stage_schema(stage)
81 for stage in ("scope", "initial", "organize", "answer", "judge")},
82 "research_schema": review_cycle.research_schema(),
83 "research_request_schema": review_cycle.research_request_schema(),
84 **{"interpretation_schema " + stage: governance.interpretation_schema(stage) for stage in ("interpret", "check")},
85 "config_schema": record_schemas.config_schema(),
86 "repair_report_schema": scheduled_operator.repair_report_schema(diagnosed=True),
87 }
88 global_nodes = set()
89 for name, module in list(sys.modules.items()):
90 if name == "do_math" or name.startswith("do_math."):
91 for value in vars(module).values():
92 global_nodes.update(id(node) for node in containers(value))
93 for name, schema in built.items():
94 with self.subTest(builder=name):
95 nodes = [id(node) for node in containers(schema)]
96 self.assertEqual(len(nodes), len(set(nodes)), "a node appears twice in the result")
97 self.assertFalse(global_nodes & set(nodes), "the result holds a module-global node")

99 def test_every_role_has_closed_schema(self):
100 """Every object with declared properties, at any depth, rejects fields it does not declare."""
101 def objects(node, path):
102 if isinstance(node, dict):
103 if "properties" in node:
104 yield path, node
105 for key, value in node.items():
106 yield from objects(value, path + "/" + key)
107 elif isinstance(node, list):
108 for index, value in enumerate(node):
109 yield from objects(value, f"{path}/{index}")
110 for role in ROLES:
111 schema = schema_for(role)
112 for path, node in objects(schema, role):
113 with self.subTest(path=path):
114 self.assertIs(node.get("additionalProperties"), False)
115 self.assertTrue({"summary", "used_sources", "questions"} <= set(schema["required"]))
116 jsonschema.Draft202012Validator.check_schema(schema)
