24 def stopped_check(self, arguments):
25 """The check reads its stop flag until the owner's judgment sets it."""
26 if not await_stop(arguments['should_stop'], timeout=5):
27 raise AssertionError('the owner control did not reach the check')

30 def run_with_control(self, engine, action, operation):
31 """Start the check, apply the control on the owner, then judge stops as a publication does."""
32 engine.formal_behavior = self.stopped_check
33 with owner_calls(engine) as calls:
34 engine._formal(action)
35 engine.control(operation, {})

38 def test_pause_resume_retains_one_action_and_reaches_identity_review(self):
39 engine, action = self.make()
40 self.run_with_control(engine, action, 'pause')
41 first = next(e for e in engine.runtime['executions'].values() if e['role'] == 'formal_check')
42 self.assertEqual(first['status'], 'ended')
43 pending = next(a for a in engine.runtime['queue'] if a['id'] == action['id'])
44 self.assertEqual(pending['continuation_of'], first['id'])
45 self.assertNotEqual(engine.runtime['activities'][first['owner_activity_id']]['status'], 'finished')
46 engine.control('resume', {})
47 engine.control('resume', {})
48 self.assertEqual(sum(a.get('id') == action['id'] for a in engine.runtime['queue']), 1)
49 engine.formal_behavior = None
50 with owner_calls(engine):
51 engine._formal(pending)
52 attempts = [e for e in engine.runtime['executions'].values() if e['role'] == 'formal_check']
53 self.assertEqual(len(attempts), 2)
54 self.assertEqual(attempts[-1]['continuation_of'], first['id'])
55 self.assertEqual(attempts[-1]['formal_status'], 'verified')
56 self.assertTrue(any(a['role'] == 'identity_review' for a in engine.runtime['queue']))

100 def test_public_pause_restart_resume_preserves_formal_attempt_lineage(self):
101 from do_math import knowledge_records
102 from research_fixtures import send_control
103 engine = ResearchEngine(self.repo, self.identifier)
104 environment = engine._formal_environment()
105 engine.instance['config']['lean_command'] = environment['toolchain']['configured_command']

107 engine.instance['execution_profiles'].update(editions['execution_profiles'])
108 engine.instance['role_profile_refs']['formal_check'] = editions['role_profile_refs']['formal_check']
109 engine.instance, _ = knowledge_records.create_candidate(engine.instance, {
110 'id': 'KN-formal-resume', 'kind': 'lemma', 'title': 'Resume fixture',
111 'statement': 'One equals one', 'summary': 'Reflexivity',
112 'lean_code': 'theorem resume_fixture : 1 = 1 := rfl', 'declaration_names': ['resume_fixture'],
113 'dependencies': []}, source_ref=engine.instance['lifecycle']['input_id'],
114 actor_ref=next(iter(engine.runtime['executions'])))
115 engine._enqueue('formal_check', 'KN-formal-resume')
116 engine._save()
117 def pause(**arguments):
118 # The owner publishes the pause; that publication sets this check's stop flag.
119 if not await_stop(arguments['should_stop'], timeout=5):
120 raise AssertionError('the published pause did not reach the check')
121 return {'status': 'stopped', 'reason': 'Explicit test pause; no Lean process'}
122 engine.formal_verifier = pause
123 with owner_calls(engine):
124 engine._formal(next(a for a in engine.runtime['queue'] if a['role'] == 'formal_check'))
125 send_control(engine, 'pause', config={})

129 send_control(restarted, 'resume', config={})
130 action = next(a for a in restarted.runtime['queue'] if a['role'] == 'formal_check')
131 self.assertEqual(action['continuation_of'], first['id'])

133 kw['source'], kw['declarations'], kw['expected_toolchain'])

135 restarted._formal(action)

137 attempts = [e for e in final.runtime['executions'].values() if e['role'] == 'formal_check']
138 self.assertEqual(len(attempts), 2)
139 self.assertTrue(any(e.get('continuation_of') == first['id'] and e['formal_status'] == 'verified' for e in attempts))
140 self.assertTrue(any(a['role'] == 'identity_review' for a in final.runtime['queue']))
