70 def test_custom_kind_reaches_worker_catalog_only_after_check_and_activation(self):
71 definition = dict(family='knowledge', token='computational_lemma',
72 meaning='An explicitly bounded computational proposition.', constraints={})
73 self.send(operation='register-kind', definition=definition)
74 def kinds():
75 # Each actual invocation receives a new immutable input workspace.
76 context = self.engine._context({'role': 'research_preparation'}, self.repo / new_id('EX'))
77 return [row['token'] for row in context['kind_definitions']['knowledge']]
78 self.assertNotIn(definition['token'], kinds())
79 # Only the central consumer of an independent check activates a kind.
80 # Use a separate v5 execution and a synthetic AI verdict.
81 from do_math import execution_lifecycle
82 from do_math.execution_profiles import profile_ref
83 engine = self.engine
84 action = next(a for a in engine.runtime['queue'] if a.get('kind_definition_id'))
85 identifier = action['kind_definition_id']
86 profile = next(p for p in engine.instance['execution_profiles'].values() if p['role'] == 'maintenance')
87 activity = activities.open_activity(engine.runtime, activity_id=new_id('AT'), activity_kind='maintenance',
88 owner_ref=identifier, input_refs=[identifier], opened_at='2026-09-20T00:00:00Z')
89 eid = new_id('EX')
90 basis = {'kind_definition': {'id': identifier, 'sha256': action['kind_definition_sha256']}}
91 execution = execution_lifecycle.create_execution(execution_id=eid, owner_activity_id=activity['id'],
92 actor_kind='ai', input_manifest=basis, profile_ref=profile_ref(profile), created_at='2026-09-20T00:00:00Z')
93 execution.update(role='maintenance', action=deepcopy(action), input_basis=basis)
94 activities.assign_execution(engine.runtime, activity['id'], execution)
95 for status in ('starting', 'running'):
96 execution_lifecycle.transition_execution(execution, status, at='2026-09-20T00:00:00Z')
97 verdict = {'definition_id': identifier, 'definition_sha256': action['kind_definition_sha256'],
98 'decision': 'approved', 'findings': []}
99 engine._apply_kind_definition_check(action, verdict, eid)
100 execution['submission_receipt'] = {'state': 'accepted', 'evidence_ref': 'synthetic:kind-check'}
101 execution_lifecycle.transition_execution(execution, 'stopping', at='2026-09-20T00:00:00Z', stop_phase='terminating')
102 execution_lifecycle.transition_execution(execution, 'ended', at='2026-09-20T00:00:00Z', outcome='output_accepted')
103 with engine.store.writer_lock():
104 engine._save()
105 engine.head, engine.instance, engine.runtime = engine.store.load()
106 self.assertIn(definition['token'], kinds())
