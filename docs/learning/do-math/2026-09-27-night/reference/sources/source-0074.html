221 def _apply_queue_plan(self, action, execution_id):
222 dispositions = self.runtime["executions"][execution_id].get("extensions", {}).get("queued_requests", [])
223 if not dispositions:
224 return
225 if action["role"] != "research_plan":
226 raise ValueError("only research planning reorders queued work")
227 _, _, original = self.store.view(self.runtime["executions"][execution_id]["input_snapshot"])
228 original_live = live_action_ids(original)
229 allowed = {a["id"] for a in original["queue"] if a.get("id") and a["role"] in WORK_ROLES and a["id"] not in original_live}
230 live = live_action_ids(self.runtime)
231 current = {a.get("id"): a for a in self.runtime["queue"]}
232 ordered, seen = [], set()
233 for row in dispositions:
234 if set(row) != {"request_id", "decision", "reason"} or row["decision"] not in {"keep", "cancel"} or not isinstance(row["reason"], str) or not row["reason"].strip():
235 raise ValueError("queued request disposition requires identity, keep/cancel and reason")
236 aid = row["request_id"]
237 if aid not in allowed or aid in seen:
238 raise ValueError("queued request was not in the planning input or is repeated")
239 seen.add(aid)
240 if aid in live or aid not in current:
241 self.runtime["history"].append({"kind": "queue_plan_not_applied", "execution_id": execution_id, **row})
242 continue
243 request = current[aid]
244 if row["decision"] == "cancel":
245 self._remove_action(request)
246 task = self.runtime["tasks"].get(request.get("target"))
247 if task and request["role"] in RESEARCH_PRODUCER_ROLES and not request.get("review_cycle_id") and task["status"] == "active":
248 task.update(status="planned", deferred_reason=row["reason"])
249 else:
250 ordered.append(request)
251 self.runtime["history"].append({"kind": "queue_plan_applied", "execution_id": execution_id, **row})
252 # Keep non-work requests in their existing positions and preserve unseen
253 # requests, including those created after the planning snapshot.
254 reorder_ids = {a["id"] for a in ordered}
255 positions = [i for i, a in enumerate(self.runtime["queue"]) if a.get("id") in reorder_ids]
256 for position, request in zip(positions, ordered):
257 self.runtime["queue"][position] = request
