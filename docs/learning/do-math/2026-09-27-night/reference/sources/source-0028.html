42def _json_document(path: str) -> tuple[dict[str, Any], str]:
43 try:
44 text = Path(path).read_bytes().decode("utf-8")
45 value = strict_json_loads(text)
46 except (OSError, UnicodeDecodeError, json.JSONDecodeError, ValueError) as error:
47 raise CliInputError(f"cannot read JSON file {path}: {error}") from error
48 if not isinstance(value, dict):
49 raise CliInputError("JSON configuration must be an object")

64def _require_git(repository: Path, create: bool = False) -> None:
65 if create:
66 if repository.exists() and (not repository.is_dir() or any(repository.iterdir())):
67 raise CliInputError("--create-repository accepts only a missing or empty directory")
68 repository.mkdir(parents=True, exist_ok=True)
69 result = subprocess.run(["git", "-C", str(repository), "init", "-q"], check=False, capture_output=True, creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0))
70 if result.returncode:
71 raise GitStateStoreError("git init failed")
72 result = subprocess.run(["git", "-C", str(repository), "rev-parse", "--git-dir"], check=False, capture_output=True, creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0))
73 if result.returncode:
74 raise CliInputError("--repository must name an existing Git repository; use init --create-repository explicitly")

226def _control(args: argparse.Namespace, action: str, *, text: str = "", config: dict[str, Any] | None = None) -> dict[str, Any]:
227 # ENGINE normalizes a source and derives the request ID; the CLI reports the ID it returns.
228 request = control(args.repository, args.instance, action, text=text, config=config or {},
229 origin=getattr(args, "origin", "user"), request_id=getattr(args, 'request_id', None),
230 idempotency_key=getattr(args, 'idempotency_key', None))
231 args._resolved_request_id = request['request_id']
232 # ENGINE's replay receipt is tied to the original publication. Do not run
233 # unrelated pending controls or replace that receipt with current state.
234 if request.get('effect') in {'applied', 'rejected'} and request.get('snapshot'):
235 return request
236 engine = ResearchEngine(args.repository, args.instance)
237 try:
238 engine.process_controls()
239 outcome = engine.control_result(request['request_id'])
240 except GitStateStoreLockError:
241 return {**request, "effect": "pending engine processing", "lock": "busy"}
242 except (ValueError, TypeError, KeyError) as error:
243 # Admission already produced a receipt. Failure applying that accepted
244 # control belongs to the implementation, not to the user's JSON parser.
245 raise RuntimeError(f"accepted request {request['request_id']} could not be processed: {error}") from error
246 if outcome is None:
247 return {**request, 'effect': 'pending engine processing', 'next_owner': 'engine'}
248 return {**request, **outcome}

267def _schedule_failure(result: dict[str, Any], error: Exception, repository: Path,
268 instance: str) -> tuple[dict[str, Any], int]:
269 mode = "install" if result.get("config", {}).get("operator_enabled", True) else "uninstall"
270 def quote(value):
271 return "'" + str(value).replace("'", "''") + "'"
272 entrypoint = Path(__file__).resolve().parents[1] / "do_math.py"
273 reconcile = ("& " + quote(sys.executable) + " " + quote(entrypoint)
274 + " operator " + mode + " --repository " + quote(repository.resolve())
275 + " --instance " + quote(instance))
276 result["operator_schedule"] = {"status": "failed", "error": str(error), "reconcile": reconcile}
277 return result, ERROR_EXIT_CODES['tool_error']

329 stream.write(canonical_json(value))

332 if command == "status":
333 from .reporting import snapshot_report, render_human_report

404 direction=args.direction if args.direction is not None else "both"), 0

461 pause = _control(args, "pause", text=scheduled_operator.hold_pause_reason(marker["id"]))
462 try:
463 cleanup = scheduled_operator.cancel_wake(args.repository, args.instance, workspace)
464 except (RuntimeError, OSError) as error:
465 cleanup = {"status": "failed", "error": str(error), "hold_still_blocks_wake": True}
466 return {"hold": marker, "pause": pause, "wake_cleanup": cleanup}, 0
467 if mode == "release":
468 released = scheduled_operator.release(workspace)
469 marker = released.get("released")
470 expected = scheduled_operator.hold_pause_reason(marker["id"]) if marker else None
471 if expected:
472 released["resume"] = _control(args, "resume", text="User released scheduled operator HOLD " + marker["id"],
473 config={"expected_user_pause_reason": expected})
474 else:
475 released["resume"] = {"effect": "not_requested", "reason": "no HOLD was released"}
476 return released, 0
477 if mode == "wake":
478 return scheduled_operator.wake(args.repository, args.instance, workspace), 0
479 if mode == "engine":
480 return scheduled_operator.run_engine(args.repository, args.instance, workspace), 0
481 return scheduled_operator.check(args.repository, args.instance, workspace, probe=mode == "probe"), 0
482 raise CliInputError("unknown command")

498def describe_commands(parser, command=None):
499 commands = {}
500 for name, subparser in _command_parsers(parser).items():
501 if command is not None and name != command:
502 continue
503 properties, required = {}, []
504 for action in subparser._actions:
505 if action.dest == 'help':
506 continue
507 scalar = {'type':'integer' if action.type is int else 'string'}
508 if action.choices is not None:
509 scalar['enum'] = list(action.choices)
510 definition = ({'type':'boolean'} if isinstance(action,argparse._StoreTrueAction) else
511 {'type':'array','items':scalar} if isinstance(action,argparse._AppendAction) or action.nargs in ('+','*') else scalar)
512 if action.default is not None and action.default != argparse.SUPPRESS:
513 definition['default'] = action.default
514 if action.help:
515 definition['description'] = action.help
516 properties[action.dest] = definition
517 if action.required:
518 required.append(action.dest)
519 schema = {'type':'object','properties':properties,'required':required,'additionalProperties':False}
520 if name == 'query':
521 schema['allOf'] = [{
522 'if': {'properties': {'files': {'const': True}}, 'required': ['files']},
523 'then': {'not': {'anyOf': [{'required': [key]} for key in ('kind', 'related_to', 'direction')]}},
524 'else': {'not': {'required': ['prefix']}},
525 }]
526 schema['dependentRequired'] = {'direction': ['related_to']}
527 groups = []
528 for group in subparser._mutually_exclusive_groups:
529 names = [action.dest for action in group._group_actions]
530 groups.append({'names':names,'required':group.required})
531 commands[name] = {'payload_schema':schema,'exclusive_groups':groups}
532 if command is not None and not commands:
533 raise CliInputError('unknown command for --describe')
534 request_schema = _request_schema(parser)
535 return {'schema_version':WIRE_VERSION,'commands':commands,'request_schema':request_schema,
536 'request_fields':list(request_schema['properties']),
537 'response_fields':[*RESPONSE_FIELDS,'result OR error'],
538 'inspection_exit_codes':{status:inspection_exit_code({'status':status})
539 for status in ('healthy','degraded','blocked','incomplete')},
540 'error_exit_codes':dict(ERROR_EXIT_CODES),
541 'caller_authority':'caller is a declared origin, not strong authentication'}

544def _json_arguments(parser, request):
545 from jsonschema import Draft202012Validator
546 errors = list(Draft202012Validator(_request_schema(parser)).iter_errors(request))
547 if errors:
548 raise CliInputError('invalid command request: ' + errors[0].message)
549 # The request schema has already fixed the command and caller to their enums.
550 command = request['command']
551 description = describe_commands(parser, command)
552 payload = request['payload']
553 errors = list(Draft202012Validator(description['commands'][command]['payload_schema']).iter_errors(payload))
554 if errors:
555 raise CliInputError('invalid command payload: ' + errors[0].message)
556 payload = dict(payload)
557 subparser = _command_parsers(parser)[command]
558 known = {action.dest:action for action in subparser._actions}
559 if 'caller' in request:
560 if 'origin' in known:
561 if 'origin' in payload and payload['origin'] != request['caller']:
562 raise CliInputError('caller differs from payload origin')
563 payload['origin'] = request['caller']
564 for key in ('request_id','idempotency_key'):
565 if key in request:
566 if key not in known or key in payload:
567 raise CliInputError('unexpected or duplicate ' + key)
568 payload[key] = request[key]
569 argv = [command]
570 for name, value in payload.items():
571 action = known[name]
572 flag = next((flag for flag in action.option_strings if flag.startswith('--')),None)
573 if isinstance(action,argparse._StoreTrueAction):
574 if value:
575 argv.append(flag)
576 elif isinstance(action,argparse._AppendAction):
577 for item in value:
578 argv.append(f"{flag}={item}")
579 elif action.nargs in ('+','*'):
580 argv.extend([flag,*map(str,value)])
581 else:
582 argv.append(f"{flag}={value}" if flag else str(value))
583 args = parser.parse_args(argv)
584 if command == 'status':
585 args.json = True
586 return args

605def main(argv: list[str] | None = None) -> int:
606 parser = build_parser()
607 argv = list(sys.argv[1:] if argv is None else argv)
608 command, identifier = None, None
609 force_json = bool(argv and argv[0] == '--request-file')
610 try:
611 if argv and argv[0] == '--describe':
612 if len(argv) > 2:
613 raise CliInputError('--describe accepts at most one command name')
614 result = describe_commands(parser,argv[1] if len(argv)==2 else None)
615 _emit(result,command='describe',request_id='QR-'+exact_content_hash(argv))
616 return 0
617 if force_json:
618 if len(argv) != 2:
619 raise CliInputError('--request-file accepts exactly one path')
620 request = _json_file(argv[1])
621 command = request.get('command')
622 args = _json_arguments(parser,request)
623 else:
624 args = parser.parse_args(argv)
625 command = args.command
626 normalized = {key:(str(value) if isinstance(value,Path) else value) for key,value in vars(args).items()
627 if key not in {'describe','request_file','json'}}
628 identifier = 'QR-' + exact_content_hash(json_safe(normalized))
629 try:
630 result, code = execute(args)
631 finally:
632 identifier = getattr(args, '_resolved_request_id', identifier)
633 except ValueError as error: # includes CliInputError
634 return _fail('input_error', error, 'user', False, command, identifier)
635 except (TypeError, KeyError) as error:

637 except (GitStateStoreConflict, GitStateStoreError, OSError, subprocess.SubprocessError) as error:

640 except RuntimeError as error: # after the store errors, which are RuntimeErrors
641 return _fail('contract_error', error, 'maintenance', False, command, identifier)
642 if isinstance(result,dict):

644 _emit(result,command=command,request_id=identifier,code=code,force_json=force_json)
645 return code
