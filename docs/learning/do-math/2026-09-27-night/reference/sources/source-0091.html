102def dependency_cycles(tasks):
103 """Return explicit task dependency cycles, including self-dependency."""
104 cycles, done, path = [], set(), []
105 def visit(identifier):
106 if identifier in path:
107 cycles.append(path[path.index(identifier):] + [identifier])
108 return
109 if identifier in done:
110 return
111 path.append(identifier)
112 for dependency in tasks[identifier].get("dependencies", []):
113 if dependency in tasks:
114 visit(dependency)
115 path.pop()
116 done.add(identifier)
117 for identifier in tasks:
118 visit(identifier)
119 return cycles
