def _seam(i):
    c = dict(components(i))
    return c["J"], c["G"]
def _fmt(x):
    return f"[{mp.nstr(mp.mpf(x.a), 16)}, {mp.nstr(mp.mpf(x.b), 16)}]"
J2, G2 = _seam(3999)
J21, G21 = _seam(4199)
